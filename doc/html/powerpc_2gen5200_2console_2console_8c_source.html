184 static int mpc5200_psc_setAttributes(
185  int minor,
186  const struct termios *t
187 )
188 {
189  int baud;
190  uint8_t csize=0, cstopb, parenb, parodd;
191  struct mpc5200_psc *psc =
192  (struct mpc5200_psc *)(&mpc5200.psc[psc_minor_to_regset[minor]]);
193 
194  /* Baud rate */
195  baud = rtems_termios_baud_to_number(t->c_ospeed);
196  if (baud > 0) {
197  /*
198  * Calculate baud rate
199  * round divider to nearest!
200  */
201  baud = (IPB_CLOCK + baud *16) / (baud * 32);
202  }
203 
204  /* Number of data bits */
205  switch ( t->c_cflag & CSIZE ) {
206  case CS5: csize = 0x00; break;
207  case CS6: csize = 0x01; break;
208  case CS7: csize = 0x02; break;
209  case CS8: csize = 0x03; break;
210  }
211 
212  /* Stop bits */
213  if(csize == 0) {
214  if(t->c_cflag & CSTOPB)
215  cstopb = 0x0F; /* Two stop bits */
216  else
217  cstopb = 0x00; /* One stop bit */
218  } else {
219  if(t->c_cflag & CSTOPB)
220  cstopb = 0x0F; /* Two stop bits */
221  else
222  cstopb = 0x07; /* One stop bit */
223  }
224 
225  /* Parity */
226  if (t->c_cflag & PARENB)
227  parenb = 0x00; /* Parity enabled on Tx and Rx */
228  else
229  parenb = 0x10; /* No parity on Tx and Rx */
230 
231  if (t->c_cflag & PARODD)
232  parodd = 0x04; /* Odd parity */
233  else
234  parodd = 0x00;
235 
236  /*
237  * Set upper timer counter
238  */
239  psc->ctur = (uint8_t) (baud >> 8);
240 
241  /*
242  * Set lower timer counter
243  */
244  psc->ctlr = (uint8_t) baud;
245 
246  /*
247  * Reset mode pointer
248  */
249  psc->cr = ((1 << 4) << 8);
250 
251  /*
252  * Set mode1 register
253  */
254  channel_info[minor].shadow_mode1 &= ~(0x1F);
255  psc->mr = channel_info[minor].shadow_mode1 | (csize | parenb | parodd);
256 
257  /*
258  * Set mode2 register
259  */
260  channel_info[minor].shadow_mode2 &= ~(0x0F);
261  psc->mr = channel_info[minor].shadow_mode2 | cstopb;
262 
263  return 0;
264 
265 }

284 static void mpc5200_psc_interrupt_handler(rtems_irq_hdl_param handle)
285 {
286  unsigned char c;
287  uint16_t isr;
288  int minor = (int)handle;
289  struct mpc5200_psc *psc =
290  (struct mpc5200_psc *)(&mpc5200.psc[psc_minor_to_regset[minor]]);
291 
292  /*
293  * get content of psc interrupt status
294  */
295  isr = psc->isr_imr;
296 
297  /*
298  * Character received?
299  */
300  if (isr & ISR_RX_RDY_FULL) {
301 
302  channel_info[minor].rx_interrupts++;
303 
304 
305 #ifndef SINGLE_CHAR_MODE
306  while(psc->rfnum) {
307 #endif
308  /*
309  * get the character
310  */
311  c = (psc->rb_tb >> 24);
312 
313  if (ttyp[minor] != NULL) {
314  rtems_termios_enqueue_raw_characters(
315  (void *)ttyp[minor], (char *)&c, (int)1);
316  channel_info[minor].rx_characters++;
317  }
318 
319 #ifndef SINGLE_CHAR_MODE
320  }
321 #endif
322 
323  }
324 
325  /*
326  * Character transmitted ?
327  */
328  if (isr & ISR_TX_RDY & channel_info[minor].shadow_imr) {
329  channel_info[minor].tx_interrupts++;
330 
331  if (ttyp[minor] != NULL) {
332  #ifndef SINGLE_CHAR_MODE
333  rtems_termios_dequeue_characters(
334  (void *)ttyp[minor], channel_info[minor].cur_tx_len);
335  channel_info[minor].tx_characters += channel_info[minor].cur_tx_len;
336  #else
337  rtems_termios_dequeue_characters((void *)ttyp[minor], (int)1);
338  channel_info[minor].tx_characters++;
339  #endif
340  }
341  }
342 
343  if(isr & ISR_ERROR) {
344  if(isr & ISR_RB)
345  channel_info[minor].breaks_detected++;
346  if(isr & ISR_FE)
347  channel_info[minor].framing_errors++;
348  if(isr & ISR_PE)
349  channel_info[minor].parity_errors++;
350  if(isr & ISR_OE)
351  channel_info[minor].overrun_errors++;
352 
353  /*
354  * Reset error status
355  */
356  psc->cr = ((4 << 4) << 8);
357  }
358 }

407 static void mpc5200_uart_psc_initialize(
408  int minor
409 )
410 {
411  uint32_t baud_divider;
412  struct mpc5200_psc *psc =
413  (struct mpc5200_psc *)(&mpc5200.psc[psc_minor_to_regset[minor]]);
414 
415  /*
416  * Check that minor number is valid
417  */
418  if ((minor < PSC1_MINOR) || (minor >= (PSC1_MINOR + NUM_PORTS)))
419  return;
420 
421  /*
422  * Clear per channel info
423  */
424  memset((void *)&channel_info[minor], 0, sizeof(struct per_channel_info));
425 
426  /*
427  * Reset receiver and transmitter
428  */
429  psc->cr = ((2 << 4) << 8);
430  psc->cr = ((3 << 4) << 8);
431 
432  /*
433  * Reset mode pointer
434  */
435  psc->cr = ((1 << 4) << 8);
436 
437  /*
438  * Set clock select register
439  */
440  psc->sr_csr = 0;
441 
442  /*
443  * Set mode1 register
444  */
445  psc->mr = channel_info[minor].shadow_mode1 = 0x33; /* 8Bit / no parity */
446 
447  /*
448  * Set mode2 register
449  */
450  psc->mr = channel_info[minor].shadow_mode2 = 7; /* 1 stop bit */
451 
452  /*
453  * Set rx FIFO alarm
454  */
455  psc->rfalarm = RX_FIFO_SIZE - 1;
456 
457  /*
458  * Set tx FIFO alarm
459  */
460  psc->tfalarm = 1;
461 
462  baud_divider =
463  (IPB_CLOCK + GEN5200_CONSOLE_BAUD *16) / (GEN5200_CONSOLE_BAUD * 32);
464 
465  /*
466  * Set upper timer counter
467  */
468  psc->ctur = baud_divider >> 16;
469 
470  /*
471  * Set lower timer counter
472  */
473 
474  psc->ctlr = baud_divider & 0x0000ffff;
475 
476  /*
477  * Disable Frame mode / set granularity 0
478  */
479  psc->tfcntl = 0;
480 
481 #ifdef UARTS_USE_TERMIOS_INT
482  /*
483  * Tie interrupt dependent routines
484  */
485  consoleIrqData.on = mpc5200_psc_enable;
486  consoleIrqData.off = mpc5200_psc_disable;
487  consoleIrqData.isOn = mpc5200_psc_isOn;
488  consoleIrqData.handle = (rtems_irq_hdl_param)minor;
489  consoleIrqData.hdl = (rtems_irq_hdl)mpc5200_psc_interrupt_handler;
490 
491  /*
492  * Tie interrupt handler
493  */
494  consoleIrqData.name = psc_minor_to_irqname[minor];
495 
496  /*
497  * Install rtems irq handler
498  */
499  assert(BSP_install_rtems_irq_handler(&consoleIrqData) == 1);
500 #endif
501 
502  /*
503  * Reset rx fifo errors Error/UF/OF
504  */
505  psc->rfstat |= 0x70;
506 
507  /*
508  * Reset tx fifo errors Error/UF/OF
509  */
510  psc->tfstat |= 0x70;
511 
512 #ifdef UARTS_USE_TERMIOS_INT
513  /*
514  * Unmask receive interrupt
515  */
516  psc->isr_imr = channel_info[minor].shadow_imr = IMR_RX_RDY_FULL;
517 #endif
518 
519  /*
520  * Enable receiver
521  */
522  psc->cr = ((1 << 0) << 8);
523 
524  /*
525  * Enable transmitter
526  */
527  psc->cr = ((1 << 2) << 8);
528 }

574 static ssize_t mpc5200_uart_write(
575  int minor,
576  const char *buf,
577  size_t len
578 )
579 {
580  struct mpc5200_psc *psc =
581  (struct mpc5200_psc *)(&mpc5200.psc[psc_minor_to_regset[minor]]);
582 
583  if (len > 0) {
584  int frame_len = len;
585  const char *frame_buf = buf;
586 
587  /*
588  * Check tx fifo space
589  */
590  if(len > (TX_FIFO_SIZE - psc->tfnum))
591  frame_len = TX_FIFO_SIZE - psc->tfnum;
592 
593 #ifndef SINGLE_CHAR_MODE
594  channel_info[minor].cur_tx_len = frame_len;
595 #else
596  frame_len = 1;
597 #endif
598 
599  /*rtems_cache_flush_multiple_data_lines( (void *)frame_buf, frame_len);*/
600 
601  while (frame_len--)
602  /* perform byte write to avoid extra NUL characters */
603  (* (volatile char *)&(psc->rb_tb)) = *frame_buf++;
604 
605  /*
606  * unmask interrupt
607  */
608  psc->isr_imr = channel_info[minor].shadow_imr |= IMR_TX_RDY;
609  } else {
610  /*
611  * mask interrupt
612  */
613  psc->isr_imr = channel_info[minor].shadow_imr &= ~(IMR_TX_RDY);
614  }
615 
616  return 0;
617 }

662 /*
663  * Initialize and register the device
664  */
665 rtems_device_driver console_initialize(
666  rtems_device_major_number major,
667  rtems_device_minor_number minor,
668  void *arg
669 )
670 {
671  rtems_status_code status;
672  rtems_device_minor_number console_minor;
673  char dev_name[] = "/dev/ttyx";
674  uint32_t tty_num = 0;
675  bool first = true;
676 
677  /*
678  * Always use and set up TERMIOS
679  */
680  console_minor = PSC1_MINOR;
681  rtems_termios_initialize();
682 
683  for (console_minor = PSC1_MINOR;
684  console_minor < PSC1_MINOR + NUM_PORTS;
685  console_minor++) {
686  /*
687  * check, whether UART is available for this board
688  */
689  if (0 != ((1 << console_minor) & (mpc5200_uart_avail_mask))) {
690  /*
691  * Do device-specific initialization and registration for Motorola IceCube
692  */
693  mpc5200_uart_psc_initialize(console_minor); /* /dev/tty0 */
694  dev_name[8] = '0' + tty_num;
695  status = rtems_io_register_name (dev_name, major, console_minor);
696  assert(status == RTEMS_SUCCESSFUL);
697 
698  #ifdef MPC5200_PSC_INDEX_FOR_GPS_MODULE
699  if (console_minor == MPC5200_PSC_INDEX_FOR_GPS_MODULE) {
700  status = rtems_io_register_name("/dev/gps", major, console_minor);
701  assert(status == RTEMS_SUCCESSFUL);
702  }
703  #endif
704 
705  if (first) {
706  first = false;
707 
708  /* Now register the RTEMS console */
709  status = rtems_io_register_name ("/dev/console", major, console_minor);
710  assert(status == RTEMS_SUCCESSFUL);
711  }
712 
713  tty_num++;
714  }
715  }
716 
717  console_initialized = true;
718  return RTEMS_SUCCESSFUL;
719 }
