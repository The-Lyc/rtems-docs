47  *
48  * this function sets the interrupt level of the specified interrupt
49  *
50  * parameters:
51  * - irq : interrupt number
52  * - prio: priority to set for this interrupt number
53  *
54  * returns: 0 if ok
55  * -1 on error

58 unsigned int sh_set_irq_priority(
59  unsigned int irq,
60  unsigned int prio )
61 {
62  uint32_t shiftcount;
63  uint32_t prioreg;
64  uint16_t temp16;
65  ISR_Level level;
66 
67  /*
68  * first check for valid interrupt
69  */
70  if (( irq > 156) || (irq < 64) || (_Hardware_isr_Table[irq] == _dummy_isp))
71  return -1;
72  /*
73  * check for valid irq priority
74  */
75  if ( prio > 15 )
76  return -1;
77 
78  /*
79  * look up appropriate interrupt priority register
80  */
81  if ( irq > 71)
82  {
83  irq = irq - 72;
84  shiftcount = 12 - ((irq & ~0x03) % 16);
85 
86  switch( irq / 16)
87  {
88  case 0: { prioreg = INTC_IPRC; break;}
89  case 1: { prioreg = INTC_IPRD; break;}
90  case 2: { prioreg = INTC_IPRE; break;}
91  case 3: { prioreg = INTC_IPRF; break;}
92  case 4: { prioreg = INTC_IPRG; break;}
93  case 5: { prioreg = INTC_IPRH; break;}
94  default: return -1;
95  }
96  }
97  else
98  {
99  shiftcount = 12 - 4 * ( irq % 4);
100  if ( irq > 67)
101  prioreg = INTC_IPRB;
102  else
103  prioreg = INTC_IPRA;
104  }
105 
106  /*
107  * Set the interrupt priority register
108  */
109  _ISR_Local_disable( level );
110 
111  temp16 = read16( prioreg);
112  temp16 &= ~( 15 << shiftcount);
113  temp16 |= prio << shiftcount;
114  write16( temp16, prioreg);
115 
116  _ISR_Local_enable( level );
117 
118  return 0;
119 }

125 void __ISR_Handler( uint32_t vector)
126 {
127  ISR_Level level;
128 
129  _ISR_Local_disable( level );
130 
131  _Thread_Dispatch_disable();
132 
133  if ( _ISR_Nest_level == 0 )
134  {
135  /* Install irq stack */
136  _old_stack_ptr = stack_ptr;
137  stack_ptr = _CPU_Interrupt_stack_high;
138  }
139 
140  _ISR_Nest_level++;
141 
142  _ISR_Local_enable( level );
143 
144  /* call isp */
145  if ( _ISR_Vector_table[ vector])
146  (*_ISR_Vector_table[ vector ])( vector );
147 
148  _ISR_Local_disable( level );
149 
150  _Thread_Dispatch_unnest( _Per_CPU_Get() );
151 
152  _ISR_Nest_level--;
153 
154  if ( _ISR_Nest_level == 0 )
155  /* restore old stack pointer */
156  stack_ptr = _old_stack_ptr;
157 
158  _ISR_Local_enable( level );
159 
160  if ( _ISR_Nest_level )
161  return;
162 
163  if ( !_Thread_Dispatch_is_enabled() ) {
164  return;
165  }
166 
167  if ( _Thread_Dispatch_necessary ) {
168  _Thread_Dispatch();
169  }
