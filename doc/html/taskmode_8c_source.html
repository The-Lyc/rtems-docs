38  RTEMS_API_Control *api;
39  ASR_Information *asr;
40  bool preempt_enabled;
41  bool needs_asr_dispatching;
42  rtems_mode old_mode;
43 
44  executing = _Thread_Get_executing();
45 
46  if ( !previous_mode_set )
47  return RTEMS_INVALID_ADDRESS;
48 
49 #if defined(RTEMS_SMP)
50  if (
51  ( mask & RTEMS_PREEMPT_MASK ) != 0
52  && !_Modes_Is_preempt( mode_set )
53  && !_Scheduler_Is_non_preempt_mode_supported(
54  _Thread_Scheduler_get_home( executing )
55  )
56  ) {
57  return RTEMS_NOT_IMPLEMENTED;
58  }
59 #endif
60 
61 #if defined(RTEMS_SMP) || CPU_ENABLE_ROBUST_THREAD_DISPATCH == TRUE
62  if (
63  ( mask & RTEMS_INTERRUPT_MASK ) != 0
64  && _Modes_Get_interrupt_level( mode_set ) != 0
65 #if CPU_ENABLE_ROBUST_THREAD_DISPATCH == FALSE
66  && _SMP_Need_inter_processor_interrupts()
67 #endif
68  ) {

70  }
71 #endif
72 
73  /*
74  * Complete all error checking before doing any operations which
75  * impact the executing thread. There should be no errors returned
76  * past this point.
77  */
78 
79  api = executing->API_Extensions[ THREAD_API_RTEMS ];
80  asr = &api->Signal;
81 
82  old_mode = (executing->is_preemptible) ? RTEMS_PREEMPT : RTEMS_NO_PREEMPT;
83 
84  if ( executing->budget_algorithm == THREAD_CPU_BUDGET_ALGORITHM_NONE )
85  old_mode |= RTEMS_NO_TIMESLICE;
86  else
87  old_mode |= RTEMS_TIMESLICE;
88 
89  old_mode |= (asr->is_enabled) ? RTEMS_ASR : RTEMS_NO_ASR;
90  old_mode |= _ISR_Get_level();
91 
92  *previous_mode_set = old_mode;
93 
94  /*
95  * These are generic thread scheduling characteristics.
96  */
97  preempt_enabled = false;
98  if ( mask & RTEMS_PREEMPT_MASK ) {
99  bool is_preempt_enabled = _Modes_Is_preempt( mode_set );
100 
101  preempt_enabled = !executing->is_preemptible && is_preempt_enabled;
102  executing->is_preemptible = is_preempt_enabled;
103  }
104 
105  if ( mask & RTEMS_TIMESLICE_MASK ) {
106  if ( _Modes_Is_timeslice(mode_set) ) {
107  executing->budget_algorithm = THREAD_CPU_BUDGET_ALGORITHM_RESET_TIMESLICE;
108  executing->cpu_time_budget =

110  } else
111  executing->budget_algorithm = THREAD_CPU_BUDGET_ALGORITHM_NONE;
112  }
113 
114  /*
115  * Set the new interrupt level
116  */
117  if ( mask & RTEMS_INTERRUPT_MASK ) {
118  _Modes_Set_interrupt_level( mode_set );
119  }
120 
121  /*
122  * This is specific to the RTEMS API
123  */
124  needs_asr_dispatching = false;
125  if ( mask & RTEMS_ASR_MASK ) {
126  bool is_asr_enabled = !_Modes_Is_asr_disabled( mode_set );
127 
128  _Thread_State_acquire( executing, &lock_context );
129 
130  if ( is_asr_enabled != asr->is_enabled ) {
131  asr->is_enabled = is_asr_enabled;
132 
133  if ( _ASR_Swap_signals( asr ) != 0 ) {
134  needs_asr_dispatching = true;
135  _Thread_Add_post_switch_action(
136  executing,
137  &api->Signal_action,
138  _Signal_Action_handler
139  );
140  }
141  }
142 
143  _Thread_State_release( executing, &lock_context );
144  }
145 
146  if ( preempt_enabled || needs_asr_dispatching ) {
147  Per_CPU_Control *cpu_self;
148 
149  cpu_self = _Thread_Dispatch_disable();
150  _Thread_State_acquire( executing, &lock_context );
151  _Scheduler_Schedule( executing );
152  _Thread_State_release( executing, &lock_context );
153  _Thread_Dispatch_direct( cpu_self );
154  }
155 
156  return RTEMS_SUCCESSFUL;
157 }
