127 /*=========================================================================*\
128 | Function: |
129 \*-------------------------------------------------------------------------*/
130 static void bestcomm_glue_irq_dispatcher
131 (
132 /*-------------------------------------------------------------------------*\
133 | Purpose: |
134 | general bestcomm interrupt handler/dispatcher |
135 +---------------------------------------------------------------------------+
136 | Input Parameters: |
137 \*-------------------------------------------------------------------------*/
138  void *arg /* irq specific handle (not used) */
139 )
140 /*-------------------------------------------------------------------------*\
141 | Return Value: |
142 | none |
143 \*=========================================================================*/
144 {
145  uint32_t pending;
146  int curr_taskno;
147 
148  pending = mpc5200.sdma.IntPend & ~mpc5200.sdma.IntMask;
149  curr_taskno = 0;
150  while (pending != 0) {
151  if ((pending & (1UL<<curr_taskno)) != 0) {
152  if (bestcomm_glue_irq_handlers[curr_taskno].handler != NULL) {
153  /*
154  * call proper handler
155  */
156  bestcomm_glue_irq_handlers[curr_taskno].handler
157  (bestcomm_glue_irq_handlers[curr_taskno].arg);
158  }
159  else {
160  /*
161  * This should never happen. we have a pending IRQ but no handler
162  * let's clear this pending bit
163  */
164  SDMA_CLEAR_IEVENT(&mpc5200.sdma.IntPend,curr_taskno);
165  }
166  /*
167  * clear this bit in our pending copy
168  * and go to next bit
169  */
170  pending &= ~(1<<curr_taskno);
171  }
172  curr_taskno++;
173  }
174 }
