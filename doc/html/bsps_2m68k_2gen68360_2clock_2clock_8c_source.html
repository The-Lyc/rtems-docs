57 /*
58  * Set up the clock hardware
59  * The rate at which the periodic interval timer
60  * can generate interrupts is almost certainly not
61  * the same as desired by the BSP configuration.
62  * Handle the difference by choosing the largest PIT
63  * interval which is less than or equal to the RTEMS
64  * interval and skipping some hardware interrupts.
65  * To reduce the jitter in the calls to RTEMS the
66  * hardware interrupt interval is never greater than
67  * the maximum non-prescaled value from the PIT.
68  *
69  * For a 25 MHz external clock the basic clock rate is
70  * 40 nsec * 128 * 4 = 20.48 usec/tick
71  */
72 extern int m360_clock_rate;
