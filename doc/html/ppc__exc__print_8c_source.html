43 static uint32_t ppc_exc_get_DAR_dflt(void)
44 {
45  if (ppc_cpu_is_60x())
46  return PPC_SPECIAL_PURPOSE_REGISTER(PPC_DAR);
47  else
48  switch (ppc_cpu_is_bookE()) {
49  default:
50  break;
51  case PPC_BOOKE_STD:
52  case PPC_BOOKE_E500:
53  return PPC_SPECIAL_PURPOSE_REGISTER(BOOKE_DEAR);
54  case PPC_BOOKE_405:
55  return PPC_SPECIAL_PURPOSE_REGISTER(PPC405_DEAR);
56  }
57  return 0xdeadbeef;
58 }

62 void BSP_printStackTrace(const BSP_Exception_frame *excPtr)
63 {
64  LRFrame f;
65  int i;
66  LRFrame sp;
67  void *lr;
68 
69  printk("Stack Trace: \n ");
70  if (excPtr) {
71  printk("IP: 0x%08" PRIxPTR ", ", excPtr->EXC_SRR0);
72  sp = (LRFrame) GET_GPR(excPtr->GPR1);
73  lr = (void *) excPtr->EXC_LR;
74  } else {
75  /* there's no macro for this */
76  __asm__ __volatile__("mr %0, 1":"=r"(sp));
77  lr = (LRFrame) ppc_link_register();
78  }
79  printk("LR: 0x%08" PRIxPTR "\n", (uintptr_t) lr);
80  for (f = (LRFrame) sp, i = 0; f->frameLink && i < STACK_CLAMP; f = f->frameLink) {
81  printk("--^ 0x%08" PRIxPTR "", (uintptr_t) (f->frameLink->lr));
82  if (!(++i % 5))
83  printk("\n");
84  }
85  if (i >= STACK_CLAMP) {
86  printk("Too many stack frames (stack possibly corrupted), giving up...\n");
87  } else {
88  if (i % 5)
89  printk("\n");
90  }
91 }
