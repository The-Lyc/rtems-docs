226 static void _RTEMS_tasks_MP_Process_packet (
227  rtems_packet_prefix *the_packet_prefix
228 )
229 {
230  RTEMS_tasks_MP_Packet *the_packet;
231  Thread_Control *the_thread;
232 
233  the_packet = (RTEMS_tasks_MP_Packet *) the_packet_prefix;
234 
235  switch ( the_packet->operation ) {

242  the_packet->Prefix.id,
243  true
244  );
245 
246  _MPCI_Return_packet( the_packet_prefix );
247  break;
248

250 
251  _Objects_MP_Close(

256  _MPCI_Return_packet( the_packet_prefix );
257  break;

276  _MPCI_Return_packet( the_packet_prefix );

287  _Thread_Executing
288  );
289  break;

292 
293  the_packet->Prefix.return_code = rtems_task_set_priority(
294  the_packet->Prefix.id,
295  the_packet->the_priority,
296  &the_packet->the_priority
297  );
298 
299  _RTEMS_tasks_MP_Send_response_packet(
300  RTEMS_TASKS_MP_SET_PRIORITY_RESPONSE,
301  _Thread_Executing
302  );
303  break;

306

308 
309  *(rtems_task_priority *)the_thread->Wait.return_argument =
