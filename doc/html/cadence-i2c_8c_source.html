87 {

136 static void cadence_i2c_next_byte(cadence_i2c_bus *bus)
137 {
138  --bus->current_msg_todo;
139  ++bus->current_msg_byte;
140 
141  if (bus->current_msg_todo == 0) {
142  i2c_msg *msg;
143 
144  ++bus->msgs;
145  --bus->msg_todo;
146 
147  msg = &bus->msgs[0];
148 
149  bus->current_msg_todo = msg->len;
150  bus->current_msg_byte = msg->buf;
151  }
152 }

154 static void cadence_i2c_write_to_fifo(
155  cadence_i2c_bus *bus,
156  volatile cadence_i2c *regs
157 )
158 {
159  uint32_t space_available;
160  uint32_t todo_now;
161  uint32_t i;
162 
163  space_available = CADENCE_I2C_FIFO_DEPTH - regs->transfer_size;
164 
165  if (bus->current_todo > space_available) {
166  todo_now = space_available;
167  } else {
168  todo_now = bus->current_todo;
169  }
170 
171  bus->current_todo -= todo_now;
172 
173  for (i = 0; i < todo_now; ++i) {
174  regs->data = *bus->current_msg_byte;
175 
176  cadence_i2c_next_byte(bus);
177  }
178 }

192 static void cadence_i2c_setup_transfer(
193  cadence_i2c_bus *bus,
194  volatile cadence_i2c *regs
195 )
196 {
197  const i2c_msg *msgs = bus->msgs;
198  uint32_t msg_todo = bus->msg_todo;
199  uint32_t i;
200  uint32_t control;
201 
202  bus->current_todo = msgs[0].len;
203  for (i = 1; i < msg_todo && (msgs[i].flags & I2C_M_NOSTART) != 0; ++i) {
204  bus->current_todo += msgs[i].len;
205  }
206 
207  regs = bus->regs;
208 
209  control = regs->control;
210  control |= CADENCE_I2C_CONTROL_CLR_FIFO;
211 
212  bus->hold = i < msg_todo;
213 
214  if (bus->hold || bus->current_todo > CADENCE_I2C_FIFO_DEPTH) {
215  control |= CADENCE_I2C_CONTROL_HOLD;
216  } else {
217  control &= ~CADENCE_I2C_CONTROL_HOLD;
218  }
219 
220  control = cadence_i2c_set_address_size(msgs, control);
221 
222  bus->read = (msgs->flags & I2C_M_RD) != 0;
223  if (bus->read) {
224  cadence_i2c_setup_read_transfer(bus, regs, control);
225  } else {
226  cadence_i2c_setup_write_transfer(bus, regs, control);
227  }
228 
229  cadence_i2c_clear_irq_status(regs);
230 
231  regs->address = CADENCE_I2C_ADDRESS(msgs->addr);
232 }

234 static void cadence_i2c_continue_read_transfer(
235  cadence_i2c_bus *bus,
236  volatile cadence_i2c *regs
237 )
238 {
239  uint32_t i;
240 
241  bus->current_todo -= CADENCE_I2C_DATA_IRQ_DEPTH;
242 
243  /*
244  * This works since CADENCE_I2C_TRANSFER_SIZE_ONCE_MAX is an integral
245  * multiple of CADENCE_I2C_DATA_IRQ_DEPTH.
246  *
247  * FIXME: Tests with a 1024 byte EEPROM show that this doesn't work. Needs
248  * further investigations with an I2C analyser or an oscilloscope.
249  */
250  if (regs->transfer_size == 0) {
251  if (bus->current_todo <= CADENCE_I2C_TRANSFER_SIZE_MAX) {
252  regs->transfer_size = bus->current_todo;
253  } else {
254  regs->transfer_size = CADENCE_I2C_TRANSFER_SIZE_ONCE_MAX;
255  }
256  }
257 
258  for (i = 0; i < CADENCE_I2C_DATA_IRQ_DEPTH; ++i) {
259  *bus->current_msg_byte = (uint8_t) regs->data;
260 
261  cadence_i2c_next_byte(bus);
262  }
263 
264  if (!bus->hold && bus->current_todo <= CADENCE_I2C_FIFO_DEPTH) {
265  regs->control &= ~CADENCE_I2C_CONTROL_HOLD;
266  }
267 }

269 static void cadence_i2c_interrupt(void *arg)
270 {
271  cadence_i2c_bus *bus = arg;
272  volatile cadence_i2c *regs = bus->regs;
273  uint32_t irqstatus = regs->irqstatus;
274  bool done = false;
275 
276  /* Clear interrupts */
277  regs->irqstatus = irqstatus;
278 
279  if ((irqstatus & (CADENCE_I2C_IXR_ARB_LOST | CADENCE_I2C_IXR_NACK)) != 0) {
280  done = true;
281  }
282 
283  if (
284  (irqstatus & CADENCE_I2C_IXR_DATA) != 0
285  && bus->read
286  && bus->current_todo >= CADENCE_I2C_DATA_IRQ_DEPTH
287  ) {
288  cadence_i2c_continue_read_transfer(bus, regs);
289  }
290 
291  if ((irqstatus & CADENCE_I2C_IXR_COMP) != 0) {
292  if (bus->read) {
293  uint32_t todo_now = bus->current_todo;
294  uint32_t i;
295 
296  for (i = 0; i < todo_now; ++i) {
297  *bus->current_msg_byte = (uint8_t) regs->data;
298 
299  cadence_i2c_next_byte(bus);
300  }
301 
302  bus->current_todo = 0;
303 
304  done = true;
305  } else {
306  if (bus->current_todo > 0) {
307  cadence_i2c_write_to_fifo(bus, regs);
308  } else {
309  done = true;
310  }
311 
312  if (!bus->hold && bus->current_todo == 0) {
313  regs->control &= ~CADENCE_I2C_CONTROL_HOLD;
314  }
315  }
316  }
317 
318  if (done) {
319  uint32_t err = irqstatus & CADENCE_I2C_IRQ_ERROR;
320 
321  if (bus->msg_todo == 0 || err != 0) {
322  rtems_status_code sc;
323 
324  cadence_i2c_disable_interrupts(regs);
325 
326  bus->irqstatus = err;
327 
328  sc = rtems_event_transient_send(bus->task_id);
329  _Assert(sc == RTEMS_SUCCESSFUL);
330  (void) sc;
331  } else {
332  cadence_i2c_setup_transfer(bus, regs);
333  }
334  }
335 }

337 static int cadence_i2c_transfer(
338  i2c_bus *base,
339  i2c_msg *msgs,
340  uint32_t msg_count
341 )
342 {
343  cadence_i2c_bus *bus = (cadence_i2c_bus *) base;
344  volatile cadence_i2c *regs;
345  rtems_status_code sc;
346  uint32_t i;
347 
348  _Assert(msg_count > 0);
349 
350  for (i = 0; i < msg_count; ++i) {
351  /* FIXME: Not sure if we can support this. */
352  if ((msgs[i].flags & I2C_M_RECV_LEN) != 0) {
353  return -EINVAL;
354  }
355  }
356 
357  bus->msgs = &msgs[0];
358  bus->msg_todo = msg_count;
359  bus->current_msg_todo = msgs[0].len;
360  bus->current_msg_byte = msgs[0].buf;
361  bus->task_id = rtems_task_self();
362 
363  regs = bus->regs;
364  cadence_i2c_setup_transfer(bus, regs);
365  regs->irqenable = CADENCE_I2C_IRQ_USED;
366 
367  sc = rtems_event_transient_receive(RTEMS_WAIT, bus->base.timeout);
368  if (sc != RTEMS_SUCCESSFUL) {
369  cadence_i2c_reset(bus);
370  rtems_event_transient_clear();
371 
372  return -ETIMEDOUT;
373  }
374 
375  return bus->irqstatus == 0 ? 0 : -EIO;
376 }

378 static int cadence_i2c_set_clock(i2c_bus *base, unsigned long clock)
379 {
380  cadence_i2c_bus *bus = (cadence_i2c_bus *) base;
381  volatile cadence_i2c *regs = bus->regs;
382  uint32_t error = 0xffffffff;
383  uint32_t best_div_a = CADENCE_I2C_DIV_A_MAX - 1;
384  uint32_t best_div_b = CADENCE_I2C_DIV_B_MAX - 1;
385  uint32_t div = bus->input_clock / (22 * clock);
386  uint32_t div_a;
387  uint32_t control;
388 
389  if (div <= 0 || div > (CADENCE_I2C_DIV_A_MAX * CADENCE_I2C_DIV_B_MAX)) {
390  return -EIO;
391  }
392 
393  for (div_a = 0; div_a < CADENCE_I2C_DIV_A_MAX; ++div_a) {
394  uint32_t a = 22 * clock * (div_a + 1);
395  uint32_t b = (bus->input_clock + a - 1) / a;
396 
397  if (b > 0 && b <= CADENCE_I2C_DIV_B_MAX) {
398  uint32_t actual_clock = bus->input_clock / (22 * (div_a + 1) * b);
399  uint32_t e = clock < actual_clock ?
400  actual_clock - clock : clock - actual_clock;
401 
402  /*
403  * Favour greater div_a values according to UG585, Zynq-7000 AP SoC
404  * Technical Reference Manual, Table 20-1: Calculated Values for Standard
405  * and High Speed SCL Clock Values".
406  */
407  if (e <= error && actual_clock <= clock) {
408  error = e;
409  best_div_a = div_a;
410  best_div_b = b - 1;
411  }
412  }
413  }
414 
415  control = regs->control;
416  control = CADENCE_I2C_CONTROL_DIV_A_SET(control, best_div_a);
417  control = CADENCE_I2C_CONTROL_DIV_B_SET(control, best_div_b);
418  regs->control = control;
419 
420  return 0;
421 }

435 int i2c_bus_register_cadence(
436  const char *bus_path,
437  uintptr_t register_base,
438  uint32_t input_clock,
439  rtems_vector_number irq
440 )
441 {
442  cadence_i2c_bus *bus;
443  rtems_status_code sc;
444  int err;
445 
446  bus = (cadence_i2c_bus *) i2c_bus_alloc_and_init(sizeof(*bus));
447  if (bus == NULL) {
448  return -1;
449  }
450 
451  bus->regs = (volatile cadence_i2c *) register_base;
452  bus->input_clock = input_clock;
453  bus->irq = irq;
454 
455  cadence_i2c_reset(bus);
456 
457  err = cadence_i2c_set_clock(&bus->base, I2C_BUS_CLOCK_DEFAULT);
458  if (err != 0) {
459  (*bus->base.destroy)(&bus->base);
460 
461  rtems_set_errno_and_return_minus_one(-err);
462  }
463 
464  sc = rtems_interrupt_handler_install(
465  irq,
466  "Cadence I2C",
467  RTEMS_INTERRUPT_UNIQUE,
468  cadence_i2c_interrupt,
469  bus
470  );
471  if (sc != RTEMS_SUCCESSFUL) {
472  (*bus->base.destroy)(&bus->base);
473 
474  rtems_set_errno_and_return_minus_one(EIO);
475  }
476 
477  bus->base.transfer = cadence_i2c_transfer;
478  bus->base.set_clock = cadence_i2c_set_clock;
479  bus->base.destroy = cadence_i2c_destroy;
480 
481  return i2c_bus_register(&bus->base, bus_path);
482 }
