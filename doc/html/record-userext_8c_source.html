35 bool _Record_Thread_create(
36  struct _Thread_Control *executing,
37  struct _Thread_Control *created
38 )
39 {
40  rtems_record_data data;
41  char name[ 2 * THREAD_DEFAULT_MAXIMUM_NAME_SIZE ];
42  rtems_record_item items[ 1 + sizeof( name ) / sizeof( data ) ];
43  size_t len;
44  size_t used;
45 
46  items[ 0 ].event = RTEMS_RECORD_THREAD_CREATE;
47  items[ 0 ].data = created->Object.id;
48 
49  len = _Thread_Get_name( created, name, sizeof( name ) );
50  used = _Record_String_to_items(
51  RTEMS_RECORD_THREAD_NAME,
52  name,
53  len,
54  &items[ 1 ],
55  RTEMS_ARRAY_SIZE( items ) - 1
56  );
57  rtems_record_produce_n( items, used + 1 );
58 
59  return true;
60 }

95 void _Record_Thread_switch(
96  struct _Thread_Control *executing,
97  struct _Thread_Control *heir
98 )
99 {
100  rtems_record_item items[ 3 ];
101 
102  items[ 0 ].event = RTEMS_RECORD_THREAD_SWITCH_OUT;
103  items[ 0 ].data = executing->Object.id;
104  items[ 1 ].event = RTEMS_RECORD_THREAD_STACK_CURRENT;
105  items[ 1 ].data =
106 #if defined(__GNUC__)
107  (uintptr_t) __builtin_frame_address( 0 )
108  - (uintptr_t) executing->Start.Initial_stack.area;
109 #else
110  0;
111 #endif
112  items[ 2 ].event = RTEMS_RECORD_THREAD_SWITCH_IN;
113  items[ 2 ].data = heir->Object.id;
114  rtems_record_produce_n( items, RTEMS_ARRAY_SIZE( items ) );
115 }
