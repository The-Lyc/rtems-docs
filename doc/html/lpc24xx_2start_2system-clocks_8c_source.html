96 unsigned lpc24xx_pllclk(void)
97 {
98  #ifdef ARM_MULTILIB_ARCH_V4
99  unsigned clksrc = GET_CLKSRCSEL_CLKSRC(CLKSRCSEL);
100  unsigned pllinclk = 0;
101  unsigned pllclk = 0;
102 
103  /* Get PLL input frequency */
104  switch (clksrc) {
105  case 0:
106  pllinclk = LPC24XX_OSCILLATOR_INTERNAL;
107  break;
108  case 1:
109  pllinclk = LPC24XX_OSCILLATOR_MAIN;
110  break;
111  case 2:
112  pllinclk = LPC24XX_OSCILLATOR_RTC;
113  break;
114  default:
115  return 0;
116  }
117 
118  /* Get PLL output frequency */
119  if ((PLLSTAT & PLLSTAT_PLLC) != 0) {
120  uint32_t pllcfg = PLLCFG;
121  unsigned n = GET_PLLCFG_NSEL(pllcfg) + 1;
122  unsigned m = GET_PLLCFG_MSEL(pllcfg) + 1;
123 
124  pllclk = (pllinclk / n) * 2 * m;
125  } else {
126  pllclk = pllinclk;
127  }
128  #else
129  volatile lpc17xx_scb *scb = &LPC17XX_SCB;
130  unsigned sysclk = lpc17xx_sysclk(scb->clksrcsel);
131  unsigned pllstat = scb->pll_0.stat;
132  unsigned pllclk = 0;
133  unsigned enabled_and_locked = LPC17XX_PLL_STAT_PLLE
134  | LPC17XX_PLL_STAT_PLOCK;
135 
136  if ((pllstat & enabled_and_locked) == enabled_and_locked) {
137  unsigned m = LPC17XX_PLL_SEL_MSEL_GET(pllstat) + 1;
138 
139  pllclk = sysclk * m;
140  }
141  #endif
142 
143  return pllclk;
144 }

159

161  cclk_in = lpc24xx_pllclk();
