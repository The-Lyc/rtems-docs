43 
44  /* Find GRLIB Interrupt controller */
45  adev = (void *)ambapp_for_each(&ambapp_plb, (OPTIONS_ALL|OPTIONS_APB_SLVS),
46  VENDOR_GAISLER, GAISLER_IRQMP,
47  ambapp_find_by_idx, NULL);
48  if (adev != NULL) {
49 
50  GRLIB_IrqCtrl_Regs = (volatile struct irqmp_regs *)DEV_TO_APB(adev)->start;
51  GRLIB_IrqCtrl_Adev = adev;
52  if ((GRLIB_IrqCtrl_Regs->ampctrl >> 28) > 0) {
53  /* IRQ Controller has support for multiple IRQ Controllers, each
54  * CPU can be routed to different Controllers, we find out which
55  * controller by looking at the IRQCTRL Select Register for this CPU.
56  * Each Controller is located at a 4KByte offset.
57  */
58  icsel = GRLIB_IrqCtrl_Regs->icsel[GRLIB_Cpu_Index/8];
59  icsel = (icsel >> ((7 - (GRLIB_Cpu_Index & 0x7)) * 4)) & 0xf;
60  GRLIB_IrqCtrl_Regs += icsel;
61  }
62  GRLIB_IrqCtrl_Regs->mask[GRLIB_Cpu_Index] = 0;
63  GRLIB_IrqCtrl_Regs->force[GRLIB_Cpu_Index] = 0;
64  GRLIB_IrqCtrl_Regs->iclear = 0xffffffff;
65 
66  /* Init Extended IRQ controller if available */
67  grlib_ext_irq_init();
68 
69  }
