82  void SystemInit(void)
83 #else /* __rtems__ */
84  void ATSAM_START_SRAM_SECTION SystemInit(void)
85 #endif /* __rtems__ */
86 {
87  uint32_t read_MOR;
88  /* Set FWS according to SYS_BOARD_MCKR configuration */
89  EFC->EEFC_FMR = EEFC_FMR_FWS(5);
90 
91  /* Before switching MAIN OSC on external crystal : enable it and don't
92  * disable at the same time RC OSC in case of if MAIN OSC is still using RC
93  * OSC
94  */
95 
96 #if ATSAM_SLOWCLOCK_USE_XTAL == 1
97  read_MOR = PMC->CKGR_MOR;
98  /* enable external crystal - enable RC OSC */
99  read_MOR |= (CKGR_MOR_KEY_PASSWD | CKGR_MOR_XT32KFME);
100  PMC->CKGR_MOR = read_MOR;
101 
102  /* Select XTAL 32k instead of internal slow RC 32k for slow clock */
103  if ((SUPC->SUPC_SR & SUPC_SR_OSCSEL) != SUPC_SR_OSCSEL_CRYST) {
104  SUPC->SUPC_CR = SUPC_CR_KEY_PASSWD | SUPC_CR_XTALSEL_CRYSTAL_SEL;
105 
106  while (!(SUPC->SUPC_SR & SUPC_SR_OSCSEL));
107  }
108 #endif
109 
110  /* Initialize main oscillator */
111  if (!(PMC->CKGR_MOR & CKGR_MOR_MOSCSEL)) {
112  PMC->CKGR_MOR = CKGR_MOR_KEY_PASSWD | SYS_BOARD_OSCOUNT | CKGR_MOR_MOSCRCEN |
113  CKGR_MOR_MOSCXTEN;
114 
115  while (!(PMC->PMC_SR & PMC_SR_MOSCXTS)) {
116  }
117  }
118 
119  /* Switch to 3-20MHz Xtal oscillator */
120  PMC->CKGR_MOR = CKGR_MOR_KEY_PASSWD | SYS_BOARD_OSCOUNT | CKGR_MOR_MOSCRCEN |
121  CKGR_MOR_MOSCXTEN | CKGR_MOR_MOSCSEL;
122 
123  while (!(PMC->PMC_SR & PMC_SR_MOSCSELS)) {
124  }
125 
126  PMC->PMC_MCKR = (PMC->PMC_MCKR & ~(uint32_t)PMC_MCKR_CSS_Msk) |
127  PMC_MCKR_CSS_MAIN_CLK;
128 
129  while (!(PMC->PMC_SR & PMC_SR_MCKRDY)) {
130  }
131 
132  /* Initialize PLLA */
133  PMC->CKGR_PLLAR = SYS_BOARD_PLLAR;
134 
135  while (!(PMC->PMC_SR & PMC_SR_LOCKA)) {
136  }
137 
138  /* Switch to main clock: DO NOT modify MDIV and CSS feild at the same access */
139  PMC->PMC_MCKR = (PMC->PMC_MCKR & ~(uint32_t)PMC_MCKR_MDIV_Msk) |
140  SYS_BOARD_MCKR_MDIV;
141  PMC->PMC_MCKR = (SYS_BOARD_MCKR & ~PMC_MCKR_CSS_Msk) | PMC_MCKR_CSS_MAIN_CLK;
142 
143  while (!(PMC->PMC_SR & PMC_SR_MCKRDY)) {
144  }
145 
146  /* Switch to PLLA */
147  PMC->PMC_MCKR = SYS_BOARD_MCKR;
148 
149  while (!(PMC->PMC_SR & PMC_SR_MCKRDY)) {
150  }
151 
152 #ifndef __rtems__
153  SystemCoreClock = CHIP_FREQ_CPU_MAX;
154 #endif /* __rtems__ */
155 }

158 void SystemCoreClockUpdate(void)
159 {
160  /* Determine clock frequency according to clock register values */
161  switch (PMC->PMC_MCKR & (uint32_t) PMC_MCKR_CSS_Msk) {
162  case PMC_MCKR_CSS_SLOW_CLK: /* Slow clock */
163  if (SUPC->SUPC_SR & SUPC_SR_OSCSEL)
164  SystemCoreClock = CHIP_FREQ_XTAL_32K;
165  else
166  SystemCoreClock = CHIP_FREQ_SLCK_RC;
167 
168  break;
169 
170  case PMC_MCKR_CSS_MAIN_CLK: /* Main clock */
171  if (PMC->CKGR_MOR & CKGR_MOR_MOSCSEL)
172  SystemCoreClock = CHIP_FREQ_XTAL_12M;
173  else {
174  SystemCoreClock = CHIP_FREQ_MAINCK_RC_4MHZ;
175 
176  switch (PMC->CKGR_MOR & CKGR_MOR_MOSCRCF_Msk) {
177  case CKGR_MOR_MOSCRCF_4_MHz:
178  break;
179 
180  case CKGR_MOR_MOSCRCF_8_MHz:
181  SystemCoreClock *= 2U;
182  break;
183 
184  case CKGR_MOR_MOSCRCF_12_MHz:
185  SystemCoreClock *= 3U;
186  break;
187 
188  default:
189  break;
190  }
191  }
192 
193  break;
194 
195  case PMC_MCKR_CSS_PLLA_CLK: /* PLLA clock */
196  if (PMC->CKGR_MOR & CKGR_MOR_MOSCSEL)
197  SystemCoreClock = CHIP_FREQ_XTAL_12M;
198  else {
199  SystemCoreClock = CHIP_FREQ_MAINCK_RC_4MHZ;
200 
201  switch (PMC->CKGR_MOR & CKGR_MOR_MOSCRCF_Msk) {
202  case CKGR_MOR_MOSCRCF_4_MHz:
203  break;
204 
205  case CKGR_MOR_MOSCRCF_8_MHz:
206  SystemCoreClock *= 2U;
207  break;
208 
209  case CKGR_MOR_MOSCRCF_12_MHz:
210  SystemCoreClock *= 3U;
211  break;
212 
213  default:
214  break;
215  }
216  }
217 
218  if ((uint32_t) (PMC->PMC_MCKR & (uint32_t) PMC_MCKR_CSS_Msk) ==
219  PMC_MCKR_CSS_PLLA_CLK) {
220  SystemCoreClock *= ((((PMC->CKGR_PLLAR) & CKGR_PLLAR_MULA_Msk) >>
221  CKGR_PLLAR_MULA_Pos) + 1U);
222  SystemCoreClock /= ((((PMC->CKGR_PLLAR) & CKGR_PLLAR_DIVA_Msk) >>
223  CKGR_PLLAR_DIVA_Pos));
224  }
225 
226  break;
227 
228  default:
229  break;
230  }
231 
232  if ((PMC->PMC_MCKR & PMC_MCKR_PRES_Msk) == PMC_MCKR_PRES_CLK_3)
233  SystemCoreClock /= 3U;
234  else
235  SystemCoreClock >>= ((PMC->PMC_MCKR & PMC_MCKR_PRES_Msk) >> PMC_MCKR_PRES_Pos);
236 }
237 #endif /* __rtems__ */

241 void system_init_flash(uint32_t ul_clk)
242 {
243  /* Set FWS for embedded Flash access according to operating frequency */
244  if (ul_clk < CHIP_FREQ_FWS_0)
245  EFC->EEFC_FMR = EEFC_FMR_FWS(0) | EEFC_FMR_CLOE;
246  else {
247  if (ul_clk < CHIP_FREQ_FWS_1)
248  EFC->EEFC_FMR = EEFC_FMR_FWS(1) | EEFC_FMR_CLOE;
249  else {
250  if (ul_clk < CHIP_FREQ_FWS_2)
251  EFC->EEFC_FMR = EEFC_FMR_FWS(2) | EEFC_FMR_CLOE;
252  else {
253  if (ul_clk < CHIP_FREQ_FWS_3)
254  EFC->EEFC_FMR = EEFC_FMR_FWS(3) | EEFC_FMR_CLOE;
255  else {
256  if (ul_clk < CHIP_FREQ_FWS_4)
257  EFC->EEFC_FMR = EEFC_FMR_FWS(4) | EEFC_FMR_CLOE;
258  else
259  EFC->EEFC_FMR = EEFC_FMR_FWS(5) | EEFC_FMR_CLOE;
260  }
261  }
262  }
263  }

272 void sysclk_enable_usb(void)
273 {
274  /* Disable FS USB clock*/
275  PMC->PMC_SCDR = PMC_SCDR_USBCLK;
276 
277  /* Enable PLL 480 MHz */
278  PMC->CKGR_UCKR = CKGR_UCKR_UPLLEN | CKGR_UCKR_UPLLCOUNT(0xF);
279 
280  /* Wait that PLL is considered locked by the PMC */
281  while (!(PMC->PMC_SR & PMC_SR_LOCKU));
282 
283  /* USB clock register: USB Clock Input is UTMI PLL */
284  PMC->PMC_USB = (PMC_USB_USBS | PMC_USB_USBDIV(USBCLK_DIV - 1));
285 
286  PMC->PMC_SCER = PMC_SCER_USBCLK;
287 }
