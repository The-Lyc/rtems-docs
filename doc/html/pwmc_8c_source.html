101 static uint16_t FindClockConfiguration(
102  uint32_t frequency,
103  uint32_t mck)
104 {
105  uint32_t divisors[11] = {1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024};
106  uint8_t divisor = 0;
107  uint32_t prescaler;
108 
109  assert(frequency <= mck);
110 
111  /* Find prescaler and divisor values */
112  prescaler = (mck / divisors[divisor]) / frequency;
113 
114  while ((prescaler > 255) && (divisor < 11)) {
115  divisor++;
116  prescaler = (mck / divisors[divisor]) / frequency;
117  }
118 
119  /* Return result */
120  if (divisor < 11) {
121  TRACE_DEBUG("Found divisor=%u and prescaler=%u for freq=%uHz\n\r",
122  divisors[divisor], prescaler, frequency);
123 
124  return prescaler | (divisor << 8);
125  } else
126  return 0;
127 }

146 void PWMC_ConfigureChannel(
147  Pwm *pPwm,
148  uint8_t channel,
149  uint32_t prescaler,
150  uint32_t alignment,
151  uint32_t polarity)
152 {
153  pPwm->PWM_CH_NUM[0].PWM_CMR = 1;
154 
155  // assert(prescaler < PWM_CMR0_CPRE_MCKB);
156  assert((alignment & (uint32_t)~PWM_CMR_CALG) == 0);
157  assert((polarity & (uint32_t)~PWM_CMR_CPOL) == 0);
158 
159  /* Disable channel (effective at the end of the current period) */
160  if ((pPwm->PWM_SR & (1 << channel)) != 0) {
161  pPwm->PWM_DIS = 1 << channel;
162 
163  while ((pPwm->PWM_SR & (1 << channel)) != 0);
164  }
165 
166  /* Configure channel */
167  pPwm->PWM_CH_NUM[channel].PWM_CMR = prescaler | alignment | polarity;
168 }

226 void PWMC_ConfigureClocks(Pwm *pPwm, uint32_t clka, uint32_t clkb, uint32_t mck)
227 {
228  uint32_t mode = 0;
229  uint32_t result;
230 
231  /* Clock A */
232  if (clka != 0) {
233  result = FindClockConfiguration(clka, mck);
234  assert(result != 0);
235  mode |= result;
236  }
237 
238  /* Clock B */
239  if (clkb != 0) {
240  result = FindClockConfiguration(clkb, mck);
241  assert(result != 0);
242  mode |= (result << 16);
243  }
244 
245  /* Configure clocks */
246  TRACE_DEBUG("Setting PWM_CLK = 0x%08X\n\r", mode);
247  pPwm->PWM_CLK = mode;
248 }
