148 static rtems_status_code rtems_bdpart_read_mbr_partition(
149  const uint8_t *data,
150  rtems_bdpart_partition **p,
151  const rtems_bdpart_partition *p_end,
152  rtems_blkdev_bnum *ep_begin
153 )
154 {
155  rtems_blkdev_bnum begin =
156  rtems_uint32_from_little_endian( data + RTEMS_BDPART_MBR_OFFSET_BEGIN);
157  rtems_blkdev_bnum size =
158  rtems_uint32_from_little_endian( data + RTEMS_BDPART_MBR_OFFSET_SIZE);
159  rtems_blkdev_bnum end = begin + size;
160  uint8_t type = data [RTEMS_BDPART_MBR_OFFSET_TYPE];
161 
162  if (type == RTEMS_BDPART_MBR_EMPTY) {
163  return RTEMS_SUCCESSFUL;
164  } else if (*p == p_end) {
165  return RTEMS_TOO_MANY;
166  } else if (begin >= end) {
167  return RTEMS_IO_ERROR;
168  } else if (type == RTEMS_BDPART_MBR_EXTENDED) {
169  if (ep_begin != NULL) {
170  *ep_begin = begin;
171  }
172  } else {
173  /* Increment partition index */
174  ++(*p);
175 
176  /* Clear partition */
177  memset( *p, 0, sizeof( rtems_bdpart_partition));
178 
179  /* Set values */
180  (*p)->begin = begin;
181  (*p)->end = end;
182  rtems_bdpart_to_partition_type( type, (*p)->type);
183  (*p)->flags = data [RTEMS_BDPART_MBR_OFFSET_FLAGS];
184  }
185 
186  return RTEMS_SUCCESSFUL;
187 }

189 static rtems_status_code rtems_bdpart_read_record(
190  rtems_disk_device *dd,
191  rtems_blkdev_bnum index,
192  rtems_bdbuf_buffer **block
193 )
194 {
195  rtems_status_code sc = RTEMS_SUCCESSFUL;
196 
197  /* Release previous block if necessary */
198  if (*block != NULL) {
199  sc = rtems_bdbuf_release( *block);
200  if (sc != RTEMS_SUCCESSFUL) {
201  return sc;
202  }
203  }
204 
205  /* Read the record block */
206  sc = rtems_bdbuf_read( dd, index, block);
207  if (sc != RTEMS_SUCCESSFUL) {
208  return sc;
209  }
210 
211  /* just in case block did not get filled in */
212  if ( *block == NULL ) {
213  return RTEMS_INVALID_ADDRESS;
214  }
215 
216  /* Check MBR signature */
217  if (!rtems_bdpart_is_valid_record( (*block)->buffer)) {
218  return RTEMS_IO_ERROR;
219  }
220 
221  return RTEMS_SUCCESSFUL;
222 }

224 rtems_status_code rtems_bdpart_read(
225  const char *disk_name,
226  rtems_bdpart_format *format,
227  rtems_bdpart_partition *pt,
228  size_t *count
229 )
230 {
231  rtems_status_code sc = RTEMS_SUCCESSFUL;
232  rtems_status_code esc = RTEMS_SUCCESSFUL;
233  rtems_bdbuf_buffer *block = NULL;
234  rtems_bdpart_partition *p = pt - 1;
235  const rtems_bdpart_partition *p_end = pt + (count != NULL ? *count : 0);
236  rtems_blkdev_bnum ep_begin = 0; /* Extended partition begin */
237  rtems_blkdev_bnum ebr = 0; /* Extended boot record block index */
238  rtems_blkdev_bnum disk_end = 0;
239  size_t i = 0;
240  const uint8_t *data = NULL;
241  int fd = -1;
242  rtems_disk_device *dd = NULL;
243 
244  /* Check parameter */
245  if (format == NULL || pt == NULL || count == NULL) {
246  return RTEMS_INVALID_ADDRESS;
247  }
248 
249  /* Set count to a save value */
250  *count = 0;
251 
252  /* Get disk data */
253  sc = rtems_bdpart_get_disk_data( disk_name, &fd, &dd, &disk_end);
254  if (sc != RTEMS_SUCCESSFUL) {
255  return sc;
256  }
257 
258  /* Read MBR */
259  sc = rtems_bdpart_read_record( dd, 0, &block);
260  if (sc != RTEMS_SUCCESSFUL) {
261  esc = sc;
262  goto cleanup;
263  }
264 
265  /* Read the first partition entry */
266  data = block->buffer + RTEMS_BDPART_MBR_OFFSET_TABLE_0;
267  sc = rtems_bdpart_read_mbr_partition( data, &p, p_end, &ep_begin);
268  if (sc != RTEMS_SUCCESSFUL) {
269  esc = sc;
270  goto cleanup;
271  }
272 
273  /* Determine if we have a MBR or GPT format */
274  if (rtems_bdpart_mbr_partition_type( p->type) == RTEMS_BDPART_MBR_GPT) {
275  esc = RTEMS_NOT_IMPLEMENTED;
276  goto cleanup;
277  }
278 
279  /* Set format */
280  format->type = RTEMS_BDPART_FORMAT_MBR;
281  format->mbr.disk_id = rtems_uint32_from_little_endian(
282  block->buffer + RTEMS_BDPART_MBR_OFFSET_DISK_ID
283  );
284  format->mbr.dos_compatibility = true;
285 
286  /* Iterate through the rest of the primary partition table */
287  for (i = 1; i < 4; ++i) {
288  data += RTEMS_BDPART_MBR_TABLE_ENTRY_SIZE;
289 
290  sc = rtems_bdpart_read_mbr_partition( data, &p, p_end, &ep_begin);
291  if (sc != RTEMS_SUCCESSFUL) {
292  esc = sc;
293  goto cleanup;
294  }
295  }
296 
297  /* Iterate through the logical partitions within the extended partition */
298  ebr = ep_begin;
299  while (ebr != 0) {
300  rtems_blkdev_bnum tmp = 0;
301 
302  /* Read EBR */
303  sc = rtems_bdpart_read_record( dd, ebr, &block);
304  if (sc != RTEMS_SUCCESSFUL) {
305  esc = sc;
306  goto cleanup;
307  }
308 
309  /* Read first partition entry */
310  sc = rtems_bdpart_read_mbr_partition(
311  block->buffer + RTEMS_BDPART_MBR_OFFSET_TABLE_0,
312  &p,
313  p_end,
314  NULL
315  );
316  if (sc != RTEMS_SUCCESSFUL) {
317  esc = sc;
318  goto cleanup;
319  }
320 
321  /* Adjust partition begin */
322  tmp = p->begin + ebr;
323  if (tmp > p->begin) {
324  p->begin = tmp;
325  } else {
326  esc = RTEMS_IO_ERROR;
327  goto cleanup;
328  }
329 
330  /* Adjust partition end */
331  tmp = p->end + ebr;
332  if (tmp > p->end) {
333  p->end = tmp;
334  } else {
335  esc = RTEMS_IO_ERROR;
336  goto cleanup;
337  }
338 
339  /* Read second partition entry for next EBR block */
340  ebr = rtems_bdpart_next_ebr(
341  block->buffer + RTEMS_BDPART_MBR_OFFSET_TABLE_1
342  );
343  if (ebr != 0) {
344  /* Adjust partition EBR block index */
345  tmp = ebr + ep_begin;
346  if (tmp > ebr) {
347  ebr = tmp;
348  } else {
349  esc = RTEMS_IO_ERROR;
350  goto cleanup;
351  }
352  }
353  }
354 
355  /* Return partition count */
356  *count = (size_t) (p - pt + 1);
357 
358 cleanup:
359 
360  if (fd >= 0) {
361  close( fd);
362  }
363 
364  if (block != NULL) {
365  rtems_bdbuf_release( block);
366  }
367 
368  return esc;
369 }
