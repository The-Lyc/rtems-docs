38 benchmark_timer_initialize (void)
39 {
40  /*
41  * Reset timers 1 and 2
42  */
43  m360.tgcr &= ~0x00FF;
44  m360.tcn1 = 0;
45  m360.tcn2 = 0;
46  m360.ter1 = 0xFFFF;
47  m360.ter2 = 0xFFFF;
48 
49  /*
50  * Cascade timers 1 and 2
51  */
52  m360.tgcr |= 0x0080;
53 
54  /*
55  * Configure timers 1 and 2 to a single 32-bit, 1 MHz timer.
56  * HARDWARE:
57  * Change the `25' to match your processor clock
58  */
59  m360.tmr2 = ((25-1) << 8) | 0x2;
60  m360.tmr1 = 0;
61 
62  /*
63  * Start the timers
64  */
65  m360.tgcr |= 0x0011;
66 }
