63 static void load_program(unsigned int *program, int size)
64 {
65  int page;
66  int word;
67  volatile unsigned int *pfpu_prog = (unsigned int *)MM_PFPU_CODEBASE;
68 
69  for (page=0;page<(PFPU_PROGSIZE/PFPU_PAGESIZE);page++) {
70  MM_WRITE(MM_PFPU_CODEPAGE, page);
71  for (word=0;word<PFPU_PAGESIZE;word++) {
72  if (size == 0) return;
73  pfpu_prog[word] = *program;
74  program++;
75  size--;
76  }
77  }

98 static rtems_status_code pfpu_execute(struct pfpu_td *td)
99 {
100  rtems_status_code sc;
101 
102  load_program(td->program, td->progsize);
103  load_registers(td->registers);
104  MM_WRITE(MM_PFPU_MESHBASE, (unsigned int)td->output);
105  MM_WRITE(MM_PFPU_HMESHLAST, td->hmeshlast);
106  MM_WRITE(MM_PFPU_VMESHLAST, td->vmeshlast);
107  MM_WRITE(MM_PFPU_CTL, PFPU_CTL_START);
108 
109  sc = rtems_semaphore_obtain(done_sem, RTEMS_WAIT, 10);
110  if (sc != RTEMS_SUCCESSFUL)
111  return sc;
112 
113  if (td->update) {
114  update_registers(td->registers);
115  if (td->invalidate) {
116  __asm__ volatile( /* Invalidate Level-1 data cache */
117  "wcsr DCC, r0\n"
118  "nop\n"
119  );
120  }
121  }
122 
123  return RTEMS_SUCCESSFUL;
124 }
