155 static rtems_isr satcan_interrupt_handler(void *v)
156 {
157  unsigned int irq;
158  unsigned int fifo;
159 
160  irq = regs->irqpend;
161 
162  if (irq & IRQ_AHB && priv->cfg->ahb_irq_callback) {
163  priv->cfg->ahb_irq_callback();
164  }
165  if (irq & IRQ_PPS && priv->cfg->pps_irq_callback) {
166  priv->cfg->pps_irq_callback();
167  }
168  if (irq & IRQ_M5 && priv->cfg->m5_irq_callback) {
169  priv->cfg->m5_irq_callback();
170  }
171  if (irq & IRQ_M4 && priv->cfg->m4_irq_callback) {
172  priv->cfg->m4_irq_callback();
173  }
174  if (irq & IRQ_M3 && priv->cfg->m3_irq_callback) {
175  priv->cfg->m3_irq_callback();
176  }
177  if (irq & IRQ_M2 && priv->cfg->m2_irq_callback) {
178  priv->cfg->m2_irq_callback();
179  }
180  if (irq & IRQ_M1 && priv->cfg->m1_irq_callback) {
181  priv->cfg->m1_irq_callback();
182  }
183  if (irq & IRQ_SYNC && priv->cfg->sync_irq_callback) {
184  priv->cfg->sync_irq_callback();
185  }
186  if (irq & IRQ_CAN) {
187  fifo = regs->satcan[SATCAN_FIFO];
188  if (!(fifo & FIFO_Empty) && priv->txactive &&
189  (((fifo & 0xff) == SATCAN_IRQ_EOD1) || ((fifo & 0xff) == SATCAN_IRQ_EOD2))) {
190  rtems_semaphore_release(priv->txsem);
191  }
192  if (priv->cfg->can_irq_callback)
193  priv->cfg->can_irq_callback(fifo);
194  }
195 }

199 static rtems_device_driver satcan_ioctl(rtems_device_major_number major, rtems_device_minor_number minor, void *arg)
200 {
201  rtems_libio_ioctl_args_t *ioarg = (rtems_libio_ioctl_args_t*)arg;
202  int *value;
203  rtems_interval *timeout;
204  satcan_regmod *regmod;
205 
206  DBG("SatCAN: IOCTL %d\n\r", ioarg->command);
207 
208  ioarg->ioctl_return = 0;
209  switch(ioarg->command) {
210  case SATCAN_IOC_DMA_2K:
211  DBG("SatCAN: ioctl: setting 2K DMA mode\n\r");
212  free(priv->dmaptr);
213  almalloc(&priv->alptr, &priv->dmaptr, ALIGN_2KMEM);
214  if (priv->dmaptr == NULL) {
215  printk("SatCAN: Failed to allocate DMA memory\n\r");
216  return RTEMS_NO_MEMORY;
217  }
218 
219  regs->membase = (unsigned int)priv->alptr;
220  regs->satcan[SATCAN_RAM_BASE] = (unsigned int)priv->alptr >> OFFSET_2K_LOW_POS;
221  regs->satcan[SATCAN_CMD1] = regs->satcan[SATCAN_CMD1] | Sel_2k_8kN;
222  break;
223 
224  case SATCAN_IOC_DMA_8K:
225  DBG("SatCAN: ioctl: setting 8K DMA mode\n\r");
226  free(priv->dmaptr);
227  almalloc(&priv->alptr, &priv->dmaptr, ALIGN_8KMEM);
228  if (priv->dmaptr == NULL) {
229  printk("SatCAN: Failed to allocate DMA memory\n\r");
230  return RTEMS_NO_MEMORY;
231  }
232 
233  regs->membase = (unsigned int)priv->alptr;
234  regs->satcan[SATCAN_RAM_BASE] = (unsigned int)priv->alptr >> OFFSET_8K_LOW_POS;
235  regs->satcan[SATCAN_CMD1] = regs->satcan[SATCAN_CMD1] & ~Sel_2k_8kN;
236  break;
237 
238  case SATCAN_IOC_GET_REG:
239  /* Get regmod structure from argument */
240  regmod = (satcan_regmod*)ioarg->buffer;
241  DBG("SatCAN: ioctl: getting register %d\n\r", regmod->reg);
242  if (regmod->reg < 0)
243  return RTEMS_INVALID_NAME;
244  else if (regmod->reg <= SATCAN_FILTER_STOP)
245  regmod->val = regs->satcan[regmod->reg];
246  else if (regmod->reg == SATCAN_WCTRL)
247  regmod->val = regs->ctrl;
248  else if (regmod->reg == SATCAN_WIPEND)
249  regmod->val = regs->irqpend;
250  else if (regmod->reg == SATCAN_WIMASK)
251  regmod->val = regs->irqmask;
252  else if (regmod->reg == SATCAN_WAHBADDR)
253  regmod->val = regs->membase;
254  else
255  return RTEMS_INVALID_NAME;
256  break;
257 
258  case SATCAN_IOC_SET_REG:
259  /* Get regmod structure from argument */
260  regmod = (satcan_regmod*)ioarg->buffer;
261  DBG("SatCAN: ioctl: setting register %d, value %x\n\r",
262  regmod->reg, regmod->val);
263  if (regmod->reg < 0)
264  return RTEMS_INVALID_NAME;
265  else if (regmod->reg <= SATCAN_FILTER_STOP)
266  regs->satcan[regmod->reg] = regmod->val;
267  else if (regmod->reg == SATCAN_WCTRL)
268  regs->ctrl = regmod->val;
269  else if (regmod->reg == SATCAN_WIPEND)
270  regs->irqpend = regmod->val;
271  else if (regmod->reg == SATCAN_WIMASK)
272  regs->irqmask = regmod->val;
273  else if (regmod->reg == SATCAN_WAHBADDR)
274  regs->membase = regmod->val;
275  else
276  return RTEMS_INVALID_NAME;
277  break;
278 
279  case SATCAN_IOC_OR_REG:
280  /* Get regmod structure from argument */
281  regmod = (satcan_regmod*)ioarg->buffer;
282  DBG("SatCAN: ioctl: or:ing register %d, with value %x\n\r",
283  regmod->reg, regmod->val);
284  if (regmod->reg < 0)
285  return RTEMS_INVALID_NAME;
286  else if (regmod->reg <= SATCAN_FILTER_STOP)
287  regs->satcan[regmod->reg] |= regmod->val;
288  else if (regmod->reg == SATCAN_WCTRL)
289  regs->ctrl |= regmod->val;
290  else if (regmod->reg == SATCAN_WIPEND)
291  regs->irqpend |= regmod->val;
292  else if (regmod->reg == SATCAN_WIMASK)
293  regs->irqmask |= regmod->val;
294  else if (regmod->reg == SATCAN_WAHBADDR)
295  regs->membase |= regmod->val;
296  else
297  return RTEMS_INVALID_NAME;
298  break;
299 
300  case SATCAN_IOC_AND_REG:
301  /* Get regmod structure from argument */
302  regmod = (satcan_regmod*)ioarg->buffer;
303  DBG("SatCAN: ioctl: masking register %d, with value %x\n\r",
304  regmod->reg, regmod->val);
305  if (regmod->reg < 0)
306  return RTEMS_INVALID_NAME;
307  else if (regmod->reg <= SATCAN_FILTER_STOP)
308  regs->satcan[regmod->reg] &= regmod->val;
309  else if (regmod->reg == SATCAN_WCTRL)
310  regs->ctrl &= regmod->val;
311  else if (regmod->reg == SATCAN_WIPEND)
312  regs->irqpend &= regmod->val;
313  else if (regmod->reg == SATCAN_WIMASK)
314  regs->irqmask &= regmod->val;
315  else if (regmod->reg == SATCAN_WAHBADDR)
316  regs->membase &= regmod->val;
317  else
318  return RTEMS_INVALID_NAME;
319  break;
320 
321  case SATCAN_IOC_EN_TX1_DIS_TX2:
322  priv->dmaen = SATCAN_DMA_ENABLE_TX1;
323  break;
324 
325  case SATCAN_IOC_EN_TX2_DIS_TX1:
326  priv->dmaen = SATCAN_DMA_ENABLE_TX2;
327  break;
328 
329  case SATCAN_IOC_GET_DMA_MODE:
330  value = (int*)ioarg->buffer;
331  *value = priv->dmamode;
332  break;
333 
334  case SATCAN_IOC_SET_DMA_MODE:
335  value = (int*)ioarg->buffer;
336  if (*value != SATCAN_DMA_MODE_USER && *value != SATCAN_DMA_MODE_SYSTEM) {
337  DBG("SatCAN: ioctl: invalid DMA mode\n\r");
338  return RTEMS_INVALID_NAME;
339  }
340  priv->dmamode = *value;
341  break;
342 
343  case SATCAN_IOC_ACTIVATE_DMA:
344  if (priv->dmamode != SATCAN_DMA_MODE_USER) {
345  DBG("SatCAN: ioctl: ACTIVATE_DMA: not in user mode\n\r");
346  return RTEMS_INVALID_NAME;
347  }
348  value = (int*)ioarg->buffer;
349  if (*value != SATCAN_DMA_ENABLE_TX1 && *value != SATCAN_DMA_ENABLE_TX2) {
350  DBG("SatCAN: ioctl: ACTIVATE_DMA: Illegal channel\n\r");
351  return RTEMS_INVALID_NAME;
352  }
353  regs->satcan[SATCAN_DMA] |= *value << 1;
354  break;
355 
356  case SATCAN_IOC_DEACTIVATE_DMA:
357  if (priv->dmamode != SATCAN_DMA_MODE_USER) {
358  DBG("SatCAN: ioctl: DEACTIVATE_DMA: not in user mode\n\r");
359  return RTEMS_INVALID_NAME;
360  }
361  value = (int*)ioarg->buffer;
362  if (*value != SATCAN_DMA_ENABLE_TX1 && *value != SATCAN_DMA_ENABLE_TX2) {
363  DBG("SatCAN: ioctl: DEACTIVATE_DMA: Illegal channel\n\r");
364  return RTEMS_INVALID_NAME;
365  }
366  regs->satcan[SATCAN_DMA] &= ~(*value << 1);
367  break;
368 
369  case SATCAN_IOC_GET_DOFFSET:
370  value = (int*)ioarg->buffer;
371  *value = priv->doff;
372  break;
373 
374  case SATCAN_IOC_SET_DOFFSET:
375  value = (int*)ioarg->buffer;
376  priv->doff = *value;
377  break;
378 
379  case SATCAN_IOC_GET_TIMEOUT:
380  timeout = (rtems_interval*)ioarg->buffer;
381  *timeout = priv->timeout;
382  break;
383 
384  case SATCAN_IOC_SET_TIMEOUT:
385  timeout = (rtems_interval*)ioarg->buffer;
386  priv->timeout = *timeout;
387  break;
388 
389  default:
390  return RTEMS_NOT_DEFINED;
391  }
392 
393  return RTEMS_SUCCESSFUL;
394 }

396 static rtems_device_driver satcan_write(rtems_device_major_number major, rtems_device_minor_number minor, void *arg)
397 {
398  int i;
399  int doff;
400  int msgindex;
401  int messages;
402  rtems_libio_rw_args_t *rw_args=(rtems_libio_rw_args_t *) arg;
403  satcan_msg *msgs;
404  rtems_status_code status;
405 
406  DBG("SatCAN: Writing %d bytes from %p\n\r",rw_args->count,rw_args->buffer);
407 
408  if ((rw_args->count < sizeof(satcan_msg)) || (!rw_args->buffer)) {
409  DBG("SatCAN: write: returning EINVAL\n\r");
410  return RTEMS_INVALID_NAME; /* EINVAL */
411  }
412 
413  messages = rw_args->count / sizeof(satcan_msg);
414  msgs = (satcan_msg*)rw_args->buffer;
415 
416  /* Check that size matches any number of satcan_msg */
417  if (rw_args->count % sizeof(satcan_msg)) {
418  DBG("SatCAN: write: count can not be evenly divided with satcan_msg size\n\r");
419  return RTEMS_INVALID_NAME; /* EINVAL */
420  }
421 
422 
423  /* DMA channel must be set if we are in system DMA mode */
424  DBG("SatCAN: write: dma channel select is %x\n\r", priv->dmaen);
425  if (!priv->dmaen && priv->dmamode == SATCAN_DMA_MODE_SYSTEM)
426  return RTEMS_INVALID_NAME; /* EINVAL */
427 
428  /* DMA must not be active */
429  if (regs->satcan[SATCAN_DMA] & (DMA_EnTx1 | DMA_EnTx2 | DMA_AutoInitDmaTx)) {
430  DBG("SatCAN: write: DMA was active\n\r");
431  rw_args->bytes_moved = 0;
432  return RTEMS_IO_ERROR; /* EIO */
433  }
434 
435  doff = regs->satcan[SATCAN_CMD1] & Sel_2k_8kN ? DMA_2K_DATA_OFFSET : DMA_8K_DATA_OFFSET;
436 
437  for (msgindex = 0; msgindex < messages; msgindex++) {
438  /* Place header in DMA area */
439  for (i = 0; i < SATCAN_HEADER_SIZE; i++) {
440  priv->alptr[priv->doff+8*msgindex+i] = msgs[msgindex].header[i];
441  }
442 
443  /* Place data in DMA area */
444  for (i = 0; i < SATCAN_PAYLOAD_SIZE; i++)
445  priv->alptr[priv->doff+doff+8*msgindex+i] = msgs[msgindex].payload[i];
446  }
447 
448  if ((priv->dmaen & SATCAN_DMA_ENABLE_TX1) || priv->dmamode == SATCAN_DMA_MODE_USER) {
449  regs->satcan[SATCAN_DMA_TX_1_CUR] = 0;
450  regs->satcan[SATCAN_DMA_TX_1_END] = messages<<3;
451  }
452 
453  if ((priv->dmaen & SATCAN_DMA_ENABLE_TX2) || priv->dmamode == SATCAN_DMA_MODE_USER) {
454  regs->satcan[SATCAN_DMA_TX_2_CUR] = 0;
455  regs->satcan[SATCAN_DMA_TX_2_END] = messages<<3;
456  }
457 
458  /* If we are in DMA user mode we are done here, otherwise we block */
459  if (priv->dmamode == SATCAN_DMA_MODE_SYSTEM) {
460  priv->txactive = 1;
461 
462  /* Enable DMA */
463  regs->satcan[SATCAN_DMA] |= priv->dmaen << 1;
464 
465  /* Wait for TX interrupt */
466  status = rtems_semaphore_obtain(priv->txsem, RTEMS_WAIT, priv->timeout);
467 
468  priv->txactive = 0;
469 
470  /* Disable activated Tx DMA */
471  regs->satcan[SATCAN_DMA] &= ~(priv->dmaen << 1);
472 
473  if (status != RTEMS_SUCCESSFUL) {
474  rw_args->bytes_moved = 0;
475  return status;
476  }
477  }
478 
479  rw_args->bytes_moved = rw_args->count;
480 
481  return RTEMS_SUCCESSFUL;
482 }

484 static rtems_device_driver satcan_read(rtems_device_major_number major, rtems_device_minor_number minor, void *arg)
485 {
486  char *buf;
487  int i;
488  int canid;
489  int messages;
490  rtems_libio_rw_args_t *rw_args = (rtems_libio_rw_args_t*)arg;
491  satcan_msg *ret;
492 
493  /* Check that there is room for the return */
494  if (rw_args->count < sizeof(satcan_msg)) {
495  DBG("SatCAN: read: length of buffer must be at least %d, current is %d\n\r",
496  sizeof(satcan_msg) + sizeof(int), rw_args->count);
497  return RTEMS_INVALID_NAME; /* -EINVAL */
498  }
499 
500  /* Check that size matches any number of satcan_msg */
501  if (rw_args->count % sizeof(satcan_msg)) {
502  DBG("SatCAN: read: count can not be evenly divided with satcan_msg size\n\r");
503  return RTEMS_INVALID_NAME; /* EINVAL */
504  }
505 
506  messages = rw_args->count / sizeof(satcan_msg);
507  ret = (satcan_msg*)rw_args->buffer;
508 
509  DBG("SatCAN: read: reading %d messages to %p\n\r", messages, ret);
510 
511  for (i = 0; i < messages; i++) {
512  canid = (ret[i].header[1] << 8) | ret[i].header[0];
513 
514  /* Copy message header from DMA header area to buffer */
515  buf = (char*)((int)priv->alptr | (canid << 3));
516  memcpy(ret[i].header, buf, SATCAN_HEADER_SIZE);
517 
518  DBG("SatCAN: read: copied header from %p to %p\n\r", buf, ret[i].header);
519 
520  /* Clear New Message Marker */
521  buf[SATCAN_HEADER_NMM_POS] = 0;
522 
523  /* Copy message payload from DMA data area to buffer */
524  buf = (char*)((int)buf |
525  (regs->satcan[SATCAN_CMD1] & Sel_2k_8kN ? DMA_2K_DATA_SELECT : DMA_8K_DATA_SELECT));
526  memcpy(ret[i].payload, buf, SATCAN_PAYLOAD_SIZE);
527 
528  DBG("SatCAN: read: copied payload from %p to %p\n\r", buf, ret[i].payload);
529  }
530  rw_args->bytes_moved = rw_args->count;
531 
532  return RTEMS_SUCCESSFUL;
533 }

556 static rtems_device_driver satcan_open(rtems_device_major_number major, rtems_device_minor_number minor, void *arg)
557 {
558  DBG("SatCAN: Opening %d\n\r",minor);
559 
560  rtems_semaphore_obtain(priv->devsem,RTEMS_WAIT, RTEMS_NO_TIMEOUT);
561  if (priv->open) {
562  rtems_semaphore_release(priv->devsem);
563  return RTEMS_RESOURCE_IN_USE; /* EBUSY */
564  }
565  priv->open = 1;
566  rtems_semaphore_release(priv->devsem);
567 
568  /* Enable AHB and CAN IRQs in wrapper and EOD1, EOD2 and CAN critical IRQs in SatCAN core */
569  regs->irqmask = MSK_AHB | MSK_CAN;
570  regs->satcan[SATCAN_INT_EN] = ((1 << SATCAN_IRQ_EOD1) | (1 << SATCAN_IRQ_EOD2) |
571  (1 << SATCAN_IRQ_CRITICAL));
572 
573  /* Select can_int as IRQ source */
574  regs->satcan[SATCAN_CMD0] = CAN_TODn_Int_sel;
575  /* CAN RX DMA Enable */
576  regs->satcan[SATCAN_DMA] = 1;
577  /* CAN RX Enable */
578  regs->satcan[SATCAN_RX] = 1;
579 
580  DBG("SatCAN: Opening %d success\n\r",minor);
581 
582  return RTEMS_SUCCESSFUL;
583 }

586 {
587  struct ambapp_ahb_info d;
588  char fs_name[20];
589  rtems_status_code status;
590
