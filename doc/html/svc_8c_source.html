164 /*
165  * Add a service program to the callout list.
166  * The dispatch routine will be called when a rpc request for this
167  * program number comes in.
168  */
169 bool_t
170 svc_register(
171  SVCXPRT *xprt,
172  u_long prog,
173  u_long vers,
174  void (*dispatch)(struct svc_req *r, SVCXPRT *xprt),
175  int protocol )
176 {
177  struct svc_callout *prev;
178  register struct svc_callout *s;
179 
180  if ((s = svc_find(prog, vers, &prev)) != NULL_SVC) {
181  if (s->sc_dispatch == dispatch)
182  goto pmap_it; /* he is registering another xptr */
183  return (FALSE);
184  }
185  s = (struct svc_callout *)mem_alloc(sizeof(struct svc_callout));
186  if (s == (struct svc_callout *)0) {
187  return (FALSE);
188  }
189  s->sc_prog = prog;
190  s->sc_vers = vers;
191  s->sc_dispatch = dispatch;
192  s->sc_next = svc_head;
193  svc_head = s;
194 pmap_it:
195  /* now register the information with the local binder service */
196  if (protocol) {
197  return (pmap_set(prog, vers, protocol, xprt->xp_port));
198  }
199  return (TRUE);
200 }

419 void
420 svc_getreqset2(
421  fd_set *readfds,
422  int width )
423 {
424  enum xprt_stat stat;
425  struct rpc_msg msg;
426  int prog_found;
427  u_long low_vers;
428  u_long high_vers;
429  struct svc_req r;
430  register SVCXPRT *xprt;
431  register int bit;
432  register int sock;
433  register fd_mask mask, *maskp;
434  char cred_area[2*MAX_AUTH_BYTES + RQCRED_SIZE];
435  msg.rm_call.cb_cred.oa_base = cred_area;
436  msg.rm_call.cb_verf.oa_base = &(cred_area[MAX_AUTH_BYTES]);
437  r.rq_clntcred = &(cred_area[2*MAX_AUTH_BYTES]);
438 
439 
440  maskp = readfds->fds_bits;
441  for (sock = 0; sock < width; sock += NFDBITS) {
442  for (mask = *maskp++; (bit = ffs(mask)); mask ^= (1 << (bit - 1))) {
443  /* sock has input waiting */
444  xprt = xports[sock + bit - 1];
445  if (xprt == NULL)
446  /* But do we control sock? */
447  continue;
448  /* now receive msgs from xprtprt (support batch calls) */
449  do {
450  if (SVC_RECV(xprt, &msg)) {
451 
452  /* now find the exported program and call it */
453  register struct svc_callout *s;
454  enum auth_stat why;
455 
456  r.rq_xprt = xprt;
457  r.rq_prog = msg.rm_call.cb_prog;
458  r.rq_vers = msg.rm_call.cb_vers;
459  r.rq_proc = msg.rm_call.cb_proc;
460  r.rq_cred = msg.rm_call.cb_cred;
461  /* first authenticate the message */
462  if ((why= _authenticate(&r, &msg)) != AUTH_OK) {
463  svcerr_auth(xprt, why);
464  goto call_done;
465  }
466  /* now match message with a registered service*/
467  prog_found = FALSE;
468  low_vers = (u_long) - 1;
469  high_vers = 0;
470  for (s = svc_head; s != NULL_SVC; s = s->sc_next) {
471  if (s->sc_prog == r.rq_prog) {
472  if (s->sc_vers == r.rq_vers) {
473  (*s->sc_dispatch)(&r, xprt);
474  goto call_done;
475  } /* found correct version */
476  prog_found = TRUE;
477  if (s->sc_vers < low_vers)
478  low_vers = s->sc_vers;
479  if (s->sc_vers > high_vers)
480  high_vers = s->sc_vers;
481  } /* found correct program */
482  }
483  /*
484  * if we got here, the program or version
485  * is not served ...
486  */
487  if (prog_found)
488  svcerr_progvers(xprt,
489  low_vers, high_vers);
490  else
491  svcerr_noprog(xprt);
492  /* Fall through to ... */
493  }
494  call_done:
495  if ((stat = SVC_STAT(xprt)) == XPRT_DIED){
496  SVC_DESTROY(xprt);
497  break;
498  }
499  } while (stat == XPRT_MOREREQS);
500  }
501  }
