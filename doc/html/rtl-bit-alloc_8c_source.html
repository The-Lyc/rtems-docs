78 static ssize_t
79 bit_find_clear (rtems_rtl_bit_alloc* balloc, size_t blocks)
80 {
81  size_t base = 0;
82  size_t clear = 0;
83  size_t b;
84 
85  for (b = 0; b < balloc->blocks; ++b)
86  {
87  if (balloc->bits[b] != 0xffffffff)
88  {
89  uint32_t word = balloc->bits[b];
90  size_t o;
91  for (o = 0; o < BITS_PER_WORD; ++o, word >>= 1)
92  {
93  if ((word & 1) == 0)
94  {
95  if (clear == 0)
96  base = (b * BITS_PER_WORD) + o;
97  ++clear;
98  if (clear == blocks)
99  return base;
100  }
101  else
102  {
103  clear = 0;
104  base = 0;
105  }
106  }
107  }
108  else
109  {
110  clear = 0;
111  base = 0;
112  }
113  }
114 
115  return -1;
116 }
