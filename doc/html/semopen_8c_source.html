82 /*
83  * sem_open
84  *
85  * Opens a named semaphore. Used in conjunction with the sem_close
86  * and sem_unlink commands.
87  *
88  * 11.2.3 Initialize/Open a Named Semaphore, P1003.1b-1993, p.221
89  *
90  * NOTE: When oflag is O_CREAT, then optional third and fourth
91  * parameters must be present.
92  */
93 sem_t *sem_open(
94  const char *name,
95  int oflag,
96  ...
97  /* mode_t mode, */
98  /* unsigned int value */
99 )
100 {
101  /*
102  * mode is set but never used. GCC gives a warning for this
103  * and we need to tell GCC not to complain. But we have to
104  * have it because we have to work through the variable
105  * arguments to get to attr.
106  */
107  mode_t mode RTEMS_UNUSED;
108 
109  va_list arg;
110  unsigned int value = 0;
111  POSIX_Semaphore_Control *the_semaphore;
112  size_t name_len;
113  Objects_Get_by_name_error error;
114  sem_t *sem;
115 
116  if ( oflag & O_CREAT ) {
117  va_start(arg, oflag);
118  mode = va_arg( arg, mode_t );
119  value = va_arg( arg, unsigned int );
120  va_end(arg);
121  }
122 
123  _Objects_Allocator_lock();
124  the_semaphore = _POSIX_Semaphore_Get_by_name( name, &name_len, &error );
125 
126  /*
127  * If the name to id translation worked, then the semaphore exists
128  * and we can just return a pointer to the id. Otherwise we may
129  * need to check to see if this is a "semaphore does not exist"
130  * or some other miscellaneous error on the name.
131  */
132 
133  if ( the_semaphore == NULL ) {
134 
135  /*
136  * Unless provided a valid name that did not already exist
137  * and we are willing to create then it is an error.
138  */
139 
140  if ( !( error == OBJECTS_GET_BY_NAME_NO_OBJECT && (oflag & O_CREAT) ) ) {
141  _Objects_Allocator_unlock();
142  rtems_set_errno_and_return_value(
143  _POSIX_Get_by_name_error( error ),
144  SEM_FAILED
145  );
146  }
147  } else {
148 
149  /*
150  * Check for existence with creation.
151  */
152 
153  if ( (oflag & (O_CREAT | O_EXCL)) == (O_CREAT | O_EXCL) ) {
154  _Objects_Allocator_unlock();
155  rtems_set_errno_and_return_value( EEXIST, SEM_FAILED );
156  }
157 
158  the_semaphore->open_count += 1;
159  _Objects_Allocator_unlock();
160  return &the_semaphore->Semaphore;
161  }
162 
163  /*
164  * At this point, the semaphore does not exist and everything has been
165  * checked. We should go ahead and create a semaphore.
166  */
167 
168  sem = _POSIX_Semaphore_Create_support(
169  name,
170  name_len,
171  value
172  );
173 
174  _Objects_Allocator_unlock();
175  return sem;
176 }
