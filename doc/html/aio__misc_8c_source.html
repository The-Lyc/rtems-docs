80  * Search and create chain of requests for given FD

85  * create - if 1 search and create

87  *

98 rtems_aio_request_chain *
99 rtems_aio_search_fd (rtems_chain_control *chain, int fildes, int create)
100 {
101  rtems_aio_request_chain *r_chain;
102  rtems_chain_node *node;
103 
104  node = rtems_chain_first (chain);
105  r_chain = (rtems_aio_request_chain *) node;
106 
107  while (r_chain->fildes < fildes && !rtems_chain_is_tail (chain, node)) {
108  node = rtems_chain_next (node);
109  r_chain = (rtems_aio_request_chain *) node;
110  }
111 
112  if (r_chain->fildes == fildes)
113  r_chain->new_fd = 0;
114  else {
115  if (create == 0)
116  r_chain = NULL;
117  else {
118  r_chain = malloc (sizeof (rtems_aio_request_chain));
119  rtems_chain_initialize_empty (&r_chain->perfd);
120  rtems_chain_initialize_node (&r_chain->next_fd);
121 
122  if (rtems_chain_is_empty (chain))
123  rtems_chain_prepend (chain, &r_chain->next_fd);
124  else
125  rtems_chain_insert (rtems_chain_previous (node), &r_chain->next_fd);

282  *
283  * Enqueue requests, and creates threads to process them
284  *
285  * Input parameters:
286  * req - see aio_misc.h
287  *
288  * Output parameters:
289  * 0 - if request was added to queue
290  * errno - otherwise

292 
293 int
294 rtems_aio_enqueue (rtems_aio_request *req)
295 {
296 
297  rtems_aio_request_chain *r_chain;
298  rtems_chain_control *chain;
299  pthread_t thid;
300  int result, policy;
301  struct sched_param param;
302 
303  /* The queue should be initialized */
304  AIO_assert (aio_request_queue.initialized == AIO_QUEUE_INITIALIZED);
305 
306  result = pthread_mutex_lock (&aio_request_queue.mutex);
307  if (result != 0) {
308  free (req);
309  return result;
310  }
311 
312  /* _POSIX_PRIORITIZED_IO and _POSIX_PRIORITY_SCHEDULING are defined,
313  we can use aio_reqprio to lower the priority of the request */
314  pthread_getschedparam (pthread_self(), &policy, &param);
315 
316  rtems_chain_initialize_node (&req->next_prio);
317  req->caller_thread = pthread_self ();
318  req->priority = param.sched_priority - req->aiocbp->aio_reqprio;
319  req->policy = policy;
320  req->aiocbp->error_code = EINPROGRESS;
321  req->aiocbp->return_value = 0;
322 
323  if ((aio_request_queue.idle_threads == 0) &&
324  aio_request_queue.active_threads < AIO_MAX_THREADS)
325  /* we still have empty places on the active_threads chain */
326  {
327  chain = &aio_request_queue.work_req;
328  r_chain = rtems_aio_search_fd (chain, req->aiocbp->aio_fildes, 1);
329 
330  if (r_chain->new_fd == 1) {
331  rtems_chain_prepend (&r_chain->perfd, &req->next_prio);
332  r_chain->new_fd = 0;
333  pthread_mutex_init (&r_chain->mutex, NULL);
334  pthread_cond_init (&r_chain->cond, NULL);
335 
336  AIO_printf ("New thread \n");
337  result = pthread_create (&thid, &aio_request_queue.attr,
338  rtems_aio_handle, (void *) r_chain);
339  if (result != 0) {
340  pthread_mutex_unlock (&aio_request_queue.mutex);
341  return result;
342  }
343  ++aio_request_queue.active_threads;
344  }
345  else {
346  /* put request in the fd chain it belongs to */
347  pthread_mutex_lock (&r_chain->mutex);
348  rtems_aio_insert_prio (&r_chain->perfd, req);
349  pthread_cond_signal (&r_chain->cond);
350  pthread_mutex_unlock (&r_chain->mutex);
351  }
352  }
353  else
354  {
355  /* the maximum number of threads has been already created
356  even though some of them might be idle.
357  The request belongs to one of the active fd chain */
358  r_chain = rtems_aio_search_fd (&aio_request_queue.work_req,
359  req->aiocbp->aio_fildes, 0);
360  if (r_chain != NULL)
361  {
362  pthread_mutex_lock (&r_chain->mutex);
363  rtems_aio_insert_prio (&r_chain->perfd, req);
364  pthread_cond_signal (&r_chain->cond);
365  pthread_mutex_unlock (&r_chain->mutex);
366 
367  } else {
368 
369  /* or to the idle chain */
370  chain = &aio_request_queue.idle_req;
371  r_chain = rtems_aio_search_fd (chain, req->aiocbp->aio_fildes, 1);
372 
373  if (r_chain->new_fd == 1) {
374  /* If this is a new fd chain we signal the idle threads that
375  might be waiting for requests */
376  AIO_printf (" New chain on waiting queue \n ");
377  rtems_chain_prepend (&r_chain->perfd, &req->next_prio);
378  r_chain->new_fd = 0;
379  pthread_mutex_init (&r_chain->mutex, NULL);
380  pthread_cond_init (&r_chain->cond, NULL);
381  } else
382  /* just insert the request in the existing fd chain */
383  rtems_aio_insert_prio (&r_chain->perfd, req);
384  if (aio_request_queue.idle_threads > 0)
385  pthread_cond_signal (&aio_request_queue.new_req);
386  }
387  }
388 
389  pthread_mutex_unlock (&aio_request_queue.mutex);
390  return 0;
391 }

393 /*
394  * rtems_aio_handle
395  *
396  * Thread processing requests
397  *
398  * Input parameters:
399  * arg - the chain for the fd to be worked on
400  *
401  * Output parameters:
402  * NULL - if error
403  */
404 
405 static void *
406 rtems_aio_handle (void *arg)
407 {
408 
409  rtems_aio_request_chain *r_chain = arg;
410  rtems_aio_request *req;
411  rtems_chain_control *chain;
412  rtems_chain_node *node;
413  int result, policy;
414  struct sched_param param;
415 
416  AIO_printf ("Thread started\n");
417 
418  while (1) {
419 
420  /* acquire the mutex of the current fd chain.
421  we don't need to lock the queue mutex since we can
422  add requests to idle fd chains or even active ones
423  if the working request has been extracted from the
424  chain */
425  result = pthread_mutex_lock (&r_chain->mutex);
426  if (result != 0)
427  return NULL;
428 
429  chain = &r_chain->perfd;
430 
431  /* If the locked chain is not empty, take the first
432  request extract it, unlock the chain and process
433  the request, in this way the user can supply more
434  requests to this fd chain */
435  if (!rtems_chain_is_empty (chain)) {
436 
437  AIO_printf ("Get new request from not empty chain\n");
438  node = rtems_chain_first (chain);
439  req = (rtems_aio_request *) node;
440 
441  /* See _POSIX_PRIORITIZE_IO and _POSIX_PRIORITY_SCHEDULING
442  discussion in rtems_aio_enqueue () */
443  pthread_getschedparam (pthread_self(), &policy, &param);
444  param.sched_priority = req->priority;
445  pthread_setschedparam (pthread_self(), req->policy, &param);
446 
447  rtems_chain_extract (node);
448 
449  pthread_mutex_unlock (&r_chain->mutex);
450 
451  switch (req->aiocbp->aio_lio_opcode) {
452  case LIO_READ:
453  AIO_printf ("read\n");
454  result = pread (req->aiocbp->aio_fildes,
455  (void *) req->aiocbp->aio_buf,
456  req->aiocbp->aio_nbytes, req->aiocbp->aio_offset);
457  break;
458 
459  case LIO_WRITE:
460  AIO_printf ("write\n");
461  result = pwrite (req->aiocbp->aio_fildes,
462  (void *) req->aiocbp->aio_buf,
463  req->aiocbp->aio_nbytes, req->aiocbp->aio_offset);
464  break;
465 
466  case LIO_SYNC:
467  AIO_printf ("sync\n");
468  result = fsync (req->aiocbp->aio_fildes);
469  break;
470 
471  default:
472  result = -1;
473  }
474  if (result == -1) {
475  req->aiocbp->return_value = -1;
476  req->aiocbp->error_code = errno;
477  } else {
478  req->aiocbp->return_value = result;
479  req->aiocbp->error_code = 0;
480  }
481 
482  // notification needed for lio
483 
484  } else {
485  /* If the fd chain is empty we unlock the fd chain
486  and we lock the queue chain, this will ensure that
487  we have at most one request comming to our fd chain
488  when we check.
489 
490  If there was no request added sleep for 3 seconds and
491  wait for a signal on chain, this will unlock the queue.
492  The fd chain is already unlocked */
493 
494  struct timespec timeout;
495 
496  AIO_printf ("Chain is empty [WQ], wait for work\n");
497 
498  pthread_mutex_unlock (&r_chain->mutex);
499  pthread_mutex_lock (&aio_request_queue.mutex);
500 
501  if (rtems_chain_is_empty (chain))
502  {
503  clock_gettime (CLOCK_REALTIME, &timeout);
504  timeout.tv_sec += 3;
505  timeout.tv_nsec = 0;
506  result = pthread_cond_timedwait (&r_chain->cond,
507  &aio_request_queue.mutex,
508  &timeout);
509 
510  /* If no requests were added to the chain we delete the fd chain from
511  the queue and start working with idle fd chains */
512  if (result == ETIMEDOUT) {
513  rtems_chain_extract (&r_chain->next_fd);
514  pthread_mutex_destroy (&r_chain->mutex);
515  pthread_cond_destroy (&r_chain->cond);
516  free (r_chain);
517 
518  /* If the idle chain is empty sleep for 3 seconds and wait for a
519  signal. The thread now becomes idle. */
520  if (rtems_chain_is_empty (&aio_request_queue.idle_req)) {
521  AIO_printf ("Chain is empty [IQ], wait for work\n");
522 
523  ++aio_request_queue.idle_threads;
524  --aio_request_queue.active_threads;
525  clock_gettime (CLOCK_REALTIME, &timeout);
526  timeout.tv_sec += 3;
527  timeout.tv_nsec = 0;
528 
529  result = pthread_cond_timedwait (&aio_request_queue.new_req,
530  &aio_request_queue.mutex,
531  &timeout);
532 
533  /* If no new fd chain was added in the idle requests
534  then this thread is finished */
535  if (result == ETIMEDOUT) {
536  AIO_printf ("Etimeout\n");
537  --aio_request_queue.idle_threads;
538  pthread_mutex_unlock (&aio_request_queue.mutex);
539  return NULL;
540  }
541  }
542  /* Otherwise move this chain to the working chain and
543  start the loop all over again */
544  AIO_printf ("Work on idle\n");
545  --aio_request_queue.idle_threads;
546  ++aio_request_queue.active_threads;
547 
548  node = rtems_chain_first (&aio_request_queue.idle_req);
549  rtems_chain_extract (node);
550 
551  r_chain = (rtems_aio_request_chain *) node;
552  rtems_aio_move_to_work (r_chain);
553 
554  }
555  }
556  /* If there was a request added in the initial fd chain then release
557  the mutex and process it */
558  pthread_mutex_unlock (&aio_request_queue.mutex);
559 
560  }
561  }
562 
563  AIO_printf ("Thread finished\n");
564  return NULL;
565 }
