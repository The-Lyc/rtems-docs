84 /*
85  * Caution : this function assumes the variable "*config"
86  * is already set and that the tables it contains are still valid
87  * and accessible.
88  */
89 static void compute_i8259_masks_from_prio (rtems_irq_global_settings* config)
90 {
91  int i;
92  int j;
93  /*
94  * Always mask at least current interrupt to prevent re-entrance
95  */
96  for (i=BSP_ISA_IRQ_LOWEST_OFFSET; i < BSP_ISA_IRQ_LOWEST_OFFSET + BSP_ISA_IRQ_NUMBER; i++) {
97  * ((unsigned short*) &irq_mask_or_tbl[i]) = (1 << i);
98  for (j = BSP_ISA_IRQ_LOWEST_OFFSET; j < BSP_ISA_IRQ_LOWEST_OFFSET + BSP_ISA_IRQ_NUMBER; j++) {
99  /*
100  * Mask interrupts at i8259 level that have a lower priority
101  */
102  if (config->irqPrioTbl [i] > config->irqPrioTbl [j]) {
103  * ((unsigned short*) &irq_mask_or_tbl[i]) |= (1 << j);
104  }
105  }
106  }
107 }

157 int BSP_setup_the_pic(rtems_irq_global_settings* config)
158 {
159  int i;
160  /*
161  * Store various code accelerators
162  */
163  default_rtems_entry = config->defaultEntry;
164  rtems_hdl_tbl = config->irqHdlTbl;
165 
166  /*
167  * set up internal tables used by rtems interrupt prologue
168  */
169 
170 #if BSP_ISA_IRQ_NUMBER > 0
171  /*
172  * start with ISA IRQ
173  */
174  compute_i8259_masks_from_prio (config);
175 
176  for (i=BSP_ISA_IRQ_LOWEST_OFFSET; i < BSP_ISA_IRQ_LOWEST_OFFSET + BSP_ISA_IRQ_NUMBER; i++) {
177  if (rtems_hdl_tbl[i].hdl != default_rtems_entry.hdl) {
178  BSP_irq_enable_at_i8259s (i);
179  }
180  else {
181  BSP_irq_disable_at_i8259s (i);
182  }
183  }
184 
185  if ( BSP_ISA_IRQ_NUMBER > 0 ) {
186  /*
187  * must enable slave pic anyway
188  */
189  BSP_irq_enable_at_i8259s (2);
190  }
191 #endif
192 
193 #if BSP_PCI_IRQ_NUMBER > 0
194  if ( ! OpenPIC )
195  return 1;
196  /*
197  * continue with PCI IRQ
198  */
199  for (i=BSP_PCI_IRQ_LOWEST_OFFSET; i < BSP_PCI_IRQ_LOWEST_OFFSET + BSP_PCI_IRQ_NUMBER ; i++) {
200  /*
201  * Note that openpic_set_priority() sets the TASK priority of the PIC
202  */
203  openpic_set_source_priority(i - BSP_PCI_IRQ_LOWEST_OFFSET,
204  config->irqPrioTbl[i]);
205  if (rtems_hdl_tbl[i].hdl != default_rtems_entry.hdl) {
206  openpic_enable_irq ((int) i - BSP_PCI_IRQ_LOWEST_OFFSET);
207  }
208  else {
209  openpic_disable_irq ((int) i - BSP_PCI_IRQ_LOWEST_OFFSET);
210  }
211  }
212 
213 #ifdef BSP_PCI_ISA_BRIDGE_IRQ
214  /*
215  * Must enable PCI/ISA bridge IRQ
216  */
217  openpic_enable_irq (BSP_PCI_ISA_BRIDGE_IRQ);
218 #endif
219 #endif
220 
221  return 1;
222 }

231 int C_dispatch_irq_handler (BSP_Exception_frame *frame, unsigned int excNum)
232 {
233  register unsigned int irq;
234 #if BSP_ISA_IRQ_NUMBER > 0
235  register unsigned isaIntr; /* boolean */
236  register unsigned oldMask = 0; /* old isa pic masks */
237  register unsigned newMask; /* new isa pic masks */
238 #endif
239 
240  if (excNum == ASM_DEC_VECTOR) {
241 
242  bsp_irq_dispatch_list(rtems_hdl_tbl, BSP_DECREMENTER, default_rtems_entry.hdl);
243 
244  return 0;
245 
246  }
247 
248 #if BSP_PCI_IRQ_NUMBER > 0
249  if ( OpenPIC ) {
250  irq = openpic_irq(0);
251  if (irq == OPENPIC_VEC_SPURIOUS) {
252  ++BSP_spuriousIntr;
253  return 0;
254  }
255 
256  /* some BSPs might want to use a different numbering... */
257  irq = irq - OPENPIC_VEC_SOURCE + BSP_PCI_IRQ_LOWEST_OFFSET;
258  } else {
259 #if BSP_ISA_IRQ_NUMBER > 0
260 #ifdef BSP_PCI_ISA_BRIDGE_IRQ
261  irq = BSP_PCI_ISA_BRIDGE_IRQ;
262 #else
263 #error "Configuration Error -- BSP with ISA + PCI IRQs MUST define BSP_PCI_ISA_BRIDGE_IRQ"
264 #endif
265 #else
266  rtems_panic("MUST have an OpenPIC if BSP has PCI IRQs but no ISA IRQs");
267  /* rtems_panic() never returns but the 'return' statement silences
268  * a compiler warning about 'irq' possibly being used w/o initialization.
269  */
270  return -1;
271 #endif
272  }
273 #endif

275 #if BSP_ISA_IRQ_NUMBER > 0
276 #ifdef BSP_PCI_ISA_BRIDGE_IRQ
277 #if 0 == BSP_PCI_IRQ_NUMBER
278 #error "Configuration Error -- BSP w/o PCI IRQs MUST NOT define BSP_PCI_ISA_BRIDGE_IRQ"
279 #endif

303 #endif
304

307 
308 #if BSP_ISA_IRQ_NUMBER > 0
309  if (isaIntr) {
310  i8259s_cache = oldMask;
311  outport_byte(PIC_MASTER_IMR_IO_PORT, i8259s_cache & 0xff);
312  outport_byte(PIC_SLAVE_IMR_IO_PORT, ((i8259s_cache & 0xff00) >> 8));
313  }
314  else
315 #endif
316  {

318 #ifdef BSP_PCI_VME_DRIVER_DOES_EOI

320  * it can re-enable the openpic while handling
321  * VME interrupts (-> VME priorities in software)
322  */
323  if (_BSP_vme_bridge_irq != irq && OpenPIC)
324 #endif
325  openpic_eoi(0);
326 #else
327  do {} while (0);
328 #endif
329  }
330  return 0;
