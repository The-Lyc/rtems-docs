151 /*
152  * BSP start routine. Called by boot_card().
153  *
154  * This routine does the bulk of the system initialization.
155  */
156 void bsp_start(void)
157 {
158  ppc_cpu_id_t myCpu;
159  ppc_cpu_revision_t myCpuRevision;
160 
161  /* Set the character output function; The application may override this */
162  BSP_output_char = __bsp_outchar_to_memory;
163 
164  printk("RTEMS %s\n", rtems_get_version_string());
165 
166  /*
167  * Get CPU identification dynamically. Note that the get_ppc_cpu_type()
168  * function stores the result in global variables so that it can be used later...
169  */
170  myCpu = get_ppc_cpu_type();
171  myCpuRevision = get_ppc_cpu_revision();
172  printk("CPU: 0x%04x, Revision: 0x%04x = %d, Name: %s\n",
173  myCpu, myCpuRevision, myCpuRevision, get_ppc_cpu_type_name(myCpu));
174 
175  /*
176  * Initialize the device driver parameters
177  */
178 
179  /* For mpc6xx clock driver: */
180  BSP_bus_frequency = 465000000;
181  BSP_processor_frequency = 465000000; /* Measured with a DPM 440 2012/8/13 */
182  BSP_time_base_divisor = 1000;
183 
184  /* Timebase register ticks/microsecond; The application may override these */
185  bsp_clicks_per_usec = BSP_bus_frequency/(BSP_time_base_divisor * 1000);
186 
187  ppc_exc_initialize();
188 
189  /* Let the user know what parameters we were compiled with */
190  printk(" Base/Start End Size\n"
191  "RAM: %p %p\n"
192  "RTEMS: %p\n"
193  "Workspace: %p %p\n"
194  "MsgArea: %p %p\n"
195  "Physical RAM %p\n",
196  RamBase, RamSize,
197  __rtems_end,
198  WorkAreaBase, __bsp_ram_end,
199  MsgAreaBase, MsgAreaSize,
200  __phy_ram_end);
201 
202  /*
203  * Initialize RTEMS IRQ system
204  */
205  BSP_rtems_irq_mngt_init(0);
206 
207  /* Continue with application-specific initialization */
208  app_bsp_start();
209 }
