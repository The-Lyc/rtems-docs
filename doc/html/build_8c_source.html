61 
62  /* For each child, increase nlink */
63  for(fd = ic->scan_dents; fd; fd = fd->next) {
64  struct jffs2_inode_cache *child_ic;
65  if (!fd->ino)
66  continue;
67

69

71  if (!child_ic) {

73  fd->name, fd->ino, ic->ino);

75  /* Clear the ic/raw union so it doesn't cause problems later. */
76  fd->ic = NULL;

81  fd->ic = child_ic;

84  * set a flag to trigger a scan later */
85  if (fd->type == DT_DIR) {
86  child_ic->flags |= INO_FLAGS_IS_DIR;
87  if (child_ic->pino_nlink > 1)
88  *dir_hardlinks = 1;
89  }
90

92  /* Can't free scan_dents so far. We might need them in pass 2 */
93  }

102 {

106  struct jffs2_full_dirent *dead_fds = NULL;
107

109

111  lists of physical nodes */
112 
113  c->flags |= JFFS2_SB_FLAG_SCANNING;
114  ret = jffs2_scan_medium(c);
115  c->flags &= ~JFFS2_SB_FLAG_SCANNING;
116  if (ret)

124  /* Now scan the directory tree, increasing nlink according to every dirent found. */
125  for_each_inode(i, c, ic) {
126  if (ic->scan_dents) {
127  jffs2_build_inode_pass1(c, ic, &dir_hardlinks);
128  cond_resched();
129  }
130  }
131 
132  dbg_fsbuild("pass 1 complete\n");
133 
134  /* Next, scan for inodes with nlink == 0 and remove them. If
135  they were directories, then decrement the nlink of their
136  children too, and repeat the scan. As that's going to be
137  a fairly uncommon occurrence, it's not so evil to do it this
138  way. Recursion bad. */
139  dbg_fsbuild("pass 2 starting\n");
140 
141  for_each_inode(i, c, ic) {
142  if (ic->pino_nlink)
143  continue;
144

148

151  while (dead_fds) {

154

156 
157  if (ic)

159  jffs2_free_full_dirent(fd);
160  }
161 
162  dbg_fsbuild("pass 2a complete\n");
163 
164  if (dir_hardlinks) {
165  /* If we detected directory hardlinks earlier, *hopefully*
166  * they are gone now because some of the links were from
167  * dead directories which still had some old dirents lying
168  * around and not yet garbage-collected, but which have
169  * been discarded above. So clear the pino_nlink field
170  * in each directory, so that the final scan below can
171  * print appropriate warnings. */
172  for_each_inode(i, c, ic) {
173  if (ic->flags & INO_FLAGS_IS_DIR)
174  ic->pino_nlink = 0;
175  }
176  }

178

180  for_each_inode(i, c, ic) {

182  fd = ic->scan_dents;
183  ic->scan_dents = fd->next;
184  /* We do use the pino_nlink field to count nlink of
185  * directories during fs build, so set it to the
186  * parent ino# now. Now that there's hopefully only
187  * one. */

191  raw node obsolete already. Just skip it. */
192  continue;
193  }
194 
195  /* We *have* to have set this in jffs2_build_inode_pass1() */
196  BUG_ON(!(fd->ic->flags & INO_FLAGS_IS_DIR));
197

200  * we don't do the check. And ic->pino_nlink still contains the nlink
201  * value (which is 1). */

205  /* Should we unlink it from its previous parent? */
206  }
207

209  fd->ic->pino_nlink = ic->ino;
210  }
211  jffs2_free_full_dirent(fd);
212  }
213  ic->scan_dents = NULL;
214  cond_resched();
215  }
216  jffs2_build_xattr_subsystem(c);
217  c->flags &= ~JFFS2_SB_FLAG_BUILDING;
218 
219  dbg_fsbuild("FS build complete\n");
220

222  jffs2_rotate_lists(c);
223 
224  ret = 0;
225 
226 exit:

228  for_each_inode(i, c, ic) {
229  while(ic->scan_dents) {
230  fd = ic->scan_dents;
231  ic->scan_dents = fd->next;
232  jffs2_free_full_dirent(fd);
233  }
234  }
235  jffs2_clear_xattr_subsystem(c);
236  }

241 static void jffs2_build_remove_unlinked_inode(struct jffs2_sb_info *c,
242  struct jffs2_inode_cache *ic,
243  struct jffs2_full_dirent **dead_fds)
244 {
245  struct jffs2_raw_node_ref *raw;
246  struct jffs2_full_dirent *fd;
247 
248  dbg_fsbuild("removing ino #%u with nlink == zero.\n", ic->ino);
249 
250  raw = ic->nodes;
251  while (raw != (void *)ic) {
252  struct jffs2_raw_node_ref *next = raw->next_in_ino;
253  dbg_fsbuild("obsoleting node at 0x%08x\n", ref_offset(raw));
254  jffs2_mark_node_obsolete(c, raw);
255  raw = next;
256  }
257 
258  if (ic->scan_dents) {
259  int whinged = 0;
260  dbg_fsbuild("inode #%u was a directory which may have children...\n", ic->ino);
261 
262  while(ic->scan_dents) {
263  struct jffs2_inode_cache *child_ic;
264 
265  fd = ic->scan_dents;
266  ic->scan_dents = fd->next;
267 
268  if (!fd->ino) {
269  /* It's a deletion dirent. Ignore it */
270  dbg_fsbuild("child \"%s\" is a deletion dirent, skipping...\n", fd->name);
271  jffs2_free_full_dirent(fd);
272  continue;
273  }
274  if (!whinged)
275  whinged = 1;
276 
277  dbg_fsbuild("removing child \"%s\", ino #%u\n", fd->name, fd->ino);
278 
279  child_ic = jffs2_get_ino_cache(c, fd->ino);
280  if (!child_ic) {
281  dbg_fsbuild("cannot remove child \"%s\", ino #%u, because it doesn't exist\n",
282  fd->name, fd->ino);
283  jffs2_free_full_dirent(fd);
284  continue;
285  }
286 
287  /* Reduce nlink of the child. If it's now zero, stick it on the
288  dead_fds list to be cleaned up later. Else just free the fd */
289  child_ic->pino_nlink--;
290 
291  if (!child_ic->pino_nlink) {
292  dbg_fsbuild("inode #%u (\"%s\") now has no links; adding to dead_fds list.\n",
293  fd->ino, fd->name);
294  fd->next = *dead_fds;
295  *dead_fds = fd;
296  } else {
297  dbg_fsbuild("inode #%u (\"%s\") has now got nlink %d. Ignoring.\n",
298  fd->ino, fd->name, child_ic->pino_nlink);
299  jffs2_free_full_dirent(fd);
300  }
301  }
302  }
303 
304  /*
305  We don't delete the inocache from the hash list and free it yet.
306  The erase code will do that, when all the nodes are completely gone.
307  */
308 }

310 static void jffs2_calc_trigger_levels(struct jffs2_sb_info *c)
311 {
312  uint32_t size;
313 
314  /* Deletion should almost _always_ be allowed. We're fairly
315  buggered once we stop allowing people to delete stuff
316  because there's not enough free space... */
317  c->resv_blocks_deletion = 2;
318 
319  /* Be conservative about how much space we need before we allow writes.
320  On top of that which is required for deletia, require an extra 2%
321  of the medium to be available, for overhead caused by nodes being
322  split across blocks, etc. */
323 
324  size = c->flash_size / 50; /* 2% of flash size */
325  size += c->nr_blocks * 100; /* And 100 bytes per eraseblock */
326  size += c->sector_size - 1; /* ... and round up */
327 
328  c->resv_blocks_write = c->resv_blocks_deletion + (size / c->sector_size);
329 
330  /* When do we let the GC thread run in the background */
331 
332  c->resv_blocks_gctrigger = c->resv_blocks_write + 1;
333 
334  /* When do we allow garbage collection to merge nodes to make
335  long-term progress at the expense of short-term space exhaustion? */
336  c->resv_blocks_gcmerge = c->resv_blocks_deletion + 1;
337 
338  /* When do we allow garbage collection to eat from bad blocks rather
339  than actually making progress? */
340  c->resv_blocks_gcbad = 0;//c->resv_blocks_deletion + 2;
341 
342  /* What number of 'very dirty' eraseblocks do we allow before we
343  trigger the GC thread even if we don't _need_ the space. When we
344  can't mark nodes obsolete on the medium, the old dirty nodes cause
345  performance problems because we have to inspect and discard them. */
346  c->vdirty_blocks_gctrigger = c->resv_blocks_gctrigger;
347  if (jffs2_can_mark_obsolete(c))
348  c->vdirty_blocks_gctrigger *= 10;
349 
350  /* If there's less than this amount of dirty space, don't bother
351  trying to GC to make more space. It'll be a fruitless task */
352  c->nospc_dirty_size = c->sector_size + (c->flash_size / 100);
353 
354  dbg_fsbuild("trigger levels (size %d KiB, block size %d KiB, %d blocks)\n",
355  c->flash_size / 1024, c->sector_size / 1024, c->nr_blocks);
356  dbg_fsbuild("Blocks required to allow deletion: %d (%d KiB)\n",
357  c->resv_blocks_deletion, c->resv_blocks_deletion*c->sector_size/1024);
358  dbg_fsbuild("Blocks required to allow writes: %d (%d KiB)\n",
359  c->resv_blocks_write, c->resv_blocks_write*c->sector_size/1024);
360  dbg_fsbuild("Blocks required to quiesce GC thread: %d (%d KiB)\n",
361  c->resv_blocks_gctrigger, c->resv_blocks_gctrigger*c->sector_size/1024);
362  dbg_fsbuild("Blocks required to allow GC merges: %d (%d KiB)\n",
363  c->resv_blocks_gcmerge, c->resv_blocks_gcmerge*c->sector_size/1024);
364  dbg_fsbuild("Blocks required to GC bad blocks: %d (%d KiB)\n",
365  c->resv_blocks_gcbad, c->resv_blocks_gcbad*c->sector_size/1024);
366  dbg_fsbuild("Amount of dirty space required to GC: %d bytes\n",
367  c->nospc_dirty_size);
368  dbg_fsbuild("Very dirty blocks before GC triggered: %d\n",
369  c->vdirty_blocks_gctrigger);
370 }
