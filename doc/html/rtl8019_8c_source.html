174 /*
175  * receive ring descriptor
176  *
177  * The National Semiconductor DS8390 Network interface controller uses
178  * the following receive ring headers. The way this works is that the
179  * memory on the interface card is chopped up into 256 bytes blocks.
180  * A contiguous portion of those blocks are marked for receive packets
181  * by setting start and end block #'s in the NIC. For each packet that
182  * is put into the receive ring, one of these headers (4 bytes each) is
183  * tacked onto the front. The first byte is a copy of the receiver status
184  * register at the time the packet was received.
185  */
186 struct ne_ring
187 {
188  unsigned char rsr; /* receiver status */
189  unsigned char next; /* pointer to next packet */
190  unsigned short count; /* bytes in packet (length + 4) */
191 };

209 static void
210 ne_read_data (struct ne_softc *sc, int addr, int len, unsigned char *p)
211 {
212  unsigned int port = sc->port;
213  unsigned int dport = port + DATAPORT;
214  outport_byte (port + CMDR, MSK_PG0 | MSK_RD2 | MSK_STA);
215  outport_byte (port + RBCR0, len);
216  outport_byte (port + RBCR1, len >> 8);
217  outport_byte (port + RSAR0, addr);
218  outport_byte (port + RSAR1, addr >> 8);
219  outport_byte (port + CMDR, MSK_PG0 | MSK_RRE | MSK_STA);
220 
221  if (sc->byte_transfers)
222  {
223  unsigned char d;
224  while (len > 0)
225  {
226  inport_byte(dport, d);
227  *p++ = d;
228  len--;
229  }
230  }
231  else /* word transfers */
232  {
233  unsigned short d;
234  while (len > 1)
235  {
236  inport_word(dport, d);
237  *p++ = d;
238  *p++ = d >> 8;
239  len -= 2;
240  }
241  if (len)
242  {
243  inport_word(dport, d);
244  *p++ = d;
245  }
246  }
247 
248  outport_byte (port + ISR, MSK_RDC);
249 }

252  signals an interrupt. It is also called at other times while

255 static void
256 ne_check_status (struct ne_softc *sc, int from_irq_handler)
257 {
258  struct ifnet *ifp = &sc->arpcom.ac_if;
259  unsigned int port = sc->port;
260  unsigned char status;
261 
262  /* It seems that we need to use a loop here, because if the NE2000
263  signals an interrupt because packet transmission is complete, and
264  then receives a packet while interrupts are disabled, it seems to
265  sometimes fail to signal the interrupt for the received packet
266  when interrupts are reenabled. (Based on the behaviour of the
267  Realtek 8019AS chip). */
268 
269  /* int count = 0; */
270  while (1)
271  {
272  inport_byte (port + ISR, status);
273 #ifdef DEBUG_NE2000
274  printk ("NE2000 status 0x%x\n", status);
275 #endif
276  if (status == 0)
277  break;
278 
279  /* ack */
280  outport_byte (port + ISR, status);
281 
282  /* Check for incoming packet overwrite. */
283  if (status & MSK_OVW)
284  {
285  ifp->if_timer = 0;
286 #ifdef DEBUG_NE
287  printk("^\n");
288 #endif
289  ++sc->stats.overruns;
290  ne_reset(sc);
291  /* Reenable device interrupts. */
292  if (from_irq_handler)
293  outport_byte(port + IMR, NE_INTERRUPTS);
294  return;
295  }
296 
297  /* Check for transmitted packet. The transmit daemon may now be
298  able to send another packet to the device. */
299  if ((status & (MSK_PTX | MSK_TXE)) != 0)
300  {
301  ifp->if_timer = 0;
302  ++sc->stats.tx_acks;
303  --sc->inuse;
304  sc->transmitting = 0;
305  if (sc->inuse > 0 || (sc->arpcom.ac_if.if_flags & IFF_OACTIVE) != 0)
306  rtems_bsdnet_event_send (sc->tx_daemon_tid, START_TRANSMIT_EVENT);
307  }
308 
309  /* Check for received packet. */
310  if ((status & (MSK_PRX | MSK_RXE)) != 0)
311  {
312  ++sc->stats.rx_acks;
313  rtems_bsdnet_event_send (sc->rx_daemon_tid, INTERRUPT_EVENT);
314  }
315 
316  /* Check for counter change. */
317  if ((status & MSK_CNT) != 0)
318  {
319  unsigned char add;
320  inport_byte (port + CNTR0, add);
321  sc->stats.rx_frame_errors += add;
322  inport_byte (port + CNTR1, add);
323  sc->stats.rx_crc_errors += add;
324  inport_byte (port + CNTR2, add);
325  sc->stats.rx_missed_errors += add;
326  }
327 
328  break;
329  }
330 
331  outport_byte (port + CMDR, MSK_PG0 | MSK_STA | MSK_RD2);
332 
333 }

367 static void
368 ne_init_hardware (struct ne_softc *sc)
369 {
370  unsigned int port = sc->port;
371  int i;
372 
373 #ifdef DEBUG_NE2000
374  printk ("ne_init_hardware()\n");
375 #endif
376 
377  /* Initialize registers. */
378 
379  /* Set interface for page 0, Remote DMA complete, Stopped */
380  outport_byte (port + CMDR, MSK_PG0 | MSK_RD2 | MSK_STP);
381 
382  /* Set FIFO threshold to 8, No auto-init Remote DMA, byte order=80x86 */
383  /* byte-wide DMA xfers */
384  if (sc->byte_transfers)
385  outport_byte (port + DCR, MSK_FT10 | MSK_BMS);
386  /* word-wide DMA xfers */
387  else
388  outport_byte (port + DCR, MSK_FT10 | MSK_BMS | MSK_WTS);
389 
390  /* Clear Remote Byte Count Registers */
391  outport_byte (port + RBCR0, 0);
392  outport_byte (port + RBCR1, 0);
393 
394  /* For the moment, don't store incoming packets in memory. */
395  outport_byte (port + RCR, MSK_MON);
396 
397  /* Place NIC in internal loopback mode */
398  outport_byte (port + TCR, MSK_LOOP);
399 
400  /* Initialize transmit/receive (ring-buffer) Page Start */
401  outport_byte (port + TPSR, NE_FIRST_TX_PAGE);
402  outport_byte (port + PSTART, NE_FIRST_RX_PAGE);
403 
404  /* Initialize Receiver (ring-buffer) Page Stop and Boundary */
405  outport_byte (port + PSTOP, NE_STOP_PAGE);
406  outport_byte (port + BNRY, NE_STOP_PAGE - 1);
407 
408  /* Clear all interrupts */
409  outport_byte (port + ISR, 0xff);
410  /* Disable all interrupts */
411  outport_byte (port + IMR, 0);
412 
413  /* Program Command Register for page 1 */
414  outport_byte (port + CMDR, MSK_PG1 | MSK_RD2 | MSK_STP);
415 
416  /* Set the Ethernet hardware address. */
417  for (i = 0; i < ETHER_ADDR_LEN; ++i)
418  outport_byte (port + PAR + i, sc->arpcom.ac_enaddr[i]);
419 
420  /* Set Current Page pointer to next_packet */
421  outport_byte (port + CURR, NE_FIRST_RX_PAGE);
422 
423  /* Clear the multicast address. */
424  for (i = 0; i < MARsize; ++i)
425  outport_byte (port + MAR + i, 0);
426 
427  /* Set page 0 registers */
428  outport_byte (port + CMDR, MSK_PG0 | MSK_RD2 | MSK_STP);
429 
430  /* accept broadcast */
431  outport_byte (port + RCR, (sc->accept_broadcasts ? MSK_AB : 0));
432 
433  /* Start interface */
434  outport_byte (port + CMDR, MSK_PG0 | MSK_RD2 | MSK_STA);
435 
436  /* Take interface out of loopback */
437  outport_byte (port + TCR, 0);
438 }

469 static void
470 ne_rx_daemon (void *arg)
471 {
472  struct ne_softc *sc = (struct ne_softc *) arg;
473  struct ifnet *ifp = &sc->arpcom.ac_if;
474  unsigned int port = sc->port;
475 
476  while (1)
477  {
478  rtems_event_set events;
479 
480  /* Wait for the interrupt handler to tell us that there is a
481  packet ready to receive. */
482  rtems_bsdnet_event_receive (INTERRUPT_EVENT,
483  RTEMS_WAIT | RTEMS_EVENT_ANY,
484  RTEMS_NO_TIMEOUT,
485  &events);
486 
487  /* Don't let the device interrupt us now. */
488  outport_byte (port + IMR, 0);
489 
490  while (1)
491  {
492  unsigned char startpage, currpage;
493  unsigned short len;
494  unsigned char next, cnt1, cnt2;
495  struct mbuf *m = NULL;
496  unsigned char *p;
497  int startaddr;
498  int toend;
499  struct ether_header *eh;
500  struct ne_ring hdr; /* ring buffer header */
501  int reset;
502 
503  inport_byte (port + BNRY, startpage);
504 
505  outport_byte (port + CMDR, MSK_PG1 | MSK_RD2);
506  inport_byte (port + CURR, currpage);
507  outport_byte (port + CMDR, MSK_PG0 | MSK_RD2);
508 
509  ++startpage;
510  if (startpage >= NE_STOP_PAGE)
511  startpage = NE_FIRST_RX_PAGE;
512 
513  if (startpage == currpage)
514  break;
515 
516 #ifdef DEBUG_NE2000
517  printk ("ne_rx_daemon: start page %x; current page %x\n",
518  startpage, currpage);
519 #endif
520 
521  reset = 0;
522 
523  /* Read the buffer header */
524  startaddr = startpage * NE_PAGE_SIZE;
525  ne_read_data(sc, startaddr, sizeof(hdr), (unsigned char *)&hdr);
526  next = hdr.next;
527 
528  if (next >= NE_STOP_PAGE)
529  next = NE_FIRST_RX_PAGE;
530 
531  /* check packet length */
532  len = hdr.count;
533  if (currpage < startpage)
534  cnt1 = currpage + (NE_STOP_PAGE - NE_FIRST_RX_PAGE) - startpage;
535  else
536  cnt1 = currpage - startpage;
537  cnt2 = len / NE_PAGE_SIZE;
538  if (len % NE_PAGE_SIZE)
539  cnt2++;
540  if (cnt1 < cnt2)
541  {
542 #ifdef DEBUG_NE
543  printk("(%x<%x:%x)\n", cnt1, cnt2, len);
544 /*
545  printk("start page 0x%x; current page 0x%x\n",
546  startpage, currpage);
547  printk("cnt1 < cnt2 (0x%x, 0x%x); len 0x%x\n",
548  cnt1, cnt2, len);
549 */
550 #endif
551  reset = 1;
552  }
553  if (len > (ETHER_MAX_LEN - ETHER_CRC_LEN + sizeof(struct ne_ring)) ||
554  len < (ETHER_MIN_LEN - ETHER_CRC_LEN + sizeof(struct ne_ring)) ||
555  len > MCLBYTES)
556  {
557 #ifdef DEBUG_NE
558  printk("(%x)", len);
559 
560  printk("start page 0x%x; current page 0x%x\n",
561  startpage, currpage);
562  printk("len out of range: 0x%x\n", len);
563  printk("stat: 0x%x, next: 0x%x\n", hdr.rsr, hdr.next);
564 #endif
565  reset = 1;
566  }
567 #ifdef DEBUG_NE
568  if (++ccc == 100)
569  { ccc = 0; reset = 1;
570  printk("T");
571  }
572 #endif
573 
574  /* reset interface */
575  if (reset)
576  {
577  printk("Reset in RX\n");
578  ne_reset(sc);
579  goto Next;
580  }
581 
582  /* The first four bytes of the length are the buffer header
583  * Just decrease them by 2 since in ARM, we have to make sure
584  * 4bytes memory access align on 4bytes
585  */
586  len -= 2;
587  startaddr += 2;
588 
589  MGETHDR (m, M_WAIT, MT_DATA);
590  MCLGET (m, M_WAIT);
591 
592  if (m == NULL)
593  panic ("ne_rx_daemon");
594 
595  m->m_pkthdr.rcvif = ifp;
596  m->m_nextpkt = 0;
597 
598  p = mtod (m, unsigned char *);
599  m->m_len = m->m_pkthdr.len = len - (sizeof(struct ether_header) + 2);
600 
601  toend = NE_STOP_PAGE * NE_PAGE_SIZE - startaddr;
602  if (toend < len)
603  {
604  ne_read_data (sc, startaddr, toend, p);
605  p += toend;
606  len -= toend;
607  startaddr = NE_FIRST_RX_PAGE * NE_PAGE_SIZE;
608  }
609 
610  if (len > 0)
611  ne_read_data (sc, startaddr, len, p);
612 
613  m->m_data +=2;
614  eh = mtod(m, struct ether_header *);
615  m->m_data += sizeof (struct ether_header);
616 
617 #ifdef DEBUG_NE
618  /*printk("[r%d]", hdr.count - sizeof(hdr));*/
619  printk("<\n");
620 #endif
621  ether_input (ifp, eh, m);
622  ++sc->stats.rx_packets;
623  outport_byte (port + BNRY, next - 1);
624  }
625 
626  if (sc->overrun) {
627  outport_byte (port + ISR, MSK_OVW);
628  outport_byte (port + TCR, 0);
629  if (sc->resend)
630  outport_byte (port + CMDR, MSK_PG0 | MSK_TXP | MSK_RD2 | MSK_STA);
631  sc->resend = 0;
632  sc->overrun = 0;
633  }
634 
635  Next:
636  /* Reenable device interrupts. */
637  outport_byte (port + IMR, NE_INTERRUPTS);
638  }
639 }

641 /* Load an NE2000 packet onto the device. */
642 
643 static void
644 ne_loadpacket (struct ne_softc *sc, struct mbuf *m)
645 {
646  unsigned int port = sc->port;
647  unsigned int dport = port + DATAPORT;
648  struct mbuf *mhold = m;
649  int leftover;
650  unsigned char leftover_data;
651  int timeout;
652  int send_cnt = 0;
653 
654 #ifdef DEBUG_NE2000
655  printk ("Uploading NE2000 packet\n");
656 #endif
657 
658  /* Reset remote DMA complete flag. */
659  outport_byte (port + ISR, MSK_RDC);
660 
661  /* Write out the count. */
662  outport_byte (port + RBCR0, m->m_pkthdr.len);
663  outport_byte (port + RBCR1, m->m_pkthdr.len >> 8);
664 
665  sc->sendlen[sc->nextavail] = m->m_pkthdr.len;
666 
667  /* Tell the device which address we want to write to. */
668  outport_byte (port + RSAR0, 0);
669  outport_byte (port + RSAR1,
670  NE_FIRST_TX_PAGE + (sc->nextavail * NE_TX_PAGES));
671 
672  /* Set up the write. */
673  outport_byte (port + CMDR, MSK_PG0 | MSK_RWR | MSK_STA);
674 
675  /* Transfer the mbuf chain to device memory. NE2000 devices require
676  that the data be transferred as words, so we need to handle odd
677  length mbufs. Never occurs if we force byte transfers. */
678 
679  leftover = 0;
680  leftover_data = '\0';
681 
682  for (; m != NULL; m = m->m_next) {
683  int len;
684  unsigned char *data;
685 
686  len = m->m_len;
687  if (len == 0)
688  continue;
689 
690  data = mtod (m, unsigned char *);
691 
692  if (leftover) {
693  unsigned char next;
694 
695  /* Data left over from previous mbuf in chain. */
696  next = *data++;
697  --len;
698  outport_word (dport, leftover_data | (next << 8));
699  send_cnt += 2;
700  leftover = 0;
701  }
702 
703  /* If using byte transfers, len always ends up as zero so
704  there are no leftovers. */
705 
706  if (sc->byte_transfers)
707  while (len > 0) {
708  outport_byte (dport, *data++);
709  len--;
710  }
711  else
712  while (len > 1) {
713  outport_word (dport, data[0] | (data[1] << 8));
714  data += 2;
715  len -= 2;
716  send_cnt += 2;
717  }
718 
719  if (len > 0)
720  {
721  leftover = 1;
722  leftover_data = *data++;
723  }
724  }
725

728  outport_word (dport, leftover_data);

734  printk("v");
735 #endif

737

739  limiting counter so that we don't wait too long. */
740  for (timeout = 0; timeout < 1000; ++timeout)
741  {
742  unsigned char status;
743 
744  inport_byte (port + ISR, status);
745 
746 #ifdef DEBUG_NE2000

753  outport_byte (port + ISR, MSK_RDC);
754  break;
755  }
756  }

759  printk ("Timed out waiting for acknowledgement of uploaded NE2000 packet\n");

804 /* The NE2000 packet transmit daemon. This task is started when the
805  NE2000 driver is initialized. */
806 
807 static void
808 ne_tx_daemon (void *arg)
809 {
810  struct ne_softc *sc = (struct ne_softc *) arg;
811  unsigned int port = sc->port;
812  struct ifnet *ifp = &sc->arpcom.ac_if;
813 
814  while (1) {
815  rtems_event_set events;
816 
817  /* Wait for a packet to be ready for sending, or for there to be
818  room for another packet in the device memory. */
819  rtems_bsdnet_event_receive (START_TRANSMIT_EVENT,
820  RTEMS_EVENT_ANY | RTEMS_WAIT,
821  RTEMS_NO_TIMEOUT,
822  &events);
823 
824 #ifdef DEBUG_NE2000
825  printk ("ne_tx_daemon\n");
826 #endif
827 
828  /* This daemon handles both uploading data onto the device and
829  telling the device to transmit data which has been uploaded.
830  These are separate tasks, because while the device is
831  transmitting one buffer we will upload another. */
832 
833  /* Don't let the device interrupt us now. */
834  outport_byte (port + IMR, 0);
835 
836  while (1) {
837  struct mbuf *m;
838 
839  /* If the device is not transmitting a packet, and we have
840  uploaded a packet, tell the device to transmit it. */
841  if (! sc->transmitting && sc->inuse > 0) {
842  sc->transmitting = 1;
843  ne_transmit (sc);
844  }
845 
846  /* If we don't have any more buffers to send, quit now. */
847  if (ifp->if_snd.ifq_head == NULL) {
848  ifp->if_flags &= ~IFF_OACTIVE;
849  break;
850  }
851 
852  /* Allocate a buffer to load data into. If there are none
853  available, quit until a buffer has been transmitted. */
854  if (sc->inuse >= NE_TX_BUFS)
855  break;
856 
857  ++sc->inuse;
858 
859  IF_DEQUEUE (&ifp->if_snd, m);
860  if (m == NULL)
861  panic ("ne_tx_daemon");
862 
863  ne_loadpacket (sc, m);
864 
865  /* Check the device status. It may have finished transmitting
866  the last packet. */
867  ne_check_status(sc, 0);
868  }
869 
870  /* Reenable device interrupts. */
871  outport_byte (port + IMR, NE_INTERRUPTS);
872  }
873 }
