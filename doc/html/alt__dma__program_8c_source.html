50 //
51 // The following section describes how the bits are used in the "flag" field:
52 //
53 
54 // [17:16] Which loop registers (LOOP0, LOOP1) are currently being used by a
55 // partially assembled program. LOOP0 is always used before LOOP1. LOOP1 is
56 // always ended before LOOP0.
57 #define ALT_DMA_PROGRAM_FLAG_LOOP0 (1UL << 16)
58 #define ALT_DMA_PROGRAM_FLAG_LOOP1 (1UL << 17)
59 #define ALT_DMA_PROGRAM_FLAG_LOOP_ALL (ALT_DMA_PROGRAM_FLAG_LOOP0 | ALT_DMA_PROGRAM_FLAG_LOOP1)
60 
61 // [18] Flag that marks LOOP0 as a forever loop. Said another way, LOOP0 is
62 // being used to execute the DMALPFE directive.
63 #define ALT_DMA_PROGRAM_FLAG_LOOP0_IS_FE (1UL << 18)
64 // [19] Flag that marks LOOP1 as a forever loop. Said another way, LOOP1 is
65 // being used to execute the DMALPFE directive.
66 #define ALT_DMA_PROGRAM_FLAG_LOOP1_IS_FE (1UL << 19)
67 
68 // [24] Flag that the first SAR has been programmed. The SAR field is valid and
69 // is the offset from the start of the buffer where SAR is located.
70 #define ALT_DMA_PROGRAM_FLAG_SAR (1UL << 24)
71 // [25] Flag that the first DAR has been programmed. The DAR field is valid and
72 // is the offset from the start of the buffer where DAR is located.
73 #define ALT_DMA_PROGRAM_FLAG_DAR (1UL << 25)
74 
75 // [31] Flag that marks the last assembled instruction as DMAEND.
76 #define ALT_DMA_PROGRAM_FLAG_ENDED (1UL << 31)

80 ALT_STATUS_CODE alt_dma_program_init(ALT_DMA_PROGRAM_t * pgm)
81 {
82  // Clear the variables that matter.
83  pgm->flag = 0;
84  pgm->code_size = 0;
85 
86  // Calculate the cache aligned start location of the buffer.
87  size_t buffer = (size_t)pgm->program;
88  size_t offset = ((buffer + ALT_DMA_PROGRAM_CACHE_LINE_SIZE - 1) & ~(ALT_DMA_PROGRAM_CACHE_LINE_SIZE - 1)) - buffer;
89 
90  // It is safe to cast to uint16_t because the extra offset can only be up to
91  // (ALT_DMA_PROGRAM_CACHE_LINE_SIZE - 1) or 31, which is within range of the
92  // uint16_t.
93  pgm->buffer_start = (uint16_t)offset;
94 
95  return ALT_E_SUCCESS;
96 }

117 ALT_STATUS_CODE alt_dma_program_validate(const ALT_DMA_PROGRAM_t * pgm)
118 {
119  // Verify that at least one instruction is in the buffer
120  if (pgm->code_size == 0)
121  {
122  return ALT_E_ERROR;
123  }
124 
125  // Verify all loops are completed.
126  if (pgm->flag & ALT_DMA_PROGRAM_FLAG_LOOP_ALL)
127  {
128  return ALT_E_ERROR;
129  }
130 
131  // Verify last item is DMAEND
132  if (!(pgm->flag & ALT_DMA_PROGRAM_FLAG_ENDED))
133  {
134  return ALT_E_ERROR;
135  }
136 
137  // Sync the DMA program to RAM.
138  void * vaddr = (void *)((uintptr_t)(pgm->program + pgm->buffer_start) & ~(ALT_CACHE_LINE_SIZE - 1));
139  size_t length = (pgm->code_size + ALT_CACHE_LINE_SIZE) & ~(ALT_CACHE_LINE_SIZE - 1);
140 
141  dprintf("DEBUG[DMAP]: Program (real) @ %p, length = 0x%x.\n", pgm->program + pgm->buffer_start, pgm->code_size);
142  dprintf("DEBUG[DMAP]: Clean: addr = %p, length = 0x%x.\n", vaddr, length);
143 
144  return alt_cache_system_clean(vaddr, length);
145 }

147 ALT_STATUS_CODE alt_dma_program_progress_reg(ALT_DMA_PROGRAM_t * pgm,
148  ALT_DMA_PROGRAM_REG_t reg,
149  uint32_t current, uint32_t * progress)
150 {
151  // Pointer to where the register is initialized in the program buffer.
152  uint8_t * buffer = NULL;
153 
154  switch (reg)
155  {
156  case ALT_DMA_PROGRAM_REG_SAR:
157  if (!(pgm->flag & ALT_DMA_PROGRAM_FLAG_SAR))
158  {
159  return ALT_E_BAD_ARG;
160  }
161  buffer = pgm->program + pgm->buffer_start + pgm->sar;
162  break;
163 
164  case ALT_DMA_PROGRAM_REG_DAR:
165  if (!(pgm->flag & ALT_DMA_PROGRAM_FLAG_DAR))
166  {
167  return ALT_E_BAD_ARG;
168  }
169  buffer = pgm->program + pgm->buffer_start + pgm->dar;
170  break;
171 
172  default:
173  return ALT_E_BAD_ARG;
174  }
175 
176  uint32_t initial =
177  (buffer[3] << 24) |
178  (buffer[2] << 16) |
179  (buffer[1] << 8) |
180  (buffer[0] << 0);
181 
182  *progress = current - initial;
183 
184  return ALT_E_SUCCESS;
185 }

521 ALT_STATUS_CODE alt_dma_program_DMALP(ALT_DMA_PROGRAM_t * pgm,
522  uint32_t iterations)
523 {
524  // For information on DMALP, see PL330, section 4.3.9.
525 
526  // Check for sufficient space in buffer
527  if ((pgm->code_size + 2) > ALT_DMA_PROGRAM_PROVISION_BUFFER_SIZE)
528  {
529  return ALT_E_BUF_OVF;
530  }
531 
532  // Verify iterations in range
533  if ((iterations == 0) || (iterations > 256))
534  {
535  return ALT_E_BAD_ARG;
536  }
537 
538  // Find suitable LOOPx register to use; construct lc mask value.
539  uint8_t lc_mask = 0;
540  switch (pgm->flag & ALT_DMA_PROGRAM_FLAG_LOOP_ALL)
541  {
542  case 0: // No LOOPx in use. Use LOOP0.
543  pgm->flag |= ALT_DMA_PROGRAM_FLAG_LOOP0;
544  pgm->loop0 = pgm->code_size + 2; // This is the first instruction after the DMALP
545  lc_mask = 0x0;
546  break;
547 
548  case ALT_DMA_PROGRAM_FLAG_LOOP0: // LOOP0 in use. Use LOOP1.
549  pgm->flag |= ALT_DMA_PROGRAM_FLAG_LOOP1;
550  pgm->loop1 = pgm->code_size + 2; // This is the first instruction after the DMALP
551  lc_mask = 0x2;
552  break;
553 
554  case ALT_DMA_PROGRAM_FLAG_LOOP_ALL: // All LOOPx in use. Report error.
555  return ALT_E_BAD_OPERATION;
556 
557  default: // Catastrophic error !!!
558  return ALT_E_ERROR;
559  }
560 
561  // Buffer of where to assemble the instruction.
562  uint8_t * buffer = pgm->program + pgm->buffer_start + pgm->code_size;
563 
564  // Assemble DMALP
565  buffer[0] = 0x20 | lc_mask;
566  buffer[1] = (uint8_t)(iterations - 1);
567 
568  // Update the code size.
569  pgm->code_size += 2;
570 
571  return ALT_E_SUCCESS;
572 }

574 ALT_STATUS_CODE alt_dma_program_DMALPEND(ALT_DMA_PROGRAM_t * pgm,
575  ALT_DMA_PROGRAM_INST_MOD_t mod)
576 {
577  // For information on DMALPEND, see PL330, section 4.3.10.
578 
579  // Check for sufficient space in buffer
580  if ((pgm->code_size + 2) > ALT_DMA_PROGRAM_PROVISION_BUFFER_SIZE)
581  {
582  return ALT_E_BUF_OVF;
583  }
584 
585  // Verify instruction modifier; construct bs, x mask value.
586  uint8_t bsx_mask = 0;
587  switch (mod)
588  {
589  case ALT_DMA_PROGRAM_INST_MOD_NONE:
590  bsx_mask = 0x0;
591  break;
592  case ALT_DMA_PROGRAM_INST_MOD_SINGLE:
593  bsx_mask = 0x1;
594  break;
595  case ALT_DMA_PROGRAM_INST_MOD_BURST:
596  bsx_mask = 0x3;
597  break;
598  default:
599  return ALT_E_BAD_ARG;
600  }
601 
602  // Determine the loop to end, if it is a forever loop; construct lc mask, nf mask, and backwards jump value.
603  uint8_t lc_mask = 0;
604  uint8_t nf_mask = 0;
605  uint16_t backwards_jump = 0;
606  switch (pgm->flag & ALT_DMA_PROGRAM_FLAG_LOOP_ALL)
607  {
608  case ALT_DMA_PROGRAM_FLAG_LOOP0: // LOOP0 in use. End LOOP0.
609 
610  backwards_jump = pgm->code_size - pgm->loop0;
611 
612  pgm->flag &= ~ALT_DMA_PROGRAM_FLAG_LOOP0;
613  pgm->loop0 = 0;
614 
615  lc_mask = 0x0;
616 
617  if (pgm->flag & ALT_DMA_PROGRAM_FLAG_LOOP0_IS_FE)
618  {
619  pgm->flag &= ~ALT_DMA_PROGRAM_FLAG_LOOP0_IS_FE;
620  }
621  else
622  {
623  nf_mask = 0x10;
624  }
625  break;
626 
627  case ALT_DMA_PROGRAM_FLAG_LOOP_ALL: // All LOOPx in use. End LOOP1.
628 
629  backwards_jump = pgm->code_size - pgm->loop1;
630 
631  pgm->flag &= ~ALT_DMA_PROGRAM_FLAG_LOOP1;
632  pgm->loop1 = 0;
633 
634  lc_mask = 0x4;
635 
636  if (pgm->flag & ALT_DMA_PROGRAM_FLAG_LOOP1_IS_FE)
637  {
638  pgm->flag &= ~ALT_DMA_PROGRAM_FLAG_LOOP1_IS_FE;
639  }
640  else
641  {
642  nf_mask = 0x10;
643  }
644  break;
645 
646  case 0: // No LOOPx in use. Report error!
647  return ALT_E_BAD_OPERATION;
648 
649  default: // Catastrophic error !!!
650  return ALT_E_ERROR;
651  }
652 
653  // Verify that the jump size is suitable
654  if (backwards_jump > 255)
655  {
656  return ALT_E_ARG_RANGE;
657  }
658 
659  // Buffer of where to assemble the instruction.
660  uint8_t * buffer = pgm->program + pgm->buffer_start + pgm->code_size;
661 
662  // Assemble DMALPEND
663  buffer[0] = 0x28 | nf_mask | lc_mask | bsx_mask;
664  buffer[1] = (uint8_t)(backwards_jump);
665 
666  // Update the code size.
667  pgm->code_size += 2;
668 
669  return ALT_E_SUCCESS;
670 }
