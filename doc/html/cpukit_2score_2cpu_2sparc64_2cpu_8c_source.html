112 /*
113  * This initializes the set of opcodes placed in each trap
114  * table entry. The routine which installs a handler is responsible
115  * for filling in the fields for the _handler address and the _vector
116  * trap type.
117  *
118  * The constants following this structure are masks for the fields which
119  * must be filled in when the handler is installed.
120  */
121 
122 /* 64-bit registers complicate this. Also, in sparc v9,
123  * each trap level gets its own set of global registers, but
124  * does not get its own dedicated register window. so we avoid
125  * using the local registers in the trap handler.
126  */
127 const CPU_Trap_table_entry _CPU_Trap_slot_template = {
128  0x89508000, /* rdpr %tstate, %g4 */
129  0x05000000, /* sethi %hh(_handler), %g2 */
130  0x8410a000, /* or %g2, %hm(_handler), %g2 */

133  0x8610c002, /* or %g3, %g2, %g3 */
134  0x81c0e000, /* jmp %g3 + %lo(_handler) */

156 /*
157  * _CPU_ISR_install_raw_handler
158  *
159  * This routine installs the specified handler as a "raw" non-executive
160  * supported trap handler (a.k.a. interrupt service routine).
161  *
162  * Input Parameters:
163  * vector - trap table entry number plus synchronous
164  * vs. asynchronous information
165  * new_handler - address of the handler to be installed
166  * old_handler - pointer to an address of the handler previously installed
167  *
168  * Output Parameters: NONE
169  * *new_handler - address of the handler previously installed
170  *
171  * NOTE:
172  *
173  * On the SPARC v9, there are really only 512 vectors. However, the executive
174  * has no easy, fast, reliable way to determine which traps are synchronous
175  * and which are asynchronous. By default, traps return to the
176  * instruction which caused the interrupt. So if you install a software
177  * trap handler as an executive interrupt handler (which is desirable since
178  * RTEMS takes care of window and register issues), then the executive needs
179  * to know that the return address is to the trap rather than the instruction
180  * following the trap.
181  *
182  * So vectors 0 through 511 are treated as regular asynchronous traps which
183  * provide the "correct" return address. Vectors 512 through 1023 are assumed
184  * by the executive to be synchronous and to require that the return be to the
185  * trapping instruction.
186  *
187  * If you use this mechanism to install a trap handler which must reexecute
188  * the instruction which caused the trap, then it should be installed as
189  * a synchronous trap. This will avoid the executive changing the return
190  * address.
191  */
192 void _CPU_ISR_install_raw_handler(
193  uint32_t vector,
194  CPU_ISR_raw_handler new_handler,
195  CPU_ISR_raw_handler *old_handler
196 )
197 {
198  uint32_t real_vector;
199  CPU_Trap_table_entry *tba;
200  CPU_Trap_table_entry *slot;
201  uint64_t u64_tba;
202  uint64_t u64_handler;
203 
204  /*
205  * Get the "real" trap number for this vector ignoring the synchronous
206  * versus asynchronous indicator included with our vector numbers.
207  */
208 
209  real_vector = SPARC_REAL_TRAP_NUMBER( vector );
210 
211  /*
212  * Get the current base address of the trap table and calculate a pointer
213  * to the slot we are interested in.
214  */
215 
216  sparc64_get_tba( u64_tba );
217 
218 /* u32_tbr &= 0xfffff000; */
219  u64_tba &= 0xffffffffffff8000; /* keep only trap base address */
220 
221  tba = (CPU_Trap_table_entry *) u64_tba;
222 
223  /* use array indexing to fill in lower bits -- require
224  * CPU_Trap_table_entry to be full-sized. */
225  slot = &tba[ real_vector ];
226 
227  /*
228  * Get the address of the old_handler from the trap table.
229  *
230  * NOTE: The old_handler returned will be bogus if it does not follow
231  * the RTEMS model.
232  */
233 
234  /* shift amount to shift of hi bits (31:10) */
235 #define HI_BITS_SHIFT 10
236 
237  /* shift amount of hm bits (41:32) */
238 #define HM_BITS_SHIFT 32
239 
240  /* shift amount of hh bits (63:42) */
241 #define HH_BITS_SHIFT 42
242 
243  /* We're only interested in bits 0-9 of the immediate field*/
244 #define IMM_MASK 0x000003FF
245 
246  if ( slot->rdpr_tstate_g4 == _CPU_Trap_slot_template.rdpr_tstate_g4 ) {
247  u64_handler =
248  (((uint64_t)((slot->sethi_of_hh_handler_to_g2 << HI_BITS_SHIFT) |
249  (slot->or_g2_hm_handler_to_g2 & IMM_MASK))) << HM_BITS_SHIFT) |
250  ((slot->sethi_of_handler_to_g3 << HI_BITS_SHIFT) |
251  (slot->jmp_to_low_of_handler_plus_g3 & IMM_MASK));
252  *old_handler = (CPU_ISR_raw_handler) u64_handler;
253  } else
254  *old_handler = 0;
255 
256  /*
257  * Copy the template to the slot and then fix it.
258  */
259 
260  *slot = _CPU_Trap_slot_template;
261 
262  u64_handler = (uint64_t) new_handler;
263 
264  /* mask for extracting %hh */
265 #define HH_BITS_MASK 0xFFFFFC0000000000
266 
267  /* mask for extracting %hm */
268 #define HM_BITS_MASK 0x000003FF00000000
269 
270  /* mask for extracting %hi */
271 #define HI_BITS_MASK 0x00000000FFFFFC00
272 
273  /* mask for extracting %lo */
274 #define LO_BITS_MASK 0x00000000000003FF
275 
276 
277  slot->mov_vector_g2 |= vector;
278  slot->sethi_of_hh_handler_to_g2 |=
279  (u64_handler & HH_BITS_MASK) >> HH_BITS_SHIFT;
280  slot->or_g2_hm_handler_to_g2 |=
281  (u64_handler & HM_BITS_MASK) >> HM_BITS_SHIFT;
282  slot->sethi_of_handler_to_g3 |=
283  (u64_handler & HI_BITS_MASK) >> HI_BITS_SHIFT;
284  slot->jmp_to_low_of_handler_plus_g3 |= (u64_handler & LO_BITS_MASK);
285 
286  /* need to flush icache after this !!! */
287 
288  /* need to flush icache in case old trap handler is in cache */
289  rtems_cache_invalidate_entire_instruction();
290 
291 }
