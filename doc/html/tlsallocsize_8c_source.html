36 uintptr_t _TLS_Get_allocation_size( void )
37 {
38  uintptr_t size;
39  uintptr_t allocation_size;
40  uintptr_t alignment;
41 
42  size = _TLS_Get_size();
43 
44  if ( size == 0 ) {
45  return 0;
46  }
47 
48  allocation_size = _TLS_Allocation_size;
49 
50  if ( allocation_size == 0 ) {
51  allocation_size = _TLS_Heap_align_up( size );
52  alignment = _TLS_Heap_align_up( (uintptr_t) _TLS_Alignment );
53 
54  /*
55  * The stack allocator does not support aligned allocations. Allocate
56  * enough to do the alignment manually.
57  */
58  if ( alignment > CPU_HEAP_ALIGNMENT ) {
59  allocation_size += alignment;
60  }
61 
62  allocation_size += _TLS_Get_thread_control_block_area_size( alignment );
63 
64 #ifndef __i386__
65  allocation_size += sizeof(TLS_Dynamic_thread_vector);
66 #endif
67 
68  _TLS_Allocation_size = allocation_size;
69  }
70 
71  return allocation_size;
72 }
