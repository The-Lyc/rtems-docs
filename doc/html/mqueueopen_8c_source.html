124 mqd_t mq_open(
125  const char *name,
126  int oflag,
127  ...
128  /* mode_t mode, */
129  /* struct mq_attr attr */
130 )
131 {
132  POSIX_Message_queue_Control *the_mq;
133  size_t name_len;
134  Objects_Get_by_name_error error;
135  mqd_t status;
136 
137  _Objects_Allocator_lock();
138  the_mq = _POSIX_Message_queue_Get_by_name( name, &name_len, &error );
139 
140  /*
141  * If the name to id translation worked, then the message queue exists
142  * and we can just return a pointer to the id. Otherwise we may
143  * need to check to see if this is a "message queue does not exist"
144  * or some other miscellaneous error on the name.
145  */
146  if ( the_mq == NULL ) {
147  va_list ap;
148  const struct mq_attr *attr;
149 
150  /*
151  * Unless provided a valid name that did not already exist
152  * and we are willing to create then it is an error.
153  */
154 
155  if ( error != OBJECTS_GET_BY_NAME_NO_OBJECT || ( oflag & O_CREAT ) == 0 ) {
156  _Objects_Allocator_unlock();
157  rtems_set_errno_and_return_value(
158  _POSIX_Get_by_name_error( error ),
159  MQ_OPEN_FAILED
160  );
161  }
162 
163  va_start( ap, oflag );
164  va_arg( ap, mode_t );
165  attr = va_arg( ap, const struct mq_attr * );
166  va_end( ap );
167 
168  if ( attr == NULL ) {
169  attr = &_POSIX_Message_queue_Default_attributes;
170  }
171 
172  /*
173  * At this point, the message queue does not exist and everything has been
174  * checked. We should go ahead and create a message queue.
175  */
176 
177  status = _POSIX_Message_queue_Create( name, name_len, oflag, attr );
178  } else {
179 
180  /*
181  * Check for existence with creation.
182  */
183 
184  if ( ( oflag & ( O_CREAT | O_EXCL ) ) == ( O_CREAT | O_EXCL ) ) {
185  _Objects_Allocator_unlock();
186  rtems_set_errno_and_return_value( EEXIST, MQ_OPEN_FAILED );
187  }
188 
189  the_mq->open_count += 1;
190  status = the_mq->Object.id;
191  }
192 
193  _Objects_Allocator_unlock();
194  return status;
195 }
