37 /* Blackfins can't just invalidate cache; they can only do flush +
38  invalidate. If the line isn't dirty then this is equivalent to
39  just an invalidate. Even if it is dirty, this should still be
40  okay since with a pure invalidate method the caller would have no
41  way to insure the dirty line hadn't been written out anyway prior
42  to the invalidate. */

44 
45  __asm__ __volatile__ ("ssync; flushinv [%0]; ssync" :: "a" (d_addr));

70 static void _CPU_cache_flush_entire_data(void) {
71  uint32_t i;
72 
73  i = 0;
74  __asm__ __volatile__ ("ssync");
75  do {
76  __asm__ __volatile__ ("flush [%0]" :: "a" (i));
77  i += CPU_DATA_CACHE_ALIGNMENT;
78  } while (i);
79  __asm__ __volatile__ ("ssync");
80 }
