241 xdrmbuf_getlong_aligned(register XDR *xdrs, register long *lp)
242 {
243  while ( (signed int)(xdrs->x_handy -= sizeof(int32_t)) < 0) {
244  if ((xdrs->x_handy += sizeof(int32_t)) == 0) {
245  /* handy was 0 on entry; request a new buffer.
246  * Coded this way, so the most frequently executed
247  * path needs only one comparison...
248  */
249  if (!xdrmbuf_next(xdrs))
250  return FALSE;
251  } else {
252  /* uh-oh an aligned long spread over two MBUFS ??
253  * let the unaligned handler deal with this rare
254  * situation.
255  */
256  return xdrmbuf_getlong_unaligned(xdrs,lp);
257  }
258  }
259  *lp = ntohl(*(int32_t *)(xdrs->x_private));
260  xdrs->x_private += sizeof(int32_t);
261 #if DEBUG & DEBUG_VERB
262  fprintf(stderr,"Got aligned long %x\n",*lp);
263 #endif

283 static bool_t
284 xdrmbuf_getlong_unaligned(
285  XDR *xdrs,
286  long *lp)
287 {
288 union {
289  int32_t l;
290  char c[sizeof(int32_t)];
291 } u;
292 
293 register int i,j;
294 register char *cp,*sp;
295 
296  i = xdrs->x_handy - sizeof(int32_t);
297 
298  /* handle the most common case first */
299  if ( i >= 0 ) {
300 
301  xdrs->x_handy = i;
302  sp = (char*)xdrs->x_private;
303  xdrs->x_private = sp + sizeof(int32_t);
304 
305 #ifdef CANDO_UNALIGNED
306  {
307  *lp = ntohl(*(int32_t *)sp);
308 # if DEBUG & DEBUG_VERB
309  fprintf(stderr,"Got unaligned long %x (%i remaining)\n",*lp, xdrs->x_handy);
310 # endif
311  return TRUE;
312  }
313 #else /* machine can't do unaligned access */
314  {
315  u.c[0] = *sp;
316  u.c[1] = *++sp;
317  u.c[2] = *++sp;
318  u.c[3] = *++sp;
319 
320  goto done;
321  }
322 #endif /* CANDO_UNALIGNED */
323  }
324 
325  /* here the messy 'crossing buffers' business starts */
326 
327 
328  j = sizeof(int32_t);
329 
330  cp = u.c-1;
331 
332  /* NOTE: on entry to this section, handy < j holds */
333  do {
334  sp = ((char*)xdrs->x_private)-1;
335 
336  if ( (i=xdrs->x_handy) >= j ) {
337  /* more data in the buffer than we need:
338  * copy everything we need and goto 'done'
339  */
340  xdrs->x_handy = i-j;
341  do {
342  *++cp = *++sp;
343  } while (--j > 0);
344  xdrs->x_private = (caddr_t)++sp;
345 
346  goto done;
347 
348  } else {
349  /* not enough data - copy as much as possible
350  * then get retrieve the next MBUF and start
351  * over
352  */
353  j-=i;
354  while (i--)
355  *++cp = *++sp;
356  if (!xdrmbuf_next(xdrs))
357  return FALSE;
358 #if DEBUG & DEBUG_VERB
359  fprintf(stderr,"getlong_unaligned: crossed mbuf boundary\n");
360 #endif
361  }
362  } while (j > 0);
363 
364 done:
365 
366  *lp = ntohl(u.l);
367 
368 #if DEBUG & DEBUG_VERB
369  fprintf(stderr,"Got unaligned long %x (%i remaining)\n",*lp, xdrs->x_handy);
370 #endif
371  return (TRUE);
372 }
