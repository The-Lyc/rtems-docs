87 rtems_am29lv160_blank (const rtems_fdisk_segment_desc* sd,
88  uint32_t device,
89  uint32_t segment,
90  uint32_t offset,
91  uint32_t size)
92 {
93  const rtems_am29lv160_config* ac = &rtems_am29lv160_configuration[device];
94  volatile uint8_t* seg_8 = ac->base;
95  volatile uint32_t* seg_32;
96  uint32_t count;
97 
98  offset += sd->offset + (segment - sd->segment) * sd->size;
99 
100  seg_8 += offset;
101 
102  count = offset & (sizeof (uint32_t) - 1);
103  size -= count;
104 
105  while (count--)
106  if (*seg_8++ != 0xff)
107  {
108 #if AM26LV160_ERROR_TRACE
109  printf ("AM26LV160: blank check error: %p = 0x%02x\n",
110  seg_8 - 1, *(seg_8 - 1));
111 #endif
112  return EIO;
113  }
114 
115  seg_32 = (volatile uint32_t*) seg_8;
116 
117  count = size / sizeof (uint32_t);
118  size -= count * sizeof (uint32_t);
119 
120  while (count--)
121  if (*seg_32++ != 0xffffffff)
122  {
123 #if AM26LV160_ERROR_TRACE
124  printf ("AM26LV160: blank check error: %p = 0x%08lx\n",
125  seg_32 - 1, *(seg_32 - 1));
126 #endif
127  return EIO;
128  }
129 
130  seg_8 = (volatile uint8_t*) seg_32;
131 
132  while (size--)
133  if (*seg_8++ != 0xff)
134  {
135 #if AM26LV160_ERROR_TRACE
136  printf ("AM26LV160: blank check error: %p = 0x%02x\n",
137  seg_8 - 1, *(seg_8 - 1));
138 #endif
139  return EIO;
140  }
141 
142  return 0;
143 }

164 static int
165 rtems_am29lv160_toggle_wait_8 (volatile uint8_t* status)
166 {
167  while (1)
168  {
169  volatile uint8_t status1 = *status;
170  volatile uint8_t status2 = *status;
171 
172  if (((status1 ^ status2) & (1 << 6)) == 0)
173  return 0;
174 
175  if ((status1 & (1 << 5)) != 0)
176  {
177  status1 = *status;
178  status2 = *status;
179 
180  if (((status1 ^ status2) & (1 << 6)) == 0)
181  return 0;
182 
183 #if AM26LV160_ERROR_TRACE
184  printf ("AM26LV160: error bit detected: %p = 0x%04x\n",

226  uint32_t offset,
