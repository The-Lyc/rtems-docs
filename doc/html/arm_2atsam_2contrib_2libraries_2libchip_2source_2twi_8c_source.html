112 void TWI_ConfigureMaster(Twihs *pTwi, uint32_t dwTwCk, uint32_t dwMCk)
113 {
114  uint32_t dwCkDiv = 0;
115  uint32_t dwClDiv;
116  uint32_t dwOk = 0;
117 
118  TRACE_DEBUG("TWI_ConfigureMaster()\n\r");
119  assert(pTwi);
120 
121  /* Reset the TWI */
122  pTwi->TWIHS_CR = TWIHS_CR_SWRST;
123 
124  /* Configure clock */
125  while (!dwOk) {
126  dwClDiv = ((dwMCk / (2 * dwTwCk)) - 4) / (1 << dwCkDiv);
127 
128  if (dwClDiv <= 255)
129  dwOk = 1;
130  else
131  dwCkDiv++;
132  }
133 
134  assert(dwCkDiv < 8);
135  TRACE_DEBUG("Using CKDIV = %u and CLDIV/CHDIV = %u\n\r", dwCkDiv, dwClDiv);
136 
137  pTwi->TWIHS_CWGR = (dwCkDiv << 16) | (dwClDiv << 8) | dwClDiv;
138 
139  /* TWI Slave Mode Disabled. */
140  pTwi->TWIHS_CR = TWIHS_CR_SVDIS;
141 
142  /* Set master mode */
143  pTwi->TWIHS_CR = TWIHS_CR_MSEN;
144 }

151 void TWI_ConfigureSlave(Twihs *pTwi, uint8_t slaveAddress)
152 {
153  /* TWI software reset */
154  pTwi->TWIHS_CR = TWIHS_CR_SWRST;
155 
156  /* Configure slave address. */
157  pTwi->TWIHS_SMR = TWIHS_SMR_SADR(slaveAddress);
158 
159  /* TWI Master Mode Disabled*/
160  pTwi->TWIHS_CR = TWIHS_CR_MSDIS;
161 
162  /* SVEN: TWI Slave Mode Enabled */
163  pTwi->TWIHS_CR = TWIHS_CR_SVEN;
164 
165  assert((pTwi->TWIHS_CR & TWIHS_CR_SVDIS) != TWIHS_CR_SVDIS);
166 }

188 void TWI_StartRead(
189  Twihs *pTwi,
190  uint8_t address,
191  uint32_t iaddress,
192  uint8_t isize)
193 {
194  assert(pTwi != NULL);
195  assert((address & 0x80) == 0);
196  assert((iaddress & 0xFF000000) == 0);
197  assert(isize < 4);
198 
199  /* Set slave address and number of internal address bytes. */
200  pTwi->TWIHS_MMR = 0;
201  pTwi->TWIHS_MMR = (isize << 8) | TWIHS_MMR_MREAD | (address << 16);
202 
203  /* Set internal address bytes */
204  pTwi->TWIHS_IADR = 0;
205  pTwi->TWIHS_IADR = iaddress;
206 
207  /* Send START condition */
208  if (0 == twi_send_stop)
209  pTwi->TWIHS_CR = TWIHS_CR_START;
210  else {
211  twi_send_stop = 0;
212  pTwi->TWIHS_CR = TWIHS_CR_START | TWIHS_CR_STOP;
213  }
214 }
