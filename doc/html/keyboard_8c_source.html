165 static void to_utf8(ushort c)
166 {
167  if (c < 0x80)
168  put_queue(c); /* 0******* */
169  else if (c < 0x800) {
170  put_queue(0xc0 | (c >> 6)); /* 110***** 10****** */
171  put_queue(0x80 | (c & 0x3f));
172  } else {

685 {

687 
688  shift_state = 0;

690  k_down[i] = 0;

693  if(atomic_load(key_down + i)) { /* skip this word if not a single bit on */

701  val = KVAL(K_SHIFT);
702  k_down[val]++;
