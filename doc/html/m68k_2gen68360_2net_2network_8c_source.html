112 /*
113  * SCC1 interrupt handler
114  */
115 static rtems_isr
116 m360Enet_interrupt_handler (rtems_vector_number v)
117 {
118  /*
119  * Frame received?
120  */
121  if ((m360.scc1.sccm & 0x8) && (m360.scc1.scce & 0x8)) {
122  m360.scc1.scce = 0x8;
123  m360.scc1.sccm &= ~0x8;
124  scc_softc[0].rxInterrupts++;
125  rtems_bsdnet_event_send (scc_softc[0].rxDaemonTid, INTERRUPT_EVENT);
126  }
127 
128  /*
129  * Buffer transmitted or transmitter error?
130  */
131  if ((m360.scc1.sccm & 0x12) && (m360.scc1.scce & 0x12)) {
132  m360.scc1.scce = 0x12;
133  m360.scc1.sccm &= ~0x12;
134  scc_softc[0].txInterrupts++;
135  rtems_bsdnet_event_send (scc_softc[0].txDaemonTid, INTERRUPT_EVENT);
136  }
137  m360.cisr = 1UL << 30; /* Clear SCC1 interrupt-in-service bit */
138 }

172  * Allocate mbuf pointers
173  */

339 /*
340  * Soak up buffer descriptors that have been sent
341  * Note that a buffer descriptor can't be retired as soon as it becomes
342  * ready. The MC68360 Errata (May 96) says that, "If an Ethernet frame is
343  * made up of multiple buffers, the user should not reuse the first buffer
344  * descriptor until the last buffer descriptor of the frame has had its
345  * ready bit cleared by the CPM".
346  */
347 static void
348 m360Enet_retire_tx_bd (struct scc_softc *sc)
349 {
350  uint16_t status;
351  int i;
352  int nRetired;
353  struct mbuf *m, *n;
354  int retries = 0;
355  int saveStatus = 0;
356 
357  i = sc->txBdTail;
358  nRetired = 0;
359  while ((sc->txBdActiveCount != 0)
360  && (((status = (sc->txBdBase + i)->status) & M360_BD_READY) == 0)) {
361  /*
362  * Check for errors which stop the transmitter.
363  */
364  if (status & (M360_BD_LATE_COLLISION |
365  M360_BD_RETRY_LIMIT |
366  M360_BD_UNDERRUN)) {
367  int j;
368 
369  if (status & M360_BD_LATE_COLLISION)
370  sc->txLateCollision++;
371  if (status & M360_BD_RETRY_LIMIT)
372  sc->txRetryLimit++;
373  if (status & M360_BD_UNDERRUN)
374  sc->txUnderrun++;
375 
376  /*
377  * Reenable buffer descriptors
378  */
379  j = sc->txBdTail;
380  for (;;) {
381  status = (sc->txBdBase + j)->status;
382  if (status & M360_BD_READY)
383  break;
384  (sc->txBdBase + j)->status = M360_BD_READY |
385  (status & (M360_BD_PAD |
386  M360_BD_WRAP |
387  M360_BD_INTERRUPT |
388  M360_BD_LAST |
389  M360_BD_TX_CRC));
390  if (status & M360_BD_LAST)
391  break;
392  if (++j == sc->txBdCount)
393  j = 0;
394  }
395 
396  /*
397  * Move transmitter back to the first
398  * buffer descriptor in the frame.
399  */
400  m360.scc1p._tbptr = m360.scc1p.tbase +
401  sc->txBdTail * sizeof (m360BufferDescriptor_t);
402 
403  /*
404  * Restart the transmitter
405  */
406  M360ExecuteRISC (M360_CR_OP_RESTART_TX | M360_CR_CHAN_SCC1);
407  continue;
408  }
409  saveStatus |= status;
410  retries += (status >> 2) & 0xF;
411  nRetired++;
412  if (status & M360_BD_LAST) {
413  /*
414  * A full frame has been transmitted.
415  * Free all the associated buffer descriptors.
416  */
417  if (saveStatus & M360_BD_DEFER)
418  sc->txDeferred++;
419  if (saveStatus & M360_BD_HEARTBEAT)
420  sc->txHeartbeat++;
421  if (saveStatus & M360_BD_CARRIER_LOST)
422  sc->txLostCarrier++;
423  saveStatus = 0;
424  sc->txRetry += retries;
425  retries = 0;
426  sc->txBdActiveCount -= nRetired;
427  while (nRetired) {
428  nRetired--;
429  m = sc->txMbuf[sc->txBdTail];
430  MFREE (m, n);
431  if (++sc->txBdTail == sc->txBdCount)
432  sc->txBdTail = 0;
433  }
434  }
435  if (++i == sc->txBdCount)
436  i = 0;
437  }
438 }

440 /*
441  * SCC reader task
442  */
443 static void
444 scc_rxDaemon (void *arg)
445 {
446  struct scc_softc *sc = (struct scc_softc *)arg;
447  struct ifnet *ifp = &sc->arpcom.ac_if;
448  struct mbuf *m;
449  uint16_t status;
450  volatile m360BufferDescriptor_t *rxBd;
451  int rxBdIndex;
452 
453  /*
454  * Allocate space for incoming packets and start reception
455  */
456  for (rxBdIndex = 0 ; ;) {
457  rxBd = sc->rxBdBase + rxBdIndex;
458  MGETHDR (m, M_WAIT, MT_DATA);
459  MCLGET (m, M_WAIT);
460  m->m_pkthdr.rcvif = ifp;
461  sc->rxMbuf[rxBdIndex] = m;
462  rxBd->buffer = mtod (m, void *);
463  if (++rxBdIndex == sc->rxBdCount) {
464  rxBd->status = M360_BD_EMPTY | M360_BD_INTERRUPT | M360_BD_WRAP;
465  break;
466  }
467  rxBd->status = M360_BD_EMPTY | M360_BD_INTERRUPT;
468  }
469 
470  /*
471  * Input packet handling loop
472  */
473  rxBdIndex = 0;
474  for (;;) {
475  rxBd = sc->rxBdBase + rxBdIndex;
476 
477  /*
478  * Wait for packet if there's not one ready
479  */
480  if ((status = rxBd->status) & M360_BD_EMPTY) {
481  /*
482  * Clear old events
483  */
484  m360.scc1.scce = 0x8;
485 
486  /*
487  * Wait for packet
488  * Note that the buffer descriptor is checked
489  * *before* the event wait -- this catches the
490  * possibility that a packet arrived between the
491  * `if' above, and the clearing of the event register.
492  */
493  while ((status = rxBd->status) & M360_BD_EMPTY) {
494  rtems_interrupt_level level;
495  rtems_event_set events;
496 
497  /*
498  * Unmask RXF (Full frame received) event
499  */
500  rtems_interrupt_disable (level);
501  m360.scc1.sccm |= 0x8;
502  rtems_interrupt_enable (level);
503 
504  rtems_bsdnet_event_receive (INTERRUPT_EVENT,
505  RTEMS_WAIT|RTEMS_EVENT_ANY,
506  RTEMS_NO_TIMEOUT,
507  &events);
508  }
509  }
510 
511  /*
512  * Check that packet is valid
513  */
514  if ((status & (M360_BD_LAST |
515  M360_BD_FIRST_IN_FRAME |
516  M360_BD_LONG |
517  M360_BD_NONALIGNED |
518  M360_BD_SHORT |
519  M360_BD_CRC_ERROR |
520  M360_BD_OVERRUN |
521  M360_BD_COLLISION)) ==
522  (M360_BD_LAST |
523  M360_BD_FIRST_IN_FRAME)) {
524  /*
525  * Pass the packet up the chain.
526  * FIXME: Packet filtering hook could be done here.
527  */
528  struct ether_header *eh;
529 
530  m = sc->rxMbuf[rxBdIndex];
531  m->m_len = m->m_pkthdr.len = rxBd->length -
532  sizeof(uint32_t) -
533  sizeof(struct ether_header);
534  eh = mtod (m, struct ether_header *);
535  m->m_data += sizeof(struct ether_header);
536  ether_input (ifp, eh, m);
537 
538  /*
539  * Allocate a new mbuf
540  */
541  MGETHDR (m, M_WAIT, MT_DATA);
542  MCLGET (m, M_WAIT);
543  m->m_pkthdr.rcvif = ifp;
544  sc->rxMbuf[rxBdIndex] = m;
545  rxBd->buffer = mtod (m, void *);
546  }
547  else {
548  /*
549  * Something went wrong with the reception
550  */
551  if (!(status & M360_BD_LAST))
552  sc->rxNotLast++;
553  if (!(status & M360_BD_FIRST_IN_FRAME))
554  sc->rxNotFirst++;
555  if (status & M360_BD_LONG)
556  sc->rxGiant++;
557  if (status & M360_BD_NONALIGNED)
558  sc->rxNonOctet++;
559  if (status & M360_BD_SHORT)
560  sc->rxRunt++;
561  if (status & M360_BD_CRC_ERROR)
562  sc->rxBadCRC++;
563  if (status & M360_BD_OVERRUN)
564  sc->rxOverrun++;
565  if (status & M360_BD_COLLISION)
566  sc->rxCollision++;
567  }
568 
569  /*
570  * Reenable the buffer descriptor
571  */
572  rxBd->status = (status & (M360_BD_WRAP | M360_BD_INTERRUPT)) | M360_BD_EMPTY;
573 
574  /*
575  * Move to next buffer descriptor
576  */
577  if (++rxBdIndex == sc->rxBdCount)
578  rxBdIndex = 0;
579  }
580 }

582 static void
583 sendpacket (struct ifnet *ifp, struct mbuf *m)
584 {
585  struct scc_softc *sc = ifp->if_softc;
586  volatile m360BufferDescriptor_t *firstTxBd, *txBd;
587  struct mbuf *l = NULL;
588  uint16_t status;
589  int nAdded;
590 
591  /*
592  * Free up buffer descriptors
593  */
594  m360Enet_retire_tx_bd (sc);
595 
596  /*
597  * Set up the transmit buffer descriptors.
598  * No need to pad out short packets since the
599  * hardware takes care of that automatically.
600  * No need to copy the packet to a contiguous buffer
601  * since the hardware is capable of scatter/gather DMA.
602  */
603  status = 0;
604  nAdded = 0;
605  txBd = firstTxBd = sc->txBdBase + sc->txBdHead;
606  while (m) {
607  /*
608  * There are more mbufs in the packet than there
609  * are transmit buffer descriptors.
610  * Coalesce into a single buffer.
611  */
612  if (nAdded == sc->txBdCount) {
613  struct mbuf *nm;
614  int j;
615  char *dest;
616 
617  /*
618  * Get the pointer to the first mbuf of the packet
619  */
620  if (sc->txBdTail != sc->txBdHead)
621  rtems_panic ("sendpacket coalesce");
622  m = sc->txMbuf[sc->txBdTail];
623 
624  /*
625  * Rescind the buffer descriptor READY bits
626  */
627  for (j = 0 ; j < sc->txBdCount ; j++)
628  (sc->txBdBase + j)->status = 0;
629 
630  /*
631  * Allocate an mbuf cluster
632  * Toss the packet if allocation fails
633  */
634  MGETHDR (nm, M_DONTWAIT, MT_DATA);
635  if (nm == NULL) {
636  sc->txCoalesceFailed++;
637  m_freem (m);
638  return;
639  }
640  MCLGET (nm, M_DONTWAIT);
641  if (nm->m_ext.ext_buf == NULL) {
642  sc->txCoalesceFailed++;
643  m_freem (m);
644  m_free (nm);
645  return;
646  }
647  nm->m_pkthdr = m->m_pkthdr;
648  nm->m_len = nm->m_pkthdr.len;
649 
650  /*
651  * Copy data from packet chain to mbuf cluster
652  */
653  sc->txCoalesced++;
654  dest = nm->m_ext.ext_buf;
655  while (m) {
656  struct mbuf *n;
657 
658  if (m->m_len) {
659  memcpy (dest, mtod(m, caddr_t), m->m_len);
660  dest += m->m_len;
661  }
662  MFREE (m, n);
663  m = n;
664  }
665 
666  /*
667  * Redo the send with the new mbuf cluster
668  */
669  m = nm;
670  nAdded = 0;
671  status = 0;
672  continue;
673  }
674 
675  /*
676  * Wait for buffer descriptor to become available.
677  */
678  if ((sc->txBdActiveCount + nAdded) == sc->txBdCount) {
679  /*
680  * Clear old events
681  */
682  m360.scc1.scce = 0x12;
683 
684  /*
685  * Wait for buffer descriptor to become available.
686  * Note that the buffer descriptors are checked
687  * *before* entering the wait loop -- this catches
688  * the possibility that a buffer descriptor became
689  * available between the `if' above, and the clearing
690  * of the event register.
691  * This is to catch the case where the transmitter
692  * stops in the middle of a frame -- and only the
693  * last buffer descriptor in a frame can generate
694  * an interrupt.
695  */
696  m360Enet_retire_tx_bd (sc);
697  while ((sc->txBdActiveCount + nAdded) == sc->txBdCount) {
698  rtems_interrupt_level level;
699  rtems_event_set events;
700 
701  /*
702  * Unmask TXB (buffer transmitted) and
703  * TXE (transmitter error) events.
704  */
705  rtems_interrupt_disable (level);
706  m360.scc1.sccm |= 0x12;
707  rtems_interrupt_enable (level);
708  rtems_bsdnet_event_receive (INTERRUPT_EVENT,
709  RTEMS_WAIT|RTEMS_EVENT_ANY,
710  RTEMS_NO_TIMEOUT,
711  &events);
712  m360Enet_retire_tx_bd (sc);
713  }
714  }
715 
716  /*
717  * The IP fragmentation routine in ip_output
718  * can produce packet fragments with zero length.
719  */
720  if (m->m_len) {
721  /*
722  * Fill in the buffer descriptor.
723  * Don't set the READY flag in the first buffer
724  * descriptor till the whole packet has been readied.
725  */
726  txBd = sc->txBdBase + sc->txBdHead;
727  txBd->buffer = mtod (m, void *);
728  txBd->length = m->m_len;
729  sc->txMbuf[sc->txBdHead] = m;
730  status = nAdded ? M360_BD_READY : 0;
731  if (++sc->txBdHead == sc->txBdCount) {
732  status |= M360_BD_WRAP;
733  sc->txBdHead = 0;
734  }
735  txBd->status = status;
736  l = m;
737  m = m->m_next;
738  nAdded++;
739  }
740  else {
741  /*
742  * Just toss empty mbufs
743  */
744  struct mbuf *n;
745  MFREE (m, n);
746  m = n;
747  if (l != NULL)
748  l->m_next = m;
749  }
750  }
751  if (nAdded) {
752  /*
753  * Send the packet
754  */
755  txBd->status = status | M360_BD_PAD | M360_BD_LAST | M360_BD_TX_CRC | M360_BD_INTERRUPT;
756  firstTxBd->status |= M360_BD_READY;
757  sc->txBdActiveCount += nAdded;
758  }

944 /*
945  * Attach an SCC driver to the system
946  */
947 int
948 rtems_scc1_driver_attach (struct rtems_bsdnet_ifconfig *config, int attaching)
949 {
950  struct scc_softc *sc;
951  struct ifnet *ifp;
952  int mtu;
953  int unitNumber;
954  char *unitName;
955 
956  /*
957  * Make sure we're really being attached
958  */
959  if (!attaching) {
960  printf ("SCC1 driver can not be detached.\n");
961  return 0;
962  }
963 
964  /*
965  * Parse driver name
966  */
967  if ((unitNumber = rtems_bsdnet_parse_driver_name (config, &unitName)) < 0)
968  return 0;
969 
970  /*
971  * Is driver free?
972  */
973  if ((unitNumber <= 0) || (unitNumber > NSCCDRIVER)) {
974  printf ("Bad SCC unit number.\n");
975  return 0;
976  }
977  sc = &scc_softc[unitNumber - 1];
978  ifp = &sc->arpcom.ac_if;
979  if (ifp->if_softc != NULL) {
980  printf ("Driver already in use.\n");
981  return 0;
982  }
983 
984  /*
985  * Process options
986  */
987  if (config->hardware_address) {
988  memcpy (sc->arpcom.ac_enaddr, config->hardware_address, ETHER_ADDR_LEN);
989  }
990  else {
991  /*
992  * The first 4 bytes of the bootstrap prom
993  * contain the value loaded into the stack
994  * pointer as part of the CPU32's hardware
995  * reset exception handler. The following
996  * 4 bytes contain the value loaded into the
997  * program counter. The boards' Ethernet
998  * address is stored in the six bytes
999  * immediately preceding this initial
1000  * program counter value.
1001  *
1002  * See start360/start360.s.
1003  */
1004  const unsigned long *ExceptionVectors;
1005  const unsigned char *entryPoint;
1006 
1007  /*
1008  * Sanity check -- assume entry point must be
1009  * within 1 MByte of beginning of boot ROM.
1010  */
1011  ExceptionVectors = (const unsigned long *)&_RomBase;
1012  entryPoint = (const unsigned char *)ExceptionVectors[1];
1013  if (((unsigned long)entryPoint - (unsigned long)ExceptionVectors)
1014  >= (1 * 1024 * 1024)) {
1015  printf ("Warning -- Ethernet address can not be found in bootstrap PROM.\n");
1016  sc->arpcom.ac_enaddr[0] = 0x08;
1017  sc->arpcom.ac_enaddr[1] = 0xF3;
1018  sc->arpcom.ac_enaddr[2] = 0x3E;
1019  sc->arpcom.ac_enaddr[3] = 0xC2;
1020  sc->arpcom.ac_enaddr[4] = 0x7E;
1021  sc->arpcom.ac_enaddr[5] = 0x38;
1022  }
1023  else {
1024  memcpy (sc->arpcom.ac_enaddr, entryPoint - ETHER_ADDR_LEN, ETHER_ADDR_LEN);
1025  }
1026  }
1027  if (config->mtu)
1028  mtu = config->mtu;
1029  else
1030  mtu = ETHERMTU;
1031  if (config->rbuf_count)
1032  sc->rxBdCount = config->rbuf_count;
1033  else
1034  sc->rxBdCount = RX_BUF_COUNT;
1035  if (config->xbuf_count)
1036  sc->txBdCount = config->xbuf_count;
1037  else
1038  sc->txBdCount = TX_BUF_COUNT * TX_BD_PER_BUF;
1039  sc->acceptBroadcast = !config->ignore_broadcast;
1040 
1041  /*
1042  * Set up network interface values
1043  */
1044  ifp->if_softc = sc;
1045  ifp->if_unit = unitNumber;
1046  ifp->if_name = unitName;
1047  ifp->if_mtu = mtu;
1048  ifp->if_init = scc_init;
1049  ifp->if_ioctl = scc_ioctl;
1050  ifp->if_start = scc_start;
1051  ifp->if_output = ether_output;
1052  ifp->if_flags = IFF_BROADCAST | IFF_SIMPLEX;
1053  if (ifp->if_snd.ifq_maxlen == 0)
1054  ifp->if_snd.ifq_maxlen = ifqmaxlen;
1055 
1056  /*
1057  * Attach the interface
1058  */
1059  if_attach (ifp);
1060  ether_ifattach (ifp);
1061  return 1;
