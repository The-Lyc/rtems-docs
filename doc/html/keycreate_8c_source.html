94 static void _POSIX_Keys_Run_destructors( Thread_Control *the_thread )
95 {
96  while ( true ) {
97  ISR_lock_Context lock_context;
98  RBTree_Node *node;
99 
100  _Objects_Allocator_lock();
101  _POSIX_Keys_Key_value_acquire( the_thread, &lock_context );
102 
103  node = _RBTree_Root( &the_thread->Keys.Key_value_pairs );
104  if ( node != NULL ) {
105  POSIX_Keys_Key_value_pair *key_value_pair;
106  pthread_key_t key;
107  void *value;
108  POSIX_Keys_Control *the_key;
109  void ( *destructor )( void * );
110 
111  key_value_pair = POSIX_KEYS_RBTREE_NODE_TO_KEY_VALUE_PAIR( node );
112  key = key_value_pair->key;
113  value = key_value_pair->value;
114  _RBTree_Extract(
115  &the_thread->Keys.Key_value_pairs,
116  &key_value_pair->Lookup_node
117  );
118 
119  _POSIX_Keys_Key_value_release( the_thread, &lock_context );
120  _POSIX_Keys_Key_value_free( key_value_pair );
121 
122  the_key = _POSIX_Keys_Get( key );
123  _Assert( the_key != NULL );
124  destructor = the_key->destructor;
125 
126  _Objects_Allocator_unlock();
127 
128  if ( destructor != NULL && value != NULL ) {
129  ( *destructor )( value );
130  }
131  } else {
132  _POSIX_Keys_Key_value_release( the_thread, &lock_context );
133  _Objects_Allocator_unlock();
134  break;
135  }
136  }
