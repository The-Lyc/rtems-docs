76 int pthread_mutex_init(
77  pthread_mutex_t *mutex,
78  const pthread_mutexattr_t *attr
79 )
80 {
81  POSIX_Mutex_Control *the_mutex;
82  const pthread_mutexattr_t *the_attr;
83  POSIX_Mutex_Protocol protocol;
84  unsigned long flags;
85  Priority_Control priority;
86  const Scheduler_Control *scheduler;
87 
88  if ( attr ) the_attr = attr;
89  else the_attr = &_POSIX_Mutex_Default_attributes;
90 
91  /* Check for NULL mutex */
92  if ( !mutex )
93  return EINVAL;
94 
95  /*
96  * The POSIX specification says:
97  *
98  * "Attempting to initialize an already initialized mutex results
99  * in undefined behavior."
100  *
101  * Trying to keep the caller from doing the create when *mutex
102  * is actually a valid ID causes grief. All it takes is the wrong
103  * value in an uninitialized variable to make this fail.
104  *
105  * Thus, we do not look at *mutex.
106  */
107 
108  if ( !the_attr->is_initialized )
109  return EINVAL;
110 
111  if ( !_POSIX_Is_valid_pshared( the_attr->process_shared ) ) {
112  return EINVAL;
113  }
114 
115  /*
116  * Determine the discipline of the mutex
117  */
118  switch ( the_attr->protocol ) {
119  case PTHREAD_PRIO_NONE:
120  protocol = POSIX_MUTEX_NO_PROTOCOL;
121  break;
122  case PTHREAD_PRIO_INHERIT:
123  protocol = POSIX_MUTEX_PRIORITY_INHERIT;
124  break;
125  case PTHREAD_PRIO_PROTECT:
126  protocol = POSIX_MUTEX_PRIORITY_CEILING;
127  break;
128  default:
129  return EINVAL;
130  }
131 
132 #if defined(_UNIX98_THREAD_MUTEX_ATTRIBUTES)
133  /*
134  * Validate the mutex type and set appropriate SuperCore mutex
135  * attributes.
136  */
137  switch ( the_attr->type ) {
138  case PTHREAD_MUTEX_NORMAL:
139  case PTHREAD_MUTEX_RECURSIVE:
140  case PTHREAD_MUTEX_ERRORCHECK:
141  case PTHREAD_MUTEX_DEFAULT:
142  break;
143 
144  default:
145  return EINVAL;
146  }
147 #endif
148 
149  the_mutex = _POSIX_Mutex_Get( mutex );
150 
151  flags = (uintptr_t) the_mutex ^ POSIX_MUTEX_MAGIC;
152  flags &= ~POSIX_MUTEX_FLAGS_MASK;
153  flags |= protocol;
154 
155  if ( the_attr->type == PTHREAD_MUTEX_RECURSIVE ) {
156  flags |= POSIX_MUTEX_RECURSIVE;
157  }
158 
159  the_mutex->flags = flags;
160 
161  if ( protocol == POSIX_MUTEX_PRIORITY_CEILING ) {
162  int prio_ceiling;
163  bool valid;
164 
165  scheduler = _Thread_Scheduler_get_home( _Thread_Get_executing() );
166  prio_ceiling = the_attr->prio_ceiling;
167 
168  if ( prio_ceiling == INT_MAX ) {
169  prio_ceiling = _POSIX_Priority_Get_maximum( scheduler );
170  }
171 
172  priority = _POSIX_Priority_To_core( scheduler, prio_ceiling, &valid );
173  if ( !valid ) {
174  return EINVAL;
175  }
176  } else {
177  priority = 0;
178  scheduler = NULL;
179  }
180 
181  _Thread_queue_Queue_initialize(
182  &the_mutex->Recursive.Mutex.Queue.Queue,
183  NULL
184  );
185  the_mutex->Recursive.nest_level = 0;
186  _Priority_Node_initialize( &the_mutex->Priority_ceiling, priority );
187  the_mutex->scheduler = scheduler;
188  return 0;
189 }
