120  __asm__ volatile (
121  ARM_SWITCH_TO_ARM

130 uint32_t _CPU_ISR_Get_level( void )
131 {
132  ARM_SWITCH_REGISTERS;
133  uint32_t level;
134 
135  __asm__ volatile (
136  ARM_SWITCH_TO_ARM
137  "mrs %[level], cpsr\n"
138  "and %[level], #" RTEMS_XSTRING( ARM_PSR_I ) "\n"
139  ARM_SWITCH_BACK
140  : [level] "=&r" (level) ARM_SWITCH_ADDITIONAL_OUTPUT
141  );
142 
143  return ( level & ARM_PSR_I ) != 0;
144 }

146 void _CPU_ISR_install_vector(
147  uint32_t vector,
148  CPU_ISR_handler new_handler,
149  CPU_ISR_handler *old_handler
150 )
151 {
152  /* Redirection table starts at the end of the vector table */
153  CPU_ISR_handler *table = (CPU_ISR_handler *) (MAX_EXCEPTIONS * 4);
154 
155  CPU_ISR_handler current_handler = table [vector];
156 
157  /* The current handler is now the old one */
158  if (old_handler != NULL) {
159  *old_handler = current_handler;
160  }
161 
162  /* Write only if necessary to avoid writes to a maybe read-only memory */
163  if (current_handler != new_handler) {
164  table [vector] = new_handler;
165  }
