113 extern void PIO_CaptureHandler(void)
114 {
115  volatile uint32_t pio_captureSr;
116  uint32_t k;
117 
118  /* Read the status register*/
119  pio_captureSr = PIOA->PIO_PCISR;
120  k = pio_captureSr;
121  pio_captureSr = k & PIOA->PIO_PCIMR;
122 
123  if (pio_captureSr & PIO_PCISR_DRDY) {
124  /* Parallel Capture Mode Data Ready */
125  if (_PioCaptureCopy->CbkDataReady != NULL)
126  _PioCaptureCopy->CbkDataReady(_PioCaptureCopy);
127  else
128  TRACE_DEBUG("IT PIO Capture Data Ready received (no callback)\n\r");
129  }
130 
131  if (pio_captureSr & PIO_PCISR_OVRE) {
132  /* Parallel Capture Mode Overrun Error */
133  if (_PioCaptureCopy->CbkOverrun != NULL)
134  _PioCaptureCopy->CbkOverrun(_PioCaptureCopy);
135  else
136  TRACE_DEBUG("IT PIO Capture Overrun Error received (no callback)\n\r");
137  }
138 
139  if (pio_captureSr & PIO_PCISR_RXBUFF) {
140  /* Reception Buffer Full */
141  if (_PioCaptureCopy->CbkBuffFull != NULL)
142  _PioCaptureCopy->CbkBuffFull(_PioCaptureCopy);
143  else {
144  TRACE_DEBUG("IT PIO Capture Reception Buffer Full received \
145  (no callback)\n\r");
146  }
147  }
148 
149  if (pio_captureSr & PIO_PCISR_ENDRX) {
150  /* End of Reception Transfer */
151  if (_PioCaptureCopy->CbkEndReception != NULL)
152  _PioCaptureCopy->CbkEndReception(_PioCaptureCopy);
153  else {
154  TRACE_DEBUG("IT PIO Capture End of Reception Transfer \
155  received (no callback)\n\r");
156  }
157  }
158 }
