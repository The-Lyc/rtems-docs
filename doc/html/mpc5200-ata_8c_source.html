25 bool ata_execute_io_command(uint8_t command, uint32_t lba, uint32_t sector_count_32)
26 {
27  assert(sector_count_32 >= 1);
28  assert(sector_count_32 <= 256);
29 
30  assert(ata_is_drive_ready_for_selection());
31 
32  ATA->write.head = (uint8_t) (0xe0 | ((lba >> 24) & 0x0f));
33 
34  ata_wait_400_nano_seconds();
35  ata_wait_for_drive_ready();
36 
37  /*
38  * It seems that the write to the ATA device registers is sometimes not
39  * successful. The write is retried until the read back values are all right
40  * or a timeout is reached.
41  */
42  bool ok = false;
43  uint8_t sector = (uint8_t) lba;
44  uint8_t cylinder_low = (uint8_t) (lba >> 8);
45  uint8_t cylinder_high = (uint8_t) (lba >> 16);
46  uint8_t sector_count = (uint8_t) sector_count_32;
47  int i;
48  for (i = 0; !ok && i < 100; ++i) {
49  ATA->write.sector = sector;
50  ATA->write.cylinder_low = cylinder_low;
51  ATA->write.cylinder_high = cylinder_high;
52  ATA->write.sector_count = sector_count;
53 
54  uint8_t actual_sector = ATA->read.sector;
55  uint8_t actual_cylinder_low = ATA->read.cylinder_low;
56  uint8_t actual_cylinder_high = ATA->read.cylinder_high;
57  uint8_t actual_sector_count = ATA->read.sector_count;
58 
59  ok = actual_cylinder_high == cylinder_high
60  && actual_cylinder_low == cylinder_low
61  && actual_sector == sector
62  && actual_sector_count == sector_count;
63  }
64 
65  if (ok) {
66  ATA->write.command = command;
67  }
68 
69  return ok;
70 }

100 static bool probe(void)

104 #ifdef MPC5200_BOARD_BRS5L

142 {
143  self->card_present = probe();

145  if (ata_driver_is_card_present(self)) {
146  create_lock(self);
147  ata_reset_device();

149  uint16_t sector_buffer[256];

151  rtems_status_code sc = ata_identify_device(0, 0, sector_buffer, &ata_device);

180 static bool transfer_pio_polled(ata_driver *self, bool read, rtems_blkdev_sg_buffer *sg, size_t sg_count)
181 {
182  bool ok = true;
183 
184  ata_sg_context sg_context;
185  ata_sg_create(&sg_context, sg, sg_count);
186  rtems_blkdev_bnum start_sector = ata_sg_get_start_sector(&sg_context);
187  rtems_blkdev_bnum sector_count = ata_sg_get_sector_count(&sg_context);
188  rtems_blkdev_bnum relative_sector = 0;
189 
190  uint8_t command = ata_read_or_write_sectors_command(read);
191 
192  while (ok && relative_sector < sector_count) {
193  rtems_blkdev_bnum remaining_sectors = sector_count - relative_sector;
194  rtems_blkdev_bnum transfer_count = ata_max_transfer_count(remaining_sectors);
195  rtems_blkdev_bnum transfer_end = relative_sector + transfer_count;
196 
197  ok = ata_execute_io_command(command, start_sector + relative_sector, transfer_count);
198 
199  rtems_blkdev_bnum transfer;
200  for (transfer = relative_sector; ok && transfer < transfer_end; ++transfer) {
201  uint16_t *current = ata_sg_get_sector_data_begin(&sg_context, transfer);
202  uint16_t *end = ata_sg_get_sector_data_end(&sg_context, current);
203 
204  ok = ata_wait_for_data_request();
205 
206  if (ok) {
207  if (read) {
208  while (current != end) {
209  *current = ATA->read.data;
210  ++current;
211  }
212  } else {
213  while (current != end) {
214  ATA->write.data = *current;
215  ++current;
216  }
217  }
218  }
219  }
220 
221  if (ok) {
222  ata_wait_for_not_busy();
223  ok = ata_check_status();
224  }
225 
226  relative_sector += ATA_PER_TRANSFER_SECTOR_COUNT_MAX;
227  }
228 
229  return ok;
230 }
