86 static uint32_t atsam_calculate_scbr(uint32_t speed_hz)
87 {
88  uint32_t scbr;
89 
90  scbr = BOARD_MCK / speed_hz;
91  if (scbr > 0x0FF) {
92  /* Best estimation we can offer with the hardware. */
93  scbr = 0x0FF;
94  }
95  if (scbr == 0) {
96  /* SCBR = 0 isn't allowed. */
97  scbr = 1;
98  }
99 
100  return scbr;
101 }

103 static void atsam_set_phase_and_polarity(uint32_t mode, uint32_t *csr)
104 {
105  uint32_t mode_mask = mode & SPI_MODE_3;
106 
107  switch(mode_mask) {
108  case SPI_MODE_0:
109  *csr |= SPI_CSR_NCPHA;
110  break;
111  case SPI_MODE_1:
112  break;
113  case SPI_MODE_2:
114  *csr |= SPI_CSR_NCPHA;
115  *csr |= SPI_CSR_CPOL;
116  break;
117  case SPI_MODE_3:
118  *csr |= SPI_CSR_CPOL;
119  break;
120  }
121  *csr |= SPI_CSR_CSAAT;
122 }

124 static void atsam_configure_spi(atsam_spi_bus *bus)
125 {
126  uint32_t scbr;
127  uint32_t csr = 0;
128  uint32_t mr;
129  uint32_t cs = bus->base.cs;
130 
131  scbr = atsam_calculate_scbr(bus->base.speed_hz);
132 
133  mr = bus->spi_mr;
134 
135  if (bus->chip_select_decode) {
136  mr |= SPI_MR_PCS(bus->base.cs);
137  mr |= SPI_MR_PCSDEC;
138  cs /= 4;
139  } else {
140  mr |= SPI_PCS(bus->base.cs);
141  }
142 
143  bus->spi_regs->SPI_MR = mr;
144 
145  csr = bus->spi_csr[cs]
146  | SPI_CSR_SCBR(scbr)
147  | SPI_CSR_BITS(bus->base.bits_per_word - 8);
148 
149  atsam_set_phase_and_polarity(bus->base.mode, &csr);
150 
151  SPI_ConfigureNPCS(bus->spi_regs, cs, csr);
152 }

184 static void atsam_spi_setup_rx_dma_desc(
185  atsam_spi_bus *bus,
186  atsam_spi_dma *dma,
187  const uint8_t *buf,
188  size_t n
189 )
190 {
191  volatile LinkedListDescriporView0 *desc;
192  uintptr_t m;
193  uintptr_t b;
194  uintptr_t a;
195  uintptr_t ae;
196  uintptr_t e;
197 
198  desc = &dma->rx_desc[0];
199  m = CPU_CACHE_LINE_BYTES - 1;
200  b = (uintptr_t) buf;
201  e = b + n;
202  a = (b + m) & ~m;
203  ae = e & ~m;
204 
205  if (n <= m) {
206  bus->rx_bounce_head_len = n;
207  bus->rx_bounce_tail_len = 0;
208 
209  desc[0].mbr_ta = (uint32_t) dma->rx_bounce_head_buf;
210  desc[0].mbr_ubc = n;
211  } else {
212  bus->rx_bounce_head_len = a - b;
213  bus->rx_bounce_tail_len = e & m;
214 
215  if ((b & m) == 0) {
216  if ((n & m) == 0) {
217  desc[0].mbr_ta = a;
218  desc[0].mbr_ubc = n;
219  } else {
220  desc[0].mbr_ta = a;
221  desc[0].mbr_ubc = (ae - a) | XDMA_UBC_NDEN_UPDATED
222  | XDMA_UBC_NVIEW_NDV0
223  | XDMA_UBC_NDE_FETCH_EN;
224  desc[1].mbr_ta = (uint32_t) dma->rx_bounce_tail_buf;
225  desc[1].mbr_ubc = n & m;
226  }
227  } else {
228  if ((e & m) == 0) {
229  desc[0].mbr_ta = (uint32_t) dma->rx_bounce_head_buf;
230  desc[0].mbr_ubc = (a - b) | XDMA_UBC_NDEN_UPDATED
231  | XDMA_UBC_NVIEW_NDV0
232  | XDMA_UBC_NDE_FETCH_EN;
233  desc[1].mbr_ta = a;
234  desc[1].mbr_ubc = ae - a;
235  } else if ((ae - a) == 0) {
236  bus->rx_bounce_head_len = n;
237  bus->rx_bounce_tail_len = 0;
238 
239  desc[0].mbr_ta = (uint32_t) dma->rx_bounce_head_buf;
240  desc[0].mbr_ubc = n;
241  } else {
242  desc[0].mbr_ta = (uint32_t) dma->rx_bounce_head_buf;
243  desc[0].mbr_ubc = (a - b) | XDMA_UBC_NDEN_UPDATED
244  | XDMA_UBC_NVIEW_NDV0
245  | XDMA_UBC_NDE_FETCH_EN;
246  desc[1].mbr_ta = a;
247  desc[1].mbr_ubc = (ae - a) | XDMA_UBC_NDEN_UPDATED
248  | XDMA_UBC_NVIEW_NDV0
249  | XDMA_UBC_NDE_FETCH_EN;
250  desc[2].mbr_ta = (uint32_t) dma->rx_bounce_tail_buf;
251  desc[2].mbr_ubc = e - ae;
252  }
253  }
254 
255  rtems_cache_invalidate_multiple_data_lines((void *) a, ae - a);
256  }
257 }

321 static int atsam_check_configure_spi(atsam_spi_bus *bus, const spi_ioc_transfer *msg)
322 {
323  if (
324  msg->mode != bus->base.mode
325  || msg->speed_hz != bus->base.speed_hz
326  || msg->bits_per_word != bus->base.bits_per_word
327  || msg->cs != bus->base.cs
328  ) {
329  if (
330  msg->bits_per_word != 8
331  || msg->mode > 3
332  || msg->speed_hz > bus->base.max_speed_hz
333  ) {
334  return -EINVAL;
335  }
336 
337  bus->base.mode = msg->mode;
338  bus->base.speed_hz = msg->speed_hz;
339  bus->base.bits_per_word = msg->bits_per_word;
340  bus->base.cs = msg->cs;
341  atsam_configure_spi(bus);
342  }
343 
344  return 0;
345 }

347 static void atsam_spi_setup_transfer(atsam_spi_bus *bus)
348 {
349  uint32_t msg_todo = bus->msg_todo;
350 
351  if (msg_todo > 0) {
352  const spi_ioc_transfer *msg;
353  int error;
354 
355  msg = bus->msg_current;
356  error = atsam_check_configure_spi(bus, msg);
357  if (error == 0) {
358  atsam_spi_start_dma_transfer(bus, msg);
359  } else {
360  bus->error = error;
361  atsam_spi_wakeup_task(bus);
362  }
363  } else {
364  atsam_spi_wakeup_task(bus);
365  }
366 }

368 static void atsam_spi_dma_callback(uint32_t ch, void *arg, uint32_t status)
369 {
370  atsam_spi_bus *bus;
371  uint32_t dma_errors;
372 
373  bus = arg;
374  dma_errors = XDMAC_CIE_DIE | XDMAC_CIE_FIE | XDMAC_CIE_RBIE | XDMAC_CIE_WBIE
375  | XDMAC_CIE_ROIE;
376 
377  if ((status & dma_errors) != 0) {
378  bus->error = -EIO;
379  }
380 
381  --bus->transfer_in_progress;
382 
383  if (bus->transfer_in_progress == 0) {
384  const spi_ioc_transfer *msg = bus->msg_current;
385 
386  if (msg->delay_usecs != bus->base.delay_usecs) {
387  uint32_t mr;
388  uint32_t mr_dlybcs;
389 
390  bus->base.delay_usecs = msg->delay_usecs;
391  mr_dlybcs = SPI_MR_DLYBCS(atsam_calculate_dlybcs(bus));
392  bus->spi_mr = mr_dlybcs | SPI_MR_MSTR | SPI_MR_MODFDIS;
393 
394  mr = bus->spi_regs->SPI_MR;
395  mr &= ~SPI_MR_DLYBCS_Msk;
396  mr |= mr_dlybcs;
397  bus->spi_regs->SPI_MR = mr;
398  }
399 
400  if (msg->cs_change) {
401  bus->spi_regs->SPI_CR = SPI_CR_LASTXFER;
402  }
403 
404  atsam_spi_copy_rx_bounce_bufs(bus, msg);
405 
406  bus->msg_current = msg + 1;
407  --bus->msg_todo;
408 
409  if (bus->error == 0) {
410  atsam_spi_setup_transfer(bus);
411  } else {
412  atsam_spi_wakeup_task(bus);
413  }
414  }
415 }

597 int spi_bus_register_atsam(
598  const char *bus_path,
599  const atsam_spi_config *config
600 )
601 {
602  atsam_spi_bus *bus;
603  size_t i;
604 
605  bus = (atsam_spi_bus *) spi_bus_alloc_and_init(sizeof(*bus));
606  if (bus == NULL) {
607  return -1;
608  }
609 
610  bus->base.transfer = atsam_spi_transfer;
611  bus->base.destroy = atsam_spi_destroy;
612  bus->base.setup = atsam_spi_setup;
613  bus->base.max_speed_hz = MAX_SPI_FREQUENCY;
614  bus->base.bits_per_word = 8;
615  bus->base.speed_hz = bus->base.max_speed_hz;
616  bus->base.cs = 1;
617  bus->spi_id = config->spi_peripheral_id;
618  bus->spi_regs = config->spi_regs;
619  bus->chip_select_decode = config->chip_select_decode;
620  bus->peripheral_clk_per_us = BOARD_MCK / 1000000;
621  bus->spi_mr = SPI_MR_MSTR | SPI_MR_MODFDIS;
622 
623  for (i = 0; i < RTEMS_ARRAY_SIZE(bus->spi_csr); ++i) {
624  if (config->dlybs_in_ns[i] != 0) {
625  bus->spi_csr[i] |= SPI_DLYBS(config->dlybs_in_ns[i], BOARD_MCK);
