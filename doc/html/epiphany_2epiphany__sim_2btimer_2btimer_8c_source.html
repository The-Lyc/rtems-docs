66 /*
67  * The following controls the behavior of benchmark_timer_read().
68  *
69  * AVG_OVEREHAD is the overhead for starting and stopping the timer. It
70  * is usually deducted from the number returned.
71  *
72  * LEAST_VALID is the lowest number this routine should trust. Numbers
73  * below this are "noise" and zero is returned.
74  */
75 
76 #define AVG_OVERHEAD 0 /* It typically takes X.X microseconds */
77  /* (Y countdowns) to start/stop the timer. */
78  /* This value is in microseconds. */
79 #define LEAST_VALID 1 /* Don't trust a clicks value lower than this */
80 
81 benchmark_timer_t benchmark_timer_read( void )
82 {
83  uint32_t timer_val = 0;
84  uint32_t total;
85 
86  __asm__ __volatile__ ("movfs %[timer_val], ctimer1 \t \n"
87  :[timer_val] "=r" (timer_val):);
88 
89  total = (0xFFFFFFFF - timer_val);
90 
91  if ( benchmark_timer_find_average_overhead == true )
92  return total;
93  else {
94  if ( total < LEAST_VALID )
95  return 0; /* below timer resolution */
96  /*
97  * Somehow convert total into microseconds
98  */
99  return (total - AVG_OVERHEAD);
100  }
101 }
