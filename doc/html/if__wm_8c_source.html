463 /*
464  * wm_reset:
465  *
466  * Reset the i82544 chip.
467  */
468 static void wm_reset(struct wm_softc *sc)
469 {
470  int i;
471 
472  /* Packet Buffer Allocation (PBA)
473  * Writing PBA sets the receive portion of the buffer.
474  * the remainder is used for the transmit buffer.
475  *
476  * 82544 has a Packet Buffer of 64K.
477  * Default allocation : PBA=40K for Rx, leaving 24K for Tx.
478  * Default for jumbo: PBA=48K for Rx, leaving 16K for Tx.
479  */
480  sc->sc_pba = sc->arpcom.ac_if.if_mtu > 8192 ? PBA_40K : PBA_48K;
481  CSR_WRITE(sc, WMREG_PBA, sc->sc_pba);
482 
483  /* device reset */
484  CSR_WRITE(sc, WMREG_CTRL, CTRL_RST);
485  rtems_bsp_delay(10000);
486 
487  for (i = 0; i < 1000; i++) {
488  if ((CSR_READ(sc, WMREG_CTRL) & CTRL_RST) == 0)
489  break;
490  rtems_bsp_delay(20);
491  }
492  if (CSR_READ(sc, WMREG_CTRL) & CTRL_RST)
493  printk("Intel 82544 1GHz reset failed to complete\n");
494 
495  sc->sc_ctrl_ext = CSR_READ(sc,WMREG_CTRL_EXT);
496  sc->sc_ctrl_ext |= CTRL_EXT_EE_RST;
497  CSR_WRITE(sc, WMREG_CTRL_EXT, sc->sc_ctrl_ext);
498  CSR_READ(sc, WMREG_STATUS);
499  /* Wait for EEPROM reload */
500  rtems_bsp_delay(2000);
501  sc->sc_ctrl= CSR_READ(sc, WMREG_CTRL);
502 }

629 /*
630  * i82544EI_sendpacket:
631  *
632  * Helper; handle transmit interrupts.
633  */
634 static void i82544EI_sendpacket(struct wm_softc *sc, struct mbuf *m)
635 {
636 
637 #ifdef WM_DEBUG_TX
638  printk("sendpacket(");
639 #endif
640 
641  if ( !(m->m_next)) { /* single buffer packet */
642  sc->txs_mbuf[sc->txq_next]= m;
643  /* Note: we currently only use 32-bit DMA addresses. */
644  sc->sc_txdescs[sc->txq_next].wtx_addr.wa_low = htole32(mtod(m, void*));
645  sc->sc_txdescs[sc->txq_next].wtx_cmdlen =htole32(TxDescCmd | m->m_len);
646  sc->txs_lastdesc= sc->txq_next;
647  sc->txq_next = WM_NEXTTX(sc->txq_next);
648  sc->txq_nactive++;
649  sc->txq_free--;
650  }
651  else /* multiple mbufs in this packet */
652  {
653  struct mbuf *mtp, *mdest;
654  volatile unsigned char *pt;
655  int len, y, loop=0;
656 
657 #ifdef WM_DEBUG_TX
658  printk("multi mbufs ");
659 #endif
660  mtp = m;
661  while ( mtp) {
662  MGETHDR(mdest, M_WAIT, MT_DATA);
663  MCLGET(mdest, M_WAIT);
664  pt = (volatile unsigned char *)mdest->m_data;
665  for ( len=0;mtp;mtp=mtp->m_next) {
666  loop++;
667  /* Each descriptor gets a 2k (MCLBYTES) buffer, although
668  * the length of each descriptor can be up to 16288 bytes.
669  * For packets which fill more than one buffer ( >2k), we
670  * chain them together.
671  * <Kate Feng> : This effective for packets > 2K
672  * The other way is effective for packets < 2K
673  */
674  if ( ((y=(len+mtp->m_len)) > sizeof(union mcluster))) {
675  printk(" >%d, use next descriptor\n", sizeof(union mcluster));
676  break;
677  }
678  memcpy((void *)pt,(char *)mtp->m_data, mtp->m_len);
679  pt += mtp->m_len;
680  len += mtp->m_len;
681  } /* end for loop */
682  mdest->m_len=len;
683  sc->txs_mbuf[sc->txq_next] = mdest;
684  /* Note: we currently only use 32-bit DMA addresses. */
685  sc->sc_txdescs[sc->txq_next].wtx_addr.wa_low = htole32(mtod(mdest, void*));
686  sc->sc_txdescs[sc->txq_next].wtx_cmdlen = htole32(TxDescCmd|mdest->m_len);
687  sc->txs_lastdesc = sc->txq_next;
688  sc->txq_next = WM_NEXTTX(sc->txq_next);
689  sc->txq_nactive ++;
690  if (sc->txq_free)
691  sc->txq_free--;
692  else
693  rtems_panic("i8254EI : no more free descriptors");
694  } /* end for while */
695  /* free old mbuf chain */
696  m_freem(m);
697  m=0;
698  } /* end multiple mbufs */
699 
700  DPRINTF(WM_DEBUG_TX,("%s: TX: desc %d: cmdlen 0x%08x\n", sc->dv_xname,
701  sc->txs_lastdesc, le32toh(sc->sc_txdescs[sc->txs_lastdesc].wtx_cmdlen)));
702  DPRINTF(WM_DEBUG_TX,("status 0x%08x\n",sc->sc_txdescs[sc->txq_fi].wtx_fields.wtxu_status));
703 
704  memBar();
705 
706  /* This is the location where software writes the first NEW descriptor */
707  CSR_WRITE(sc,WMREG_TDT, sc->txq_next);
708 
709  DPRINTF(WM_DEBUG_TX,("%s: addr 0x%08x, TX: TDH %d, TDT %d\n",sc->dv_xname,
710  le32toh(sc->sc_txdescs[sc->txs_lastdesc].wtx_addr.wa_low), CSR_READ(sc,WMREG_TDH),
711  CSR_READ(sc,WMREG_TDT)));
712 
713  DPRINTF(WM_DEBUG_TX,("%s: TX: finished transmitting packet, job %d\n",
714  sc->dv_xname, sc->txq_next));
715 
716 }

784 static void i82544EI_rx(struct wm_softc *sc)
785 {
786  struct ifnet *ifp = &sc->arpcom.ac_if;
787  struct mbuf *m;
788  int i, len;
789  uint8_t status, errors;
790  struct ether_header *eh;
791 
792 #ifdef WM_DEBUG
793  printk("i82544EI_rx()\n");
794 #endif
795 
796  for (i = sc->sc_rxptr;; i = WM_NEXTRX(i)) {
797  DPRINTF(WM_DEBUG_RX, ("%s: RX: checking descriptor %d\n",
798  sc->dv_xname, i));
799 
800  status = sc->sc_rxdescs[i].wrx_status;
801  if ((status & WRX_ST_DD) == 0) break; /* descriptor not done */
802 
803  errors = sc->sc_rxdescs[i].wrx_errors;
804  len = le16toh(sc->sc_rxdescs[i].wrx_len);
805  m = sc->rxs_mbuf[i];
806  if (sc->sc_rxdiscard) {
807  printk("RX: discarding contents of descriptor %d\n", i);
808  wm_init_rxdesc(sc, i);
809  if (status & WRX_ST_EOP) {
810  /* Reset our state. */
811  printk("RX: resetting rxdiscard -> 0\n");
812  sc->sc_rxdiscard = 0;
813  }
814  continue;
815  }
816 
817  /*
818  * If an error occurred, update stats and drop the packet.
819  */
820  if (errors &(WRX_ER_CE|WRX_ER_SE|WRX_ER_SEQ|WRX_ER_CXE|WRX_ER_RXE)) {
821  ifp->if_ierrors++;
822  if (errors & WRX_ER_SE)
823  printk("%s: symbol error\n",sc->dv_xname);
824  else if (errors & WRX_ER_SEQ)
825  printk("%s: receive sequence error\n",sc->dv_xname);
826  else if (errors & WRX_ER_CE)
827  printk("%s: CRC error\n",sc->dv_xname);
828  m_freem(m);
829  goto give_it_back;
830  }
831 
832  /*
833  * No errors. Receive the packet.
834  *
835  * Note, we have configured the chip to include the
836  * CRC with every packet.
837  */
838  m->m_len = m->m_pkthdr.len = len - sizeof(struct ether_header);
839 
840  DPRINTF(WM_DEBUG_RX,("%s: RX: buffer at %p len %d\n",
841  sc->dv_xname, m->m_data, len));
842 
843 
844  eh = mtod (m, struct ether_header *);
845  m->m_data += sizeof(struct ether_header);
846  ether_input (ifp, eh, m);
847  /* Pass it on. */
848  ifp->if_ipackets++;
849 
850  give_it_back:
851  /* Add a new receive buffer to the ring.*/
852  if (wm_add_rxbuf(sc, i) != 0) {
853  /*
854  * Failed, throw away what we've done so
855  * far, and discard the rest of the packet.
856  */
857  printk("Failed in wm_add_rxbuf(), drop packet\n");
858  ifp->if_ierrors++;
859  wm_init_rxdesc(sc, i);
860  if ((status & WRX_ST_EOP) == 0)
861  sc->sc_rxdiscard = 1;
862  m_freem(m);
863  }
864  } /* end for */
865 
866  /* Update the receive pointer. */
867  sc->sc_rxptr = i;
868  DPRINTF(WM_DEBUG_RX, ("%s: RX: rxptr -> %d\n", sc->dv_xname, i));
869 }

1607 {

1611  struct ifnet *ifp=&sc->arpcom.ac_if;

1613 #ifdef WM_DEBUG
1614  printk("i82544EI_daemon()\n");
1615 #endif
1616

1618  * since that has PRIORITY_INVERSION enabled, our
1619  * subsequent call to bsdnet_event_receive() will
1620  * _not_ release it. It's still in posession of our
1621  * owner.
1622  * This is different from how killing this task
1623  * is handled.
1624  */
1625 
1626  for (;;) {
1627  /* sleep until there's work to be done */
1628  /* Note: bsdnet_event_receive() acquires
1629  * the global bsdnet semaphore for
1630  * mutual exclusion.
1631  */
1632  rtems_bsdnet_event_receive(ALL_EVENTS,
1633  RTEMS_WAIT | RTEMS_EVENT_ANY,
1634  RTEMS_NO_TIMEOUT,
1635  &events);
1636  if (KILL_EVENT & events) break;
1637

1650  }
1651  else {
1652  i82544EI_txq_done(sc);
1653  break;
1654  }
1655  }

1657  * - either because there's no free buffer
1658  * (m=0 initializer && !sc->txq_free)
1659  * - or there's nothing to send (IF_DEQUEUE
1660  * returned 0
1661  */
1662  } while (m);
1663

1667  if (events & ERR_EVENT) i82544EI_error(sc);

1684  if (!m) break;
1685  m_freem(m);
1686  }

1688  * networking semaphore
1689  */
