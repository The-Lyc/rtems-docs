45 static bool thread_names_visitor( rtems_tcb *tcb, void *arg )
46 {
47  dump_context *ctx;
48  char name[ 2 * THREAD_DEFAULT_MAXIMUM_NAME_SIZE ];
49  size_t n;
50  size_t i;
51  rtems_record_item item;
52  rtems_record_data data;
53 
54  ctx = arg;
55  item.event = RTEMS_RECORD_THREAD_ID;
56  item.data = tcb->Object.id;
57  dump_chunk( ctx, &item, sizeof( item ) );
58 
59  n = _Thread_Get_name( tcb, name, sizeof( name ) );
60  i = 0;
61 
62  while ( i < n ) {
63  size_t j;
64 
65  data = 0;
66 
67  for ( j = 0; i < n && j < sizeof( data ); ++j ) {
68  rtems_record_data c;
69 
70  c = (unsigned char) name[ i ];
71  data |= c << ( j * 8 );
72  ++i;
73  }
74 
75  item.event = RTEMS_RECORD_THREAD_NAME;
76  item.data = data;
77  dump_chunk( ctx, &item, sizeof( item ) );
78  }
79 
80  return false;
81 }
