38 static Objects_Maximum
39 T_objects_count(Objects_APIs api, uint16_t cls)
40 {
41  const Objects_Information *information;
42  Objects_Maximum count;
43 
44  information = _Objects_Get_information(api, cls);
45 
46  _RTEMS_Lock_allocator();
47 
48  if (information != NULL) {
49  count = _Objects_Active_count(information);
50  } else {
51  count = 0;
52  }
53 
54  _RTEMS_Unlock_allocator();
55 
56  return count;
57 }

59 static void
60 T_objects_check(Objects_APIs api, uint16_t cls,
61  Objects_Maximum *expected, const char *name)
62 {
63  Objects_Maximum count;
64  int32_t delta;
65 
66  count = T_objects_count(api, cls);
67  delta = (int32_t)count - (int32_t)*expected;
68 
69  if (delta != 0) {
70  *expected = count;
71  T_check_true(false, NULL, "%s leak (%" PRIi32 ")", name, delta);
72  }
73 }

387 static ssize_t
388 T_get_active_posix_key_value_pairs(void)
389 {
390  ssize_t count;
391  Objects_Id id;
392  POSIX_Keys_Control *the_key;
393 
394  count = 0;
395  id = OBJECTS_ID_INITIAL_INDEX;
396 
397  while ((the_key = T_get_next_posix_key(&id)) != NULL ) {
398  count += (ssize_t)
399  _Chain_Node_count_unprotected(&the_key->Key_value_pairs);
400  _Objects_Allocator_unlock();
401  }
402 
403  return count;
404 }
