73 /*
74  * Set initial CACR mode, mainly enables branch/instruction/data cache. The
75  * FPU must be switched on in the BSP startup code since the
76  * _Thread_Start_multitasking() will restore the floating-point context of the
77  * initialization task if necessary.
78  */
79 static const uint32_t BSP_CACR_INIT = MCF548X_CACR_DEC /* enable data cache */
80  | MCF548X_CACR_BEC /* enable branch cache */
81  | MCF548X_CACR_IEC /* enable instruction cache */
82  | MCF548X_CACR_DDCM(DCACHE_ON_WRIGHTTHROUGH)
83  /* set data cache mode to write-through */
84  | MCF548X_CACR_DESB /* enable data store buffer */
85  | MCF548X_CACR_DDSP /* data access only in supv. mode */
86  | MCF548X_CACR_IDSP; /* instr. access only in supv. mode */

176 /*
177  * bsp_start
178  *
179  * This routine does the bulk of the system initialisation.
180  */
181 void bsp_start( void )
182 {
183  /* Initialize CACR shadow register */
184  _CPU_cacr_shadow = BSP_CACR_INIT;
185 
186  /*
187  * Load the shadow variable of CACR with initial mode and write it to the
188  * CACR. Interrupts are still disabled, so there is no need for surrounding
189  * rtems_interrupt_enable() / rtems_interrupt_disable().
190  */
191  m68k_set_cacr( _CPU_cacr_shadow);
192 
193  /*
194  * do mapping of acr's and/or mmu
195  */
196  acr_mmu_mapping();
197 }
