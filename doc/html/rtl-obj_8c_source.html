201 bool
202 rtems_rtl_parse_name (const char* name,
203  const char** aname,
204  const char** oname,
205  off_t* ooffset)
206 {
207  const char* laname = NULL;
208  const char* loname = NULL;
209  const char* colon;
210  const char* end;
211 
212  /*
213  * Parse the name to determine if the object file is part of an archive or it
214  * is an object file. If an archive check the name for a '@' to see if the
215  * archive contains an offset.
216  *
217  * Note, if an archive the object file oofset may be know but the
218  * object file is not. Leave the object name as a NULL.
219  */
220  end = name + strlen (name);
221  colon = strrchr (name, ':');
222  if (colon == NULL || colon < strrchr(name, '/'))
223  colon = end;
224 
225  loname = rtems_rtl_alloc_new (RTEMS_RTL_ALLOC_OBJECT, colon - name + 1, true);
226  if (!loname)
227  {
228  rtems_rtl_set_error (ENOMEM, "no memory for object file name");
229  return false;
230  }
231 
232  memcpy ((void*) loname, name, colon - name);
233 
234  /*
235  * If the pointers match there is no ':' delimiter.
236  */
237  if (colon != end)
238  {
239  const char* at;
240 
241  /*
242  * The file name is an archive and the object file name is next after the
243  * delimiter. Move the pointer to the archive name.
244  */
245  laname = loname;
246  ++colon;
247 
248  /*
249  * See if there is a '@' to delimit an archive offset for the object in the
250  * archive.
251  */
252  at = strchr (colon, '@');
253 
254  if (at == NULL)
255  at = end;
256 
257 
258  loname = rtems_rtl_alloc_new (RTEMS_RTL_ALLOC_OBJECT, at - colon + 1, true);
259  if (!loname)
260  {
261  rtems_rtl_alloc_del (RTEMS_RTL_ALLOC_OBJECT, (void*) laname);
262  rtems_rtl_set_error (ENOMEM, "no memory for object file name");
263  return false;
264  }
265 
266  memcpy ((void*) loname, colon, at - colon);
267 
268  if (at != end)
269  {
270  /*
271  * The object name has an archive offset. If the number
272  * does not parse 0 will be returned and the archive will be
273  * searched.
274  */
275  *ooffset = strtoul (at + 1, 0, 0);
276  }
277  }
278 
279  *oname = loname;
280  *aname = laname;
281  return true;
282 }

642 {

645

647  return false;

658  size_t d;

660  {

662  free_slot = &(depends->depends[d]);

664  return false;

681  free_slot = NULL;

844 {

865  sync_ctx->end_va - sync_ctx->start_va + 1);

868  }
869 
870  sync_ctx->end_va = sect->base + sect->size;

923 static int
924 rtems_rtl_obj_sections_linked_to_order (rtems_rtl_obj* obj,
925  int section,
926  uint32_t visited_mask)
927 {
928  rtems_chain_control* sections = &obj->sections;
929  rtems_chain_node* node = rtems_chain_first (sections);
930  /*
931  * Find the section being linked-to. If the linked-to link field is 0 we have
932  * the end and the section's order is the position we are after.
933  */
934  while (!rtems_chain_is_tail (sections, node))
935  {
936  rtems_rtl_obj_sect* sect = (rtems_rtl_obj_sect*) node;
937  if (sect->section == section)
938  {
939  const uint32_t mask = sect->flags & RTEMS_RTL_OBJ_SECT_TYPES;
940  int order = 0;
941  if (sect->link != 0)
942  {
943  /*
944  * Have we already visited this type of section? Avoid nesting for
945  * ever.
946  */
947  if ((sect->flags & visited_mask) != 0)
948  {
949  rtems_rtl_set_error (errno, "section link loop");
950  return -1;
951  }
952  return rtems_rtl_obj_sections_linked_to_order (obj,
953  sect->link,
954  visited_mask | mask);
955  }
956  node = rtems_chain_first (sections);
957  while (!rtems_chain_is_tail (sections, node))
958  {
959  sect = (rtems_rtl_obj_sect*) node;
960  if ((sect->flags & mask) == mask)
961  {
962  if (sect->section == section)
963  return order;
964  ++order;
965  }
966  node = rtems_chain_next (node);
967  }
968  }
969  node = rtems_chain_next (node);
970  }
971  rtems_rtl_set_error (errno, "section link not found");
972  return -1;
973 }
