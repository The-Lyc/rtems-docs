104 static int spi_bus_ioctl(
105  rtems_libio_t *iop,
106  ioctl_command_t command,
107  void *arg
108 )
109 {
110  spi_bus *bus = IMFS_generic_get_context_by_iop(iop);
111  int err;
112  uint32_t previous;
113 
114  spi_bus_obtain(bus);
115 
116  switch (command) {
117  case SPI_BUS_OBTAIN:
118  spi_bus_obtain(bus);
119  err = 0;
120  break;
121  case SPI_BUS_RELEASE:
122  spi_bus_release(bus);
123  err = 0;
124  break;
125  case SPI_IOC_RD_MODE:
126  *(uint8_t *) arg = bus->mode;
127  err = 0;
128  break;
129  case SPI_IOC_RD_MODE32:
130  *(uint32_t *) arg = bus->mode;
131  err = 0;
132  break;
133  case SPI_IOC_RD_LSB_FIRST:
134  *(uint8_t *) arg = bus->lsb_first;
135  err = 0;
136  break;
137  case SPI_IOC_RD_BITS_PER_WORD:
138  *(uint8_t *) arg = bus->bits_per_word;
139  err = 0;
140  break;
141  case SPI_IOC_RD_MAX_SPEED_HZ:
142  *(uint32_t *) arg = bus->speed_hz;
143  err = 0;
144  break;
145  case SPI_IOC_WR_MODE:
146  previous = bus->mode;
147  bus->mode = *(uint8_t *) arg;
148  err = (*bus->setup)(bus);
149 
150  if (err != 0) {
151  bus->mode = previous;
152  }
153 
154  break;
155  case SPI_IOC_WR_MODE32:
156  previous = bus->mode;
157  bus->mode = *(uint32_t *) arg;
158  err = (*bus->setup)(bus);
159 
160  if (err != 0) {
161  bus->mode = previous;
162  }
163 
164  break;
165  case SPI_IOC_WR_LSB_FIRST:
166  previous = bus->lsb_first;
167  bus->lsb_first = (*(uint8_t *) arg) != 0;
168  err = (*bus->setup)(bus);
169 
170  if (err != 0) {
171  bus->lsb_first = (bool) previous;
172  }
173 
174  break;
175  case SPI_IOC_WR_BITS_PER_WORD:
176  previous = bus->bits_per_word;
177  bus->bits_per_word = *(uint8_t *) arg;
178  err = (*bus->setup)(bus);
179 
180  if (err != 0) {
181  bus->bits_per_word = (uint8_t) previous;
182  }
183 
184  break;
185  case SPI_IOC_WR_MAX_SPEED_HZ:
186  previous = bus->speed_hz;
187  bus->speed_hz = *(uint32_t *) arg;
188  err = (*bus->setup)(bus);
189 
190  if (err != 0) {
191  bus->speed_hz = previous;
192  }
193 
194  break;
195  default:
196  if (IOCBASECMD(command) == IOCBASECMD(_IOW(SPI_IOC_MAGIC, 0, 0))) {
197  err = (*bus->transfer)(
198  bus,
199  arg,
200  IOCPARM_LEN(command) / sizeof(struct spi_ioc_transfer)
201  );
202  } else {
203  err = -EINVAL;
204  }
205  break;
206  }
207 
208  spi_bus_release(bus);
209 
210  if (err == 0) {
211  return 0;
212  } else {
213  rtems_set_errno_and_return_minus_one(-err);
214  }
215 }
