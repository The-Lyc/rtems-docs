35 void pic_remap(uint8_t offset1, uint8_t offset2)
36 {
37  uint8_t a1, a2;
38 
39  /* save masks */
40  a1 = inport_byte(PIC1_DATA);
41  a2 = inport_byte(PIC2_DATA);
42 
43  /* start the initialization sequence in cascade mode */
44  outport_byte(PIC1_COMMAND, PIC_ICW1_INIT | PIC_ICW1_ICW4);
45  stub_io_wait();
46  outport_byte(PIC2_COMMAND, PIC_ICW1_INIT | PIC_ICW1_ICW4);
47  stub_io_wait();
48  /* ICW2: Master PIC vector offset */
49  outport_byte(PIC1_DATA, offset1);
50  stub_io_wait();
51  /* ICW2: Slave PIC vector offset */
52  outport_byte(PIC2_DATA, offset2);
53  stub_io_wait();
54  /* ICW3: tell Master PIC that there is a slave PIC at IRQ2 (0000 0100) */
55  outport_byte(PIC1_DATA, 4);
56  stub_io_wait();
57  /* ICW3: tell Slave PIC its cascade identity (0000 0010) */
58  outport_byte(PIC2_DATA, 2);
59  stub_io_wait();
60 
61  outport_byte(PIC1_DATA, PIC_ICW4_8086);
62  stub_io_wait();
63  outport_byte(PIC2_DATA, PIC_ICW4_8086);
64  stub_io_wait();
65 
66  /* restore saved masks. */
67  outport_byte(PIC1_DATA, a1);
68  outport_byte(PIC2_DATA, a2);
69 }

72 {
