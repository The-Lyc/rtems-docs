318 {
319  uint32_t byte1, byte2, byte3, byte4, swapped;
320 
321  byte4 = (value >> 24) & 0xff;
322  byte3 = (value >> 16) & 0xff;
323  byte2 = (value >> 8) & 0xff;
324  byte1 = value & 0xff;
325 
326  swapped = (byte1 << 24) | (byte2 << 16) | (byte3 << 8) | byte4;
