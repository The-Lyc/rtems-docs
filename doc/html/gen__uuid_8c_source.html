153 static int get_random_fd(void)
154 {
155  struct timeval tv;
156  static int fd = -2;
157  int i;
158 
159  if (fd == -2) {
160  gettimeofday(&tv, 0);
161 #ifndef _WIN32
162  fd = open("/dev/urandom", O_RDONLY);
163  if (fd == -1)
164  fd = open("/dev/random", O_RDONLY | O_NONBLOCK);
165  if (fd >= 0) {
166  i = fcntl(fd, F_GETFD);
167  if (i >= 0)
168  fcntl(fd, F_SETFD, i | FD_CLOEXEC);
169  }
170 #endif
171  srand((getpid() << ((sizeof(pid_t)*CHAR_BIT)>>1)) ^ getuid() ^ tv.tv_sec ^ tv.tv_usec);
172 #ifdef DO_JRAND_MIX
173  jrand_seed[0] = getpid() ^ (tv.tv_sec & 0xFFFF);
174  jrand_seed[1] = getppid() ^ (tv.tv_usec & 0xFFFF);
175  jrand_seed[2] = (tv.tv_sec ^ tv.tv_usec) >> 16;
176 #endif
177  }
178  /* Crank the random number generator a few times */
179  gettimeofday(&tv, 0);
180  for (i = (tv.tv_sec ^ tv.tv_usec) & 0x1F; i > 0; i--)
181  rand();
182  return fd;
183 }

186 /*
187  * Generate a series of random bytes. Use /dev/urandom if possible,
188  * and if not, use srandom/random.
189  */
190 static void get_random_bytes(void *buf, int nbytes)
191 {
192  int i, n = nbytes, fd = get_random_fd();
193  int lose_counter = 0;
194  unsigned char *cp = (unsigned char *) buf;
195 #ifdef DO_JRAND_MIX
196  unsigned short tmp_seed[3];
197 #endif
198 
199  if (fd >= 0) {
200  while (n > 0) {
201  i = read(fd, cp, n);
202  if (i <= 0) {
203  if (lose_counter++ > 16)
204  break;
205  continue;
206  }
207  n -= i;
208  cp += i;
209  lose_counter = 0;
210  }
211  }
212 
213  /*
214  * We do this all the time, but this is the only source of
215  * randomness if /dev/random/urandom is out to lunch.
216  */
217  for (cp = buf, i = 0; i < nbytes; i++)
218  *cp++ ^= (rand() >> 7) & 0xFF;
219 #ifdef DO_JRAND_MIX
220  memcpy(tmp_seed, jrand_seed, sizeof(tmp_seed));
221  jrand_seed[2] = jrand_seed[2] ^ syscall(__NR_gettid);
222  for (cp = buf, i = 0; i < nbytes; i++)
223  *cp++ ^= (jrand48(tmp_seed) >> 7) & 0xFF;
224  memcpy(jrand_seed, tmp_seed,
225  sizeof(jrand_seed)-sizeof(unsigned short));
226 #endif
227 
228  return;
229 }

266 #endif /* HAVE_SA_LEN*/

269  if (sd < 0) {

300  * XXX we don't have a way of getting the hardware

302  */
303  close(sd);
304  return 0;
305 #endif /* HAVE_NET_IF_DL_H */
306 #endif /* SIOCGENADDR */
307 #endif /* SIOCGIFHWADDR */
308  if (!a[0] && !a[1] && !a[2] && !a[3] && !a[4] && !a[5])

317 #endif

324 static int get_clock(uint32_t *clock_high, uint32_t *clock_low,
325  uint16_t *ret_clock_seq, int *num)
326 {
327  THREAD_LOCAL int adjustment = 0;
328  THREAD_LOCAL struct timeval last = {0, 0};
329  THREAD_LOCAL int state_fd = -2;
330  THREAD_LOCAL FILE *state_f;
331  THREAD_LOCAL uint16_t clock_seq;
332  struct timeval tv;
333  struct flock fl;
334  uint64_t clock_reg;
335  mode_t save_umask;
336  int len;
337 
338  if (state_fd == -2) {
339  save_umask = umask(0);
340  state_fd = open("/var/lib/libuuid/clock.txt",
341  O_RDWR|O_CREAT, 0660);
342  (void) umask(save_umask);
343  state_f = fdopen(state_fd, "r+");
344  if (!state_f) {
345  close(state_fd);
346  state_fd = -1;
347  }
348  }
349  fl.l_type = F_WRLCK;
350  fl.l_whence = SEEK_SET;
351  fl.l_start = 0;
352  fl.l_len = 0;
353  fl.l_pid = 0;
354  if (state_fd >= 0) {
355  rewind(state_f);
356  while (fcntl(state_fd, F_SETLKW, &fl) < 0) {
357  if ((errno == EAGAIN) || (errno == EINTR))
358  continue;
359  fclose(state_f);
360  close(state_fd);
361  state_fd = -1;
362  break;
363  }
364  }
365  if (state_fd >= 0) {
366  unsigned int cl;
367  utime_t tv1;
368  unsigned long tv2;
369  int a;
370 
371  if (fscanf(state_f, "clock: %04x tv: %" SCNutime_t " %lu adj: %d\n",
372  &cl, &tv1, &tv2, &a) == 4) {
373  clock_seq = cl & 0x3FFF;
374  last.tv_sec = tv1;
375  last.tv_usec = tv2;
376  adjustment = a;
377  }
378  }
379 
380  if ((last.tv_sec == 0) && (last.tv_usec == 0)) {
381  get_random_bytes(&clock_seq, sizeof(clock_seq));
382  clock_seq &= 0x3FFF;
383  gettimeofday(&last, 0);
384  last.tv_sec--;
385  }
386 
387 try_again:
388  gettimeofday(&tv, 0);
389  if ((tv.tv_sec < last.tv_sec) ||
390  ((tv.tv_sec == last.tv_sec) &&
391  (tv.tv_usec < last.tv_usec))) {
392  clock_seq = (clock_seq+1) & 0x3FFF;
393  adjustment = 0;
394  last = tv;
395  } else if ((tv.tv_sec == last.tv_sec) &&
396  (tv.tv_usec == last.tv_usec)) {
397  if (adjustment >= MAX_ADJUSTMENT)
398  goto try_again;
399  adjustment++;
400  } else {
401  adjustment = 0;
402  last = tv;
403  }
404 
405  clock_reg = ((uint64_t) tv.tv_usec)*10 + adjustment;
406  clock_reg += ((uint64_t) tv.tv_sec)*10000000;
407  clock_reg += (((uint64_t) 0x01B21DD2) << 32) + 0x13814000;
408 
409  if (num && (*num > 1)) {
410  adjustment += *num - 1;
411  last.tv_usec += adjustment / 10;
412  adjustment = adjustment % 10;
413  last.tv_sec += last.tv_usec / 1000000;
414  last.tv_usec = last.tv_usec % 1000000;
415  }
416 
417  if (state_fd > 0) {
418  rewind(state_f);
419  len = fprintf(state_f,
420  "clock: %04x tv: %016" PRIutime_t " %08lu adj: %08d\n",
421  clock_seq, last.tv_sec, last.tv_usec, adjustment);
422  fflush(state_f);
423  if (ftruncate(state_fd, len) < 0) {
424  fprintf(state_f, " \n");
425  fflush(state_f);
426  }
427  rewind(state_f);
428  fl.l_type = F_UNLCK;
429  fcntl(state_fd, F_SETLK, &fl);
430  }
431 
432  *clock_high = clock_reg >> 32;
433  *clock_low = clock_reg;
434  *ret_clock_seq = clock_seq;
435  return 0;
436 }

568 void uuid__generate_time(uuid_t out, int *num)
569 {
570  static unsigned char node_id[6];
571  static int has_init = 0;
572  struct uuid uu;
573  uint32_t clock_mid;
574 
575  if (!has_init) {
576  if (get_node_id(node_id) <= 0) {
577  get_random_bytes(node_id, 6);
578  /*
579  * Set multicast bit, to prevent conflicts
580  * with IEEE 802 addresses obtained from
581  * network cards
582  */
583  node_id[0] |= 0x01;
584  }
585  has_init = 1;
586  }
587  get_clock(&clock_mid, &uu.time_low, &uu.clock_seq, num);
588  uu.clock_seq |= 0x8000;
589  uu.time_mid = (uint16_t) clock_mid;
590  uu.time_hi_and_version = ((clock_mid >> 16) & 0x0FFF) | 0x1000;
591  memcpy(uu.node, node_id, 6);
592  uuid_pack(&uu, out);
593 }
