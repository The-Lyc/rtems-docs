36 rtems_status_code rtems_task_create(
37  rtems_name name,
38  rtems_task_priority initial_priority,
39  size_t stack_size,
40  rtems_mode initial_modes,
41  rtems_attribute attribute_set,
42  rtems_id *id
43 )
44 {
45  Thread_Control *the_thread;
46  Thread_Configuration config;
47 #if defined(RTEMS_MULTIPROCESSING)
48  Objects_MP_Control *the_global_object = NULL;
49  bool is_global;
50 #endif
51  bool status;
52  rtems_attribute the_attribute_set;
53  bool valid;
54  RTEMS_API_Control *api;
55  ASR_Information *asr;
56 
57  if ( !id )
58  return RTEMS_INVALID_ADDRESS;
59 
60  if ( !rtems_is_name_valid( name ) )
61  return RTEMS_INVALID_NAME;
62 
63  /*
64  * Core Thread Initialize insures we get the minimum amount of
65  * stack space.
66  */
67 
68  /*
69  * Fix the attribute set to match the attributes which
70  * this processor (1) requires and (2) is able to support.
71  * First add in the required flags for attribute_set
72  * Typically this might include FP if the platform
73  * or application required all tasks to be fp aware.
74  * Then turn off the requested bits which are not supported.
75  */
76 
77  the_attribute_set = _Attributes_Set( attribute_set, ATTRIBUTES_REQUIRED );
78  the_attribute_set =
79  _Attributes_Clear( the_attribute_set, ATTRIBUTES_NOT_SUPPORTED );
80 
81  memset( &config, 0, sizeof( config ) );
82  config.budget_algorithm = _Modes_Is_timeslice( initial_modes ) ?
83  THREAD_CPU_BUDGET_ALGORITHM_RESET_TIMESLICE
84  : THREAD_CPU_BUDGET_ALGORITHM_NONE,
85  config.isr_level = _Modes_Get_interrupt_level( initial_modes );
86  config.name.name_u32 = name;
87  config.is_fp = _Attributes_Is_floating_point( the_attribute_set );
88  config.is_preemptible = _Modes_Is_preempt( initial_modes );
89  config.stack_size = _Stack_Ensure_minimum( stack_size );
90  config.stack_size = _Stack_Extend_size( config.stack_size, config.is_fp );
91 
92  /*
93  * Validate the RTEMS API priority and convert it to the core priority range.
94  */
95 
96  if ( !_Attributes_Is_system_task( the_attribute_set ) ) {
97  if ( initial_priority == PRIORITY_MINIMUM ) {
98  return RTEMS_INVALID_PRIORITY;
99  }
100  }
101 
102  config.scheduler = _Thread_Scheduler_get_home( _Thread_Get_executing() );
103 
104  config.priority = _RTEMS_Priority_To_core(
105  config.scheduler,
106  initial_priority,
107  &valid
108  );
109  if ( !valid ) {
110  return RTEMS_INVALID_PRIORITY;
111  }
112 
113 #if defined(RTEMS_MULTIPROCESSING)
114  if ( !_System_state_Is_multiprocessing ) {
115  the_attribute_set = _Attributes_Clear( the_attribute_set, RTEMS_GLOBAL );
116  }
117 
118  is_global = _Attributes_Is_global( the_attribute_set );
119 #endif
120 
121  /*
122  * Allocate the thread control block and -- if the task is global --
123  * allocate a global object control block.
124  *
125  * NOTE: This routine does not use the combined allocate and open
126  * global object routine (_Objects_MP_Allocate_and_open) because
127  * this results in a lack of control over when memory is allocated
128  * and can be freed in the event of an error.
129  */
130  the_thread = _RTEMS_tasks_Allocate();
131 
132  if ( !the_thread ) {
133  _Objects_Allocator_unlock();
134  return RTEMS_TOO_MANY;
135  }
136 
137 #if defined(RTEMS_MULTIPROCESSING)
138  if ( is_global ) {
139  the_global_object = _Objects_MP_Allocate_global_object();
140 
141  if ( _Objects_MP_Is_null_global_object( the_global_object ) ) {
142  _RTEMS_tasks_Free( the_thread );
143  _Objects_Allocator_unlock();
144  return RTEMS_TOO_MANY;
145  }
146  }
147 #endif
148 
149  config.stack_area = _Stack_Allocate( config.stack_size );
150  config.allocated_stack = config.stack_area;
151  status = ( config.stack_area != NULL );
152 
153  /*
154  * Initialize the core thread for this task.
155  */
156 
157  if ( status ) {
158  status = _Thread_Initialize(
159  &_RTEMS_tasks_Information,
160  the_thread,
161  &config
162  );
163  }
164 
165  if ( !status ) {
166 #if defined(RTEMS_MULTIPROCESSING)
167  if ( is_global )
168  _Objects_MP_Free_global_object( the_global_object );
169 #endif
170  _RTEMS_tasks_Free( the_thread );
171  _Objects_Allocator_unlock();
172  return RTEMS_UNSATISFIED;
173  }
174 
175  api = the_thread->API_Extensions[ THREAD_API_RTEMS ];
176  asr = &api->Signal;
177 
178  asr->is_enabled = _Modes_Is_asr_disabled(initial_modes) ? false : true;
179 
180  *id = the_thread->Object.id;
181 
182 #if defined(RTEMS_MULTIPROCESSING)
183  the_thread->is_global = is_global;
184  if ( is_global ) {
185 
186  _Objects_MP_Open(
187  &_RTEMS_tasks_Information.Objects,
188  the_global_object,
189  name,
190  the_thread->Object.id
191  );
192 
193  _RTEMS_tasks_MP_Send_process_packet(
194  RTEMS_TASKS_MP_ANNOUNCE_CREATE,
195  the_thread->Object.id,
196  name
197  );
198 
199  }
200 #endif
201 
202  _Objects_Allocator_unlock();
203  return RTEMS_SUCCESSFUL;
204 }
