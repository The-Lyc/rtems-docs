160 
161 
162  cause = inl(ETH0_EICR);
163  outl( ~cause,ETH0_EICR); /* clear the ICR */
164 
165  if ( (!cause) || (cause & 0x803d00)) {
166  sc->intr_errsts[sc->intr_err_ptr2++]=cause;
167  sc->intr_err_ptr2 %=INTR_ERR_SIZE; /* Till Straumann */
168  events |= ERR_EVENT;
