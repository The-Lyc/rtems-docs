39 # define TOHOST_CMD(dev, cmd, payload) \
40  (((uint64_t)(dev) << 56) | ((uint64_t)(cmd) << 48) | (uint64_t)(payload))
41 
42 #define FROMHOST_DEV(fromhost_value) ((uint64_t)(fromhost_value) >> 56)
43 #define FROMHOST_CMD(fromhost_value) ((uint64_t)(fromhost_value) << 8 >> 56)
44 #define FROMHOST_DATA(fromhost_value) ((uint64_t)(fromhost_value) << 16 >> 16)

51 static void __check_fromhost(void)
52 {
53  uint64_t fh = fromhost;
54  if (!fh) {
55  return;
56  }
57  fromhost = 0;
58 
59  // this should be from the console
60  assert(FROMHOST_DEV(fh) == 1);
61  switch (FROMHOST_CMD(fh)) {
62  case 0:
63  htif_console_buf = 1 + (uint8_t)FROMHOST_DATA(fh);
64  break;
65  case 1:
66  break;
67  default:
68  assert(0);
69  }
70 }

72 static void __set_tohost(uintptr_t dev, uintptr_t cmd, uintptr_t data)
73 {
74  while (tohost) {
75  __check_fromhost();
76  }
77  tohost = TOHOST_CMD(dev, cmd, data);
78 }

80 int htif_console_getchar(rtems_termios_device_context *base)
81 {
82  __check_fromhost();
83  int ch = htif_console_buf;
84  if (ch >= 0) {
85  htif_console_buf = -1;
86  __set_tohost(1, 0, 0);
87  }
88 
89  return ch - 1;
90 }
