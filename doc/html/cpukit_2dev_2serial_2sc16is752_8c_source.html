74 static void set_mcr_dll_dlh(
75  sc16is752_context *ctx,
76  uint8_t mcr,
77  uint32_t divisor
78 )
79 {
80  bool sleep_mode = is_sleep_mode_enabled(ctx);
81  uint8_t dll = (uint8_t)divisor;
82  uint8_t dlh = (uint8_t)(divisor >> 8);
83 
84  if (sleep_mode) {
85  set_sleep_mode(ctx, false);
86  }
87 
88  ctx->lcr |= SC16IS752_LCR_ENABLE_DIVISOR;
89  write_reg(ctx, SC16IS752_LCR, &ctx->lcr, 1);
90 
91  write_reg(ctx, SC16IS752_MCR, &mcr, 1);
92  write_reg(ctx, SC16IS752_DLH, &dlh, 1);
93  write_reg(ctx, SC16IS752_DLL, &dll, 1);
94 
95  ctx->lcr &= ~SC16IS752_LCR_ENABLE_DIVISOR;
96  write_reg(ctx, SC16IS752_LCR, &ctx->lcr, 1);
97 
98  if (sleep_mode) {
99  set_sleep_mode(ctx, true);
100  }
101 }

128 static bool set_baud(sc16is752_context *ctx, rtems_termios_baud_t baud)
129 {
130  uint32_t freq = ctx->input_frequency;
131  uint8_t mcr;
132  uint32_t divisor;
133 
134  read_reg(ctx, SC16IS752_MCR, &mcr, 1);
135 
136  divisor = freq / baud / 16;
137  if (divisor > 0xFFFF){
138  divisor = (freq / (4 * baud)) / 16;
139  if (divisor > 0xFFFF){
140  return false;
141  } else {
142  mcr |= SC16IS752_MCR_PRESCALE_NEEDED;
143  }
144  } else {
145  mcr &= ~SC16IS752_MCR_PRESCALE_NEEDED;
146  }
147 
148  set_mcr_dll_dlh(ctx, mcr, divisor);
149  return true;
150 }

152 static bool sc16is752_set_attributes(
153  rtems_termios_device_context *base,
154  const struct termios *term
155 )
156 {
157  sc16is752_context *ctx = (sc16is752_context *)base;
158  rtems_termios_baud_t baud;
159 
160  ctx->lcr = 0;
161 
162  baud = rtems_termios_baud_to_number(term->c_ospeed);
163 
164  if (baud > 0) {
165  if (!set_baud(ctx, baud)){
166  return false;
167  }
168 
169  ctx->efcr &= ~SC16IS752_EFCR_TX_DISABLE;
170 
171  if ((term->c_cflag & CREAD) == 0){
172  ctx->efcr |= SC16IS752_EFCR_RX_DISABLE;
173  } else {
174  ctx->efcr &= ~SC16IS752_EFCR_RX_DISABLE;
175  }
176  } else {
177  ctx->efcr |= SC16IS752_EFCR_RX_DISABLE | SC16IS752_EFCR_TX_DISABLE;
178  }
179 
180  write_reg(ctx, SC16IS752_EFCR, &ctx->efcr, 1);
181 
182  switch (term->c_cflag & CSIZE) {
183  case CS5:
184  ctx->lcr |= SC16IS752_LCR_CHRL_5_BIT;
185  break;
186  case CS6:
187  ctx->lcr |= SC16IS752_LCR_CHRL_6_BIT;
188  break;
189  case CS7:
190  ctx->lcr |= SC16IS752_LCR_CHRL_7_BIT;
191  break;
192  case CS8:
193  ctx->lcr |= SC16IS752_LCR_CHRL_8_BIT;
194  break;
195  }
196 
197  if ((term->c_cflag & PARENB) != 0){
198  ctx->lcr |= SC16IS752_LCR_SET_PARITY;
199  if ((term->c_cflag & PARODD) != 0) {
200  ctx->lcr &= ~SC16IS752_LCR_EVEN_PARITY;
201  } else {
202  ctx->lcr |= SC16IS752_LCR_EVEN_PARITY;
203  }
204  } else {
205  ctx->lcr &= ~SC16IS752_LCR_SET_PARITY;
206  }
207 
208  if ((term->c_cflag & CSTOPB) != 0) {
209  ctx->lcr |= SC16IS752_LCR_2_STOP_BIT;
210  } else {
211  ctx->lcr &= ~SC16IS752_LCR_2_STOP_BIT;
212  }
213 
214  write_reg(ctx, SC16IS752_LCR, &ctx->lcr, 1);
215  return true;
216 }

218 static bool sc16is752_first_open(
219  rtems_termios_tty *tty,
220  rtems_termios_device_context *base,
221  struct termios *term,
222  rtems_libio_open_close_args_t *args
223 )
224 {
225  bool ok;
226  uint8_t fcr;
227  uint8_t efcr;
228 
229  (void)args;
230  sc16is752_context *ctx = (sc16is752_context *)base;
231 
232  ctx->tty = tty;
233 
234  ok = (*ctx->first_open)(ctx);
235  if (!ok) {
236  return ok;
237  }
238 
239  set_efr(ctx, SC16IS752_EFR_ENHANCED_FUNC_ENABLE);
240 
241  efcr = 0;
242 
243  switch (ctx->mode) {
244  case SC16IS752_MODE_RS485_RTS_INV:
245  efcr |= SC16IS752_EFCR_RTSINVER;
246  /* Fall through */
247  case SC16IS752_MODE_RS485_RTS:
248  efcr |= SC16IS752_EFCR_RTSCON;
249  /* Fall through */
250  case SC16IS752_MODE_RS485:
251  efcr |= SC16IS752_EFCR_RS485_ENABLE;
252  break;
253  default:
254  break;
255  }
256 
257  ctx->efcr = efcr;
258  write_reg(ctx, SC16IS752_EFCR, &ctx->efcr, 1);
259 
260  fcr = SC16IS752_FCR_FIFO_EN
261  | SC16IS752_FCR_RX_FIFO_RST
262  | SC16IS752_FCR_TX_FIFO_RST
263  | SC16IS752_FCR_RX_FIFO_TRG_8
264  | SC16IS752_FCR_TX_FIFO_TRG_32;
265  write_reg(ctx, SC16IS752_FCR, &fcr, 1);
266 
267  fcr = SC16IS752_FCR_FIFO_EN
268  | SC16IS752_FCR_RX_FIFO_TRG_8
269  | SC16IS752_FCR_TX_FIFO_TRG_32;
270  write_reg(ctx, SC16IS752_FCR, &fcr, 1);
271 
272  set_tlr(ctx, 0);
273 
274  ctx->ier = SC16IS752_IER_RHR;
275  write_reg(ctx, SC16IS752_IER, &ctx->ier, 1);
276 
277  rtems_termios_set_initial_baud(tty, 115200);
278  ok = sc16is752_set_attributes(base, term);
279  if (!ok) {
280  return ok;
281  }
282 
283  ok = (*ctx->install_irq)(ctx);
284  return ok;
285 }

300 static void sc16is752_write(
301  rtems_termios_device_context *base,
302  const char *buf,
303  size_t len
304 )
305 {
306  sc16is752_context *ctx = (sc16is752_context *)base;
307 
308  if (len > 0) {
309  ctx->ier |= SC16IS752_IER_THR;
310  len = MIN(len, 32);
311  ctx->tx_in_progress = (uint8_t)len;
312  write_reg(ctx, SC16IS752_THR, (const uint8_t *)&buf[0], len);
313  write_reg(ctx, SC16IS752_IER, &ctx->ier, 1);
314  } else {
315  ctx->tx_in_progress = 0;
316  ctx->ier &= ~SC16IS752_IER_THR;
317  write_reg(ctx, SC16IS752_IER, &ctx->ier, 1);
318  }
319 }

321 static void sc16is752_get_modem_bits(sc16is752_context *ctx, int *bits)
322 {
323  *bits = 0;
324  uint8_t msr;
325  uint8_t mcr;
326 
327  read_reg(ctx, SC16IS752_MSR, &msr, 1);
328  read_reg(ctx, SC16IS752_MCR, &mcr, 1);
329 
330  if (msr & SC16IS752_MSR_CTS) {
331  *bits |= TIOCM_CTS;
332  }
333  if (msr & SC16IS752_MSR_DSR) {
334  *bits |= TIOCM_DSR;
335  }
336  if (msr & SC16IS752_MSR_RI) {
337  *bits |= TIOCM_RI;
338  }
339  if (msr & SC16IS752_MSR_CD) {
340  *bits |= TIOCM_CD;
341  }
342  if ((mcr & SC16IS752_MCR_DTR) == 0) {
343  *bits |= TIOCM_DTR;
344  }
345  if ((mcr & SC16IS752_MCR_RTS) == 0) {
346  *bits |= TIOCM_RTS;
347  }
348 }

350 static void sc16is752_set_modem_bits(
351  sc16is752_context *ctx, int *bits, int set, int clear
352 )
353 {
354  uint8_t mcr;
355 
356  read_reg(ctx, SC16IS752_MCR, &mcr, 1);
357 
358  if (bits != NULL) {
359  if ((*bits & TIOCM_DTR) == 0) {
360  mcr |= SC16IS752_MCR_DTR;
361  } else {
362  mcr &= ~SC16IS752_MCR_DTR;
363  }
364 
365  if ((*bits & TIOCM_RTS) == 0) {
366  mcr |= SC16IS752_MCR_RTS;
367  } else {
368  mcr &= ~SC16IS752_MCR_RTS;
369  }
370  }
371 
372  if ((set & TIOCM_DTR) != 0) {
373  mcr &= ~SC16IS752_MCR_DTR;
374  }
375  if ((set & TIOCM_RTS) != 0) {
376  mcr &= ~SC16IS752_MCR_RTS;
377  }
378  if ((clear & TIOCM_DTR) != 0) {
379  mcr |= SC16IS752_MCR_DTR;
380  }
381  if ((clear & TIOCM_RTS) != 0) {
382  mcr |= SC16IS752_MCR_RTS;
383  }
384 
385  write_reg(ctx, SC16IS752_MCR, &mcr, 1);
386 }

388 static int sc16is752_ioctl(
389  rtems_termios_device_context *base,
390  ioctl_command_t request,
391  void *buffer
392 )
393 {
394  sc16is752_context *ctx = (sc16is752_context *)base;
395  uint8_t regval;
396 
397  switch (request) {
398  case SC16IS752_SET_SLEEP_MODE:
399  set_sleep_mode(ctx, *(int *)buffer != 0);
400  break;
401  case SC16IS752_GET_SLEEP_MODE:
402  *(int *)buffer = is_sleep_mode_enabled(ctx);
403  break;
404  case SC16IS752_SET_IOCONTROL:
405  regval = (*(uint8_t *)buffer) & ~SC16IS752_IOCONTROL_SRESET;
406  write_reg(ctx, SC16IS752_IOCONTROL, &regval, 1);
407  break;
408  case SC16IS752_GET_IOCONTROL:
409  read_reg(ctx, SC16IS752_IOCONTROL, (uint8_t *)buffer, 1);
410  break;
411  case SC16IS752_SET_IODIR:
412  write_reg(ctx, SC16IS752_IODIR, (uint8_t *)buffer, 1);
413  break;
414  case SC16IS752_GET_IODIR:
415  read_reg(ctx, SC16IS752_IODIR, (uint8_t *)buffer, 1);
416  break;
417  case SC16IS752_SET_IOSTATE:
418  write_reg(ctx, SC16IS752_IOSTATE, (uint8_t *)buffer, 1);
419  break;
420  case SC16IS752_GET_IOSTATE:
421  read_reg(ctx, SC16IS752_IOSTATE, (uint8_t *)buffer, 1);
422  break;
423  case SC16IS752_SET_EFCR:
424  write_reg(ctx, SC16IS752_EFCR, (uint8_t *)buffer, 1);
425  break;
426  case SC16IS752_GET_EFCR:
427  read_reg(ctx, SC16IS752_EFCR, (uint8_t *)buffer, 1);
428  break;
429  case TIOCMGET:
430  sc16is752_get_modem_bits(ctx, (int *)buffer);
431  break;
432  case TIOCMSET:
433  sc16is752_set_modem_bits(ctx, (int *)buffer, 0, 0);
434  break;
435  case TIOCMBIS:
436  sc16is752_set_modem_bits(ctx, NULL, *(int *)buffer, 0);
437  break;
438  case TIOCMBIC:
439  sc16is752_set_modem_bits(ctx, NULL, 0, *(int *)buffer);
440  break;
441  default:
442  rtems_set_errno_and_return_minus_one(EINVAL);
443  }
444 
445  return 0;
446 }

457 void sc16is752_interrupt_handler(void *arg)
458 {
459  sc16is752_context *ctx = (sc16is752_context *)arg;
460  uint8_t data[2];
461  uint8_t iir;
462 
463  read_2_reg(ctx, SC16IS752_IIR, SC16IS752_RXLVL, data);
464  iir = data[0];
465 
466  if ((iir & SC16IS752_IIR_TX_INTERRUPT) != 0 && ctx->tx_in_progress > 0) {
467  rtems_termios_dequeue_characters(ctx->tty, ctx->tx_in_progress);
468  }
469 
470  if ((iir & SC16IS752_IIR_RX_INTERRUPT) != 0) {
471  uint8_t buf[SC16IS752_FIFO_DEPTH];
472  uint8_t rxlvl = data[1];
473 
474  rxlvl = MIN(rxlvl, SC16IS752_FIFO_DEPTH);
475  read_reg(ctx, SC16IS752_RHR, &buf[0], rxlvl);
476  rtems_termios_enqueue_raw_characters(ctx->tty, (const char *)&buf[0], rxlvl);
477  }
