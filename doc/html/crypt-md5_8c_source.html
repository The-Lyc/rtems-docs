46 crypt_md5_r(const char *pw, const char *salt, struct crypt_data *data)
47 {
48  MD5_CTX ctx,ctx1;
49  unsigned long l;
50  int sl, pl;
51  u_int i;
52  u_char final[MD5_SIZE];
53  const char *sp, *ep;
54  char *passwd = &data->buffer[0], *p;
55  static const char magic[] = "$1$";
56 
57  /* Refine the Salt first */
58  sp = salt;
59 
60  /* If it starts with the magic string, then skip that */
61  if(!strncmp(sp, magic, strlen(magic)))
62  sp += strlen(magic);
63 
64  /* It stops at the first '$', max 8 chars */
65  for(ep = sp; *ep && *ep != '$' && ep < (sp + 8); ep++)
66  continue;
67 
68  /* get the length of the true salt */
69  sl = ep - sp;
70 
71  MD5Init(&ctx);
72 
73  /* The password first, since that is what is most unknown */
74  MD5Update(&ctx, (const u_char *)pw, strlen(pw));
75 
76  /* Then our magic string */
77  MD5Update(&ctx, (const u_char *)magic, strlen(magic));
78 
79  /* Then the raw salt */
80  MD5Update(&ctx, (const u_char *)sp, (u_int)sl);
81 
82  /* Then just as many characters of the MD5(pw,salt,pw) */
83  MD5Init(&ctx1);
84  MD5Update(&ctx1, (const u_char *)pw, strlen(pw));
85  MD5Update(&ctx1, (const u_char *)sp, (u_int)sl);
86  MD5Update(&ctx1, (const u_char *)pw, strlen(pw));
87  MD5Final(final, &ctx1);
88  for(pl = (int)strlen(pw); pl > 0; pl -= MD5_SIZE)
89  MD5Update(&ctx, (const u_char *)final,
90  (u_int)(pl > MD5_SIZE ? MD5_SIZE : pl));
91 
92  /* Don't leave anything around in vm they could use. */
93  memset(final, 0, sizeof(final));
94 
95  /* Then something really weird... */
96  for (i = strlen(pw); i; i >>= 1)
97  if(i & 1)
98  MD5Update(&ctx, (const u_char *)final, 1);
99  else
100  MD5Update(&ctx, (const u_char *)pw, 1);
101 
102  /* Now make the output string */
103  strcpy(passwd, magic);
104  strncat(passwd, sp, (u_int)sl);
105  strcat(passwd, "$");
106 
107  MD5Final(final, &ctx);
108 
109  /*
110  * and now, just to make sure things don't run too fast
111  * On a 60 Mhz Pentium this takes 34 msec, so you would
112  * need 30 seconds to build a 1000 entry dictionary...
113  */
114  for(i = 0; i < 1000; i++) {
115  MD5Init(&ctx1);
116  if(i & 1)
117  MD5Update(&ctx1, (const u_char *)pw, strlen(pw));
118  else
119  MD5Update(&ctx1, (const u_char *)final, MD5_SIZE);
120 
121  if(i % 3)
122  MD5Update(&ctx1, (const u_char *)sp, (u_int)sl);
123 
124  if(i % 7)
125  MD5Update(&ctx1, (const u_char *)pw, strlen(pw));
126 
127  if(i & 1)
128  MD5Update(&ctx1, (const u_char *)final, MD5_SIZE);
129  else
130  MD5Update(&ctx1, (const u_char *)pw, strlen(pw));
131  MD5Final(final, &ctx1);
132  }
133 
134  p = passwd + strlen(passwd);
135 
136  l = (final[ 0]<<16) | (final[ 6]<<8) | final[12];
137  _crypt_to64(p, l, 4); p += 4;
138  l = (final[ 1]<<16) | (final[ 7]<<8) | final[13];
139  _crypt_to64(p, l, 4); p += 4;
140  l = (final[ 2]<<16) | (final[ 8]<<8) | final[14];
141  _crypt_to64(p, l, 4); p += 4;
142  l = (final[ 3]<<16) | (final[ 9]<<8) | final[15];
143  _crypt_to64(p, l, 4); p += 4;
144  l = (final[ 4]<<16) | (final[10]<<8) | final[ 5];
145  _crypt_to64(p, l, 4); p += 4;
146  l = final[11];
147  _crypt_to64(p, l, 2); p += 2;
148  *p = '\0';
149 
150  /* Don't leave anything around in vm they could use. */
151  memset(final, 0, sizeof(final));
152 
153  return (passwd);
154 }
