242 static void thread_names_produce(
243  thread_names_context *ctx,
244  rtems_record_event event,
245  rtems_record_data data
246 )
247 {
248  size_t i;
249 
250  i = ctx->index;
251  ctx->items[ i ].event = RTEMS_RECORD_TIME_EVENT( 0, event );
252  ctx->items[ i ].data = data;
253 
254  if (i == RTEMS_ARRAY_SIZE(ctx->items) - 1) {
255  ctx->index = 0;
256  (void) write( ctx->fd, ctx->items, sizeof( ctx->items ) );
257  } else {
258  ctx->index = i + 1;
259  }
260 }

262 static bool thread_names_visitor( rtems_tcb *tcb, void *arg )
263 {
264  thread_names_context *ctx;
265  char name[ 2 * THREAD_DEFAULT_MAXIMUM_NAME_SIZE ];
266  size_t n;
267  size_t i;
268  rtems_record_data data;
269 
270  ctx = arg;
271  thread_names_produce( ctx, RTEMS_RECORD_THREAD_ID, tcb->Object.id );
272  n = _Thread_Get_name( tcb, name, sizeof( name ) );
273  i = 0;
274 
275  while ( i < n ) {
276  size_t j;
277 
278  data = 0;
279 
280  for ( j = 0; i < n && j < sizeof( data ); ++j ) {
281  rtems_record_data c;
282 
283  c = (unsigned char) name[ i ];
284  data |= c << ( j * 8 );
285  ++i;
286  }
287 
288  thread_names_produce( ctx, RTEMS_RECORD_THREAD_NAME, data );
289  }
290 
291  return false;
292 }

307 void rtems_record_server( uint16_t port, rtems_interval period )
308 {
309  rtems_status_code sc;
310  rtems_id self;
311  rtems_id timer;
312  struct sockaddr_in addr;
313  int sd;
314  int rv;
315 
316  sd = -1;
317  self = rtems_task_self();
318 
319  sc = rtems_timer_create( rtems_build_name( 'R', 'C', 'R', 'D' ), &timer );
320  if ( sc != RTEMS_SUCCESSFUL ) {
321  return;
322  }
323 
324  sd = socket( PF_INET, SOCK_STREAM, 0 );
325  if (sd < 0) {
326  goto error;
327  }
328 
329  memset( &addr, 0, sizeof( addr ) );
330  addr.sin_family = AF_INET;
331  addr.sin_port = htons( port );
332  addr.sin_addr.s_addr = htonl( INADDR_ANY );
333 
334  rv = bind( sd, (const struct sockaddr *) &addr, sizeof( addr ) );
335  if (rv != 0) {
336  goto error;
337  }
338 
339  rv = listen( sd, 0 );
340  if (rv != 0) {
341  goto error;
342  }
343 
344  while ( true ) {
345  int cd;
346  bool written;
347  ssize_t n;
348 
349  cd = accept( sd, NULL, NULL );
350 
351  if ( cd < 0 ) {
352  break;
353  }
354 
355  wait( RTEMS_NO_WAIT );
356  (void) rtems_timer_fire_after( timer, period, wakeup_timer, &self );
357  send_header( cd );
358  send_thread_names( cd );
359 
360  while ( true ) {
361  n = rtems_record_writev( cd, &written );
362 
363  if ( written && n <= 0 ) {
364  break;
365  }
366 
367  wait( RTEMS_WAIT );
368  }
369 
370  (void) rtems_timer_cancel( timer );
371  (void) close( cd );
372  }
373 
374 error:
375 
376  (void) close( sd );
377  (void) rtems_timer_delete( timer );
378 }
