56 {

60  bool terminate = false;
61 
62  /*
63  * scan through all fstab entries;
64  */
65  while (!terminate &&
66  (fstab_idx < fstab_count)) {
67  tmp_rc = 0;
68  /*
69  * create mount point
70  */
71  if (tmp_rc == 0) {
72  tmp_rc = rtems_mkdir(fstab_ptr->target, S_IRWXU | S_IRWXG | S_IRWXO);
73  if (tmp_rc != 0) {
74  if (0 != (fstab_ptr->report_reasons & FSMOUNT_MNTPNT_CRTERR)) {
75  fprintf(stdout,"fsmount: creation of mount point \"%s\" failed: %s\n",
76  fstab_ptr->target,
77  strerror(errno));
78  }
79  if (0 != (fstab_ptr->abort_reasons & FSMOUNT_MNTPNT_CRTERR)) {
80  terminate = true;
81  rc = tmp_rc;
82  }
83  }
84  }
85  /*
86  * mount device to given mount point
87  */
88  if (tmp_rc == 0) {
89  tmp_rc = mount(fstab_ptr->source,
90  fstab_ptr->target,
91  fstab_ptr->type,
92  fstab_ptr->options,
93  NULL);
94  if (tmp_rc != 0) {
95  if (0 != (fstab_ptr->report_reasons & FSMOUNT_MNT_FAILED)) {
96  fprintf(stdout,"fsmount: mounting of \"%s\" to"
97  " \"%s\" failed: %s\n",
98  fstab_ptr->source,
99  fstab_ptr->target,
100  strerror(errno));
101  }
102  if (0 != (fstab_ptr->abort_reasons & FSMOUNT_MNT_FAILED)) {
103  terminate = true;
104  rc = tmp_rc;
105  }
106  }
107  else {
108  if (0 != (fstab_ptr->report_reasons & FSMOUNT_MNT_OK)) {
109  fprintf(stdout,"fsmount: mounting of \"%s\" to"
110  " \"%s\" succeeded\n",
111  fstab_ptr->source,
112  fstab_ptr->target);
113  }
114  if (0 != (fstab_ptr->abort_reasons & FSMOUNT_MNT_OK)) {
115  terminate = true;
116  }
117  }
118  }
119  /*
120  * proceed to next entry
121  */
122  if (!terminate) {
123  fstab_ptr++;
124  fstab_idx++;
125  }
126  }
127  if (fail_idx != NULL) {
128  *fail_idx = fstab_idx;
129  }
