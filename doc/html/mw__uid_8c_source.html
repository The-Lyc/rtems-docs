90 {

93  int wait = (timeout != 0);

95 
96  if (timeout == (unsigned long) -1) {
97  ticks = RTEMS_NO_TIMEOUT;
98  } else if (timeout && ticks == 0) {

102  ticks = 1;
103  }
104 
105  status = rtems_message_queue_receive(

110  ticks
111  );
112
