16 /* This code is a crude memory manager for early boot for LinuxPPC.
17  * As such, it does not try to perform many optimiztions depending
18  * on the processor, it only uses features which are common to
19  * all processors (no BATs...).
20  *
21  * On PreP platorms (the only ones on which it works for now),
22  * it maps 1:1 all RAM/ROM and I/O space as claimed by the
23  * residual data. The holes between these areas can be virtually
24  * remapped to any of these, since for some functions it is very handy
25  * to have virtually contiguous but physically discontiguous memory.
26  *
27  * Physical memory allocation is also very crude, since it's only
28  * designed to manage a small number of large chunks. For valloc/vfree
29  * and palloc/pfree, the unit of allocation is the 4kB page.
30  *
31  * The salloc/sfree has been added after tracing gunzip and seeing
32  * how it performed a very large number of small allocations.
33  * For these the unit of allocation is 8 bytes (the s stands for
34  * small or subpage). This memory is cleared when allocated.
35  *
36  */

67 /* The LSB of the firstpte entries on map lists other than mappings
68  * are constants which can be checked for debugging. All these constants
69  * have bit of weight 4 set, this bit is zero in the mappings list entries.
70  * Actually firstpte&7 value is:
71  * - 0 or 1 should not happen
72  * - 2 for RW actual virtual->physical mappings
73  * - 3 for RO actual virtual->physical mappings
74  * - 6 for free areas to be suballocated by salloc
75  * - 7 for salloc'ated areas
76  * - 4 or 5 for all others, in this case firtpte & 63 is
77  * - 4 for unused maps (on the free list)
78  * - 12 for free physical memory
79  * - 13 for physical memory in use
80  * - 20 for free virtual address space
81  * - 21 for allocated virtual address space
82  * - 28 for physical memory space suballocated by salloc
83  * - 29 for physical memory that can't be freed
84  */

132 /* The handler used for all exceptions although for now it is only
133  * designed to properly handle MMU interrupts to fill the hash table.
134  */
135 void _handler(int vec, ctxt *p) {
136  map *area;
137  struct _mm_private *mm = (struct _mm_private *) bd->mm_private;
138  u_long vaddr, cause;
139  if (vec==4 || vec==7) { /* ISI exceptions are different */
140  vaddr = p->nip;
141  cause = p->msr;
142  } else { /* Valid for DSI and alignment exceptions */
143  vaddr = _read_PPC_DAR();
144  cause = _read_DSISR();
145  }
146 
147  if (vec==3 || vec==4) {
148  /* Panic if the fault is not PTE not found. */
149  if (!(cause & 0x40000000)) {
150  MMUon();
151  printk("\nPanic: vector=%x, cause=%lx\n", vec, cause);
152  hang("Memory protection violation at ", vaddr, p);
153  }
154 
155  for(area=mm->mappings; area; area=area->next) {
156  if(area->base<=vaddr && vaddr<=area->end) break;
157  }
158 
159  if (area) {
160  u_long hash, vsid, rpn;
161  hash_entry volatile *hte, *_hte1;
162  u_int i, alt=0, flushva;
163 
164  vsid = _read_SR((void *)vaddr);
165  rpn = (vaddr&PAGE_MASK)-area->base+area->firstpte;
166  hash = vsid<<6;
167  hash ^= (vaddr>>(PAGE_SHIFT-6))&0x3fffc0;
168  hash &= mm->hashmask;
169  /* Find an empty entry in the PTEG, else
170  * replace a random one.
171  */
172  hte = (hash_entry *) ((u_long)(mm->sdr1)+hash);
173  for (i=0; i<8; i++) {
174  if (hte[i].key>=0) goto found;
175  }
176  hash ^= mm->hashmask;
177  alt = 0x40; _hte1 = hte;
178  hte = (hash_entry *) ((u_long)(mm->sdr1)+hash);
179 
180  for (i=0; i<8; i++) {
181  if (hte[i].key>=0) goto found;
182  }
183  alt = 0;
184  hte = _hte1;
185  /* Chose a victim entry and replace it. There might be
186  * better policies to choose the victim, but in a boot
187  * loader we want simplicity as long as it works.
188  *
189  * We would not need to invalidate the TLB entry since
190  * the mapping is still valid. But this would be a mess
191  * when unmapping so we make sure that the TLB is a
192  * subset of the hash table under all circumstances.
193  */
194  i = mm->hashcnt;
195  mm->hashcnt = (mm->hashcnt+1)%8;
196  /* Note that the hash is already complemented here ! */
197  flushva = (~(hash<<9)^((hte[i].key)<<5)) &0x3ff000;
198  if (hte[i].key&0x40) flushva^=0x3ff000;
199  flushva |= ((hte[i].key<<21)&0xf0000000)
200  | ((hte[i].key<<22)&0x0fc00000);
201  hte[i].key=0;
202  asm volatile("sync; tlbie %0; sync" : : "r" (flushva));
203  found:
204  hte[i].rpn = rpn;
205  asm volatile("eieio": : );
206  hte[i].key = 0x80000000|(vsid<<7)|alt|
207  ((vaddr>>22)&0x3f);
208  return;
209  } else {
210  MMUon();
211  printk("\nPanic: vector=%x, cause=%lx\n", vec, cause);
212  hang("\nInvalid memory access attempt at ", vaddr, p);
213  }
214  } else {
215  MMUon();
216  printk(
217  "\nPanic: vector=%d, dsisr=%lx, faultaddr =%lx, "
218  "msr=%lx opcode=%x\n", vec,
219  cause, p->nip, p->msr, * ((unsigned int*) p->nip) );
220  if (vec == 7) {
221  unsigned int* ptr = ((unsigned int*) p->nip) - 4 * 10;
222  for (; ptr <= (((unsigned int*) p->nip) + 4 * 10); ptr ++)
223  printk("Hexdecimal code at address %p = %x\n", ptr, *ptr);
224  }
225  hang("Program or alignment exception at ", vaddr, p);
226  }
227 }

241 /* Sorted insertion in linked list */
242 static
243 int insert_map(map **head, map *p) {
244  map *q = *head;
245  if (!p) return 0;
246  if (q && (q->base < p->base)) {
247  for(;q->next && q->next->base<p->base; q = q->next);
248  if ((q->end >= p->base) ||
249  (q->next && p->end>=q->next->base)) {
250  free_map(p);
251  printk("Overlapping areas!\n");
252  return 1;
253  }
254  p->next = q->next;
255  q->next = p;
256  } else { /* Insert at head */
257  if (q && (p->end >= q->base)) {
258  free_map(p);
259  printk("Overlapping areas!\n");
260  return 1;
261  }
262  p->next = q;
263  *head = p;
264  }
265  return 0;
266 }

290  map *p, *q = *head;

303 static inline
304 map * alloc_map_page(void) {
305  map *from, *p;
306  struct _mm_private *mm = (struct _mm_private *) bd->mm_private;
307 
308  /* printk("Allocating new map page !"); */
309  /* Get the highest page */
310  for (from=mm->physavail; from && from->next; from=from->next);
311  if (!from) return NULL;
312 
313  from->end -= PAGE_SIZE;
314 
315  mm->freemaps = (map *) (from->end+1);
316 
317  for(p=mm->freemaps; p<mm->freemaps+PAGE_SIZE/sizeof(map)-1; p++) {
318  p->next = p+1;
319  p->firstpte = MAP_FREE;
320  }
321  (p-1)->next=0;
322 
323  /* Take the last one as pointer to self and insert
324  * the map into the permanent map list.
325  */
326 
327  p->firstpte = MAP_PERM_PHYS;
328  p->base=(u_long) mm->freemaps;
329  p->end = p->base+PAGE_SIZE-1;
330 
331  insert_map(&mm->physperm, p);
332 
333  if (from->end+1 == from->base)
334  free_map(remove_map(&mm->physavail, from));
335 
336  return mm->freemaps;
337 }

368 /* These routines are used to find the free memory zones to avoid
369  * overlapping destructive copies when initializing.
370  * They work from the top because of the way we want to boot.
371  * In the following the term zone refers to the memory described
372  * by one or several contiguous so called segments in the
373  * residual data.
374  */
375 #define STACK_PAGES 2
376 static inline u_long
377 find_next_zone(RESIDUAL *res, u_long lowpage, u_long flags) {
378  u_long i, newmin=0, size=0;
379  for(i=0; i<res->ActualNumMemSegs; i++) {
380  if (res->Segs[i].Usage & flags
381  && res->Segs[i].BasePage<lowpage
382  && res->Segs[i].BasePage>newmin) {
383  newmin=res->Segs[i].BasePage;
384  size=res->Segs[i].PageCount;
385  }
386  }
387  return newmin+size;
388 }

390 static inline u_long
391 find_zone_start(RESIDUAL *res, u_long highpage, u_long flags) {
392  u_long i;
393  int progress;
394  do {
395  progress=0;
396  for (i=0; i<res->ActualNumMemSegs; i++) {
397  if ( (res->Segs[i].BasePage+res->Segs[i].PageCount
398  == highpage)
399  && res->Segs[i].Usage & flags) {
400  highpage=res->Segs[i].BasePage;
401  progress=1;
402  }
403  }
404  } while(progress);
405  return highpage;
406 }

471 /* This routine is the first C code called with very little stack space!
472  * Its goal is to find where the boot image can be moved. This will
473  * be the highest address with enough room.
474  */
475 int early_setup(u_long image_size) {
476  register RESIDUAL *res = bd->residual;
477  u_long minpages = PAGE_ALIGN(image_size)>>PAGE_SHIFT;
478 
479  if ( residual_fw_is_qemu( res ) ) {
480  /* save command-line - QEMU firmware sets R6/R7 to
481  * commandline start/end (NON-PReP STD)
482  */
483  int len = bd->r7 - bd->r6;
484  if ( len > 0 ) {
485  if ( len > sizeof(bd->cmd_line) - 1 )
486  len = sizeof(bd->cmd_line) - 1;
487  codemove(bd->cmd_line, bd->r6, len, bd->cache_lsize);
488  bd->cmd_line[len] = 0;
489  }
490  }
491 
492  /* Fix residual if we are loaded by Motorola NT firmware */
493  if ( res && res->VitalProductData.FirmwareSupplier == 0x10000 )
494  fix_residual( res );
495 
496  /* FIXME: if OF we should do something different */
497  if( !bd->of_entry && res &&
498  res->ResidualLength <= sizeof(RESIDUAL) && res->Version == 0 ) {
499  u_long lowpage=ULONG_MAX, highpage;
500  u_long imghigh=0, stkhigh=0;
501  /* Find the highest and large enough contiguous zone
502  consisting of free and BootImage sections. */
503  /* Find 3 free areas of memory, one for the main image, one
504  * for the stack (STACK_PAGES), and page one to put the map
505  * structures. They are allocated from the top of memory.
506  * In most cases the stack will be put just below the image.
507  */
508  while((highpage =
509  find_next_zone(res, lowpage, BootImage|Free))) {
510  lowpage=find_zone_start(res, highpage, BootImage|Free);
511  if ((highpage-lowpage)>minpages &&
512  highpage>imghigh) {
513  imghigh=highpage;
514  highpage -=minpages;
515  }
516  if ((highpage-lowpage)>STACK_PAGES &&
517  highpage>stkhigh) {
518  stkhigh=highpage;
519  highpage-=STACK_PAGES;
520  }
521  }
522 
523  bd->image = (void *)((imghigh-minpages)<<PAGE_SHIFT);
524  bd->stack=(void *) (stkhigh<<PAGE_SHIFT);
525 
526  /* The code mover is put at the lowest possible place
527  * of free memory. If this corresponds to the loaded boot
528  * partition image it does not matter because it overrides
529  * the unused part of it (x86 code).
530  */
531  bd->mover=(void *) (lowpage<<PAGE_SHIFT);
532 
533  /* Let us flush the caches in all cases. After all it should
534  * not harm even on 601 and we don't care about performance.
535  * Right now it's easy since all processors have a line size
536  * of 32 bytes. Once again residual data has proved unreliable.
537  */
538  bd->cache_lsize = 32;
539  }
540  /* For now we always assume that it's succesful, we should
541  * handle better the case of insufficient memory.
542  */
543  return 0;
544 }

546 void * valloc(u_long size) {
547  map *p, *q;
548  struct _mm_private * mm = (struct _mm_private *) bd->mm_private;
549 
550  if (size==0) return NULL;
551  size=PAGE_ALIGN(size)-1;
552  for (p=mm->virtavail; p; p=p->next) {
553  if (p->base+size <= p->end) break;
554  }
555  if(!p) return NULL;
556  q=alloc_map();
557  q->base=p->base;
558  q->end=q->base+size;
559  q->firstpte=MAP_USED_VIRT;
560  insert_map(&mm->virtused, q);
561  if (q->end==p->end) free_map(remove_map(&mm->virtavail, p));
562  else p->base += size+1;
563  return (void *)q->base;
564 }

566 static
567 void vflush(map *virtmap) {
568  struct _mm_private * mm = (struct _mm_private *) bd->mm_private;
569  u_long i, limit=(mm->hashmask>>3)+8;
570  hash_entry volatile *p=(hash_entry *) mm->sdr1;
571 
572  /* PTE handling is simple since the processor never update
573  * the entries. Writable pages always have the C bit set and
574  * all valid entries have the R bit set. From the processor
575  * point of view the hash table is read only.
576  */
577  for (i=0; i<limit; i++) {
578  if (p[i].key<0) {
579  u_long va;
580  va = ((i<<9)^((p[i].key)<<5)) &0x3ff000;
581  if (p[i].key&0x40) va^=0x3ff000;
582  va |= ((p[i].key<<21)&0xf0000000)
583  | ((p[i].key<<22)&0x0fc00000);
584  if (va>=virtmap->base && va<=virtmap->end) {
585  p[i].key=0;
586  asm volatile("sync; tlbie %0; sync" : :
587  "r" (va));
588  }
589  }
590  }
591 }

593 void vfree(void *vaddr) {
594  map *physmap, *virtmap; /* Actual mappings pertaining to this vm */
595  struct _mm_private * mm = (struct _mm_private *) bd->mm_private;
596 
597  /* Flush memory queues */
598  asm volatile("sync": : : "memory");
599 
600  virtmap = remove_map_at(&mm->virtused, vaddr);
601  if (!virtmap) return;
602 
603  /* Remove mappings corresponding to virtmap */
604  for (physmap=mm->mappings; physmap; ) {
605  map *nextmap=physmap->next;
606  if (physmap->base>=virtmap->base
607  && physmap->base<virtmap->end) {
608  free_map(remove_map(&mm->mappings, physmap));
609  }
610  physmap=nextmap;
611  }
612 
613  vflush(virtmap);
614 
615  virtmap->firstpte= MAP_FREE_VIRT;
616  insert_map(&mm->virtavail, virtmap);
617  coalesce_maps(mm->virtavail);
618 }

620 void vunmap(void *vaddr) {
621  map *physmap, *virtmap; /* Actual mappings pertaining to this vm */
622  struct _mm_private *mm = (struct _mm_private *) bd->mm_private;
623 
624  /* Flush memory queues */
625  asm volatile("sync": : : "memory");
626 
627  /* vaddr must be within one of the vm areas in use and
628  * then must correspond to one of the physical areas
629  */
630  for (virtmap=mm->virtused; virtmap; virtmap=virtmap->next) {
631  if (virtmap->base<=(u_long)vaddr &&
632  virtmap->end>=(u_long)vaddr) break;
633  }
634  if (!virtmap) return;

722 void mm_init(u_long image_size)
723 {
724  u_long lowpage=ULONG_MAX, highpage;
725  struct _mm_private *mm = (struct _mm_private *) bd->mm_private;
726  RESIDUAL * res=bd->residual;
727  int i;
728  map *p;
729 
730  /* The checks are simplified by the fact that the image
731  * and stack area are always allocated at the upper end
732  * of a free block.
733  */
734  while((highpage = find_next_zone(res, lowpage, BootImage|Free))) {
735  lowpage=find_zone_start(res, highpage, BootImage|Free);
736  if ( ( ((u_long)bd->image+PAGE_ALIGN(image_size))>>PAGE_SHIFT)
737  == highpage) {
738  highpage=(u_long)(bd->image)>>PAGE_SHIFT;
739  add_perm_map((u_long)bd->image, image_size);
740  }
741  if ( (( u_long)bd->stack>>PAGE_SHIFT) == highpage) {
742  highpage -= STACK_PAGES;
743  add_perm_map(highpage<<PAGE_SHIFT,
744  STACK_PAGES*PAGE_SIZE);
745  }
746  /* Protect the interrupt handlers that we need ! */
747  if (lowpage<2) lowpage=2;
748  /* Check for the special case of full area! */
749  if (highpage>lowpage) {
750  p = alloc_map();
751  p->base = lowpage<<PAGE_SHIFT;
752  p->end = (highpage<<PAGE_SHIFT)-1;
753  p->firstpte=MAP_FREE_PHYS;
754  insert_map(&mm->physavail, p);
755  }
756  }
757 
758  /* Allocate the hash table */
759  mm->sdr1=__palloc(0x10000, PA_PERM|16);
760  _write_SDR1((u_long)mm->sdr1);
761  memset(mm->sdr1, 0, 0x10000);
762  mm->hashmask = 0xffc0;
763 
764  /* Setup the segment registers as we want them */
765  for (i=0; i<16; i++) _write_SR(i, (void *)(i<<28));
766  /* Create the maps for the physical memory, firwmarecode does not
767  * seem to be necessary. ROM is mapped read-only to reduce the risk
768  * of reprogramming it because it's often Flash and some are
769  * amazingly easy to overwrite.
770  */
771  create_identity_mappings(BootImage|Free|FirmwareCode|FirmwareHeap|
772  FirmwareStack, PTE_RAM);
773  create_identity_mappings(SystemROM, PTE_ROM);
774  create_identity_mappings(IOMemory|SystemIO|SystemRegs|
775  PCIAddr|PCIConfig|ISAAddr, PTE_IO);

780  codemove((void *) 0x300, _handler_glue, 0x100, bd->cache_lsize);
781  codemove((void *) 0x400, _handler_glue, 0x100, bd->cache_lsize);
782  codemove((void *) 0x600, _handler_glue, 0x100, bd->cache_lsize);

831 /* first/last area fit, flags is a power of 2 indicating the required
832  * alignment. The algorithms are stupid because we expect very little
833  * fragmentation of the areas, if any. The unit of allocation is the page.
834  * The allocation is by default performed from higher addresses down,
835  * unless flags&PA_LOW is true.
836  */
837 
838 void * __palloc(u_long size, int flags)
839 {
840  u_long mask = ((1<<(flags&PA_ALIGN_MASK))-1);
841  map *newmap, *frommap, *p, *splitmap=0;
842  map **queue;
843  u_long qflags;
844  struct _mm_private *mm = (struct _mm_private *) bd->mm_private;
845 
846  /* Asking for a size which is not a multiple of the alignment
847  is likely to be an error. */
848 
849  if (size & mask) return NULL;
850  size = PAGE_ALIGN(size);
851  if(!size) return NULL;
852 
853  if (flags&PA_SUBALLOC) {
854  queue = &mm->sallocphys;
855  qflags = MAP_SUBS_PHYS;
856  } else if (flags&PA_PERM) {
857  queue = &mm->physperm;
858  qflags = MAP_PERM_PHYS;
859  } else {
860  queue = &mm->physused;
861  qflags = MAP_USED_PHYS;
862  }
863  /* We need to allocate that one now so no two allocations may attempt
864  * to take the same memory simultaneously. Alloc_map_page does
865  * not call back here to avoid infinite recursion in alloc_map.
866  */
867 
868  if (mask&PAGE_MASK) {
869  splitmap=alloc_map();
870  if (!splitmap) return NULL;
871  }
872 
873  for (p=mm->physavail, frommap=NULL; p; p=p->next) {
874  u_long high = p->end;
875  u_long limit = ((p->base+mask)&~mask) + size-1;
876  if (high>=limit && ((p->base+mask)&~mask)+size>p->base) {
877  frommap = p;
878  if (flags&PA_LOW) break;
879  }
880  }
881 
882  if (!frommap) {
883  if (splitmap) free_map(splitmap);
884  return NULL;
885  }
886 
887  newmap=alloc_map();
888 
889  if (flags&PA_LOW) {
890  newmap->base = (frommap->base+mask)&~mask;
891  } else {
892  newmap->base = (frommap->end +1 - size) & ~mask;
893  }
894 
895  newmap->end = newmap->base+size-1;
896  newmap->firstpte = qflags;
897 
898  /* Add a fragment if we don't allocate until the end. */
899 
900  if (splitmap) {
901  splitmap->base=newmap->base+size;
902  splitmap->end=frommap->end;
903  splitmap->firstpte= MAP_FREE_PHYS;
904  frommap->end=newmap->base-1;
905  } else if (flags & PA_LOW) {
906  frommap->base=newmap->base+size;
907  } else {
908  frommap->end=newmap->base-1;
909  }
910 
911  /* Remove a fragment if it becomes empty. */
912  if (frommap->base == frommap->end+1) {
913  free_map(remove_map(&mm->physavail, frommap));
914  }
915 
916  if (splitmap) {
917  if (splitmap->base == splitmap->end+1) {
918  free_map(remove_map(&mm->physavail, splitmap));
919  } else {
920  insert_map(&mm->physavail, splitmap);
921  }
922  }
923 
924  insert_map(queue, newmap);
925  return (void *) newmap->base;
926 
927 }
