89

93  return false;
94  }
95 
96  if (cache->fd == fd)
97  {
98  if (offset >= cache->file_size)
99  {
100  rtems_rtl_set_error (EINVAL, "offset past end of file: offset=%i size=%i",
101  (int) offset, (int) cache->file_size);
102  return false;
103  }
104 
105  /*
106  * We sometimes are asked to read strings of a length we do not know.
107  */
108  if ((offset + *length) > cache->file_size)
109  {
110  *length = cache->file_size - offset;
111  if (rtems_rtl_trace (RTEMS_RTL_TRACE_CACHE))
112  printf ("rtl: cache: %2d: truncate length=%d\n", fd, (int) *length);
113 
114  }
115  }
116 
117  while (true)
118  {
119  size_t buffer_offset = 0;
120  size_t buffer_read = cache->size;
121 
122  /*
123  * Is the data in the cache for this file ?
124  */
125  if (fd == cache->fd)
126  {
127  /*
128  * Do not read past the end of the file.
129  */
130  if ((offset + buffer_read) > cache->file_size)
131  buffer_read = cache->file_size - offset;
132 
133  /*
134  * Is any part of the data in the cache ?
135  */
136  if ((offset >= cache->offset) &&
137  (offset < (cache->offset + cache->level)))
138  {
139  size_t size;
140 
141  buffer_offset = offset - cache->offset;
142  size = cache->level - buffer_offset;
143 
144  /*
145  * Return the location of the data in the cache.
146  */
147  *buffer = cache->buffer + buffer_offset;
148 
149  /*
150  * Is all the data in the cache or just a part ?
151  */
152  if (*length <= size)
153  return true;
154 
155  if (rtems_rtl_trace (RTEMS_RTL_TRACE_CACHE))
156  printf ("rtl: cache: %2d: copy-down: buffer_offset=%d size=%d level=%d\n",
157  fd, (int) buffer_offset, (int) size, (int) cache->level);
158 
159  /*
160  * Copy down the data in the buffer and then fill the remaining space
161  * with as much data we are able to read.
162  */
163  memmove (cache->buffer, cache->buffer + buffer_offset, size);
164 
165  cache->offset = offset;
166  cache->level = size;
167  buffer_read = cache->size - cache->level;
168  buffer_offset = size;
169 
170  /*
171  * Do not read past the end of the file.
172  */
173  if ((offset + buffer_offset + buffer_read) > cache->file_size)
174  buffer_read = cache->file_size - (offset + buffer_offset);
175  }
176  }
177 
178  if (rtems_rtl_trace (RTEMS_RTL_TRACE_CACHE))
179  printf ("rtl: cache: %2d: seek: offset=%" PRIdoff_t " buffer_offset=%zu"
180  " read=%zu cache=[%" PRIdoff_t ",%" PRIdoff_t "] "
181  "dist=%" PRIdoff_t "\n",
182  fd, offset + buffer_offset, buffer_offset, buffer_read,
183  offset, offset + buffer_read,
184  (cache->file_size - offset));
185 
186  if (lseek (fd, offset + buffer_offset, SEEK_SET) < 0)
187  {
188  rtems_rtl_set_error (errno, "file seek failed");
189  return false;
190  }
191 
192  /*
193  * Loop reading the data from the file until either an error or 0 is
194  * returned and if data has been read check if the amount is what we
195  * want. If not it is an error. A POSIX read can read data in fragments.
196  */
197 
198  cache->level = buffer_offset + buffer_read;
199 
200  while (buffer_read)
201  {
202  int r = read (fd, cache->buffer + buffer_offset, buffer_read);
203  if (r < 0)
204  {
205  rtems_rtl_set_error (errno, "file read failed");
206  return false;
207  }
208  if ((r == 0) && buffer_read)
209  {
210  if (rtems_rtl_trace (RTEMS_RTL_TRACE_CACHE))
211  printf ("rtl: cache: %2d: read: past end by=%d\n", fd, (int) buffer_read);
212  cache->level = cache->level - buffer_read;
213  buffer_read = 0;
214  }
215  else
216  {
217  buffer_read -= r;
218  buffer_offset += r;
219  }
220  }
221 
222  cache->offset = offset;
223 
224  if (cache->fd != fd)
225  {
226  cache->fd = fd;
227 
228  if (fstat (cache->fd, &sb) < 0)
229  {
230  rtems_rtl_set_error (errno, "file stat failed");
231  return false;
232  }
233 
234  cache->file_size = sb.st_size;
235  }
236  }
237 
238  return false;
239 }
