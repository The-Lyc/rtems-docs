116 static void ComputeLockRange(uint32_t dwStart, uint32_t dwEnd,
117  uint32_t *pdwActualStart, uint32_t *pdwActualEnd)
118 {
119  Efc *pStartEfc;
120  Efc *pEndEfc;
121  uint16_t wStartPage;
122  uint16_t wEndPage;
123  uint16_t wNumPagesInRegion;
124  uint16_t wActualStartPage;
125  uint16_t wActualEndPage;
126 
127  /* Convert start and end address in page numbers */
128  EFC_TranslateAddress(&pStartEfc, dwStart, &wStartPage, 0);
129  EFC_TranslateAddress(&pEndEfc, dwEnd, &wEndPage, 0);
130 
131  /* Find out the first page of the first region to lock */
132  wNumPagesInRegion = IFLASH_LOCK_REGION_SIZE / IFLASH_PAGE_SIZE;
133  wActualStartPage = wStartPage - (wStartPage % wNumPagesInRegion);
134  wActualEndPage = wEndPage;
135 
136  if ((wEndPage % wNumPagesInRegion) != 0)
137  wActualEndPage += wNumPagesInRegion - (wEndPage % wNumPagesInRegion);
138 
139  /* Store actual page numbers */
140  EFC_ComputeAddress(pStartEfc, wActualStartPage, 0, pdwActualStart);
141  EFC_ComputeAddress(pEndEfc, wActualEndPage, 0, pdwActualEnd);
142  TRACE_DEBUG("Actual lock range is 0x%06X - 0x%06X\n\r",
143  (unsigned int)*pdwActualStart, (unsigned int)*pdwActualEnd);
144 }

222 extern uint32_t FLASHD_ErasePages(uint32_t dwAddress, uint32_t dwPageNum)
223 {
224  Efc *pEfc;
225  uint16_t wPage;
226  uint16_t wOffset;
227  uint32_t dwError;
228  static uint32_t dwFarg;
229 
230  assert((dwAddress >= IFLASH_ADDR)
231  || (dwAddress <= (IFLASH_ADDR + IFLASH_SIZE)));
232 
233  /* Translate write address */
234  EFC_TranslateAddress(&pEfc, dwAddress, &wPage, &wOffset);
235 
236  /* Get FARG field for EPA command:
237  * The first page to be erased is specified in the FARG[15:2] field of
238  * the MC_FCR register. The first page number must be modulo 4, 8,16 or 32
239  * according to the number of pages to erase at the same time.
240  *
241  * The 2 lowest bits of the FARG field define the number of pages to
242  * be erased (FARG[1:0]).
243  */
244  if (dwPageNum == 32) {
245  wPage &= ~(32u - 1u);
246  dwFarg = (wPage) | 3; /* 32 pages */
247  } else if (dwPageNum == 16) {
248  wPage &= ~(16u - 1u);
249  dwFarg = (wPage) | 2; /* 16 pages */
250  } else if (dwPageNum == 8) {
251  wPage &= ~(8u - 1u);
252  dwFarg = (wPage) | 1; /* 8 pages */
253  } else {
254  wPage &= ~(4u - 1u);
255  dwFarg = (wPage) | 0; /* 4 pages */
256  }
257 
258  dwError = EFC_PerformCommand(pEfc, EFC_FCMD_EPA, dwFarg, _dwUseIAP);
259 
260  return dwError;
261 }

274 extern uint32_t FLASHD_Write(uint32_t dwAddress,
275  const void *pvBuffer, uint32_t dwSize)
276 {
277  Efc *pEfc;
278  uint16_t page;
279  uint16_t offset;
280  uint32_t writeSize;
281  uint32_t pageAddress;
282  uint16_t padding;
283  uint32_t dwError;
284  uint32_t dwIdx;
285  uint32_t *pAlignedDestination;
286  uint8_t *pucPageBuffer = (uint8_t *)_pdwPageBuffer;
287 
288  assert(pvBuffer);
289  assert(dwAddress >= IFLASH_ADDR);
290  assert((dwAddress + dwSize) <= (IFLASH_ADDR + IFLASH_SIZE));
291 
292  /* Translate write address */
293  EFC_TranslateAddress(&pEfc, dwAddress, &page, &offset);
294 
295  /* Write all pages */
296  while (dwSize > 0) {
297  /* Copy data in temporary buffer to avoid alignment problems */
298  writeSize = min((uint32_t)IFLASH_PAGE_SIZE - offset, dwSize);
299  EFC_ComputeAddress(pEfc, page, 0, &pageAddress);
300  padding = IFLASH_PAGE_SIZE - offset - writeSize;
301 
302  /* Pre-buffer data */
303  memcpy(pucPageBuffer, (void *) pageAddress, offset);
304 
305  /* Buffer data */
306  memcpy(pucPageBuffer + offset, pvBuffer, writeSize);
307 
308  /* Post-buffer data */
309  memcpy(pucPageBuffer + offset + writeSize,
310  (void *) (pageAddress + offset + writeSize), padding);
311 
312  /* Write page
313  * Writing 8-bit and 16-bit data is not allowed and may
314  lead to unpredictable data corruption
315  */
316  pAlignedDestination = (uint32_t *)pageAddress;
317 
318  for (dwIdx = 0; dwIdx < (IFLASH_PAGE_SIZE / sizeof(uint32_t)); ++ dwIdx) {
319  *pAlignedDestination++ = _pdwPageBuffer[dwIdx];
320  memory_barrier()
321  }
322 
323  /* Cache coherence operation before flash write*/
324  SCB_CleanDCache_by_Addr((uint32_t *)pageAddress, IFLASH_PAGE_SIZE);
325 
326  /* Note: It is not possible to use Erase and write Command (EWP) on all Flash
327  (this command is available on the First 2 Small Sector, 16K Bytes).
328  For the next block, Erase them first then use Write page command. */
329 
330  /* Send writing command */
331  dwError = EFC_PerformCommand(pEfc, EFC_FCMD_WP, page, _dwUseIAP);
332 
333  if (dwError)
334  return dwError;
335 
336  /* Progression */
337  pvBuffer = (void *)((uint32_t) pvBuffer + writeSize);
338  dwSize -= writeSize;
339  page++;
340  offset = 0;
341  }
342 
343  return 0;
344 }

356 extern uint32_t FLASHD_Lock(uint32_t start, uint32_t end,
357  uint32_t *pActualStart, uint32_t *pActualEnd)
358 {
359  Efc *pEfc;
360  uint32_t actualStart, actualEnd;
361  uint16_t startPage, endPage;
362  uint32_t dwError;
363  uint16_t numPagesInRegion = IFLASH_LOCK_REGION_SIZE / IFLASH_PAGE_SIZE;
364 
365  /* Compute actual lock range and store it */
366  ComputeLockRange(start, end, &actualStart, &actualEnd);
367 
368  if (pActualStart != NULL)
369  *pActualStart = actualStart;
370 
371  if (pActualEnd != NULL)
372  *pActualEnd = actualEnd;
373 
374  /* Compute page numbers */
375  EFC_TranslateAddress(&pEfc, actualStart, &startPage, 0);
376  EFC_TranslateAddress(0, actualEnd, &endPage, 0);
377 
378  /* Lock all pages */
379  while (startPage < endPage) {
380  dwError = EFC_PerformCommand(pEfc, EFC_FCMD_SLB, startPage, _dwUseIAP);
381 
382  if (dwError)
383  return dwError;
384 
385  startPage += numPagesInRegion;
386  }
387 
388  return 0;
389 }

441 extern uint32_t FLASHD_IsLocked(uint32_t start, uint32_t end)
442 {
443  uint32_t i, j;
444  Efc *pEfc;
445  uint16_t startPage, endPage;
446  uint8_t startRegion, endRegion;
447  uint32_t numPagesInRegion;
448  uint32_t status[IFLASH_NB_OF_LOCK_BITS / 32u];
449  uint32_t numLockedRegions = 0;
450 
451  assert(end >= start);
452  assert((start >= IFLASH_ADDR) && (end <= IFLASH_ADDR + IFLASH_SIZE));
453 
454  /* Compute page numbers */
455  EFC_TranslateAddress(&pEfc, start, &startPage, 0);
456  EFC_TranslateAddress(0, end, &endPage, 0);
457 
458  /* Compute region numbers */
459  numPagesInRegion = IFLASH_LOCK_REGION_SIZE / IFLASH_PAGE_SIZE;
460  startRegion = startPage / numPagesInRegion;
461  endRegion = endPage / numPagesInRegion;
462 
463  if ((endPage % numPagesInRegion) != 0)
464  endRegion++;
465 
466  /* Retrieve lock status */
467  EFC_PerformCommand(pEfc, EFC_FCMD_GLB, 0, _dwUseIAP);
468 
469  for (i = 0; i < (IFLASH_NB_OF_LOCK_BITS / 32u); i++)
470  status[i] = EFC_GetResult(pEfc);
471 
472  /* Check status of each involved region */
473  while (startRegion < endRegion) {
474  i = startRegion / 32u;
475  j = startRegion % 32u;
476 
477  if ((status[i] & (1 << j)) != 0)
478  numLockedRegions++;
479 
480  startRegion++;
481  }
482 
483  return numLockedRegions;
484 }

547 #ifdef __ICCARM__
548  extern __ramfunc uint32_t FLASHD_ReadUniqueID(uint32_t *pdwUniqueID)
549 #else
550  __attribute__ ((section (".ramfunc")))
551  uint32_t FLASHD_ReadUniqueID(uint32_t *pdwUniqueID)
552 #endif
553 {
554  uint32_t status;
555 
556  if (pdwUniqueID == NULL)
557  return 1;
558 
559  pdwUniqueID[0] = 0;
560  pdwUniqueID[1] = 0;
561  pdwUniqueID[2] = 0;
562  pdwUniqueID[3] = 0;
563 
564  /* Send the Start Read unique Identifier command (STUI) by writing the Flash
565  Command Register with the STUI command.*/
566  EFC->EEFC_FCR = EEFC_FCR_FKEY_PASSWD | EFC_FCMD_STUI;
567 
568  /* When the Unique Identifier is ready to be read, the FRDY bit in the Flash
569  Programming Status Register (EEFC_FSR) falls. */
570  do {
571  status = EFC->EEFC_FSR;
572  } while ((status & EEFC_FSR_FRDY) == EEFC_FSR_FRDY);
573 
574  /* The Unique Identifier is located in the first 128 bits of the Flash
575  memory mapping. So, at the address 0x400000-0x40000F. */
576  pdwUniqueID[0] = *(uint32_t *)IFLASH_ADDR;
577  pdwUniqueID[1] = *(uint32_t *)(IFLASH_ADDR + 4);
578  pdwUniqueID[2] = *(uint32_t *)(IFLASH_ADDR + 8);
579  pdwUniqueID[3] = *(uint32_t *)(IFLASH_ADDR + 12);
580 
581  /* To stop the Unique Identifier mode, the user needs to send the Stop Read
582  unique Identifier command (SPUI) by writing the Flash Command Register
583  with the SPUI command. */
584  EFC->EEFC_FCR = EEFC_FCR_FKEY_PASSWD | EFC_FCMD_SPUI;
585 
586  /* When the Stop read Unique Unique Identifier command (SPUI) has been
587  performed, the FRDY bit in the Flash Programming Status Register (EEFC_FSR)
588  rises. */
589  do {
590  status = EFC->EEFC_FSR;
591  } while ((status & EEFC_FSR_FRDY) != EEFC_FSR_FRDY);
592 
593  return 0;
594 }
