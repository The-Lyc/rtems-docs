255 rtems_trace_buffering_shell_trace (int argc, char *argv[])
256 {
257  uint32_t* trace_buffer;
258  uint32_t records;
259  uint32_t traces;
260  uint32_t r;
261  size_t start = 0;
262  size_t end = 40;
263  size_t count;
264  uint64_t last_sample = 0;
265 
266  if (!rtems_trace_buffering_present ())
267  return rtems_trace_buffering_no_trace_buffer_code ();
268 
269  trace_buffer = rtems_trace_buffering_buffer ();
270  records = rtems_trace_buffering_buffer_in ();
271  traces = rtems_trace_names_size ();
272 
273  if (argc > 1)
274  {
275  if (argc > 3)
276  return rtems_trace_buffering_wrong_number_of_args ();
277 
278  if (argv[1][0] == '+')
279  {
280  if (argc > 2)
281  return rtems_trace_buffering_wrong_number_of_args ();
282  end = strtoul (argv[1] + 1, 0, 0);
283  if (end == 0)
284  {
285  printf("error: invalid number of lines\n");
286  return 1;
287  }
288  ++end;
289  }
290  else
291  {
292  start = strtoul (argv[1], 0, 0);
293  if (start >= records)
294  {
295  printf ("error: start record out of range (max %" PRIu32 ")\n", records);
296  return 1;
297  }
298  }
299 
300  end += start;
301 
302  if (argc == 3)
303  {
304  if (argv[2][0] == '+')
305  {
306  end = strtoul (argv[2] + 1, 0, 0);
307  if (end == 0)
308  {
309  printf("error: invalid number of lines\n");
310  return 1;
311  }
312  end += start + 1;
313  }
314  else
315  {
316  end = strtoul (argv[2], 0, 0);
317  if (end < start)
318  {
319  printf ("error: end record before start\n");
320  return 1;
321  }
322  else if (end > records)
323  {
324  printf ("error: end record out of range (max %" PRIu32 ")\n", records);
325  }
326  }
327  }
328  }
329 
330  rtems_trace_buffering_banner ("trace");
331 
332  if (!rtems_trace_buffering_finished ())
333  {
334  printf("tracing still running\n");
335  return 0;
336  }
337 
338  printf(" Trace buffer: %p\n", trace_buffer);
339  printf(" Words traced: %" PRIu32 "\n", records);
340  printf(" Traces: %" PRIu32 "\n", traces);
341 
342  count = 0;
343  r = 0;
344 
345  while ((r < records) && (count < end))
346  {
347  const uint32_t header = trace_buffer[r];
348  const uint32_t func_index = header & 0xffff;
349  const uint32_t len = (header >> 16) & 0x0fff;
350  const uint32_t task_id = trace_buffer[r + 1];
351  const uint32_t task_status = trace_buffer[r + 2];
352  const uint64_t when = (((uint64_t) trace_buffer[r + 4]) << 32) | trace_buffer[r + 5];
353  const uint8_t* argv = (uint8_t*) &trace_buffer[r + 6];
354  const bool ret = (header & (1 << 30)) == 0 ? false : true;
355  const bool irq = (header & (1 << 31)) == 0 ? false : true;
356 
357  if (count > start)
358  {
359  const rtems_trace_sig* sig = rtems_trace_signatures (func_index);
360 
361  rtems_trace_buffering_print_timestamp (when);
362  printf (" %10" PRIu32 " %c%08" PRIx32 " [%3" PRIu32 "/%3" PRIu32 "] %c %s",
363  (uint32_t) (when - last_sample),
364  irq ? '*' : ' ', task_id, (task_status >> 8) & 0xff, task_status & 0xff,
365  ret ? '<' : '>', rtems_trace_names (func_index));
366 
367  if (sig->argc)
368  {
369  if (ret)
370  {
371  if (sig->args[0].size)
372  printf(" => ");
373  rtems_trace_buffering_print_arg (&sig->args[0], argv);
374  }
375  else
376  {
377  size_t a;
378  printf("(");
379  for (a = 1; a < sig->argc; ++a)
380  {
381  if (a > 1)
382  printf (", ");
383  rtems_trace_buffering_print_arg (&sig->args[a], argv);
384  argv += sig->args[a].size;
385  }
386  printf(")");
387  }
388  }
389 
390  printf("\n");
391  }
392 
393  r += ((len - 1) / sizeof (uint32_t)) + 1;
394  last_sample = when;
395  ++count;
396  }
397 
398  return 0;
399 }

683  return r;
684  }
685  }
