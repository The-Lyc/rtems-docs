63 /*
64  * The `ID sequence' is really just snapshots of an 8-bit CRC register as 0
65  * bits are shifted in. Different board types use different polynomials.
66  */
67 void
68 elink_idseq(u_char p)
69 {
70  register int i;
71  register u_char c;
72 
73  c = 0xff;
74  for (i = 255; i; i--) {
75  outb(ELINK_ID_PORT, c);
76  if (c & 0x80) {
77  c <<= 1;
78  c ^= p;
79  } else
80  c <<= 1;
81  }
82 }
