54  for(j=0; j< 7-strlen(dumpregs[i].name); j++) printk(" ");
55 #if (__mips == 1 ) || (__mips == 32)

58 #elif __mips == 3
59  printk(" %08" PRIu32 "", frame_u32[2 * dumpregs[i].offset + 1] );
