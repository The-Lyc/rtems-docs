109 static ssize_t
110 rtems_rfs_rtems_file_read (rtems_libio_t* iop,
111  void* buffer,
112  size_t count)
113 {
114  rtems_rfs_file_handle* file = rtems_rfs_rtems_get_iop_file_handle (iop);
115  rtems_rfs_pos pos;
116  uint8_t* data = buffer;
117  ssize_t read = 0;
118  int rc;
119 
120  if (rtems_rfs_rtems_trace (RTEMS_RFS_RTEMS_DEBUG_FILE_READ))
121  printf("rtems-rfs: file-read: handle:%p count:%zd\n", file, count);
122 
123  rtems_rfs_rtems_lock (rtems_rfs_file_fs (file));
124 
125  pos = iop->offset;
126 
127  if (pos < rtems_rfs_file_size (file))
128  {
129  while (count)
130  {
131  size_t size;
132 
133  rc = rtems_rfs_file_io_start (file, &size, true);
134  if (rc > 0)
135  {
136  read = rtems_rfs_rtems_error ("file-read: read: io-start", rc);
137  break;
138  }
139 
140  if (size == 0)
141  break;
142 
143  if (size > count)
144  size = count;
145 
146  memcpy (data, rtems_rfs_file_data (file), size);
147 
148  data += size;
149  count -= size;
150  read += size;
151 
152  rc = rtems_rfs_file_io_end (file, size, true);
153  if (rc > 0)
154  {
155  read = rtems_rfs_rtems_error ("file-read: read: io-end", rc);
156  break;
157  }
158  }
159  }
160 
161  if (read >= 0)
162  iop->offset = pos + read;
163 
164  rtems_rfs_rtems_unlock (rtems_rfs_file_fs (file));
165 
166  return read;
167 }

177 static ssize_t
178 rtems_rfs_rtems_file_write (rtems_libio_t* iop,
179  const void* buffer,
180  size_t count)
181 {
182  rtems_rfs_file_handle* file = rtems_rfs_rtems_get_iop_file_handle (iop);
183  rtems_rfs_pos pos;
184  rtems_rfs_pos file_size;
185  const uint8_t* data = buffer;
186  ssize_t write = 0;
187  int rc;
188 
189  if (rtems_rfs_rtems_trace (RTEMS_RFS_RTEMS_DEBUG_FILE_WRITE))
190  printf("rtems-rfs: file-write: handle:%p count:%zd\n", file, count);
191 
192  rtems_rfs_rtems_lock (rtems_rfs_file_fs (file));
193 
194  pos = iop->offset;
195  file_size = rtems_rfs_file_size (file);
196  if (pos > file_size)
197  {
198  /*
199  * If the iop position is past the physical end of the file we need to set
200  * the file size to the new length before writing. The
201  * rtems_rfs_file_io_end() will grow the file subsequently.
202  */
203  rc = rtems_rfs_file_set_size (file, pos);
204  if (rc)
205  {
206  rtems_rfs_rtems_unlock (rtems_rfs_file_fs (file));
207  return rtems_rfs_rtems_error ("file-write: write extend", rc);
208  }
209 
210  rtems_rfs_file_set_bpos (file, pos);
211  }
212  else if (pos < file_size && rtems_libio_iop_is_append(iop))
213  {
214  pos = file_size;
215  rc = rtems_rfs_file_seek (file, pos, &pos);
216  if (rc)
217  {
218  rtems_rfs_rtems_unlock (rtems_rfs_file_fs (file));
219  return rtems_rfs_rtems_error ("file-write: write append seek", rc);
220  }
221  }
222 
223  while (count)
224  {
225  size_t size = count;
226 
227  rc = rtems_rfs_file_io_start (file, &size, false);
228  if (rc)
229  {
230  /*
231  * If we have run out of space and have written some data return that
232  * amount first as the inode will have accounted for it. This means
233  * there was no error and the return code from can be ignored.
234  */
235  if (!write)
236  write = rtems_rfs_rtems_error ("file-write: write open", rc);
237  break;
238  }
239 
240  if (size > count)
241  size = count;
242 
243  memcpy (rtems_rfs_file_data (file), data, size);
244 
245  data += size;
246  count -= size;
247  write += size;
248 
249  rc = rtems_rfs_file_io_end (file, size, false);
250  if (rc)
251  {
252  write = rtems_rfs_rtems_error ("file-write: write close", rc);
253  break;
254  }
255  }
256 
257  if (write >= 0)
258  iop->offset = pos + write;
259 
260  rtems_rfs_rtems_unlock (rtems_rfs_file_fs (file));
261 
262  return write;
263 }

293

300  }
301 
302  rtems_rfs_rtems_unlock (rtems_rfs_file_fs (file));
303 
304  return new_offset;
