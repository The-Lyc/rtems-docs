210 {
211  rtems_status_code sc = RTEMS_SUCCESSFUL;
212  int arg_error = 0;
213  int i;
214  char *tmp_ptr;
215  int nm_argc = 0;
216  /*
217  * set filter and selector to default
218  */
219  memset(filter,0,sizeof(*filter));

221  memset(selector,0,sizeof(*selector));
222  selector->show_all = true;
223 
224  /*
225  * scan arguments
226  */
227  for (i = 1;
228  (i < argc) && (arg_error == 0);
229  i++) {
230  if (argv[i][0] == '-') {
231  /*
232  * modifier arguments
233  */
234  switch(tolower(argv[i][1])) {
235  /*
236  * selection, which bdbufs to show
237  */
238  case 'm': /* only show bdbufs modified */
239  filter->bdbuf_modified = true ;
240  filter->bdbuf_all = false;
241  break;
242  case 'i': /* only show bdbufs in progress*/
243  filter->bdbuf_in_progress = true ;
244  filter->bdbuf_all = false;
245  break;
246  case 'v': /* only show bdbufs, which have valid data*/
247  filter->bdbuf_actual = true ;
248  filter->bdbuf_all = false;
249  break;
250  case 'u': /* only show bdbufs, which are in use */
251  filter->bdbuf_used = true ;
252  filter->bdbuf_all = false;
253  break;
254  case 'p': /* only show bdbufs, which belong to pool <n> */
255  filter->pool_id = strtol(argv[i]+2,&tmp_ptr,0);
256  if (tmp_ptr == argv[i]+2) { /* no conversion performed... */
257  arg_error = i;
258  }
259  filter->bdbuf_all = false;
260  break;
261  /*
262  * selection, what fields to show
263  */
264  case 'n': /* show bdbuf node_chain */
265  selector->show_node_chain = true ;
266  selector->show_all = false;
267  break;
268  case 'd': /* show device */
269  selector->show_dev = true ;
270  selector->show_all = false;
271  break;
272  case 'b': /* show blocknum */
273  selector->show_blocknum = true ;
274  selector->show_all = false;
275  break;
276  case 'e': /* show bdbuf error status */
277  selector->show_error = true ;
278  selector->show_all = false;
279  break;
280  case 's': /* show bdbuf state */
281  selector->show_state = true ;
282  selector->show_all = false;
283  break;
284  case 'c': /* show bdbuf use count */
285  selector->show_use_count = true ;
286  selector->show_all = false;
287  break;
288  case 'l': /* show bdbuf pool id */
289  selector->show_pool_id = true ;
290  selector->show_all = false;
291  break;
292  case 't': /* show bdbuf transfer sema */
293  selector->show_sema = true ;
294  break;
295  default:
296  arg_error = i;
297  break;
298  }
299  }
300  else {
301  /*
302  * non-modifier arguments
303  */
304  switch(++nm_argc) {
305  default: /* no further arguments defined */
306  arg_error = i;
307  break;
308  }
309  }
310  }
311  if (arg_error) {
312  printf("%s: unknown argument %s\n",argv[0],argv[arg_error]);
313  sc = RTEMS_NOT_DEFINED;
314  }
315  return sc;
316 }
