36 static rtems_isr interrupt_handler(rtems_vector_number n)
37 {
38  unsigned char msg[8];
39  int i;
40 
41  lm32_interrupt_ack(1 << MM_IRQ_USB);
42 
43  while(mouse_consume != COMLOC_MEVT_PRODUCE) {

46  rtems_message_queue_send(event_q, msg, 4);

49

141 static void load_firmware(const unsigned char *firmware, int length)
142 {
143  int i, nwords;
144  volatile unsigned int *usb_dmem
145  = (volatile unsigned int *)MM_SOFTUSB_DMEM_BASE;
146  volatile unsigned int *usb_pmem
147  = (volatile unsigned int *)MM_SOFTUSB_PMEM_BASE;
148 
149  MM_WRITE(MM_SOFTUSB_CONTROL, SOFTUSB_CONTROL_RESET);
150  for(i=0;i<SOFTUSB_DMEM_SIZE/4;i++)
151  usb_dmem[i] = 0;
152  for(i=0;i<SOFTUSB_PMEM_SIZE/2;i++)
153  usb_pmem[i] = 0;
154  nwords = (length+1)/2;
155  for(i=0;i<nwords;i++)
156  usb_pmem[i] = ((unsigned int)(firmware[2*i]))
157  |((unsigned int)(firmware[2*i+1]) << 8);
158  MM_WRITE(MM_SOFTUSB_CONTROL, 0);
159 }
