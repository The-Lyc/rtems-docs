113 void bsp_interrupt_dispatch(void)
114 {
115  uint32_t icciar = READ_SR(ICC_IAR1);
116  rtems_vector_number vector = GIC_CPUIF_ICCIAR_ACKINTID_GET(icciar);
117  rtems_vector_number spurious = 1023;
118 
119  if (vector != spurious) {
120  uint32_t psr = _ARMV4_Status_irq_enable();
121  bsp_interrupt_handler_dispatch(vector);
122 
123  _ARMV4_Status_restore(psr);
124 
125  WRITE_SR(ICC_EOIR1, icciar);
126  }
127 }

159 static inline uint32_t get_id_count(volatile gic_dist *dist)
160 {
161  uint32_t id_count = GIC_DIST_ICDICTR_IT_LINES_NUMBER_GET(dist->icdictr);

169 static void init_cpu_interface(void)
170 {
171  uint32_t sre_value = 0x7;
172  WRITE_SR(ICC_SRE, sre_value);
173  WRITE_SR(ICC_PMR, GIC_CPUIF_ICCPMR_PRIORITY(0xff));
174  WRITE_SR(ICC_BPR0, GIC_CPUIF_ICCBPR_BINARY_POINT(0x0));
175 
176  volatile gic_redist *redist = ARM_GIC_REDIST;
177  uint32_t waker = redist->icrwaker;
178  uint32_t waker_mask = GIC_REDIST_ICRWAKER_PROCESSOR_SLEEP;
179  waker &= ~waker_mask;
180  redist->icrwaker = waker;
181 
182  /* Set interrupt group to 1NS for SGI/PPI interrupts routed through the redistributor */
183  volatile gic_sgi_ppi *sgi_ppi = ARM_GIC_SGI_PPI;
184  sgi_ppi->icspigrpr[0] = 0xffffffff;
185  sgi_ppi->icspigrpmodr[0] = 0;
186  for (int id = 0; id < 32; id++) {
187  sgi_ppi->icspiprior[id] = PRIORITY_DEFAULT;
188  }
189 
190  /* Enable interrupt groups 0 and 1 */
191  WRITE_SR(ICC_IGRPEN0, 0x1);
192  WRITE_SR(ICC_IGRPEN1, 0x1);
193  WRITE_SR(ICC_CTLR, 0x0);
194 }

196 rtems_status_code bsp_interrupt_facility_initialize(void)
197 {
198  volatile gic_dist *dist = ARM_GIC_DIST;
199  uint32_t id_count = get_id_count(dist);
200  uint32_t id;
201 
202  arm_cp15_set_exception_handler(
203  ARM_EXCEPTION_IRQ,
204  _ARMV4_Exception_interrupt
205  );
206 
207  dist->icddcr = GIC_DIST_ICDDCR_ARE_NS | GIC_DIST_ICDDCR_ARE_S
208  | GIC_DIST_ICDDCR_ENABLE_GRP1S | GIC_DIST_ICDDCR_ENABLE_GRP1NS
209  | GIC_DIST_ICDDCR_ENABLE_GRP0;
210 
211  for (id = 0; id < id_count; id += 32) {
212  /* Disable all interrupts */
213  dist->icdicer[id / 32] = 0xffffffff;
214 
215  /* Set interrupt group to 1NS for all interrupts */
216  dist->icdigr[id / 32] = 0xffffffff;
217  dist->icdigmr[id / 32] = 0;
218  }
219 
220  for (id = 0; id < id_count; ++id) {
221  gic_id_set_priority(dist, id, PRIORITY_DEFAULT);
222  }
223 
224  for (id = 32; id < id_count; ++id) {
225  gic_id_set_targets(dist, id, 0x01);
226  }
227 
228  init_cpu_interface();
229  return RTEMS_SUCCESSFUL;
230 }

249 )
250 {
251  rtems_status_code sc = RTEMS_SUCCESSFUL;
252 
253  if (bsp_interrupt_is_valid_vector(vector)) {
254  if (vector < 32) {
255  volatile gic_sgi_ppi *sgi_ppi = ARM_GIC_SGI_PPI;
256  sgi_ppi->icspiprior[vector] = priority;
257  } else {
258  volatile gic_dist *dist = ARM_GIC_DIST;
259  gic_id_set_priority(dist, vector, priority);

261  } else {
262  sc = RTEMS_INVALID_ID;
263  }
264 
265  return sc;
