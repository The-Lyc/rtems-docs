147 static int ambapp_grlib_init1(struct drvmgr_dev *dev)
148 {
149  struct ambapp_config *config;
150 
151  dev->priv = NULL;
152  dev->name = "GRLIB AMBA PnP";
153 
154  DBG("AMBAPP GRLIB: intializing\n");
155 
156  config = grlib_malloc(sizeof(*config));
157  if ( !config )
158  return RTEMS_NO_MEMORY;
159 
160  config->ops = &ambapp_grlib_ops;
161  config->maps_up = DRVMGR_TRANSLATE_ONE2ONE;
162  config->maps_down = DRVMGR_TRANSLATE_ONE2ONE;
163  config->abus = drv_mgr_grlib_config->abus;
164  config->resources = drv_mgr_grlib_config->resources;
165  config->funcs = ambapp_grlib_funcs;
166  config->bus_type = DRVMGR_BUS_TYPE_AMBAPP;
167 
168  /* Initialize the generic part of the AMBA Bus */
169  return ambapp_bus_register(dev, config);
