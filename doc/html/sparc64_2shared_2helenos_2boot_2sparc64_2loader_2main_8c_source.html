136 static void detect_architecture(void)
137 {
138  phandle root = ofw_find_device("/");
139  char compatible[COMPATIBLE_PROP_MAXLEN];
140 
141  if (ofw_get_property(root, "compatible", compatible,
142  COMPATIBLE_PROP_MAXLEN) <= 0) {
143  printk("Unable to determine architecture, default: sun4u.\n");
144  architecture = COMPATIBLE_SUN4U;
145  return;
146  }
147 
148  if (strcmp(compatible, "sun4v") == 0) {
149  architecture = COMPATIBLE_SUN4V;
150  } else {
151  /*
152  * As not all sun4u machines have "sun4u" in their "compatible"
153  * OBP property (e.g. Serengeti's OBP "compatible" property is
154  * "SUNW,Serengeti"), we will by default fallback to sun4u if
155  * an unknown value of the "compatible" property is encountered.
156  */
157  architecture = COMPATIBLE_SUN4U;
158  }
159 }

241 static void bootstrap_sun4v(void)
242 {
243  /*
244  * When SILO booted, the OBP had established a virtual to physical
245  * memory mapping. This mapping is not an identity (because the
246  * physical memory starts on non-zero address) - this is not
247  * surprising. But! The mapping even does not map virtual address
248  * 0 onto the starting address of the physical memory, but onto an
249  * address which is 0x400000 bytes higher. The reason is that the
250  * OBP had already used the memory just at the beginning of the
251  * physical memory, so that memory cannot be used by SILO (nor
252  * bootloader). As for now, we solve it by a nasty workaround:
253  * we pretend that the physical memory starts 0x400000 bytes further
254  * than it actually does (and hence pretend that the physical memory
255  * is 0x400000 bytes smaller). Of course, the value 0x400000 will most
256  * probably depend on the machine and OBP version (the workaround now
257  * works on Simics). A solution would be to inspect the "available"
258  * property of the "/memory" node to find out which parts of memory
259  * are used by OBP and redesign the algorithm of copying
260  * kernel/init tasks/ramdisk from the bootable image to memory
261  * (which we must do anyway because of issues with claiming the memory
262  * on Serengeti).
263  */
264  bootinfo.physmem_start += 0x400000;
265  bootinfo.memmap.zones[0].start += 0x400000;
266  bootinfo.memmap.zones[0].size -= 0x400000;
267 #if 0
268  printf("The sun4v init finished.");
269 #endif
270 }
