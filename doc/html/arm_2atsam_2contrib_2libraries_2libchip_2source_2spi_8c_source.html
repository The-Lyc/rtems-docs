241 extern void SPI_Write(Spi *spi, uint32_t dwNpcs, uint16_t wData)
242 {
243  /* Send data */
244  while ((spi->SPI_SR & SPI_SR_TXEMPTY) == 0);
245 
246  spi->SPI_TDR = wData | SPI_PCS(dwNpcs);
247 
248  while ((spi->SPI_SR & SPI_SR_TDRE) == 0);
249 }

260 extern void SPI_WriteLast(Spi *spi, uint32_t dwNpcs, uint16_t wData)
261 {
262  /* Send data */
263  while ((spi->SPI_SR & SPI_SR_TXEMPTY) == 0);
264 
265  spi->SPI_TDR = wData | SPI_PCS(dwNpcs) | SPI_TDR_LASTXFER;
266 
267  while ((spi->SPI_SR & SPI_SR_TDRE) == 0);
268 }
