47 void SUPC_SelectExtCrystal32K(void)
48 {
49  PMC_EnableXT32KFME();
50 
51  /* Select XTAL 32k instead of internal slow RC 32k for slow clock */
52  if ((SUPC->SUPC_SR & SUPC_SR_OSCSEL) != SUPC_SR_OSCSEL_CRYST) {
53  SUPC->SUPC_CR = SUPC_CR_KEY_PASSWD | SUPC_CR_XTALSEL_CRYSTAL_SEL;
54 
55  while (!(SUPC->SUPC_SR & SUPC_SR_OSCSEL));
56  }
57 }
