92 static void PIO_Interrupt(Pio *pPio, uint32_t id)
93 {
94  uint32_t status;
95  uint32_t status_save;
96  size_t i;
97 
98  do {
99  status = pPio->PIO_ISR;
100  status &= pPio->PIO_IMR;
101  status_save = status;
102 
103  for (i = 0; status != 0 && i < MAX_INTERRUPT_SOURCES; ++i) {
104  const InterruptSource *is = &_aIntSources[i];
105  const Pin *pin = is->pPin;;
106 
107  if (pin->id == id) {
108  uint32_t mask = pin->mask;
109 
110  if ((status & mask) != 0) {
111  status &= ~mask;
112  (*is->handler)(pin, is->arg);
113  }
114  }
115  }
116  } while (status_save != 0);
117 }

264 void PIO_ConfigureIt(const Pin *pPin, void (*handler)(const Pin *, void *arg),
265  void *arg)
266 {
267  InterruptSource *pSource;
268  rtems_interrupt_level level;
269 
270  TRACE_DEBUG("PIO_ConfigureIt()\n\r");
271 
272  rtems_interrupt_disable(level);
273 
274  if (_dwNumSources == MAX_INTERRUPT_SOURCES) {
275  bsp_fatal(ATSAM_FATAL_PIO_CONFIGURE_IT);
276  }
277 
278  pSource = &(_aIntSources[_dwNumSources]);
279  pSource->pPin = pPin;
280  pSource->handler = handler;
281  pSource->arg = arg;
282 
283  _dwNumSources++;
284 
285  rtems_interrupt_enable(level);
286 
287  /* Define new source */
288  TRACE_DEBUG("PIO_ConfigureIt: Defining new source #%d.\n\r", _dwNumSources);
289 }

299 rtems_status_code PIO_RemoveIt(const Pin *pPin,
300  void (*handler)(const Pin *, void *arg), void *arg)
301 {
302  InterruptSource *pSource;
303  rtems_interrupt_level level;
304  rtems_status_code sc = RTEMS_UNSATISFIED;
305  uint32_t i;
306 
307  TRACE_DEBUG("PIO_RemoveIt()\n\r");
308 
309  rtems_interrupt_disable(level);
310 
311  for(i = 0; i < _dwNumSources; ++i) {
312  pSource = &(_aIntSources[_dwNumSources]);
313  if(pSource->pPin == pPin &&
314  pSource->handler == handler &&
315  pSource->arg == arg) {
316  if(i + 1 < _dwNumSources) {
317  TRACE_DEBUG("PIO_RemoveIt: Remove #%d.\n\r", i);
318  /* Move remaining sources */
319  memcpy(pSource,
320  &(_aIntSources[i+1]),
321  sizeof(_aIntSources[0])*(_dwNumSources-i-1)
322  );
323  }
324  _dwNumSources--;
325  sc = RTEMS_SUCCESSFUL;
326  }
327  }
328 
329  rtems_interrupt_enable(level);
330 
331  return sc;
332 }
