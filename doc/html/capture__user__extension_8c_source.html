79 static inline void capture_record (rtems_tcb* tcb, uint32_t events)
80 {
81  rtems_capture_record_lock_context rec_context;
82 
83  if (rtems_capture_filter (tcb, events))
84  return;
85 
86  if (!rtems_capture_task_recorded (tcb))
87  rtems_capture_record_task (tcb);
88 
89  rtems_capture_record_open (tcb, events, 0, &rec_context);
90  rtems_capture_record_close (&rec_context);
91 }

94 rtems_status_code rtems_capture_user_extension_open(void)
95 {
96  rtems_status_code sc;
97  rtems_name name;
98  int index;
99 
100  /*
101  * Register the user extension handlers for the CAPture Engine.
102  */
103  name = rtems_build_name ('C', 'A', 'P', 'E');
104  sc = rtems_extension_create (name, &capture_extensions, &capture_id);
105  if (sc != RTEMS_SUCCESSFUL)
106  capture_id = 0;
107  else {
108  index = rtems_object_id_get_index (capture_id);
109  rtems_capture_set_extension_index (index);
110  }
111 
112  return sc;
113 }

122 /*
123  * This function is called when a task is created.
124  */
125 static bool
126 rtems_capture_create_task (rtems_tcb* ct,
127  rtems_tcb* nt)
128 {
129  /*
130  * The task pointers may not be known as the task may have
131  * been created before the capture engine was open. Add them.
132  */
133  if (!rtems_capture_task_initialized (ct))
134  rtems_capture_initialize_task (ct);
135 
136  /*
137  * Create the new task's capture control block.
138  */
139  rtems_capture_initialize_task (nt);
140 
141  if (rtems_capture_trigger_fired (ct, nt, RTEMS_CAPTURE_CREATE))
142  {
143  capture_record (ct, RTEMS_CAPTURE_CREATED_BY_EVENT);
144  capture_record (nt, RTEMS_CAPTURE_CREATED_EVENT);
145  }
146 
147  return 1 == 1;
148 }

274 /*
275  * This function is called when a context is switched.
276  */
277 static void
278 rtems_capture_switch_task (rtems_tcb* ct,
279  rtems_tcb* ht)
280 {
281  uint32_t flags = rtems_capture_get_flags();
282 
283  /*
284  * Only perform context switch trace processing if tracing is
285  * enabled.
286  */
287  if (flags & RTEMS_CAPTURE_ON)
288  {
289  rtems_capture_time time;
290  if (!rtems_capture_task_initialized (ct))
291  rtems_capture_initialize_task (ct);
292 
293  if (!rtems_capture_task_initialized (ht))
294  rtems_capture_initialize_task (ht);
295 
296  /*
297  * Update the execution time. Assume the time will not overflow
298  * for now. This may need to change.
299  */
300  rtems_capture_get_time (&time);
301 
302  if (rtems_capture_trigger_fired (ct, ht, RTEMS_CAPTURE_SWITCH))
303  {
304  capture_record (ct, RTEMS_CAPTURE_SWITCHED_OUT_EVENT);
305  capture_record (ht, RTEMS_CAPTURE_SWITCHED_IN_EVENT);
306  }
307  }
