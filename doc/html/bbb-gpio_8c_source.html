148 uint32_t rtems_gpio_bsp_interrupt_line(rtems_vector_number vector)
149 {
150  uint32_t event_status;
151  uint8_t bank_nr = 0;
152 
153  /* Following loop will get the bank number from vector number */
154  while (bank_nr < GPIO_BANK_COUNT && vector != gpio_bank_vector[bank_nr])
155  {
156  bank_nr++;
157  }
158 
159  /* Retrieve the interrupt event status. */
160  event_status = mmio_read(bbb_reg(bank_nr, AM335X_GPIO_IRQSTATUS_0));
161 
162  /* Clear the interrupt line. */
163  mmio_write(
164  (bbb_reg(bank_nr, AM335X_GPIO_IRQSTATUS_0)), event_status);
165 
166  return event_status;
167 }

169 rtems_status_code rtems_gpio_bsp_enable_interrupt(
170  uint32_t bank,
171  uint32_t pin,
172  rtems_gpio_interrupt interrupt
173 ) {
174 
175  /* Enable IRQ generation for the specific pin */
176  mmio_set(bbb_reg(bank, AM335X_GPIO_IRQSTATUS_SET_0), BIT(pin));
177 
178  switch ( interrupt ) {
179  case FALLING_EDGE:
180  /* Enables asynchronous falling edge detection. */
181  mmio_set(bbb_reg(bank, AM335X_GPIO_FALLINGDETECT), BIT(pin));
182  break;
183  case RISING_EDGE:
184  /* Enables asynchronous rising edge detection. */
185  mmio_set(bbb_reg(bank, AM335X_GPIO_RISINGDETECT), BIT(pin));
186  break;
187  case BOTH_EDGES:
188  /* Enables asynchronous falling edge detection. */
189  mmio_set(bbb_reg(bank, AM335X_GPIO_FALLINGDETECT), BIT(pin));
190 
191  /* Enables asynchronous rising edge detection. */
192  mmio_set(bbb_reg(bank, AM335X_GPIO_RISINGDETECT), BIT(pin));
193  break;
194  case LOW_LEVEL:
195  /* Enables pin low level detection. */
196  mmio_set(bbb_reg(bank, AM335X_GPIO_LEVELDETECT0), BIT(pin));
197  break;
198  case HIGH_LEVEL:
199  /* Enables pin high level detection. */
200  mmio_set(bbb_reg(bank, AM335X_GPIO_LEVELDETECT1), BIT(pin));
201  break;
202  case BOTH_LEVELS:
203  /* Enables pin low level detection. */
204  mmio_set(bbb_reg(bank, AM335X_GPIO_LEVELDETECT0), BIT(pin));
205 
206  /* Enables pin high level detection. */
207  mmio_set(bbb_reg(bank, AM335X_GPIO_LEVELDETECT1), BIT(pin));
208  break;

210  default:

278 rtems_status_code rtems_gpio_bsp_multi_select(
279  rtems_gpio_multiple_pin_select *pins,
280  uint32_t pin_count,
281  uint32_t select_bank
282 ) {
283  uint32_t register_address;
284  uint32_t select_register;
285  uint8_t i;
286 
287  register_address = gpio_bank_addrs[select_bank] + AM335X_GPIO_OE;
288 
289  select_register = REG(register_address);
290 
291  for ( i = 0; i < pin_count; ++i ) {
292  if ( pins[i].function == DIGITAL_INPUT ) {
293  select_register |= BIT(pins[i].pin_number);
294  } else if ( pins[i].function == DIGITAL_OUTPUT ) {
295  select_register &= ~BIT(pins[i].pin_number);
296  } else { /* BSP_SPECIFIC function. */
297  return RTEMS_NOT_DEFINED;
298  }
299  }
300 
301  REG(register_address) = select_register;
302 
303  return RTEMS_SUCCESSFUL;
304 }
