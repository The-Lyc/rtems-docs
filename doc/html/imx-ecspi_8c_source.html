132 static uint32_t imx_ecspi_conreg_divider(imx_ecspi_bus *bus, uint32_t speed_hz)
133 {
134  uint32_t post;
135  uint32_t pre;
136  uint32_t clk_in;
137 
138  clk_in = bus->base.max_speed_hz;
139 
140  if (clk_in > speed_hz) {
141  post = fls((int) clk_in) - fls((int) speed_hz);
142 
143  if (clk_in > (speed_hz << post)) {
144  ++post;
145  }
146 
147  /* We have 2^4 == 16, use the pre-divider for this factor */
148  post = MAX(4, post) - 4;
149 
150  if (post <= 0xf) {
151  pre = howmany(clk_in, speed_hz << post) - 1;
152  } else {
153  post = 0xf;
154  pre = 0xf;
155  }
156  } else {
157  post = 0;
158  pre = 0;
159  }
160 
161  return IMX_ECSPI_CONREG_POST_DIVIDER(post)
162  | IMX_ECSPI_CONREG_PRE_DIVIDER(pre);
163 }

165 static void imx_ecspi_config(
166  imx_ecspi_bus *bus,
167  volatile imx_ecspi *regs,
168  uint32_t speed_hz,
169  uint8_t bits_per_word,
170  uint32_t mode,
171  uint8_t cs
172 )
173 {
174  uint32_t conreg;
175  uint32_t testreg;
176  uint32_t configreg;
177  uint32_t dmareg;
178  uint32_t cs_bit;
179 
180  conreg = IMX_ECSPI_CONREG_CHANNEL_MODE(0xf)
181  | IMX_ECSPI_CONREG_SMC | IMX_ECSPI_CONREG_EN;
182  testreg = regs->testreg;
183  configreg = regs->configreg;
184  dmareg = regs->dmareg;
185  cs_bit = 1U << cs;
186 
187  conreg |= imx_ecspi_conreg_divider(bus, speed_hz);
188  conreg |= IMX_ECSPI_CONREG_CHANNEL_SELECT(cs);
189 
190  configreg |= IMX_ECSPI_CONFIGREG_SS_CTL(cs_bit);
191 
192  if ((mode & SPI_CPHA) != 0) {
193  configreg |= IMX_ECSPI_CONFIGREG_SCLK_PHA(cs_bit);
194  } else {
195  configreg &= ~IMX_ECSPI_CONFIGREG_SCLK_PHA(cs_bit);
196  }
197 
198  if ((mode & SPI_CPOL) != 0) {
199  configreg |= IMX_ECSPI_CONFIGREG_SCLK_POL(cs_bit);
200  configreg |= IMX_ECSPI_CONFIGREG_SCLK_CTL(cs_bit);
201  } else {
202  configreg &= ~IMX_ECSPI_CONFIGREG_SCLK_POL(cs_bit);
203  configreg &= ~IMX_ECSPI_CONFIGREG_SCLK_CTL(cs_bit);
204  }
205 
206  if ((mode & SPI_CS_HIGH) != 0) {
207  configreg |= IMX_ECSPI_CONFIGREG_SS_POL(cs_bit);
208  } else {
209  configreg &= ~IMX_ECSPI_CONFIGREG_SS_POL(cs_bit);
210  }
211 
212  if ((mode & SPI_LOOP) != 0) {
213  testreg |= IMX_ECSPI_TESTREG_LBC;
214  } else {
215  testreg &= ~IMX_ECSPI_TESTREG_LBC;
216  }
217 
218  dmareg = IMX_ECSPI_DMAREG_TX_THRESHOLD_SET(dmareg, IMX_ECSPI_FIFO_SIZE/2);
219 
220  regs->conreg = conreg;
221  regs->testreg = testreg;
222  regs->dmareg = dmareg;
223  regs->configreg = configreg;
224 
225  bus->conreg = conreg;
226  bus->speed_hz = speed_hz;
227  bus->bits_per_word = bits_per_word;
228  bus->mode = mode;
229  bus->cs = cs;
230 
231  /* FIXME: Clock change delay */
232 }

234 static void imx_ecspi_set_push_pop(
235  imx_ecspi_bus *bus,
236  uint32_t len,
237  uint8_t bits_per_word
238 )
239 {
240  uint32_t conreg;
241 
242  conreg = bus->conreg;
243 
244  if (len % 4 == 0 && len <= IMX_ECSPI_FIFO_SIZE) {
245  conreg |= IMX_ECSPI_CONREG_BURST_LENGTH((len * 8) - 1);
246 
247  bus->push = imx_ecspi_push_uint32_t_swap;
248  bus->pop = imx_ecspi_pop_uint32_t_swap;
249  } else {
250  conreg |= IMX_ECSPI_CONREG_BURST_LENGTH(bits_per_word - 1);
251 
252  if (bits_per_word <= 8) {
253  bus->push = imx_ecspi_push_uint8_t;
254  bus->pop = imx_ecspi_pop_uint8_t;
255  } else if (bits_per_word <= 16) {
256  bus->push = imx_ecspi_push_uint16_t;
257  bus->pop = imx_ecspi_pop_uint16_t;
258  } else {
259  bus->push = imx_ecspi_push_uint32_t;
260  bus->pop = imx_ecspi_pop_uint32_t;
261  }
262  }
263 
264  bus->regs->conreg = conreg;
265 }

267 static void imx_ecspi_next_msg(imx_ecspi_bus *bus, volatile imx_ecspi *regs)
268 {
269  if (bus->msg_todo > 0) {
270  const spi_ioc_transfer *msg;
271 
272  msg = bus->msg;
273 
274  if (
275  msg->speed_hz != bus->speed_hz
276  || msg->bits_per_word != bus->bits_per_word
277  || msg->mode != bus->mode
278  || msg->cs != bus->cs
279  ) {
280  imx_ecspi_config(
281  bus,
282  regs,
283  msg->speed_hz,
284  msg->bits_per_word,
285  msg->mode,
286  msg->cs
287  );
288  }
289 
290  bus->todo = msg->len;
291  bus->rx_buf = msg->rx_buf;
292  bus->tx_buf = msg->tx_buf;
293  imx_ecspi_set_push_pop(bus, msg->len, msg->bits_per_word);
294  imx_ecspi_push(bus, regs);
295  regs->intreg = IMX_ECSPI_TE | IMX_ECSPI_TDR;
296  } else {
297  regs->intreg = 0;
298  imx_ecspi_done(bus);
299  }
300 }

302 static void imx_ecspi_interrupt(void *arg)
303 {
304  imx_ecspi_bus *bus;
305  volatile imx_ecspi *regs;
306 
307  bus = arg;
308  regs = bus->regs;
309 
310  while (imx_ecspi_is_rx_fifo_not_empty(regs)) {
311  (*bus->pop)(bus, regs);
312  --bus->in_transfer;
313  }
314 
315  if (bus->todo > 0) {
316  imx_ecspi_push(bus, regs);
317  } else if (bus->in_transfer > 0) {
318  regs->intreg = IMX_ECSPI_RR;
319  } else {
320  --bus->msg_todo;
321  ++bus->msg;
322  imx_ecspi_next_msg(bus, regs);
323  }
324 }
