61 /* Returns a zeroed out table. */
62 static unsigned long *table_alloc(int size){
63  unsigned long *addr=next_avail;
64  if(((unsigned long)next_avail + size) > MAX_TABLE_ADDR){
65  return 0;
66  }
67  memset((void *)addr,0, size);
68  next_avail =(unsigned long *)((unsigned long)next_avail + size);
69  return addr;

73  void page_table_init();
74 
75  This should transparently map the first 4 Meg of ram. Caching is
76  turned off from 0x00000000 to 0x00020000 (this region is used by
77  162Bug and contains the page tables). From 0x00020000 to 0x00400000
78  we are using copy back caching. DTTR0 and ITTR0 are set up to
79  directly translate from 0x80000000-0xffffffff with caching turned
80  off and serialized. Addresses between 0x400000 and 0x80000000 are
81  illegal.
82 */
83 void page_table_init(){
84 
85  /* put everything in a known state */
86  page_table_teardown();
87 
88  root_table=table_alloc(ROOT_TABLE_SIZE);
89 
90  /* First set up TTR.
91  base address = 0x80000000
92  address mask = 0x7f
93  Ignore FC2 for match.
94  Noncachable.
95  Not write protected.*/
96  __asm__ volatile ("movec %0,%%dtt0\n\
97  movec %0,%%itt0"
98  :: "d" (0x807fc040));
99 
100  /* Point urp and srp at root page table. */
101  __asm__ volatile ("movec %0,%%urp\n\
102  movec %0,%%srp"
103  :: "d" (BASE_TABLE_ADDR));
104 
105  page_table_map((void *)0,0x20000, CACHE_NONE);
106  page_table_map((void *)0x20000,0x400000-0x20000,CACHE_COPYBACK);
107 
108  /* Turn on paging with a 4 k page size.*/
109  __asm__ volatile ("movec %0,%%tc"
110  :: "d" (0x8000));
111 
112  /* Turn on the cache. */
113  __asm__ volatile ("movec %0,%%cacr"
114  :: "d" (0x80008000));
115 }

117 void page_table_teardown(){
118  next_avail=(unsigned long *)BASE_TABLE_ADDR;
119  /* Turn off paging. Turn off the cache. Flush the cache. Tear down
120  the transparent translations. */
121  __asm__ volatile ("movec %0,%%tc\n\
122  movec %0,%%cacr\n\
123  cpusha %%bc\n\
124  movec %0,%%dtt0\n\
125  movec %0,%%itt0\n\
126  movec %0,%%dtt1\n\
127  movec %0,%%itt1"
128  :: "d" (0) );
129 }

131 /* Identity maps addr to addr+size with caching cache_type. */
132 int page_table_map(void *addr, unsigned long size, int cache_type){
133  unsigned long *pointer_table;
134  unsigned long *page_table;
135  unsigned long root_index, pointer_index, page_index;
136  /* addr must be a multiple of 4k */
137  if((unsigned long)addr & 0xfff){
138  return PTM_BAD_ADDR;
139  }
140  /* size must also be a multiple of 4k */
141  if(size & 0xfff){
142  return PTM_BAD_SIZE;
143  }
144  /* check for valid cache type */
145  if( (cache_type>CACHE_NONE) || (cache_type<CACHE_WRITE_THROUGH)){
146  return PTM_BAD_CACHE;
147  }
148 
149  while(size){
150  root_index=(unsigned long)addr;
151  root_index >>= 25;
152  root_index &= 0x7f;
153 
154  if(root_table[root_index]){
155  pointer_table =
156  (unsigned long *) (root_table[root_index] & 0xfffffe00);
157  }else{
158  if(!(pointer_table=table_alloc(POINTER_TABLE_SIZE))){
159  return PTM_NO_TABLE_SPACE;
160  }
161  root_table[root_index]=((unsigned long)pointer_table) + 0x03;
162  }
163 
164  pointer_index=(unsigned long)addr;
165  pointer_index >>=18;
166  pointer_index &= 0x7f;
167 
168  if(pointer_table[pointer_index]){
169  page_table =
170  (unsigned long *) (pointer_table[pointer_index] &
171  0xffffff00);
172  }else{
173  if(!(page_table=table_alloc(PAGE_TABLE_SIZE))){
174  return PTM_NO_TABLE_SPACE;
175  }
176  pointer_table[pointer_index]=
177  ((unsigned long)page_table) + 0x03;
178  }
179 
180  page_index=(unsigned long)addr;
181  page_index >>=12;
182  page_index &= 0x3f;
183 
184  page_table[page_index] =
185  ((unsigned long) addr & 0xfffff000) + 0x03 + (cache_type << 5);
186 
187  size -= 4096;
188  addr = (void *) ((unsigned long)addr + 4096);
189  }
190 
191  /* Flush the ATC. Push and invalidate the cache. */
192  __asm__ volatile ("pflusha\n\
193  cpusha %bc");
194 
195  return PTM_SUCCESS;
196 }
