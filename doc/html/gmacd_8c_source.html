51 __STATIC_INLINE int fixed_mod(int a, int b)
52 {
53  int rem = a % b;
54 
55  while (rem < 0)
56  rem += b;
57 
58  return rem;
59 }

117 static void GMACD_ResetTx(sGmacd *pDrv, gmacQueList_t queIdx)
118 {
119  Gmac *pHw = pDrv->pHw;
120  uint8_t *pTxBuffer = pDrv->queueList[queIdx].pTxBuffer;
121  sGmacTxDescriptor *pTd = pDrv->queueList[queIdx].pTxD;
122  uint32_t Index;
123  uint32_t Address;
124 
125  /* Disable TX */
126  GMAC_TransmitEnable(pHw, 0);
127 
128  /* Setup the TX descriptors. */
129  GCIRC_CLEAR(pDrv->queueList[queIdx].wTxHead, pDrv->queueList[queIdx].wTxTail);
130 
131  for (Index = 0; Index < pDrv->queueList[queIdx].wTxListSize; Index++) {
132  Address = (uint32_t)(&(pTxBuffer[Index *
133  pDrv->queueList[queIdx].wTxBufferSize]));
134  pTd[Index].addr = Address;
135  pTd[Index].status.val = (uint32_t)GMAC_TX_USED_BIT;
136  }
137 
138  pTd[pDrv->queueList[queIdx].wTxListSize - 1].status.val =
139  GMAC_TX_USED_BIT | GMAC_TX_WRAP_BIT;
140 
141  /* Transmit Buffer Queue Pointer Register */
142 
143  GMAC_SetTxQueue(pHw, (uint32_t)pTd, queIdx);

184 static void GMACD_TxCompleteHandler(sGmacd *pGmacd, gmacQueList_t qId)
185 {
186  Gmac *pHw = pGmacd->pHw;
187  sGmacTxDescriptor *pTxTd;
188  fGmacdTransferCallback fTxCb;
189  uint32_t tsr;
190 
191  /* Clear status */
192  tsr = GMAC_GetTxStatus(pHw);
193  GMAC_ClearTxStatus(pHw, tsr);
194 
195  while (!GCIRC_EMPTY(
196  pGmacd->queueList[qId].wTxHead, pGmacd->queueList[qId].wTxTail)) {
197  pTxTd = &pGmacd->queueList[qId].pTxD[pGmacd->queueList[qId].wTxTail];
198 
199  /* Exit if frame has not been sent yet:
200  * On TX completion, the GMAC set the USED bit only into the
201  * very first buffer descriptor of the sent frame.
202  * Otherwise it updates this descriptor with status error bits.
203  * This is the descriptor write back.
204  */
205  if ((pTxTd->status.val & GMAC_TX_USED_BIT) == 0)
206  break;
207 
208  /* Process all buffers of the current transmitted frame */
209  while ((pTxTd->status.val & GMAC_TX_LAST_BUFFER_BIT) == 0) {
210  GCIRC_INC(pGmacd->queueList[qId].wTxTail,
211  pGmacd->queueList[qId].wTxListSize);
212  pTxTd = &pGmacd->queueList[qId].pTxD[pGmacd->queueList[qId].wTxTail];
213  memory_sync();
214  }
215 
216  /* Notify upper layer that a frame has been sent */
217  fTxCb = pGmacd->queueList[qId].fTxCbList[pGmacd->queueList[qId].wTxTail];
218 
219  if (fTxCb)
220  fTxCb(tsr);
221 
222  /* Go to next frame */
223  GCIRC_INC(pGmacd->queueList[qId].wTxTail, pGmacd->queueList[qId].wTxListSize);
224  }
225 
226  /* If a wakeup has been scheduled, notify upper layer that it can
227  send other packets, send will be successful. */
228  if (pGmacd->queueList[qId].fWakupCb &&
229  GCIRC_SPACE(pGmacd->queueList[qId].wTxHead,
230  pGmacd->queueList[qId].wTxTail,
231  pGmacd->queueList[qId].wTxListSize) >=
232  pGmacd->queueList[qId].bWakeupThreshold)
233  pGmacd->queueList[qId].fWakupCb();
234 }

241 static void GMACD_TxErrorHandler(sGmacd *pGmacd, gmacQueList_t qId)
242 {
243  Gmac *pHw = pGmacd->pHw;
244  sGmacTxDescriptor *pTxTd;
245  fGmacdTransferCallback fTxCb;
246  uint32_t tsr;
247 
248  /* Clear TXEN bit into the Network Configuration Register:
249  * this is a workaround to recover from TX lockups that
250  * occur on sama5d3 gmac (r1p24f2) when using scatter-gather.
251  * This issue has never been seen on sama5d4 gmac (r1p31).
252  */
253  GMAC_TransmitEnable(pHw, 0);
254 
255  /* The following step should be optional since this function is called
256  * directly by the IRQ handler. Indeed, according to Cadence
257  * documentation, the transmission is halted on errors such as
258  * too many retries or transmit under run.
259  * However it would become mandatory if the call of this function
260  * were scheduled as a task by the IRQ handler (this is how Linux
261  * driver works). Then this function might compete with GMACD_Send().
262  *
263  * Setting bit 10, tx_halt, of the Network Control Register is not enough:
264  * We should wait for bit 3, tx_go, of the Transmit Status Register to
265  * be cleared at transmit completion if a frame is being transmitted.
266  */
267  GMAC_TransmissionHalt(pHw);
268 
269  while (GMAC_GetTxStatus(pHw) & GMAC_TSR_TXGO);
270 
271  /* Treat frames in TX queue including the ones that caused the error. */
272  while (!GCIRC_EMPTY(pGmacd->queueList[qId].wTxHead,
273  pGmacd->queueList[qId].wTxTail)) {
274  int tx_completed = 0;
275  pTxTd = &pGmacd->queueList[qId].pTxD[pGmacd->queueList[qId].wTxTail];
276 
277  /* Check USED bit on the very first buffer descriptor to validate
278  * TX completion.
279  */
280  if (pTxTd->status.val & GMAC_TX_USED_BIT)
281  tx_completed = 1;
282 
283  /* Go to the last buffer descriptor of the frame */
284  while ((pTxTd->status.val & GMAC_TX_LAST_BUFFER_BIT) == 0) {
285  GCIRC_INC(pGmacd->queueList[qId].wTxTail,
286  pGmacd->queueList[qId].wTxListSize);
287  pTxTd = &pGmacd->queueList[qId].pTxD[pGmacd->queueList[qId].wTxTail];
288  }
289 
290  /* Notify upper layer that a frame status */
291  fTxCb = pGmacd->queueList[qId].fTxCbList[pGmacd->queueList[qId].wTxTail];
292 
293  if (fTxCb)
294  fTxCb(tx_completed ? GMAC_TSR_TXCOMP : 0);
295 
296  // TODO: which error to notify?
297 
298  /* Go to next frame */
299  GCIRC_INC(pGmacd->queueList[qId].wTxTail, pGmacd->queueList[qId].wTxListSize);
300  }
301 
302  /* Reset TX queue */
303  GMACD_ResetTx(pGmacd, qId);
304 
305  /* Clear status */
306  tsr = GMAC_GetTxStatus(pHw);
307  GMAC_ClearTxStatus(pHw, tsr);
308 
309  /* Now we are ready to start transmission again */
310  GMAC_TransmitEnable(pHw, 1);
311 
312  if (pGmacd->queueList[qId].fWakupCb)
313  pGmacd->queueList[qId].fWakupCb();
314 }

336 void GMACD_Handler(sGmacd *pGmacd, gmacQueList_t queIdx)
337 {
338  Gmac *pHw = pGmacd->pHw;
339  uint32_t isr;
340  uint32_t rsr;
341 
342  /* Interrupt Status Register is cleared on read */
343  while ((isr = GMAC_GetItStatus(pHw, queIdx)) != 0) {
344  /* Sync Frame Received - PTP */
345  if (0u != (isr & GMAC_ISR_SFR)) {
346  rsr = GMAC_ISR_SFR;
347  memory_barrier();
348 
349  /* Invoke callbacks */
350  if (pGmacd->queueList[queIdx].fRxCb)
351  pGmacd->queueList[queIdx].fRxCb(rsr);
352  else {
353  }
354  } else {
355  }
356 
357  /* Peer Delay Request Frame Received - PTP */
358  if (0u != (isr & GMAC_ISR_PDRQFR)) {
359  rsr = GMAC_ISR_PDRQFR;
360  memory_barrier();
361 
362  /* Invoke callbacks */
363  if (pGmacd->queueList[queIdx].fRxCb)
364  pGmacd->queueList[queIdx].fRxCb(rsr);
365  else {
366  }
367  } else {
368  }
369 
370  /* Peer Delay Response Frame Received - PTP */
371  if (0u != (isr & GMAC_ISR_PDRSFR)) {
372 
373  rsr = GMAC_ISR_PDRSFR;
374  memory_barrier();
375 
376  /* Invoke callbacks */
377  if (pGmacd->queueList[queIdx].fRxCb)
378  pGmacd->queueList[queIdx].fRxCb(rsr);
379  else {
380  }
381  } else {
382  }
383 
384  if (0u != (isr & GMAC_ISR_TSU)) {
385  /* Invoke call back with flag set to TSU comparison interrupt */
386  rsr = GMAC_ISR_TSU;
387  memory_barrier();
388 
389  /* Invoke callbacks */
390  if (pGmacd->queueList[queIdx].fRxCb)
391  pGmacd->queueList[queIdx].fRxCb(rsr);
392  else {
393  }
394  } else {
395  }
396 
397  /* RX packet */
398  if (isr & GMAC_INT_RX_STATUS_BITS) {
399  /* Clear status */
400  rsr = GMAC_GetRxStatus(pHw);
401  GMAC_ClearRxStatus(pHw, rsr);
402 
403  /* Invoke callback */
404  if (pGmacd->queueList[queIdx].fRxCb)
405  pGmacd->queueList[queIdx].fRxCb(rsr);
406  }
407 
408  /* TX error */
409  if (isr & GMAC_INT_TX_STATUS_ERR_BITS) {
410  GMACD_TxErrorHandler(pGmacd, queIdx);
411  break;
412  }
413 
414 #ifndef PTP_1588_TX_DISABLE
415 
416  /* Transmit of SYNC / PDELAY_REQ / PDELAY_RSP */
417  if (0u != (isr & isrMasks[gPtpMsgTxQue[ptpTxQueReadIdx]])) {
418  /* Invoke callback */
419  /* Check if it is possible for multiple messages to be triggered
420  within a single isr. If so, a loop may be needed to validate the top
421  of the queue with the actual interrupt that has been triggered */
422  /* while (0u != (isr & (GMAC_IMR_SFT | GMAC_IMR_PDRQFT | GMAC_IMR_PDRSFT))) { */
423  if (pGmacd->queueList[queIdx].fTxPtpEvtCb) {
424  switch (gPtpMsgTxQue[ptpTxQueReadIdx]) {
425  case SYNC_MSG_TYPE:
426  pGmacd->queueList[queIdx].fTxPtpEvtCb
427  (gPtpMsgTxQue[ptpTxQueReadIdx],
428  GMAC_GetTxEvtFrameSec(pHw),
429  GMAC_GetTxEvtFrameNsec(pHw),
430  gPtpMsgTxSeqId[ptpTxQueReadIdx]);
431  isr &= GMAC_IMR_SFT;
432  break;
433 
434  case PDELAY_REQ_TYPE:
435  pGmacd->queueList[queIdx].fTxPtpEvtCb
436  (gPtpMsgTxQue[ptpTxQueReadIdx],
437  GMAC_GetTxPeerEvtFrameSec(pHw),
438  GMAC_GetTxPeerEvtFrameNsec(pHw),
439  gPtpMsgTxSeqId[ptpTxQueReadIdx]);
440  isr &= GMAC_IMR_PDRQFT;
441  break;
442 
443  case PDELAY_RESP_TYPE:
444  pGmacd->queueList[queIdx].fTxPtpEvtCb
445  (gPtpMsgTxQue[ptpTxQueReadIdx],
446  GMAC_GetTxPeerEvtFrameSec(pHw),
447  GMAC_GetTxPeerEvtFrameNsec(pHw),
448  gPtpMsgTxSeqId[ptpTxQueReadIdx]);
449  isr &= GMAC_IMR_PDRSFT;
450  break;
451 
452  default:
453  /* Only for Peer messages & sync messages */
454  break;
455  };
456  } else {
457  }
458 
459  ptpTxQueReadIdx++;
460  ptpTxQueReadIdx &= (EFRS_BUFFER_LEN - 1);
461 
462  } else {
463  /* if (0u != (isr & isrMasks[gPtpMsgTxQue[ptpTxQueReadIdx]])) */
464  }
465 
466 #endif /* #ifndef PTP_1588_TX_DISABLE */
467 
468  /* TX packet */
469  if (isr & GMAC_IER_TCOMP)
470  GMACD_TxCompleteHandler(pGmacd, queIdx);
471 
472  if (isr & GMAC_IER_HRESP)
473  TRACE_ERROR("HRESP\n\r");
474  }
475 }

551 uint8_t GMACD_InitTransfer(sGmacd *pGmacd, const sGmacInit *pInit,
552  gmacQueList_t queIdx)
553 {
554  Gmac *pHw = pGmacd->pHw;
555  uint8_t *pRxBuffer = pInit->pRxBuffer;
556  sGmacRxDescriptor *pRxD = pInit->pRxD;
557  uint16_t wRxBufferSize = pInit->wRxBufferSize;
558  uint16_t wRxSize = pInit->wRxSize;
559  uint8_t *pTxBuffer = pInit->pTxBuffer;
560  sGmacTxDescriptor *pTxD = pInit->pTxD;
561  uint16_t wTxBufferSize = pInit->wTxBufferSize;
562  uint16_t wTxSize = pInit->wTxSize;
563  fGmacdTransferCallback *pTxCb = pInit->pTxCb;
564  uint32_t dwDmaCfg;
565 
566  if (wRxSize <= 1 || wTxSize <= 1 || pTxCb == NULL) return GMACD_PARAM;
567 
568  if (!wRxBufferSize || wRxBufferSize > 16 * 1024 || wRxBufferSize & 0x3f)
569  return GMACD_PARAM;
570 
571  if (!wTxBufferSize)
572  return GMACD_PARAM;
573 
574  if (pInit->bIsGem) {
575  if (!queIdx) {
576  dwDmaCfg = (GMAC_DCFGR_DRBS(wRxBufferSize >> 6))
577  | GMAC_DCFGR_RXBMS(3) | GMAC_DCFGR_TXPBMS;
578 
579  switch (pInit->bDmaBurstLength) {
580  case 16:
581  dwDmaCfg |= GMAC_DCFGR_FBLDO_INCR16;
582  break;
583 
584  case 8:
585  dwDmaCfg |= GMAC_DCFGR_FBLDO_INCR8;
586  break;
587 
588  case 4:
589  dwDmaCfg |= GMAC_DCFGR_FBLDO_INCR4;
590  break;
591 
592  case 1:
593  dwDmaCfg |= GMAC_DCFGR_FBLDO_SINGLE;
594  break;
595 
596  default:
597  return GMACD_PARAM;
598  }
599  } else
600  dwDmaCfg = (GMAC_RBSRPQ_RBS(wRxBufferSize >> 6));
601 
602  GMAC_SetDMAConfig(pHw, dwDmaCfg, queIdx);
603  }
604 
605  pGmacd->queueList[queIdx].wRxBufferSize = wRxBufferSize;
606  pGmacd->queueList[queIdx].wTxBufferSize = wTxBufferSize;
607 
608  /* Assign RX buffers */
609  if (((uint32_t)pRxBuffer & 0x7)
610  || ((uint32_t)pRxD & 0x7)) {
611  wRxSize --;
612  TRACE_DEBUG("RX list address adjusted\n\r");
613  }
614 
615  pGmacd->queueList[queIdx].pRxBuffer = (uint8_t *)((uint32_t)pRxBuffer &
616  0xFFFFFFF8);
617  pGmacd->queueList[queIdx].pRxD = (sGmacRxDescriptor *)((
618  uint32_t)pRxD & 0xFFFFFFF8);
619  pGmacd->queueList[queIdx].wRxListSize = wRxSize;
620 
621  /* Assign TX buffers */
622  if (((uint32_t)pTxBuffer & 0x7)
623  || ((uint32_t)pTxD & 0x7)) {
624  wTxSize --;
625  TRACE_DEBUG("TX list address adjusted\n\r");
626  }
627 
628  pGmacd->queueList[queIdx].pTxBuffer = (uint8_t *)((uint32_t)pTxBuffer &
629  0xFFFFFFF8);
630  pGmacd->queueList[queIdx].pTxD = (sGmacTxDescriptor *)((
631  uint32_t)pTxD & 0xFFFFFFF8);
632  pGmacd->queueList[queIdx].wTxListSize = wTxSize;
633  pGmacd->queueList[queIdx].fTxCbList = pTxCb;
634 
635  /* Reset TX & RX */
636  GMACD_ResetRx(pGmacd, queIdx);
637  GMACD_ResetTx(pGmacd, queIdx);
638 
639  /* Setup the interrupts for RX/TX completion (and errors) */
640  switch (queIdx) {
641  case GMAC_QUE_0:
642  /* YBP: Que 0 should be configured last so as to enable transmit and
643  Receive in the NCR register */
644 
645  /* Enable Rx and Tx, plus the status register. */
646  GMAC_TransmitEnable(pHw, 1);
647  GMAC_ReceiveEnable(pHw, 1);
648  GMAC_StatisticsWriteEnable(pHw, 1);
649 
650  GMAC_EnableIt(pHw,
651  GMAC_INT_RX_BITS |
652  GMAC_INT_TX_BITS |
653  GMAC_INT_TX_ERR_BITS, GMAC_QUE_0);

713  uint16_t wTxPos, wTxHead;
714  int i;
715

717 
718  /* Check parameter */
719  if (!sgl->len) {
720  TRACE_ERROR("%s:: ethernet frame is empty.\r\n", __FUNCTION__);
721  return GMACD_PARAM;
722  }
723 
724  if (sgl->len >= pGmacd->queueList[queIdx].wTxListSize) {
725  TRACE_ERROR("%s: ethernet frame has too many buffers.\r\n", __FUNCTION__);
726  return GMACD_PARAM;
727  }
728 
729  /* Check available space */
730  if (GCIRC_SPACE(pGmacd->queueList[queIdx].wTxHead,
731  pGmacd->queueList[queIdx].wTxTail,
732  pGmacd->queueList[queIdx].wTxListSize) < (int)sgl->len)
733  return GMACD_TX_BUSY;
734 
735  /* Tag end of TX queue */
736  wTxHead = fixed_mod(pGmacd->queueList[queIdx].wTxHead + sgl->len,
737  pGmacd->queueList[queIdx].wTxListSize);
738  wTxPos = wTxHead;
739  pGmacd->queueList[queIdx].fTxCbList[wTxPos] = NULL;
740  pTxTd = &pTd[wTxPos];
741  pTxTd->status.val = GMAC_TX_USED_BIT;
742 
743  /* Update buffer descriptors in reverse order to avoid a race
744  * condition with hardware.
745  */
746  for (i = (int)(sgl->len - 1); i >= 0; --i) {
747  const sGmacSG *sg = &sgl->sg[i];
748  uint32_t status;
749 
750  if (sg->size > pGmacd->queueList[queIdx].wTxBufferSize) {
751  TRACE_ERROR("%s: buffer size is too big.\r\n", __FUNCTION__);
752  return GMACD_PARAM;
753  }
754 
755  if (wTxPos == 0)
756  wTxPos = pGmacd->queueList[queIdx].wTxListSize - 1;
757  else
758  wTxPos--;
759 
760  /* Reset TX callback */
761  pGmacd->queueList[queIdx].fTxCbList[wTxPos] = NULL;
762 
763  pTxTd = &pTd[wTxPos];
764 #ifdef GMAC_ZERO_COPY
765 
768  pTxTd->addr = (uint32_t)sg->pBuffer;
769 
770 #else
771 
772  /* Copy data into transmission buffer */
773  if (sg->pBuffer && sg->size) {
774  memcpy((void *)pTxTd->addr, sg->pBuffer, sg->size);
775  SCB_CleanDCache_by_Addr((void *)pTxTd->addr, sg->size);
776  }
777 
778 #endif
779 
780  /* Compute buffer descriptor status word */
781  status = sg->size & GMAC_LENGTH_FRAME;
782 
783  if (i == (int)(sgl->len - 1)) {
784  status |= GMAC_TX_LAST_BUFFER_BIT;
785  pGmacd->queueList[queIdx].fTxCbList[wTxPos] = fTxCb;
786  }
787 
788  if (wTxPos == pGmacd->queueList[queIdx].wTxListSize - 1)
789  status |= GMAC_TX_WRAP_BIT;
790 
791  /* Update buffer descriptor status word: clear USED bit */
792  pTxTd->status.val = status;
793 
794  memory_sync();
795  }
796 
797  /* Update TX ring buffer pointers */
798  pGmacd->queueList[queIdx].wTxHead = wTxHead;
799  /* Now start to transmit if it is not already done */
800 
801  GMAC_TransmissionStart(pHw);
802  return GMACD_OK;
803 }

885 uint8_t GMACD_Poll(sGmacd *pGmacd,
886  uint8_t *pFrame,
887  uint32_t frameSize,
888  uint32_t *pRcvSize,
889  gmacQueList_t queIdx)
890 {
891 
892  uint16_t bufferLength;
893  uint32_t tmpFrameSize = 0;
894  uint8_t *pTmpFrame = 0;
895  uint32_t tmpIdx = pGmacd->queueList[queIdx].wRxI;
896  volatile sGmacRxDescriptor *pRxTd =
897  &pGmacd->queueList[queIdx].pRxD[pGmacd->queueList[queIdx].wRxI];
898 
899  uint8_t isFrame = 0;
900 
901  if (pFrame == NULL) return GMACD_PARAM;
902 
903  /* Set the default return value */
904  *pRcvSize = 0;
905 
906  /* Process received RxTd */
907  while ((pRxTd->addr.val & GMAC_RX_OWNERSHIP_BIT) == GMAC_RX_OWNERSHIP_BIT) {
908  /* A start of frame has been received, discard previous fragments */
909  if ((pRxTd->status.val & GMAC_RX_SOF_BIT) == GMAC_RX_SOF_BIT) {
910  /* Skip previous fragment */
911  while (tmpIdx != pGmacd->queueList[queIdx].wRxI) {
912  pRxTd =
913  &pGmacd->queueList[queIdx].pRxD[pGmacd->queueList[queIdx].wRxI];
914  pRxTd->addr.val &= ~(GMAC_RX_OWNERSHIP_BIT);
915  GCIRC_INC(pGmacd->queueList[queIdx].wRxI,
916  pGmacd->queueList[queIdx].wRxListSize);
917  }
918 
919  pTmpFrame = pFrame;
920  tmpFrameSize = 0;
921  /* Start to gather buffers in a frame */
922  isFrame = 1;
923  }
924 
925  /* Increment the pointer */
926  GCIRC_INC(tmpIdx, pGmacd->queueList[queIdx].wRxListSize);
927 
928  /* Copy data in the frame buffer */
929  if (isFrame) {
930  if (tmpIdx == pGmacd->queueList[queIdx].wRxI) {
931  TRACE_INFO("no EOF (Invalid of buffers too small)\n\r");
932 
933  do {
934  pRxTd =
935  &pGmacd->queueList[queIdx].pRxD[pGmacd->queueList[queIdx].wRxI];
936  pRxTd->addr.val &= ~(GMAC_RX_OWNERSHIP_BIT);
937  GCIRC_INC(pGmacd->queueList[queIdx].wRxI,
938  pGmacd->queueList[queIdx].wRxListSize);
939  } while (tmpIdx != pGmacd->queueList[queIdx].wRxI);
940 
941  return GMACD_RX_NULL;
942  }
943 
944  /* Copy the buffer into the application frame */
945  bufferLength = pGmacd->queueList[queIdx].wRxBufferSize;
946 
947  if ((tmpFrameSize + bufferLength) > frameSize)
948  bufferLength = frameSize - tmpFrameSize;
949 
950  SCB_InvalidateDCache_by_Addr((void *)(pRxTd->addr.val & GMAC_ADDRESS_MASK) ,
951  bufferLength);
952  memcpy(pTmpFrame, (void *)(pRxTd->addr.val & GMAC_ADDRESS_MASK),
953  bufferLength);
954  pTmpFrame += bufferLength;
955  tmpFrameSize += bufferLength;
956 
957  /* An end of frame has been received, return the data */
958  if ((pRxTd->status.val & GMAC_RX_EOF_BIT) == GMAC_RX_EOF_BIT) {
959  /* Frame size from the GMAC */
960  *pRcvSize = (pRxTd->status.val & GMAC_LENGTH_FRAME);
961 
962  /* Application frame buffer is too small all data have not been
963  copied */
964  if (tmpFrameSize < *pRcvSize)
965  return GMACD_SIZE_TOO_SMALL;
966 
967  TRACE_DEBUG("packet %d-%d (%d)\n\r",
968  pGmacd->queueList[queIdx].wRxI, tmpIdx, *pRcvSize);
969 
970  /* All data have been copied in the application frame buffer
971  => release TD */
972  while (pGmacd->queueList[queIdx].wRxI != tmpIdx) {
973  pRxTd =
974  &pGmacd->queueList[queIdx].pRxD[pGmacd->queueList[queIdx].wRxI];
975  pRxTd->addr.val &= ~(GMAC_RX_OWNERSHIP_BIT);
976  GCIRC_INC(pGmacd->queueList[queIdx].wRxI,
977  pGmacd->queueList[queIdx].wRxListSize);
978  }
979 
980  return GMACD_OK;
981  }
982  } else {
983  /* SOF has not been detected, skip the fragment */
984  pRxTd->addr.val &= ~(GMAC_RX_OWNERSHIP_BIT);
985  pGmacd->queueList[queIdx].wRxI = tmpIdx;
986  }
987 
988  /* Process the next buffer */
989  pRxTd = &pGmacd->queueList[queIdx].pRxD[tmpIdx];
990  }
991 
992  return GMACD_RX_NULL;
993 }
