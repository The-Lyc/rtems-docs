75 /*
76  * Set up the clock hardware
77  *
78  * We need to have 1 interrupt every 10,000 microseconds
79  * XLB clock 100 MHz / MCF548X_SLT_SLTCNT0 = XLB clock/100
80  */
81 #define Clock_driver_support_initialize_hardware() \
82  do { \
83  bsp_interrupt_vector_enable(CLOCK_IRQ); \
84  MCF548X_SLT_SLTCNT0 = get_CPU_clock_speed() \
85  / 1000 \
86  * rtems_configuration_get_microseconds_per_tick() \
87  / 1000; \
88  MCF548X_SLT_SCR0 |= (MCF548X_SLT_SCR_TEN | MCF548X_SLT_SCR_RUN | MCF548X_SLT_SCR_IEN); \
89  } while (0)
