346 /****************************************************************************************/
347 /* alt_gpio_port_config() configures a group of GPIO signals with the same parameters. */
348 /* Allows for configuring all parameters of a given port at one time. */
349 /****************************************************************************************/
350 
351 ALT_STATUS_CODE alt_gpio_port_config(ALT_GPIO_PORT_t gpio_pid,
352  uint32_t mask, ALT_GPIO_PIN_DIR_t dir, ALT_GPIO_PIN_TYPE_t type,
353  ALT_GPIO_PIN_POL_t pol, ALT_GPIO_PIN_DEBOUNCE_t debounc,
354  uint32_t data)
355 {
356  ALT_STATUS_CODE ret;
357 
358  // set all affected GPIO bits to inputs
359  ret = alt_gpio_port_datadir_set(gpio_pid, mask, ALT_GPIO_ALLORNONE(ALT_GPIO_PIN_INPUT));
360  // the ALT_GPIO_ALLORNONE() macro expands the zero or one bit to the 29-bit GPIO word
361 
362  // set trigger type
363  if (ret == ALT_E_SUCCESS)
364  {
365  ret = alt_gpio_port_int_type_set(gpio_pid, mask, ALT_GPIO_ALLORNONE(type));
366  }
367 
368  // set polarity
369  if (ret == ALT_E_SUCCESS)
370  {
371  alt_gpio_port_int_pol_set(gpio_pid, mask, ALT_GPIO_ALLORNONE(pol));
372  }
373 
374  // set debounce
375  if (ret == ALT_E_SUCCESS)
376  {
377  alt_gpio_port_debounce_set(gpio_pid, mask, ALT_GPIO_ALLORNONE(debounc));
378  }
379 
380  // set data output(s)
381  if (ret == ALT_E_SUCCESS)
382  {
383  alt_gpio_port_data_write(gpio_pid, mask, ALT_GPIO_ALLORNONE(data));
384  }
385 
386  if (ret == ALT_E_SUCCESS)
387  {
388  // set data direction of one or more bits to select output
389  ret = alt_gpio_port_datadir_set(gpio_pid, mask, ALT_GPIO_ALLORNONE(dir));
390  }
391 
392  return ret;
393 }

583 /* Returns the configuration parameters of a given GPIO bit. */
584 /****************************************************************************************/

586 ALT_STATUS_CODE alt_gpio_bitconfig_get(ALT_GPIO_1BIT_t signal_num,
587  ALT_GPIO_CONFIG_RECORD_t *config)
588 {
589  ALT_STATUS_CODE ret = ALT_E_ERROR;
590  ALT_GPIO_PORT_t pid;
591  uint32_t mask, shift;
592 
593  if ((config != NULL) && (signal_num != ALT_END_OF_GPIO_SIGNALS) && (signal_num <= ALT_LAST_VALID_GPIO_BIT))
594  {
595  pid = alt_gpio_bit_to_pid(signal_num);
596  shift = alt_gpio_bit_to_port_pin(signal_num);
597  if ((pid != ALT_GPIO_PORT_UNKNOWN) && (shift <= ALT_GPIO_BIT_MAX))
598  {
599  config->signal_number = signal_num;
600  mask = 0x00000001 << shift;
601  config->direction = (alt_gpio_port_datadir_get(pid, mask) == 0) ? ALT_GPIO_PIN_INPUT : ALT_GPIO_PIN_OUTPUT;
602  config->type = (alt_gpio_port_int_type_get(pid, mask) == 0) ? ALT_GPIO_PIN_LEVEL_TRIG_INT : ALT_GPIO_PIN_EDGE_TRIG_INT;
603 
604  // save the following data whatever the state of config->direction
605  config->polarity = (alt_gpio_port_int_pol_get(pid, mask) == 0) ? ALT_GPIO_PIN_ACTIVE_LOW : ALT_GPIO_PIN_ACTIVE_HIGH;
606  config->debounce = (alt_gpio_port_debounce_get(pid, mask) == 0) ? ALT_GPIO_PIN_NODEBOUNCE : ALT_GPIO_PIN_DEBOUNCE;
607  config->data = (alt_gpio_port_data_read(pid, mask) == 0) ? ALT_GPIO_PIN_DATAZERO : ALT_GPIO_PIN_DATAONE;
608  ret = ALT_E_SUCCESS;
609  }
610  }

615 /****************************************************************************************/
616 /* alt_gpio_group_config() configures a list of GPIO bits. The GPIO bits do not have */
617 /* to be configured the same, as was the case for the mask version of this function, */
618 /* alt_gpio_port_config(). Each bit may be configured differently and bits may be */
619 /* listed in any order. */
620 /****************************************************************************************/
621 
622 ALT_STATUS_CODE alt_gpio_group_config(ALT_GPIO_CONFIG_RECORD_t* config_array, uint32_t len)
623 {
624  ALT_STATUS_CODE ret = ALT_E_ERROR;
625 
626  if (config_array != NULL)
627  {
628  if (config_array->signal_number == ALT_END_OF_GPIO_SIGNALS) { ret = ALT_E_SUCCESS; }
629  // catches the condition where the pointers are good, but the
630  // first index is the escape character - which isn't an error
631  else
632  {
633  for (; (len-- > 0) && (config_array->signal_number != ALT_END_OF_GPIO_SIGNALS) && (config_array != NULL); config_array++)
634  {
635  ret = alt_gpio_bit_config(config_array->signal_number,
636  config_array->direction, config_array->type, config_array->polarity,
637  config_array->debounce, config_array->data);
638  if ((config_array->direction == ALT_GPIO_PIN_OUTPUT) && (ret == ALT_E_SUCCESS))
639  {
640  // if the pin is set to be an output, set it to the correct value
641  alt_gpio_port_data_write(alt_gpio_bit_to_pid(config_array->signal_number),
642  0x1 << alt_gpio_bit_to_port_pin(config_array->signal_number),
643  ALT_GPIO_ALLORNONE(config_array->data));
644  // ret should retain the value returned by alt_gpio_bit_config() above
645  // and should not be changed by the alt_gpio_port_data_write() call.
646  }
647  if (((ret != ALT_E_SUCCESS) && (config_array->signal_number <= ALT_LAST_VALID_GPIO_BIT))
648  || ((ret == ALT_E_SUCCESS) && (config_array->signal_number > ALT_LAST_VALID_GPIO_BIT)))
649  {
650  ret = ALT_E_ERROR;
651  break;
652  }
653  }
654  }
655  }
656  return ret;
657 }

743 /****************************************************************************************/
744 /* A useful utility function. Extracts the GPIO signal (pin) mask from the supplied */
745 /* GPIO Signal Index Number. */
746 /****************************************************************************************/
747 
748 ALT_GPIO_PORTBIT_t alt_gpio_bit_to_port_pin(ALT_GPIO_1BIT_t pin_num)
749 {

751  else if (pin_num <= ALT_GPIO_EOPB) { pin_num -= (ALT_GPIO_EOPA + 1); }
752  else if (pin_num <= ALT_GPIO_EOPC) { pin_num -= (ALT_GPIO_EOPB + 1); }

758 /****************************************************************************************/
759 /* A useful utility function. Extracts the GPIO Signal Index Number from the supplied */
760 /* GPIO port ID and signal mask. If passed a bitmask composed of more than one signal, */
761 /* the signal number of the lowest bitmask presented is returned. */
762 /****************************************************************************************/
763 
764 ALT_GPIO_1BIT_t alt_gpio_port_pin_to_bit(ALT_GPIO_PORT_t pid,
765  uint32_t bitmask)
766 {
767  uint32_t i;
768 
769  for (i=0; i <= ALT_GPIO_BITNUM_MAX ;i++)
770  {
771  if (bitmask & 0x00000001)
772  {
773  if (pid == ALT_GPIO_PORTA) {}
774  else if (pid == ALT_GPIO_PORTB) { i += ALT_GPIO_EOPA + 1; }
775  else if (pid == ALT_GPIO_PORTC) { i += ALT_GPIO_EOPB + 1; }
776  else { return ALT_END_OF_GPIO_SIGNALS; }
777  return (ALT_GPIO_1BIT_t) i;
778  }
779  bitmask >>= 1;
780  }
781  return ALT_END_OF_GPIO_SIGNALS;
782 }
