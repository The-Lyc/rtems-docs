30 /* Before UART driver has registered (or when no UART is available), calls to
31  * printk that gets to bsp_out_char() will be filling data into the
32  * pre_printk_dbgbuf[] buffer, hopefully the buffer can help debugging the
33  * early BSP boot.. At least the last printk() will be caught.
34  */
35 static char pre_printk_dbgbuf[32] = {0};

38 /* Initialize the BSP system debug console layer. It will scan AMBA Plu&Play
39  * for a debug APBUART and enable RX/TX for that UART.
40  */
41 static void bsp_debug_uart_init(void)
42 {
43  int i;
44  struct ambapp_dev *adev;
45  struct ambapp_apb_info *apb;
46 
47  /* Update leon3_debug_uart_index to index used as debug console. Let user
48  * select Debug console by setting leon3_debug_uart_index. If the BSP is to
49  * provide the default UART (leon3_debug_uart_index==0):
50  * non-MP: APBUART[0] is debug console
51  * MP: LEON CPU index select UART
52  */
53  if (leon3_debug_uart_index == 0) {
54 #if defined(RTEMS_MULTIPROCESSING)
55  leon3_debug_uart_index = LEON3_Cpu_Index;
56 #else
57  leon3_debug_uart_index = 0;
58 #endif
59  } else {
60  leon3_debug_uart_index--; /* User selected dbg-console */
61  }
62 
63  /* Find APBUART core for System Debug Console */
64  i = leon3_debug_uart_index;
65  adev = (void *)ambapp_for_each(&ambapp_plb, (OPTIONS_ALL|OPTIONS_APB_SLVS),
66  VENDOR_GAISLER, GAISLER_APBUART,
67  ambapp_find_by_idx, (void *)&i);
68  if (adev) {
69  /* Found a matching debug console, initialize debug uart if present
70  * for printk
71  */
72  apb = (struct ambapp_apb_info *)adev->devinfo;
73  leon3_debug_uart = (struct apbuart_regs *)apb->start;
74  leon3_debug_uart->ctrl |= APBUART_CTRL_RE | APBUART_CTRL_TE;
75  leon3_debug_uart->status = 0;
76  }
77 }

85 /* putchar/getchar for printk */
86 static void bsp_out_char(char c)
87 {
88  if (leon3_debug_uart == NULL) {
89  /* Local debug buffer when UART driver has not registered */
90  pre_printk_dbgbuf[pre_printk_pos++] = c;
91  pre_printk_pos = pre_printk_pos & (sizeof(pre_printk_dbgbuf)-1);
92  return;
93  }
94 
95  apbuart_outbyte_polled(leon3_debug_uart, c, 1, 1);
96 }
