110 static void resolve_hold_back(
111  rtems_record_client_context *ctx,
112  rtems_record_client_per_cpu *per_cpu
113 )
114 {
115  if ( per_cpu->hold_back ) {
116  uint32_t last_tail;
117  uint32_t last_head;
118  uint32_t last_capacity;
119  uint32_t new_head;
120  uint32_t new_content;
121  uint32_t begin_index;
122  uint32_t index;
123  uint32_t first;
124  uint32_t last;
125  uint32_t delta;
126  uint64_t uptime;
127 
128  last_head = per_cpu->head[ per_cpu->tail_head_index ];
129  last_tail = per_cpu->tail[ per_cpu->tail_head_index ];
130  new_head = per_cpu->head[ per_cpu->tail_head_index ^ 1 ];
131  last_capacity = ( last_tail - last_head - 1 ) & ( ctx->count - 1 );
132  new_content = new_head - last_head;
133 
134  if ( new_content > last_capacity ) {
135  begin_index = new_content - last_capacity;
136  per_cpu->overflow += begin_index;
137  } else {
138  begin_index = 0;
139  }
140 
141  if ( begin_index >= per_cpu->item_index ) {
142  per_cpu->item_index = 0;
143  return;
144  }
145 
146  per_cpu->hold_back = false;
147 
148  first = RTEMS_RECORD_GET_TIME( per_cpu->items[ begin_index ].event );
149  last = first;
150  delta = 0;
151  uptime = 0;
152 
153  for ( index = begin_index; index < per_cpu->item_index; ++index ) {
154  const rtems_record_item_64 *item;
155  rtems_record_event event;
156  uint32_t time;
157 
158  item = &per_cpu->items[ index ];
159  event = RTEMS_RECORD_GET_EVENT( item->event );
160  time = RTEMS_RECORD_GET_TIME( item->event );
161  delta += ( time - last ) & TIME_MASK;
162  last = time;
163 
164  if (
165  event == RTEMS_RECORD_UPTIME_LOW
166  && index + 1 < per_cpu->item_index
167  && RTEMS_RECORD_GET_EVENT( ( item + 1 )->event )
168  == RTEMS_RECORD_UPTIME_HIGH
169  ) {
170  uptime = (uint32_t) item->data;
171  uptime += ( item + 1 )->data << 32;
172  break;
173  }
174  }
175 
176  per_cpu->uptime.bt = uptime - ( ( delta * ctx->to_bt_scaler ) >> 31 );
177  per_cpu->uptime.time_at_bt = first;
178  per_cpu->uptime.time_last = first;
179  per_cpu->uptime.time_accumulated = 0;
180 
181  if ( per_cpu->overflow > 0 ) {
182  call_handler(
183  ctx,
184  per_cpu->uptime.bt,
185  RTEMS_RECORD_PER_CPU_OVERFLOW,
186  per_cpu->overflow
187  );
188  per_cpu->overflow = 0;
189  }
190 
191  for ( index = begin_index; index < per_cpu->item_index; ++index ) {
192  const rtems_record_item_64 *item;
193 
194  item = &per_cpu->items[ index ];
195  visit( ctx, item->event, item->data );
196  }
197  }
198 }

200 static void process_per_cpu_head(
201  rtems_record_client_context *ctx,
202  rtems_record_client_per_cpu *per_cpu,
203  uint64_t data
204 )
205 {
206  uint32_t last_tail;
207  uint32_t last_head;
208  uint32_t last_capacity;
209  uint32_t new_tail;
210  uint32_t new_head;
211  uint32_t new_content;
212  uint32_t content;
213 
214  new_tail = per_cpu->tail[ per_cpu->tail_head_index ];
215  new_head = (uint32_t) data;
216  content = new_head - new_tail;
217 
218  per_cpu->head[ per_cpu->tail_head_index ] = new_head;
219  per_cpu->tail_head_index ^= 1;
220 
221  if ( content >= ctx->count ) {
222  /*
223  * This is a complete ring buffer overflow, the server will detect this
224  * also. It sets the tail to the head plus one and sends us all the
225  * content. This reduces the ring buffer capacity to zero. So, during
226  * transfer, new events will overwrite items in transfer. This is handled
227  * by resolve_hold_back().
228  */
229  per_cpu->tail[ per_cpu->tail_head_index ^ 1 ] = new_head + 1;
230  signal_overflow( per_cpu, content - ctx->count + 1 );
231  return;
232  }
233 
234  last_tail = per_cpu->tail[ per_cpu->tail_head_index ];
235  last_head = per_cpu->head[ per_cpu->tail_head_index ];
236 
237  if ( last_tail == last_head ) {
238  if ( per_cpu->uptime.bt == 0 ) {
239  /*
240  * This is a special case during initial ramp up. We hold back the items
241  * to deduce the uptime of the first item via resolve_hold_back().
242  */
243  per_cpu->hold_back = true;
244  } else {
245  resolve_hold_back( ctx, per_cpu );
246  }
247 
248  return;
249  }
250 
251  last_capacity = ( last_tail - last_head - 1 ) & ( ctx->count - 1 );
252  new_content = new_head - last_head;
253 
254  if ( new_content <= last_capacity || per_cpu->hold_back ) {
255  resolve_hold_back( ctx, per_cpu );
256  return;
257  }
258 
259  signal_overflow( per_cpu, new_content - last_capacity );
260 }

262 static rtems_record_client_status process_per_cpu_count(
263  rtems_record_client_context *ctx,
264  uint64_t data
265 )
266 {
267  size_t per_cpu_items;
268  rtems_record_item_64 *items;
269  uint32_t cpu;
270 
271  if ( ctx->count != 0 ) {
272  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_DOUBLE_PER_CPU_COUNT );
273  }
274 
275  if ( ctx->cpu_count == 0 ) {
276  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_NO_CPU_MAX );
277  }
278 
279  ctx->count = (uint32_t) data;
280 
281  /*
282  * The ring buffer capacity plus two items for RTEMS_RECORD_PROCESSOR and
283  * RTEMS_RECORD_PER_CPU_TAIL.
284  */
285  per_cpu_items = ctx->count + 1;
286 
287  items = malloc( per_cpu_items * ctx->cpu_count * sizeof( *items ) );
288 
289  if ( items == NULL ) {
290  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_NO_MEMORY );
291  }
292 
293  for ( cpu = 0; cpu < ctx->cpu_count; ++cpu ) {
294  ctx->per_cpu[ cpu ].items = items;
295  items += per_cpu_items;
296  }
297 
298  return RTEMS_RECORD_CLIENT_SUCCESS;
299 }

301 static rtems_record_client_status hold_back(
302  rtems_record_client_context *ctx,
303  rtems_record_client_per_cpu *per_cpu,
304  uint32_t time_event,
305  rtems_record_event event,
306  uint64_t data
307 )
308 {
309  if ( event != RTEMS_RECORD_PER_CPU_HEAD ) {
310  uint32_t item_index;
311 
312  item_index = per_cpu->item_index;
313 
314  if ( item_index <= ctx->count ) {
315  per_cpu->items[ item_index ].event = time_event;
316  per_cpu->items[ item_index ].data = data;
317  per_cpu->item_index = item_index + 1;
318  } else {
319  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_PER_CPU_ITEMS_OVERFLOW );
320  }
321  } else {
322  return call_handler( ctx, 0, RTEMS_RECORD_GET_EVENT( time_event ), data );
323  }
324 
325  return RTEMS_RECORD_CLIENT_SUCCESS;
326 }

328 static uint64_t time_bt(
329  const rtems_record_client_context *ctx,
330  rtems_record_client_per_cpu *per_cpu,
331  uint32_t time
332 )
333 {
334  uint64_t bt;
335 
336  if ( time != 0 ) {
337  uint32_t delta;
338 
339  delta = ( time - per_cpu->uptime.time_last ) & TIME_MASK;
340  per_cpu->uptime.time_last = time;
341  per_cpu->uptime.time_accumulated += delta;
342  bt = ( per_cpu->uptime.time_accumulated * ctx->to_bt_scaler ) >> 31;
343  bt += per_cpu->uptime.bt;
344  } else {
345  bt = 0;
346  }
347 
348  return bt;
349 }

351 static rtems_record_client_status visit(
352  rtems_record_client_context *ctx,
353  uint32_t time_event,
354  uint64_t data
355 )
356 {
357  rtems_record_client_per_cpu *per_cpu;
358  uint32_t time;
359  rtems_record_event event;
360  rtems_record_client_status status;
361 
362  per_cpu = &ctx->per_cpu[ ctx->cpu ];
363  time = RTEMS_RECORD_GET_TIME( time_event );
364  event = RTEMS_RECORD_GET_EVENT( time_event );
365 
366  switch ( event ) {
367  case RTEMS_RECORD_PROCESSOR:
368  if ( data >= ctx->cpu_count ) {
369  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_UNSUPPORTED_CPU );
370  }
371 
372  ctx->cpu = (uint32_t) data;
373  per_cpu = &ctx->per_cpu[ ctx->cpu ];
374  break;
375  case RTEMS_RECORD_UPTIME_LOW:
376  per_cpu->uptime.bt = (uint32_t) data;
377  per_cpu->uptime.time_at_bt = time;
378  per_cpu->uptime.time_last = time;
379  per_cpu->uptime.time_accumulated = 0;
380  time = 0;
381  break;
382  case RTEMS_RECORD_UPTIME_HIGH:
383  per_cpu->uptime.bt += data << 32;
384  time = 0;
385  break;
386  case RTEMS_RECORD_PER_CPU_TAIL:
387  per_cpu->tail[ per_cpu->tail_head_index ] = (uint32_t) data;
388  break;
389  case RTEMS_RECORD_PER_CPU_HEAD:
390  process_per_cpu_head( ctx, per_cpu, data );
391  break;
392  case RTEMS_RECORD_PROCESSOR_MAXIMUM:
393  if ( data >= RTEMS_RECORD_CLIENT_MAXIMUM_CPU_COUNT ) {
394  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_UNSUPPORTED_CPU_MAX );
395  }
396 
397  if ( ctx->cpu_count != 0 ) {
398  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_DOUBLE_CPU_MAX );
399  }
400 
401  ctx->cpu_count = (uint32_t) data + 1;
402  break;
403  case RTEMS_RECORD_PER_CPU_COUNT:
404  status = process_per_cpu_count( ctx, data );
405 
406  if ( status != RTEMS_RECORD_CLIENT_SUCCESS ) {
407  return status;
408  }
409 
410  break;
411  case RTEMS_RECORD_FREQUENCY:
412  set_to_bt_scaler( ctx, (uint32_t) data );
413  break;
414  case RTEMS_RECORD_VERSION:
415  if ( data != RTEMS_RECORD_THE_VERSION ) {
416  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_UNSUPPORTED_VERSION );
417  }
418 
419  break;
420  default:
421  break;
422  }
423 
424  if ( per_cpu->hold_back ) {
425  return hold_back( ctx, per_cpu, time_event, event, data );
426  }
427 
428  return call_handler( ctx, time_bt( ctx, per_cpu, time ), event, data );
429 }

432  rtems_record_client_context *ctx,
433  const void *buf,
434  size_t n
435 )
436 {
437  while ( n > 0 ) {
438  size_t m;
439  char *pos;
440 
441  m = ctx->todo < n ? ctx->todo : n;
442  pos = ctx->pos;
443  pos = memcpy( pos, buf, m );
444  n -= m;
445  buf = (char *) buf + m;
446 
447  if ( m == ctx->todo ) {
448  rtems_record_client_status status;
449 
450  ctx->todo = sizeof( ctx->item.format_32 );
451  ctx->pos = &ctx->item.format_32;
452 
453  status = visit(
454  ctx,
455  ctx->item.format_32.event,
456  ctx->item.format_32.data
457  );
458 
459  if ( status != RTEMS_RECORD_CLIENT_SUCCESS ) {
460  return status;
461  }
462  } else {
463  ctx->todo -= m;
464  ctx->pos = pos + m;
465  }
466  }
467 
468  return RTEMS_RECORD_CLIENT_SUCCESS;

591 static rtems_record_client_status consume_init(
592  rtems_record_client_context *ctx,
593  const void *buf,
594  size_t n
595 )
596 {
597  while ( n > 0 ) {
598  size_t m;
599  char *pos;
600 
601  m = ctx->todo < n ? ctx->todo : n;
602  pos = ctx->pos;
603  pos = memcpy( pos, buf, m );
604  n -= m;
605  buf = (char *) buf + m;
606 
607  if ( m == ctx->todo ) {
608  uint32_t magic;
609 
610  magic = ctx->header[ 1 ];
611 
612  switch ( ctx->header[ 0 ] ) {
613 #if __BYTE_ORDER__ == __ORDER_LITTLE_ENDIAN__
614  case RTEMS_RECORD_FORMAT_LE_32:
615  ctx->todo = sizeof( ctx->item.format_32 );
616  ctx->pos = &ctx->item.format_32;
617  ctx->consume = consume_32;
618  ctx->data_size = 4;
619  break;
620  case RTEMS_RECORD_FORMAT_LE_64:
621  ctx->todo = sizeof( ctx->item.format_64 );
622  ctx->pos = &ctx->item.format_64;
623  ctx->consume = consume_64;
624  ctx->data_size = 8;
625  break;
626  case RTEMS_RECORD_FORMAT_BE_32:
627  ctx->todo = sizeof( ctx->item.format_32 );
628  ctx->pos = &ctx->item.format_32;
629  ctx->consume = consume_swap_32;
630  ctx->data_size = 4;
631  magic = __builtin_bswap32( magic );
632  break;
633  case RTEMS_RECORD_FORMAT_BE_64:
634  ctx->todo = sizeof( ctx->item.format_64 );
635  ctx->pos = &ctx->item.format_64;
636  ctx->consume = consume_swap_64;
637  ctx->data_size = 8;
638  magic = __builtin_bswap32( magic );
639  break;
640 #elif __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
641  case RTEMS_RECORD_FORMAT_LE_32:
642  ctx->todo = sizeof( ctx->item.format_32 );
643  ctx->pos = &ctx->item.format_32;
644  ctx->consume = consume_swap_32;
645  ctx->data_size = 4;
646  magic = __builtin_bswap32( magic );
647  break;
648  case RTEMS_RECORD_FORMAT_LE_64:
649  ctx->todo = sizeof( ctx->item.format_64 );
650  ctx->pos = &ctx->item.format_64;
651  ctx->consume = consume_swap_64;
652  ctx->data_size = 8;
653  magic = __builtin_bswap32( magic );
654  break;
655  case RTEMS_RECORD_FORMAT_BE_32:
656  ctx->todo = sizeof( ctx->item.format_32 );
657  ctx->pos = &ctx->item.format_32;
658  ctx->consume = consume_32;
659  ctx->data_size = 4;
660  break;
661  case RTEMS_RECORD_FORMAT_BE_64:
662  ctx->todo = sizeof( ctx->item.format_64 );
663  ctx->pos = &ctx->item.format_64;
664  ctx->consume = consume_64;
665  ctx->data_size = 8;
666  break;
667 #else
668 #error "unexpected __BYTE_ORDER__"
669 #endif
670  default:
671  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_UNKNOWN_FORMAT );
672  }
673 
674  if ( magic != RTEMS_RECORD_MAGIC ) {
675  return error( ctx, RTEMS_RECORD_CLIENT_ERROR_INVALID_MAGIC );
676  }
677 
678  return rtems_record_client_run( ctx, buf, n );
679  } else {
680  ctx->todo -= m;
681  ctx->pos = pos + m;
682  }
683  }
684 
685  return RTEMS_RECORD_CLIENT_SUCCESS;
686 }

712 void rtems_record_client_destroy(
713  rtems_record_client_context *ctx
714 )
