1 /*
2  * monlib.c -
3  * This file is part of the monitor code, but it is actually linked into
4  * the application. It is built with (but not linked with) the monitor,
5  * then the monlib.o file is linked with the application.
6  * The only requirement on the application is that it know where the address
7  * of the monCom function is in the monitor's space.
8  * The monCom function will be accessible in some "well-known" way (processor
9  * and platform dependent) so that this will not be a problem.
10  *
11  * This monlib.c file is a replacement for the older mechanism that was
12  * a bit more error-prone... A table of function pointers existed at some
13  * well-known location in the monitor, and the content of that table was
14  * assumed to also be "well-known". This new version only assumes that the
15  * pointer to monCom is well-known; everything else will work based on the
16  * fact that the monitor and application will share the monlib.h header
17  * file.
18  *

25  * so that all may benefit from the changes. In addition, notification back

211  *
212  * monConnect():
213  * This must be the first call by the application code to talk to the
214  * monitor. It is expecting three incoming function pointers:
215  *
216  * mon: Points to the monitor's _moncom function;
217  * This is a "well-known" address because the monitor and
218  * application code (two separately linked binaries) must
219  * know it.
220  * lock: Points to a function in the application code that will be
221  * used by the monitor as a lock-out function (some kind of
222  * semaphore in the application).
223  * unlock: Points to a function in the application code that will be
224  * used by the monitor as an un-lock-out function (undo whatever
225  * lock-out mechanism was done by lock).
226  */

319 /* ignorelock:
320  * Used as a back-door to disable the monLock()/monUnlock() stuff.
321  * This is useful if the application CLI falls through to the monitor's
322  * CLI and you are using the "call" command in the monitor to execute some
323  * function that has a mon_xxx function in it. In this case, the fact that
324  * the application has fallen through to the monitor means that the lock
325  * is already active, so when the function tries to call some other mon_xxx
326  * function it won't be able to because of the lock already being set.
327  *
328  * With these functions in the application space, the user can do the
329  * following:
330  * call %DisableLock
331  * call %Func_with_monXXX_in_it
332  * call %EnableLock
333  *
334  * Note that this is NOT to be used by application code, it is simply a
335  * back-door mechanism to allow "call" from the CLI to invoke functions
336  * that have mon_XXX functionality in them.
337  */
338 static int ignorelock = 0;
339

354 /* monLock() & monUnlock():
355  * Used by all of the wrapper functions below this point to call
356  * the function pointed to by _monlock & _monunlock function pointers
357  * (if set).
358  * These functions must test both the function pointer and the state
359  * of the ignorelock variable. The function DisableMonLock() sets the
360  * ignorelock variable to 2 because it is being executed through "call"
361  * which means that the lock is active.
362  */
363 static void
364 monLock(void)
365 {
366  if (_monlock) {
367  switch(ignorelock) {
368  case 1:
369  break;
370  case 2:
371  ignorelock--;
372  break;
373  default:
374  _monlock();
375  break;
376  }
377  }
378 }
379 
380 static void
381 monUnlock(void)
382 {
383  if (_monunlock) {
384  switch(ignorelock) {
385  case 1:
386  break;
387  case 2:
388  ignorelock--;
389  default:
390  _monunlock();
391  break;
392  }
393  }
394 }
