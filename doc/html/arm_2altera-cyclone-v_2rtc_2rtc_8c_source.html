21 /*
22  * Driver for the DS1339 RTC (Maxim Semiconductors) -> RTC1
23  * and the M41ST87 RTC (ST Microelectronics) -> RTC2
24  *
25  * Please note the following points:
26  * - The day of week is ignored.
27  * - The century bit is interpreted the following way:
28  * - century not set: TOD_BASE_YEAR .. 1999
29  * - century set: 2000 .. 2099
30  * - century not set: 2100 .. (TOD_BASE_YEAR + 200)
31  */

171 static void ds1339_set_time(ds1339_time_t* time,
172  unsigned int second,
173  unsigned int minute,
174  unsigned int hour,
175  unsigned int day,
176  unsigned int month,
177  unsigned int year)
178 {
179 
180  unsigned int tens;
181  unsigned int ones;
182  uint8_t century = 0;
183 
184  tens = second / 10;
185  ones = second % 10;
186  time->seconds = tens << 4 | ones;
187 
188  tens = minute / 10;
189  ones = minute % 10;
190  time->minutes = tens << 4 | ones;
191 
192  tens = hour / 10;
193  ones = hour % 10;
194  time->hours = tens << 4 | ones;
195 
196  /* Weekday is not used. Therefore it can be set to an arbitrary valid value */
197  time->weekday = 1;
198 
199  tens = day / 10;
200  ones = day % 10;
201  time->date = tens << 4 | ones;
202 
203  tens = month / 10;
204  ones = month % 10;
205  if ((year >= 2000) && (year < 2100))
206  century = DS1339_MONTH_CENTURY;
207  time->month = century | tens << 4 | ones;
208 
209  tens = (year % 100) / 10;
210  ones = year % 10;
211  time->year = tens << 4 | ones;

305 static void altera_cyclone_v_ds1339_initialize(int minor)
306 {
307 
308  rtems_status_code sc = RTEMS_SUCCESSFUL;
309  uint8_t status = 0;
310 
311  /* Check RTC valid */
312  sc = ds1339_read(DS1339_ADDR_STATUS, &status, sizeof(status));
313  assert(sc == RTEMS_SUCCESSFUL);
314 
315  if (status & DS1339_STATUS_OSF)
316  {
317  /* RTC has been stopped. Initialise it. */
318  ds1339_time_t time;
319 
320  uint8_t write = DS1339_CTRL_DEFAULT;
321  sc = ds1339_write(DS1339_ADDR_CTRL, &write, sizeof(write));
322  assert(sc == RTEMS_SUCCESSFUL);
323 
324  write = DS1339_STATUS_CLEAR;
325  sc = ds1339_write(DS1339_ADDR_STATUS, &write, sizeof(write));
326  assert(sc == RTEMS_SUCCESSFUL);
327 
328  ds1339_set_time(&time, 0, 0, 0, 1, 1, TOD_BASE_YEAR);
329  sc = ds1339_write(DS1339_ADDR_TIME, &time, sizeof(time));
330  assert(sc == RTEMS_SUCCESSFUL);
331  }

387 {
388

390  uint8_t buf;
391

398  sc = ds1339_read(0x20, &buf, 1);
