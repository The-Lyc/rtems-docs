26 static void _Heap_Free_block( Heap_Control *heap, Heap_Block *block )
27 {
28  Heap_Statistics *const stats = &heap->stats;
29  Heap_Block *first_free;
30 
31  /* Statistics */
32  ++stats->used_blocks;
33  --stats->frees;
34 
35  /*
36  * The _Heap_Free() will place the block to the head of free list. We want
37  * the new block at the end of the free list. So that initial and earlier
38  * areas are consumed first.
39  */
40  _Heap_Free( heap, (void *) _Heap_Alloc_area_of_block( block ) );
41  _Heap_Protection_free_all_delayed_blocks( heap );
42  first_free = _Heap_Free_list_first( heap );
43  _Heap_Free_list_remove( first_free );
44  _Heap_Free_list_insert_before( _Heap_Free_list_tail( heap ), first_free );
45 }

119 uintptr_t _Heap_Extend(
120  Heap_Control *heap,
121  void *extend_area_begin_ptr,
122  uintptr_t extend_area_size,
123  uintptr_t unused RTEMS_UNUSED
124 )
125 {
126  Heap_Statistics *const stats = &heap->stats;
127  Heap_Block *const first_block = heap->first_block;
128  Heap_Block *start_block = first_block;
129  Heap_Block *merge_below_block = NULL;
130  Heap_Block *merge_above_block = NULL;
131  Heap_Block *link_below_block = NULL;
132  Heap_Block *link_above_block = NULL;
133  Heap_Block *extend_first_block = NULL;
134  Heap_Block *extend_last_block = NULL;
135  uintptr_t const page_size = heap->page_size;
136  uintptr_t const min_block_size = heap->min_block_size;
137  uintptr_t const extend_area_begin = (uintptr_t) extend_area_begin_ptr;
138  uintptr_t const extend_area_end = extend_area_begin + extend_area_size;
139  uintptr_t const free_size = stats->free_size;
140  uintptr_t extend_first_block_size = 0;
141  uintptr_t extended_size = 0;
142  bool extend_area_ok = false;
143 
144  if ( extend_area_end < extend_area_begin ) {
145  return 0;
146  }
147 
148  extend_area_ok = _Heap_Get_first_and_last_block(
149  extend_area_begin,
150  extend_area_size,
151  page_size,
152  min_block_size,
153  &extend_first_block,
154  &extend_last_block
155  );
156  if (!extend_area_ok ) {
157  /* For simplicity we reject extend areas that are too small */
158  return 0;
159  }
160 
161  do {
162  uintptr_t const sub_area_begin = (start_block != first_block) ?
163  (uintptr_t) start_block : heap->area_begin;
164  uintptr_t const sub_area_end = start_block->prev_size;
165  Heap_Block *const end_block =
166  _Heap_Block_of_alloc_area( sub_area_end, page_size );
167 
168  if (
169  sub_area_end > extend_area_begin && extend_area_end > sub_area_begin
170  ) {
171  return 0;
172  }
173 
174  if ( extend_area_end == sub_area_begin ) {
175  merge_below_block = start_block;
176  } else if ( extend_area_end < sub_area_end ) {
177  link_below_block = start_block;
178  }
179 
180  if ( sub_area_end == extend_area_begin ) {
181  start_block->prev_size = extend_area_end;
182 
183  merge_above_block = end_block;
184  } else if ( sub_area_end < extend_area_begin ) {
185  link_above_block = end_block;
186  }
187 
188  start_block = _Heap_Block_at( end_block, _Heap_Block_size( end_block ) );
189  } while ( start_block != first_block );
190 
191  if ( extend_area_begin < heap->area_begin ) {
192  heap->area_begin = extend_area_begin;
193  } else if ( heap->area_end < extend_area_end ) {
194  heap->area_end = extend_area_end;
195  }
196 
197  extend_first_block_size =
198  (uintptr_t) extend_last_block - (uintptr_t) extend_first_block;
199 
200  extend_first_block->prev_size = extend_area_end;
201  extend_first_block->size_and_flag =
202  extend_first_block_size | HEAP_PREV_BLOCK_USED;
203  _Heap_Protection_block_initialize( heap, extend_first_block );
204 
205  extend_last_block->prev_size = extend_first_block_size;
206  extend_last_block->size_and_flag = 0;
207  _Heap_Protection_block_initialize( heap, extend_last_block );
208 
209  if ( (uintptr_t) extend_first_block < (uintptr_t) heap->first_block ) {
210  heap->first_block = extend_first_block;
211  } else if ( (uintptr_t) extend_last_block > (uintptr_t) heap->last_block ) {
212  heap->last_block = extend_last_block;
213  }
214 
215  if ( merge_below_block != NULL ) {
216  _Heap_Merge_below( heap, extend_area_begin, merge_below_block );
217  } else if ( link_below_block != NULL ) {
218  _Heap_Link_below(
219  link_below_block,
220  extend_last_block
221  );
222  }
223 
224  if ( merge_above_block != NULL ) {
225  _Heap_Merge_above( heap, merge_above_block, extend_area_end );
226  } else if ( link_above_block != NULL ) {
227  _Heap_Link_above(
228  link_above_block,
229  extend_first_block,
230  extend_last_block
231  );
232  }
233 
234  if ( merge_below_block == NULL && merge_above_block == NULL ) {
235  _Heap_Free_block( heap, extend_first_block );
236  }
237 
238  _Heap_Set_last_block_size( heap );
239 
240  extended_size = stats->free_size - free_size;
241 
242  /* Statistics */
243  stats->size += extended_size;
244 
245  return extended_size;
246 }
