51 benchmark_timer_t benchmark_timer_read(void)
52 {
53  uint32_t ticks;
54  uint32_t total;
55 
56  ticks = MM_READ(MM_TIMER1_COUNTER);
57  if (ticks == 0xffffffff)
58  printk("Timer overflow!\n");
59 
60  total = ticks / (MM_READ(MM_FREQUENCY) / 1000000);
61 
62  if (benchmark_timer_find_average_overhead)
63  return total;
64  else
65  {
66  if (total < LEAST_VALID)
67  return 0;
68 
69  return (total - AVG_OVERHEAD);
70  }
