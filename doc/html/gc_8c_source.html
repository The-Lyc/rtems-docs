557  if((start >> PAGE_SHIFT) < ((end-1) >> PAGE_SHIFT)) {
558  /* It crosses a page boundary. Therefore, it must be a hole. */
559  ret = jffs2_garbage_collect_hole(c, jeb, f, fn, start, end);
560  } else {
561  /* It could still be a hole. But we GC the page this way anyway */
562  ret = jffs2_garbage_collect_dnode(c, jeb, f, fn, start, end);
563  }
564  goto upnout;
565  }
566 
567  /* Wasn't a dnode. Try dirent */
568  for (fd = f->dents; fd; fd=fd->next) {
569  if (fd->raw == raw)
570  break;
571  }
572 
573  if (fd && fd->ino) {
574  ret = jffs2_garbage_collect_dirent(c, jeb, f, fd);
575  } else if (fd) {
576  ret = jffs2_garbage_collect_deletion_dirent(c, jeb, f, fd);
577  } else {
578  pr_warn("Raw node at 0x%08x wasn't in node lists for ino #%u\n",
579  ref_offset(raw), f->inocache->ino);
580  if (ref_obsolete(raw)) {
581  pr_warn("But it's obsolete so we don't mind too much\n");
582  } else {
583  jffs2_dbg_dump_node(c, ref_offset(raw));
584  BUG();
585  }
586  }
587  upnout:
588  mutex_unlock(&f->sem);
589 
590  return ret;
591 }

593 static int jffs2_garbage_collect_pristine(struct jffs2_sb_info *c,
594  struct jffs2_inode_cache *ic,
595  struct jffs2_raw_node_ref *raw)
596 {
597  union jffs2_node_union *node;
598  size_t retlen;
599  int ret;
600  uint32_t phys_ofs, alloclen;
601  uint32_t crc, rawlen;
602  int retried = 0;
603 
604  jffs2_dbg(1, "Going to GC REF_PRISTINE node at 0x%08x\n",
605  ref_offset(raw));
606 
607  alloclen = rawlen = ref_totlen(c, c->gcblock, raw);
608 
609  /* Ask for a small amount of space (or the totlen if smaller) because we
610  don't want to force wastage of the end of a block if splitting would
611  work. */
612  if (ic && alloclen > sizeof(struct jffs2_raw_inode) + JFFS2_MIN_DATA_LEN)
613  alloclen = sizeof(struct jffs2_raw_inode) + JFFS2_MIN_DATA_LEN;
614 
615  ret = jffs2_reserve_space_gc(c, alloclen, &alloclen, rawlen);
616  /* 'rawlen' is not the exact summary size; it is only an upper estimation */
617 
618  if (ret)
619  return ret;
620 
621  if (alloclen < rawlen) {
622  /* Doesn't fit untouched. We'll go the old route and split it */
623  return -EBADFD;
624  }
625 
626  node = kmalloc(rawlen, GFP_KERNEL);
627  if (!node)
628  return -ENOMEM;
629 
630  ret = jffs2_flash_read(c, ref_offset(raw), rawlen, &retlen, (char *)node);
631  if (!ret && retlen != rawlen)
632  ret = -EIO;
633  if (ret)
634  goto out_node;
635 
636  crc = crc32(0, node, sizeof(struct jffs2_unknown_node)-4);
637  if (je32_to_cpu(node->u.hdr_crc) != crc) {
638  pr_warn("Header CRC failed on REF_PRISTINE node at 0x%08x: Read 0x%08x, calculated 0x%08x\n",
639  ref_offset(raw), je32_to_cpu(node->u.hdr_crc), crc);
640  goto bail;
641  }
642 
643  switch(je16_to_cpu(node->u.nodetype)) {
644  case JFFS2_NODETYPE_INODE:
645  crc = crc32(0, node, sizeof(node->i)-8);
646  if (je32_to_cpu(node->i.node_crc) != crc) {
647  pr_warn("Node CRC failed on REF_PRISTINE data node at 0x%08x: Read 0x%08x, calculated 0x%08x\n",
648  ref_offset(raw), je32_to_cpu(node->i.node_crc),
649  crc);
650  goto bail;
651  }
652 
653  if (je32_to_cpu(node->i.dsize)) {
654  crc = crc32(0, node->i.data, je32_to_cpu(node->i.csize));
655  if (je32_to_cpu(node->i.data_crc) != crc) {
656  pr_warn("Data CRC failed on REF_PRISTINE data node at 0x%08x: Read 0x%08x, calculated 0x%08x\n",
657  ref_offset(raw),
658  je32_to_cpu(node->i.data_crc), crc);
659  goto bail;
660  }
661  }
662  break;
663 
664  case JFFS2_NODETYPE_DIRENT:
665  crc = crc32(0, node, sizeof(node->d)-8);
666  if (je32_to_cpu(node->d.node_crc) != crc) {
667  pr_warn("Node CRC failed on REF_PRISTINE dirent node at 0x%08x: Read 0x%08x, calculated 0x%08x\n",
668  ref_offset(raw),
669  je32_to_cpu(node->d.node_crc), crc);
670  goto bail;
671  }
672 
673  if (strnlen(node->d.name, node->d.nsize) != node->d.nsize) {
674  pr_warn("Name in dirent node at 0x%08x contains zeroes\n",
675  ref_offset(raw));
676  goto bail;
677  }
678 
679  if (node->d.nsize) {
680  crc = crc32(0, node->d.name, node->d.nsize);
681  if (je32_to_cpu(node->d.name_crc) != crc) {
682  pr_warn("Name CRC failed on REF_PRISTINE dirent node at 0x%08x: Read 0x%08x, calculated 0x%08x\n",
683  ref_offset(raw),
684  je32_to_cpu(node->d.name_crc), crc);
685  goto bail;
686  }
687  }
688  break;
689  default:
690  /* If it's inode-less, we don't _know_ what it is. Just copy it intact */
691  if (ic) {
692  pr_warn("Unknown node type for REF_PRISTINE node at 0x%08x: 0x%04x\n",
693  ref_offset(raw), je16_to_cpu(node->u.nodetype));
694  goto bail;
695  }
696  }
697 
698  /* OK, all the CRCs are good; this node can just be copied as-is. */
699  retry:
700  phys_ofs = write_ofs(c);
701 
702  ret = jffs2_flash_write(c, phys_ofs, rawlen, &retlen, (char *)node);
703 
704  if (ret || (retlen != rawlen)) {
705  pr_notice("Write of %d bytes at 0x%08x failed. returned %d, retlen %zd\n",
706  rawlen, phys_ofs, ret, retlen);
707  if (retlen) {
708  jffs2_add_physical_node_ref(c, phys_ofs | REF_OBSOLETE, rawlen, NULL);
709  } else {
710  pr_notice("Not marking the space at 0x%08x as dirty because the flash driver returned retlen zero\n",
711  phys_ofs);
712  }
713  if (!retried) {
714  /* Try to reallocate space and retry */
715  uint32_t dummy;
716  struct jffs2_eraseblock *jeb = &c->blocks[phys_ofs / c->sector_size];
717 
718  retried = 1;
719 
720  jffs2_dbg(1, "Retrying failed write of REF_PRISTINE node.\n");
721 
722  jffs2_dbg_acct_sanity_check(c,jeb);
723  jffs2_dbg_acct_paranoia_check(c, jeb);
724 
725  ret = jffs2_reserve_space_gc(c, rawlen, &dummy, rawlen);
726  /* this is not the exact summary size of it,
727  it is only an upper estimation */
728 
729  if (!ret) {
730  jffs2_dbg(1, "Allocated space at 0x%08x to retry failed write.\n",
731  phys_ofs);
732 
733  jffs2_dbg_acct_sanity_check(c,jeb);
734  jffs2_dbg_acct_paranoia_check(c, jeb);
735 
736  goto retry;
737  }
738  jffs2_dbg(1, "Failed to allocate space to retry failed write: %d!\n",
739  ret);
740  }
741 
742  if (!ret)
743  ret = -EIO;
744  goto out_node;
745  }
746  jffs2_add_physical_node_ref(c, phys_ofs | REF_PRISTINE, rawlen, ic);
747 
748  jffs2_mark_node_obsolete(c, raw);
749  jffs2_dbg(1, "WHEEE! GC REF_PRISTINE node at 0x%08x succeeded\n",
750  ref_offset(raw));
751 
752  out_node:
753  kfree(node);
754  return ret;
755  bail:
756  ret = -EBADFD;
757  goto out_node;
758 }

761  struct jffs2_inode_info *f, struct jffs2_full_dnode *fn)

852  out:

854  kfree(mdata);
855  return ret;

903 static int jffs2_garbage_collect_deletion_dirent(struct jffs2_sb_info *c, struct jffs2_eraseblock *jeb,
904  struct jffs2_inode_info *f, struct jffs2_full_dirent *fd)
905 {
906  struct jffs2_full_dirent **fdp = &f->dents;
907  int found = 0;
908 
909  /* On a medium where we can't actually mark nodes obsolete
910  pernamently, such as NAND flash, we need to work out
911  whether this deletion dirent is still needed to actively
912  delete a 'real' dirent with the same name that's still
913  somewhere else on the flash. */
914  if (!jffs2_can_mark_obsolete(c)) {
915  struct jffs2_raw_dirent *rd;
916  struct jffs2_raw_node_ref *raw;
917  int ret;
918  size_t retlen;
919  int name_len = strlen(fd->name);
920  uint32_t name_crc = crc32(0, fd->name, name_len);
921  uint32_t rawlen = ref_totlen(c, jeb, fd->raw);
922 
923  rd = kmalloc(rawlen, GFP_KERNEL);
924  if (!rd)
925  return -ENOMEM;
926 
927  /* Prevent the erase code from nicking the obsolete node refs while
928  we're looking at them. I really don't like this extra lock but
929  can't see any alternative. Suggestions on a postcard to... */
930  mutex_lock(&c->erase_free_sem);
931 
932  for (raw = f->inocache->nodes; raw != (void *)f->inocache; raw = raw->next_in_ino) {
933 
934  cond_resched();
935 
936  /* We only care about obsolete ones */
937  if (!(ref_obsolete(raw)))
938  continue;
939 
940  /* Any dirent with the same name is going to have the same length... */
941  if (ref_totlen(c, NULL, raw) != rawlen)
942  continue;
943 
944  /* Doesn't matter if there's one in the same erase block. We're going to
945  delete it too at the same time. */
946  if (SECTOR_ADDR(raw->flash_offset) == SECTOR_ADDR(fd->raw->flash_offset))
947  continue;
948 
949  jffs2_dbg(1, "Check potential deletion dirent at %08x\n",
950  ref_offset(raw));
951 
952  /* This is an obsolete node belonging to the same directory, and it's of the right
953  length. We need to take a closer look...*/
954  ret = jffs2_flash_read(c, ref_offset(raw), rawlen, &retlen, (char *)rd);
955  if (ret) {
956  pr_warn("%s(): Read error (%d) reading obsolete node at %08x\n",
957  __func__, ret, ref_offset(raw));
958  /* If we can't read it, we don't need to continue to obsolete it. Continue */
959  continue;
960  }
961  if (retlen != rawlen) {
962  pr_warn("%s(): Short read (%zd not %u) reading header from obsolete node at %08x\n",
963  __func__, retlen, rawlen,
964  ref_offset(raw));
965  continue;
966  }
967 
968  if (je16_to_cpu(rd->nodetype) != JFFS2_NODETYPE_DIRENT)
969  continue;
970 
971  /* If the name CRC doesn't match, skip */
972  if (je32_to_cpu(rd->name_crc) != name_crc)
973  continue;
974 
975  /* If the name length doesn't match, or it's another deletion dirent, skip */
976  if (rd->nsize != name_len || !je32_to_cpu(rd->ino))
977  continue;
978 
979  /* OK, check the actual name now */
980  if (memcmp(rd->name, fd->name, name_len))
981  continue;
982 
983  /* OK. The name really does match. There really is still an older node on
984  the flash which our deletion dirent obsoletes. So we have to write out
985  a new deletion dirent to replace it */
986  mutex_unlock(&c->erase_free_sem);
987 
988  jffs2_dbg(1, "Deletion dirent at %08x still obsoletes real dirent \"%s\" at %08x for ino #%u\n",
989  ref_offset(fd->raw), fd->name,
990  ref_offset(raw), je32_to_cpu(rd->ino));
991  kfree(rd);
992 
993  return jffs2_garbage_collect_dirent(c, jeb, f, fd);
994  }
995 
996  mutex_unlock(&c->erase_free_sem);
997  kfree(rd);
998  }
999 
1000  /* FIXME: If we're deleting a dirent which contains the current mtime and ctime,
1001  we should update the metadata node with those times accordingly */
1002 
1003  /* No need for it any more. Just mark it obsolete and remove it from the list */
1004  while (*fdp) {
1005  if ((*fdp) == fd) {
1006  found = 1;
1007  *fdp = fd->next;
1008  break;
1009  }
1010  fdp = &(*fdp)->next;
1011  }
1012  if (!found) {
1013  pr_warn("Deletion dirent \"%s\" not found in list for ino #%u\n",
1014  fd->name, f->inocache->ino);
1015  }
1016  jffs2_mark_node_obsolete(c, fd->raw);
1017  jffs2_free_full_dirent(fd);
1018  return 0;
1019 }

1021 static int jffs2_garbage_collect_hole(struct jffs2_sb_info *c, struct jffs2_eraseblock *jeb,
1022  struct jffs2_inode_info *f, struct jffs2_full_dnode *fn,
1023  uint32_t start, uint32_t end)

1061  pr_warn("%s: Node at 0x%08x had CRC 0x%08x which doesn't match calculated CRC 0x%08x\n",
1062  __func__, ref_offset(fn->raw),
1063  je32_to_cpu(ri.node_crc), crc);
1064  /* FIXME: We could possibly deal with this by writing new holes for each frag */
1065  pr_warn("Data in the range 0x%08x to 0x%08x of inode #%u will be lost\n",
1066  start, end, f->inocache->ino);
1067  goto fill;
1068  }
1069  if (ri.compr != JFFS2_COMPR_ZERO) {
