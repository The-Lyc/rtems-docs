97 /*
98  * Determine the source of the interrupt and dispatch the correct handler.
99  */
100 void bsp_interrupt_dispatch(void)
101 {
102  unsigned int pend;
103  unsigned int pend_bit;
104 
105  rtems_vector_number vector = 255;
106 
107 #ifdef RTEMS_SMP
108  uint32_t cpu_index_self = _SMP_Get_current_processor();
109  uint32_t local_source = BCM2835_REG(BCM2836_IRQ_SOURCE_REG(cpu_index_self));
110 
111  if ( local_source & BCM2836_IRQ_SOURCE_MBOX3 ) {
112  /* reset mailbox 3 contents to zero */
113  BCM2835_REG(BCM2836_MAILBOX_3_READ_CLEAR_BASE + 0x10 * cpu_index_self) = 0xffffffff;
114  _SMP_Inter_processor_interrupt_handler(
115  _Per_CPU_Get_by_index(cpu_index_self)
116  );
117  }
118  if ( cpu_index_self != 0 )
119  return;
120 #endif /* RTEMS_SMP */
121 
122  pend = BCM2835_REG(BCM2835_IRQ_BASIC);
123  if ( pend & BCM2835_IRQ_BASIC_SPEEDUP_USED_BITS ) {
124  pend_bit = ffs(pend) - 1;
125  vector = bcm2835_irq_speedup_table[pend_bit];
126  } else {
127  pend = BCM2835_REG(BCM2835_IRQ_PENDING1);
128  if ( pend != 0 ) {
129  pend_bit = ffs(pend) - 1;
130  vector = pend_bit;
131  } else {
132  pend = BCM2835_REG(BCM2835_IRQ_PENDING2);
133  if ( pend != 0 ) {
134  pend_bit = ffs(pend) - 1;
135  vector = pend_bit + 32;
136  }
137  }
138  }
139 
140  if ( vector < 255 )
141  {
142  bsp_interrupt_handler_dispatch(vector);
143  }
144 }
