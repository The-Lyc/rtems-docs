380 /**********************************************************************************
381  *
382  * DESCRIPTION:
383  * We get eeprom data from the id_port given an offset into the eeprom.
384  * Basically; after the ID_sequence is sent to all of the cards; they enter
385  * the ID_CMD state where they will accept command requests. 0x80-0xbf loads
386  * the eeprom data. We then read the port 16 times and with every read; the
387  * cards check for contention (ie: if one card writes a 0 bit and another
388  * writes a 1 bit then the host sees a 0. At the end of the cycle; each card
389  * compares the data on the bus; if there is a difference then that card goes
390  * into ID_WAIT state again). In the meantime; one bit of data is returned in
391  * the AX register which is conveniently returned to us by inb(). Hence; we
392  * read 16 times getting one bit of data with each read.
393  *
394  * RETURNS: 16 bit word from the EEPROM
395  *
396  **********************************************************************************/
397 static int get_eeprom_data( int id_port, int offset )
398 {
399  int i, data = 0;
400  outb(id_port, 0x80 + offset);
401  Wait_X_ms( 1 );
402  for (i = 0; i < 16; i++)
403  data = (data << 1) | (inw(id_port) & 1);
404  return( data );
405 }

770  * DESCRIPTION:
771  * This function looks for a 3COM card 3c5x9 in an isa bus. If a board is found, it
772  * returns a structure describing the caracteristics of the card. It returns zero when
773  * card can not be found.
774  *
775  * RETURNS: 0 - fail - could not find a card...
776  * <> description of the card.
777  *

779 static struct ep_board *ep_look_for_board_at( struct isa_device *is )
780 {

782  int count = 0;
783

785  {
786  /* Come here just one time */
787  ep_current_tag--;
788 
789  /* Look for the ISA boards. Init and leave them actived */
790  outb(id_port, 0);
791  outb(id_port, 0);
792

794  elink_reset();

796  for (i = 0; i < EP_MAX_BOARDS; i++)
797  {
798  outb(id_port, 0);
799  outb(id_port, 0);
800  elink_idseq(0xCF);
801 
802  data = get_eeprom_data(id_port, EEPROM_MFG_ID);
803  if (data != MFG_ID)
804  break;

807  for (j = 0; j < 3; j++)
808  get_eeprom_data(id_port, j);
809 
810  /* and save this address for later use */
811 
812  for (j = 0; j < 3; j++)
813  ep_board[ep_boards].eth_addr[j] = get_eeprom_data(id_port, j);

819  ep_board[ep_boards].epb_addr =
820  (get_eeprom_data(id_port, EEPROM_ADDR_CFG) & 0x1f) * 0x100 + 0x40d0;
821 #else
822  ep_board[ep_boards].epb_addr =
823  (get_eeprom_data(id_port, EEPROM_ADDR_CFG) & 0x1f) * 0x10 + 0x200;
824  if (ep_board[ep_boards].epb_addr > 0x3E0)
825  /* Board in EISA configuration mode */
826  continue;
827 #endif /* PC98 */
828

830  outb(id_port, ACTIVATE_ADAPTER_TO_CONFIG);

833  ep_current_tag--;
834  }

863  ep_board[i].epb_used = 1;

866  else

874  is->id_unit, IS_BASE );
875  }
876  ep_board[i].epb_used = 1;
877  return &ep_board[i];
878  }

881 /**********************************************************************************
882  *
883  * DESCRIPTION:
884  * This routine checks if there card installed on the machine.
885  *
886  * RETURNS: 0 - no card founded.
887  * 16 - size of the IO range for the card.
888  *
889  **********************************************************************************/

891 {

893  struct ep_board *epb;
894  u_short k;
895

897  if( (epb = ep_look_for_board_at(is)) == 0 )
898  return (0);

901  sc->ep_io_addr = epb->epb_addr;

904  /*
905  * The iobase was found and MFG_ID was 0x6d50. PROD_ID should be
906  * 0x9[0-f]50 (IBM-PC)
907  * 0x9[0-f]5[0-f] (PC-98)

911 #ifdef PC98

914 #else

917 #endif

926  *

930  * 2. Device was configured with 'irq xxx'
931  * In this case we set up the board to use specified interrupt
932  *
933  */
934 
935  if (is->id_irq == 0)
936  { /* irq? */
937  is->id_irq = ( k == 2 ) ? 9 : k;
938  }
939 
940  sc->stat = 0; /* 16 bit access */
941

943

1073  inb(BASE + EP_W1_TX_STATUS);
1074

1076  outw(BASE + EP_COMMAND, ACK_INTR | 0xff);

1079 
1080  outw(BASE + EP_COMMAND, SET_INTR_MASK | S_5_INTS);

1104  */

1122  }

1182 static const char padmap[] = {0, 3, 2, 1};

1184 /**********************************************************************************
1185  *
1186  * DESCRIPTION: Routine to transmit frames to the card.
1187  *
1188  * RETURNS: nothing.
1189  *
1190  **********************************************************************************/
1191 static void epstart( struct ifnet *ifp )
1192 {
1193  register struct ep_softc *sc = ifp->if_softc;
1194  register u_int len;
1195  register struct mbuf *m;
1196  struct mbuf *top;
1197  int pad;
1198 
1199  while( inw(BASE + EP_STATUS) & S_COMMAND_IN_PROGRESS )
1200  ;
1201 startagain:
1202  /* printk( "S-" ); */
1203 
1204  /* Sneak a peek at the next packet */
1205  m = ifp->if_snd.ifq_head;
1206  if (m == 0)
1207  {
1208  ifp->if_flags &= ~IFF_OACTIVE;
1209  return;
1210  }
1211 
1212  for( len = 0, top = m; m; m = m->m_next )
1213  len += m->m_len;
1214 
1215  pad = padmap[ len & 3 ];
1216 
1217  /*
1218  * The 3c509 automatically pads short packets to minimum ethernet length,
1219  * but we drop packets that are too large. Perhaps we should truncate
1220  * them instead?
1221  */
1222  if( len + pad > ETHER_MAX_LEN )
1223  {
1224  /* packet is obviously too large: toss it */
1225  ++ifp->if_oerrors;
1226  IF_DEQUEUE( &ifp->if_snd, m );
1227  m_freem( m );
1228  goto readcheck;
1229  }
1230  if (inw(BASE + EP_W1_FREE_TX) < len + pad + 4)
1231  {
1232  /* no room in FIFO */
1233  outw(BASE + EP_COMMAND, SET_TX_AVAIL_THRESH | (len + pad + 4));
1234  /* make sure */
1235  if (inw(BASE + EP_W1_FREE_TX) < len + pad + 4)
1236  {
1237  ifp->if_flags |= IFF_OACTIVE;
1238  return;
1239  }
1240  }
1241  IF_DEQUEUE( &ifp->if_snd, m );
1242  outw(BASE + EP_W1_TX_PIO_WR_1, len);
1243  outw(BASE + EP_W1_TX_PIO_WR_1, 0x0); /* Second dword meaningless */
1244 
1245  for (top = m; m != 0; m = m->m_next)
1246  {
1247  if( ep_ftst(F_ACCESS_32_BITS ) )
1248  {
1249  outsl( BASE + EP_W1_TX_PIO_WR_1, mtod(m, uint8_t *), m->m_len / 4 );
1250  if( m->m_len & 3 )
1251  outsb(BASE + EP_W1_TX_PIO_WR_1, mtod(m, uint8_t *) + (m->m_len & (~3)), m->m_len & 3 );
1252  }
1253  else
1254  {
1255  outsw( BASE + EP_W1_TX_PIO_WR_1, mtod(m, uint8_t *), m->m_len / 2 );
1256  if( m->m_len & 1 )
1257  outb( BASE + EP_W1_TX_PIO_WR_1, *(mtod(m, uint8_t *) + m->m_len - 1) );
1258  }
1259  }
1260  while( pad-- )
1261  {
1262  outb(BASE + EP_W1_TX_PIO_WR_1, 0); /* Padding */
1263  }
1264  ifp->if_timer = 2;
1265  ifp->if_opackets++;
1266  m_freem(top);
1267 
1268 /* goto startagain; */
1269  /*
1270  * Is another packet coming in? We don't want to overflow the tiny RX
1271  * fifo.
1272  */
1273 readcheck:
1274  if( inw(BASE + EP_W1_RX_STATUS) & RX_BYTES_MASK )
1275  {
1276  /*
1277  * we check if we have packets left, in that case we prepare to come
1278  * back later
1279  */
1280  if( ifp->if_snd.ifq_head )
1281  {
1282  outw(BASE + EP_COMMAND, SET_TX_AVAIL_THRESH | 8);
1283  }
1284  return;
1285  }
1286  goto startagain;
1287 }

1289 /**********************************************************************************
1290  *
1291  * DESCRIPTION: Routine to read frames from the card.
1292  *
1293  * RETURNS: nothing.
1294  *
1295  **********************************************************************************/
1296 static void epread( register struct ep_softc *sc )
1297 {
1298  struct ether_header *eh;
1299  struct mbuf *top, *mcur, *m;
1300  struct ifnet *ifp;
1301  int lenthisone;
1302 
1303  short rx_fifo2, status;
1304  register short rx_fifo;
1305 
1306  ifp = &sc->arpcom.ac_if;
1307  status = inw( BASE + EP_W1_RX_STATUS );
1308 
1309 read_again:
1310 
1311  if (status & ERR_RX)
1312  {
1313  ++ifp->if_ierrors;
1314  if( status & ERR_RX_OVERRUN )
1315  {
1316  /*
1317  * we can think the rx latency is actually greather than we
1318  * expect
1319  */
1320  if( ep_ftst(F_RX_FIRST) )
1321  sc->rx_overrunf++;
1322  else
1323  sc->rx_overrunl++;
1324 
1325  }
1326  goto out;
1327  }
1328  rx_fifo = rx_fifo2 = status & RX_BYTES_MASK;
1329 
1330  if( ep_ftst( F_RX_FIRST ) )
1331  {
1332  MGETHDR( m, M_DONTWAIT, MT_DATA );
1333  if( !m )
1334  goto out;
1335  if( rx_fifo >= MINCLSIZE )
1336  MCLGET( m, M_DONTWAIT );
1337  sc->top = sc->mcur = top = m;
1338 #define EROUND ((sizeof(struct ether_header) + 3) & ~3)
1339 #define EOFF (EROUND - sizeof(struct ether_header))
1340  top->m_data += EOFF;
1341 
1342  /* Read what should be the header. */
1343  insw(BASE + EP_W1_RX_PIO_RD_1, mtod(top, uint8_t *), sizeof(struct ether_header) / 2);
1344  top->m_len = sizeof(struct ether_header);
1345  rx_fifo -= sizeof(struct ether_header);
1346  sc->cur_len = rx_fifo2;
1347  }
1348  else
1349  {
1350  /* come here if we didn't have a complete packet last time */
1351  top = sc->top;
1352  m = sc->mcur;
1353  sc->cur_len += rx_fifo2;
1354  }
1355 
1356  /* Reads what is left in the RX FIFO */
1357  while (rx_fifo > 0)
1358  {
1359  lenthisone = min( rx_fifo, M_TRAILINGSPACE(m) );
1360  if( lenthisone == 0 )
1361  { /* no room in this one */
1362  mcur = m;
1363  MGET(m, M_WAIT, MT_DATA);
1364  if (!m)
1365  goto out;
1366  if (rx_fifo >= MINCLSIZE)
1367  MCLGET(m, M_WAIT);
1368  m->m_len = 0;
1369  mcur->m_next = m;
1370  lenthisone = min(rx_fifo, M_TRAILINGSPACE(m));
