59 BSP_START_TEXT_SECTION void bsp_start_hook_1( void )
60 {
61  uint32_t addr_filt_start;
62  uint32_t addr_filt_end;
63 
64  /* Disable ECC. Preloader respectively UBoot enable ECC.
65  But they do run without interrupts. Our BSP will enable interrupts
66  and get spurious ECC error interrupts. Thus we disasable ECC
67  until we either know about a better handling or Altera has modified
68  it's SDRAM settings to not create possibly false ECC errors */
69  uint32_t ctlcfg = alt_read_word( ALT_SDR_CTL_CTLCFG_ADDR );
70  ctlcfg &= ALT_SDR_CTL_CTLCFG_ECCEN_CLR_MSK;
71  alt_write_word( ALT_SDR_CTL_CTLCFG_ADDR, ctlcfg );
72 
73  /* Perform L3 remap register programming first by setting the desired new MPU
74  address space 0 mapping. Assume BOOTROM in order to be able to boot the
75  second core. */
76  alt_addr_space_remap(
77  ALT_ADDR_SPACE_MPU_ZERO_AT_BOOTROM,
78  ALT_ADDR_SPACE_NONMPU_ZERO_AT_SDRAM,
79  ALT_ADDR_SPACE_H2F_ACCESSIBLE,
80  ALT_ADDR_SPACE_LWH2F_ACCESSIBLE );
81 
82  /* Next, adjust the L2 cache address filtering range. Set the start address
83  * to the default reset value and retain the existing end address
84  * configuration. */
85  alt_l2_addr_filter_cfg_get( &addr_filt_start, &addr_filt_end );
86 
87  if ( addr_filt_start != BSPSTART_L2_CACHE_ADDR_FILTERING_START_RESET ) {
88  alt_l2_addr_filter_cfg_set( BSPSTART_L2_CACHE_ADDR_FILTERING_START_RESET,
89  addr_filt_end );
90  }
91 
92  arm_a9mpcore_start_hook_1();
93  bsp_start_copy_sections();
94  setup_mmu_and_cache();
95 #ifndef RTEMS_SMP
96  /* Enable unified L2 cache */
97  rtems_cache_enable_data();
98 #endif
99  bsp_start_clear_bss();
100 }
