83 static void HardFault_reason(void)
84 {
85  uint32_t CFSRValue;
86  TRACE_DEBUG("In Hard Fault Handler\n\r");
87  TRACE_DEBUG("SCB->HFSR = 0x%08x\n\r", SCB->HFSR);
88 
89  if ((SCB->HFSR & SCB_HFSR_DEBUGEVT_Msk)) {
90  TRACE_DEBUG("Debug Event Hard Fault\n\r");
91  TRACE_DEBUG("SCB->DFSR = 0x%08x\n", SCB->DFSR);
92  }
93 
94  if ((SCB->HFSR & SCB_HFSR_VECTTBL_Msk)) {
95  TRACE_DEBUG("Fault was due to vector table read on \
96  exception processing\n\r");
97  }
98 
99  // Forced HardFault
100  if ((SCB->HFSR & SCB_HFSR_FORCED_Msk)) {
101  TRACE_DEBUG("Forced Hard Fault\n\r");
102  TRACE_DEBUG("SCB->CFSR = 0x%08x\n\r", SCB->CFSR);
103 
104  // Usage Fault
105  if ((SCB->CFSR & SCB_CFSR_USGFAULTSR_Msk)) {
106  CFSRValue = SCB->CFSR;
107  TRACE_DEBUG("Usage fault: ");
108  CFSRValue >>= SCB_CFSR_USGFAULTSR_Pos;
109 
110  if ((CFSRValue & (1 << 9)))
111  TRACE_DEBUG("Divide by zero\n\r");
112 
113  if ((CFSRValue & (1 << 8)))
114  TRACE_DEBUG("Unaligned access error\n\r");
115 
116  if ((CFSRValue & (1 << 3)))
117  TRACE_DEBUG("Coprocessor access error\n\r");
118 
119  if ((CFSRValue & (1 << 2)))
120  TRACE_DEBUG("Integrity check error on EXC_RETURN\n\r");
121  }
122 
123  // Bus Fault
124  if ((SCB->CFSR & SCB_CFSR_BUSFAULTSR_Msk)) {
125  CFSRValue = SCB->CFSR;
126  TRACE_DEBUG("Bus fault: ");
127  CFSRValue >>= SCB_CFSR_BUSFAULTSR_Pos;
128 
129  if ((CFSRValue & (1 << 7)) && (CFSRValue & (1 << 1))) {
130  TRACE_DEBUG("Precise data access error. Bus Fault Address \
131  Register is: %x \n\r", SCB->BFAR);
132  }
133 
134  if ((CFSRValue & (1 << 4)))
135  TRACE_DEBUG("Bus fault has occurred on exception entry\n\r");
136 
137  if ((CFSRValue & (1 << 3)))
138  TRACE_DEBUG("bus fault has occurred on exception return\n\r");
139 
140  if ((CFSRValue & (1 << 2)))
141  TRACE_DEBUG("Imprecise data access error\n\r");
142 
143  if ((CFSRValue & (1 << 0))) {
144  TRACE_DEBUG("This bit indicates a bus fault on an instruction \
145  pre-fetch. \n\r");
146  }
147  }
148  }
149 
150  // MemoryFault
151  if ((SCB->CFSR & SCB_CFSR_MEMFAULTSR_Msk)) {
152  CFSRValue = SCB->CFSR;
153  TRACE_DEBUG("Memory fault: ");
154  CFSRValue >>= SCB_CFSR_MEMFAULTSR_Pos;
155 
156  if ((CFSRValue & (1 << 9)) != 0)
157  TRACE_DEBUG("Divide by zero\n\r");
158  }
159 
160  __ISB();
161  __DMB();
162  __ASM volatile("BKPT #01");
163 }
