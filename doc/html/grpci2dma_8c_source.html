286 /* This functions adds a channel descriptor to the DMA channel
287  * linked list. It assumes that someone has check the input
288  * parameters already.
289  */
290 STATIC int grpci2dma_channel_list_add(struct grpci2_bd_chan * list,
291  struct grpci2_bd_chan * chan)
292 {
293  DBG("Adding channel (0x%08x) to GRPCI2 DMA driver\n", (unsigned int) chan);
294 
295  /* Add channel to the linnked list */
296  if (list == chan) {
297  /* No previous channels. Finish. */
298  return GRPCI2DMA_ERR_OK;
299  } else {
300  /* Get next chan from list */
301  struct grpci2_bd_chan * nchan = grpci2dma_channel_bd_get_next(list);
302  /* Close the circular linked list */
303  grpci2dma_channel_bd_set_next(chan,nchan);
304  /* Attach the new channel in the middle */
305  grpci2dma_channel_bd_set_next(list, chan);
306  return GRPCI2DMA_ERR_OK;
307  }
308 }

310 /* This functions removes a channel descriptor from the DMA channel
311  * linked list. It assumes that someone has check the input
312  * parameters already.
313  * It returns 0 if successfull. Otherwise,
314  * it can return:
315  * - ERROR: Different causes:
316  * x Number of channels is corrupted.
317  */
318 STATIC int grpci2dma_channel_list_remove(struct grpci2_bd_chan * chan)
319 {
320  DBG("Removing channel (0x%08x) from GRPCI2 DMA driver\n",
321  (unsigned int) chan);
322 
323  /* Remove channel from the linnked list */
324  struct grpci2_bd_chan * nchan = grpci2dma_channel_bd_get_next(chan);
325  if (nchan != chan){
326  /* There are more channels */
327  /* Since this is a circular linked list, we need to find last channel
328  * and update the pointer to the next element */
329  /* Use index to avoid having an infinite loop in case of corrupted
330  * channels */
331  struct grpci2_bd_chan * new_first_chan = nchan;
332  struct grpci2_bd_chan * curr_chan;
333  int i=1;
334  while((nchan != chan) && (i<MAX_DMA_CHANS)){
335  curr_chan = nchan;
336  nchan = grpci2dma_channel_bd_get_next(curr_chan);
337  i++;
338  }
339  if (nchan != chan) {
340  DBG("Maximum DMA channels exceeded. Maybe corrupted?\n");
341  return GRPCI2DMA_ERR_ERROR;
342  } else {
343  /* Update the pointer */
344  grpci2dma_channel_bd_set_next(curr_chan, new_first_chan);
345  return GRPCI2DMA_ERR_OK;
346  }
347  }else{
348  /* There are no more channels */
349  return GRPCI2DMA_ERR_OK;
350  }
351 }

353 /* This functions adds a data descriptor to the channel's data
354  * linked list. The function assumes, that the data descriptor
355  * points to either a DISABLED_DESCRIPTOR or to linked list of
356  * data descriptors that ends with a DISABLED_DESCRIPTOR.
357  * It returns the number of active data descriptors
358  * if successfull. Otherwise, it can return:
359  * - ERROR: Different causes:
360  * x Number of channels is corrupted.
361  * x Last linked list element is not pointing to the first.
362  */
363 STATIC int grpci2dma_data_list_add(struct grpci2_bd_chan * chan,
364  struct grpci2_bd_data * data, struct grpci2_bd_data * last_chan_data)
365 {
366  DBG("Adding data (0x%08x) to channel (0x%08x)\n",
367  (unsigned int) data, (unsigned int) chan);
368 
369  /* Add data to the linnked list */
370  /* 1st- Get current data */
371  struct grpci2_bd_data * first_data = grpci2dma_channel_bd_get_data(chan);
372  if (first_data == NULL) {
373  /* Channel should always be pointing to a disabled descriptor */
374  DBG("Channel not pointing to disabled descpriptor\n");
375  return GRPCI2DMA_ERR_ERROR;
376  } else if (first_data == DISABLED_DESCRIPTOR){
377  /* No previous data. Assign this one and finish. */
378  grpci2dma_channel_bd_set_data(chan, data);
379  return GRPCI2DMA_ERR_OK;
380  } else {
381  /* Let's add the data to the last data pointer added to this channel */
382  /* Attach the new data */
383  grpci2dma_data_bd_set_next(last_chan_data, data);
384  /* 2nd- Let's check again to make sure that the DMA did not finished
385  * while we were inserting the new data */
386  first_data = grpci2dma_channel_bd_get_data(chan);
387  if (first_data == DISABLED_DESCRIPTOR){
388  grpci2dma_channel_bd_set_data(chan, data);
389  }
390  return GRPCI2DMA_ERR_OK;
391  }
392 }

394 /* This functions removes a data descriptor from the channel's data
395  * linked list. Note that in a normal execution, the DMA will remove
396  * the data descriptors from the linked list, so there is no need to
397  * use this function. It returns 0 if successfull. Otherwise,
398  * it can return:
399  * - WRONGPTR: The chan (or data) pointer is either NULL or not aligned to
400  * 0x10.
401  * - STOPDMA: The DMA is running, cannot add channels while DMA is running.
402  * - TOOMANY: The max number of data is reached.
403  * - ERROR: Different causes:
404  * x There are no free channel id numbers.
405  * x Number of channels is corrupted.
406  * x Last linked list element is not pointing to the first.
407  */
408 UNUSED STATIC int grpci2dma_data_list_remove(struct grpci2_bd_chan * chan,
409  struct grpci2_bd_data * data)
410 {
411  DBG("Removing data (0x%08x) from channel (0x%08x)\n",
412  (unsigned int) data, (unsigned int) chan);
413 
414  /* Remove data from the linked list */
415  /* 1st- Get current DMA data */
416  struct grpci2_bd_data * first_data = grpci2dma_channel_bd_get_data(chan);
417  if (first_data == NULL) {
418  /* Channel should always be pointing to a disabled descriptor */
419  DBG("Channel not pointing to disabled descpriptor\n");
420  return GRPCI2DMA_ERR_ERROR;
421  } else if (first_data == DISABLED_DESCRIPTOR){
422  /* No previous data. Cannot detach */
423  DBG("No data to detach.\n");
424  return GRPCI2DMA_ERR_NOTFOUND;
425  } else {
426  /* 2nd- Already available data, let's find the data */
427  if (first_data == data) {
428  /* 3rd- It is the first one. */
429  struct grpci2_bd_data *current = first_data;
430  struct grpci2_bd_data *next = grpci2dma_data_bd_get_next(current);
431  if (next != DISABLED_DESCRIPTOR){
432  /* There are more data */
433  /* Set channel next data descriptor to data*/
434  grpci2dma_channel_bd_set_data(chan, next);
435  /* Update the removed data */
436  grpci2dma_data_bd_set_next(data, DISABLED_DESCRIPTOR);
437  return GRPCI2DMA_ERR_OK;
438  }else{
439  /* No more data */
440  /* Clear DMA NBD */
441  grpci2dma_channel_bd_set_data(chan, DISABLED_DESCRIPTOR);
442  /* Update the removed data */
443  grpci2dma_data_bd_set_next(data, DISABLED_DESCRIPTOR);
444  return GRPCI2DMA_ERR_OK;
445  }
446  } else {
447  /* It is not the first data. Let's find it */
448  struct grpci2_bd_data * current = first_data;
449  struct grpci2_bd_data * next = grpci2dma_data_bd_get_next(current);
450  while( (next != data) && (next != DISABLED_DESCRIPTOR) &&
451  (next != NULL)){
452  current = next;
453  next = grpci2dma_data_bd_get_next(current);
454  }
455  if (next != data) {
456  DBG("Maximum DMA data exceeded. Maybe corrupted?\n");
457  return GRPCI2DMA_ERR_NOTFOUND;
458  } else {
459  /* Detach the data */
460  next = grpci2dma_data_bd_get_next(data);
461  grpci2dma_data_bd_set_next(current, next);
462  /* Update the removed data */
463  grpci2dma_data_bd_set_next(data, DISABLED_DESCRIPTOR);
464  return GRPCI2DMA_ERR_OK;
465  }
466  }
467  }

922 /* Start a channel */
923 STATIC int grpci2dma_channel_start(int chan_no, int options)
924 {
925  struct grpci2dma_priv *priv = grpci2dmapriv;
926  struct grpci2_bd_chan *chan;
927  SPIN_IRQFLAGS(irqflags);
928 
929  /* Get chan pointer */
930  chan = priv->channel[chan_no].ptr;
931 
932  /* Check if channel is active */
933  if (priv->channel[chan_no].active){
934  /* nothing to do */
935  return GRPCI2DMA_ERR_OK;
936  }
937 
938  /* Get the max descriptor count */
939  unsigned int desccnt;
940  if (options == 0){
941  /* Default */
942  desccnt = 0xffff;
943  }else{
944  desccnt = options & 0xffff;
945  }
946 
947  /* Start the channel by enabling it.
948  * HWNOTE: In GRPCI2 this bit does not work as it is supposed.
949  * So we better add/remove the channel from the active linked
950  * list. */
951  grpci2dma_channel_bd_enable(chan, desccnt);
952  priv->channel[chan_no].active = 1;
953  priv->nactive++;
954  /* Get active linked list */
955  struct grpci2_bd_chan * list = grpci2dma_channel_get_active_list();
956  if (list == NULL){
957  /* No previous channels. New list */
958  list = chan;
959  }
960  /* Add channel from the linked list */
961  if (grpci2dma_channel_list_add(list, chan) < 0){
962  return GRPCI2DMA_ERR_ERROR;
963  }
964 
965  /* Increase NUMCH in DMA ctrl */
966  SPIN_LOCK_IRQ(&priv->devlock, irqflags);
967  grpci2dma_ctrl_numch_set( (priv->nactive? priv->nactive -1:0));
968 
969  /* Check if DMA is active */
970  if (!grpci2dma_active()){
971  /* Start DMA */
972  grpci2dma_ctrl_start(chan);
973  }
974  SPIN_UNLOCK_IRQ(&priv->devlock, irqflags);
975 
976  DBG("Channel %d started (0x%08x)\n", chan_no, (unsigned int) chan);
977 
978  return GRPCI2DMA_ERR_OK;
979 }

981 /* Stop a channel */
982 STATIC int grpci2dma_channel_stop(int chan_no)
983 {
984  struct grpci2dma_priv *priv = grpci2dmapriv;
985  struct grpci2_bd_chan *chan;
986  SPIN_IRQFLAGS(irqflags);
987  int resume;
988 
989  /* Get chan pointer */
990  chan = priv->channel[chan_no].ptr;
991 
992  /* Check if channel is active */
993  if (!priv->channel[chan_no].active){
994  /* nothing to do */
995  return GRPCI2DMA_ERR_OK;
996  }
997 
998  /* First remove channel from the linked list */
999  if (grpci2dma_channel_list_remove(chan) < 0){
1000  return GRPCI2DMA_ERR_ERROR;
1001  }
1002 
1003  /* Update driver struct */
1004  priv->channel[chan_no].active = 0;
1005  priv->nactive--;
1006 
1007  /* Check if DMA is active and it the removed
1008  * channel is the active */
1009  resume = 0;
1010  SPIN_LOCK_IRQ(&priv->devlock, irqflags);
1011  if (grpci2dma_active() && (grpci2dma_ctrl_active() == (unsigned int)chan)){
1012  /* We need to stop the DMA */
1013  grpci2dma_ctrl_stop();
1014  SPIN_UNLOCK_IRQ(&priv->devlock, irqflags);
1015  /* Wait until DMA stops */
1016  while (grpci2dma_active()){}
1017  /* We need to check later to resume the DMA */
1018  resume = 1;
1019  }else{
1020  SPIN_UNLOCK_IRQ(&priv->devlock, irqflags);
1021  }
1022 
1023 
1024  /* Now either the DMA is stopped, or it is processing
1025  * a different channel and the removed channel is no
1026  * longer in the linked list */
1027 
1028  /* Now is safe to update the removed channel */
1029  grpci2dma_channel_bd_set_next(chan, chan);
1030 
1031  /* Stop the channel by disabling it.
1032  * HWNOTE: In GRPCI2 this bit does not work as it is supposed.
1033  * So we better remove the channel from the active linked
1034  * list. */
1035  grpci2dma_channel_bd_disable(chan);
1036 
1037  /* Point channel to disabled descriptor */
1038  grpci2dma_channel_bd_set_data(chan, DISABLED_DESCRIPTOR);
1039 
1040  DBG("Channel %d stoped (0x%08x)\n", chan_no, (unsigned int) chan);
1041 
1042  /* Decrease NUMCH in DMA ctrl */
1043  SPIN_LOCK_IRQ(&priv->devlock, irqflags);
1044  grpci2dma_ctrl_numch_set( (priv->nactive? priv->nactive -1:0));
1045 
1046  /* Reactivate DMA only if we stopped */
1047  if (resume){
1048  /* We have two options, either we stopped when the active
1049  * channel was still the active one, or we stopped when
1050  * the active channel was a different one */
1051  if (grpci2dma_ctrl_active() == (unsigned int) chan){
1052  /* In this case, we need to start the DMA with
1053  * any active channel on the list */
1054  int i;
1055  for (i=0; i<MAX_DMA_CHANS; i++){
1056  if (priv->channel[i].active){
1057  grpci2dma_ctrl_start(priv->channel[i].ptr);
1058  break;
1059  }
1060  }
1061  }else{
1062  /* In this case, we need to resume the DMA operation */
1063  /* HWNOTE: The GRPCI2 core does not update the channel next
1064  * data descriptor if we stopped a channel. This means that
1065  * we need to resume the DMA from the descriptor is was,
1066  * by only setting the enable bit, and not changing the
1067  * base register */
1068  grpci2dma_ctrl_resume();
1069  }
1070  }
1071  SPIN_UNLOCK_IRQ(&priv->devlock, irqflags);
1072 
1073  return GRPCI2DMA_ERR_OK;
1074 }

1169 /* PCI DMA Interrupt handler, called when there is a PCI DMA interrupt */
1170 STATIC void grpci2dma_isr(void *arg)
1171 {
1172  struct grpci2dma_priv *priv = arg;
1173  SPIN_ISR_IRQFLAGS(irqflags);
1174  unsigned int ctrl = grpci2dma_ctrl_status();
1175  /* Clear Interrupts */
1176  SPIN_LOCK(&priv->devlock, irqflags);
1177  grpci2dma_ctrl_interrupt_clear();
1178  SPIN_UNLOCK(&priv->devlock, irqflags);
1179  unsigned int sts = (ctrl & DMACTRL_CHIRQ) >> DMACTRL_CHIRQ_BIT;
1180  unsigned int errsts = (ctrl & DMACTRL_ERR);
1181 
1182  /* Error interrupt */
1183  if(errsts){
1184  /* Find which channels had the error.
1185  * The GRPCI2DMA core does not indicate which channel
1186  * had the error, so we need to get 1st the base descriptor register
1187  * and see if it a channel. If is not a channel, then the active
1188  * channel register tells us which channel is.
1189  * After having the channel we need to find out which channel was. */
1190  struct grpci2_bd_chan * chan =
1191  (struct grpci2_bd_chan *) grpci2dma_ctrl_base();
1192  /* Check if the base is a channel descriptor */
1193  if ((BD_READ(&chan->ctrl) & BD_CHAN_TYPE) != BD_CHAN_TYPE_DMA){
1194  /* Is not a channel, so the channel is in the channel active
1195  * register */
1196  chan = (struct grpci2_bd_chan *) grpci2dma_ctrl_active();
1197  }
1198  int i;
1199  for (i=0; i<MAX_DMA_CHANS; i++){
1200  if (chan == priv->channel[i].ptr){
1201  /* Found */
1202  if (priv->channel[i].isr != NULL){
1203  (priv->channel[i].isr)(priv->channel[i].isr_arg,i,errsts);
1204  }else{
1205  printk("Unhandled GRPCI2 DMA error interrupt, sts:0x%02x\n", errsts);
1206  }
1207  break;
1208  }
1209  }
1210  if (i == MAX_DMA_CHANS){
1211  printk("Unhandled GRPCI2 DMA error interrupt , sts:0x%02x\n", errsts);
1212  }
1213  }
1214 
1215  /* Normal packet interrupt */
1216  int cid=0;
1217  /* Find which channels have interrupts */
1218  while(sts){
1219  /* Find if current channel has an interrupt*/
1220  if(sts & 0x1){
1221  /* Find if current channel has an ISR */
1222  if (priv->channel[cid].isr != NULL){
1223  (priv->channel[cid].isr)(
1224  priv->channel[cid].isr_arg, cid, errsts);
1225  }else{
1226  printk("Unhandled GRPCI2 DMA interrupt in channel %d, sts:0x%02x\n", cid, 0);
1227  }

1641 /* Transfer_size =0 means maximum */
1642 int grpci2dma_prepare(
1643  uint32_t pci_start, uint32_t ahb_start, int dir, int endianness,
1644  int size, void * dataptr, int index, int ndata, int transfer_size)
1645 {

1647 
1648  /* Check data pointer */
1649  if ((data == NULL) ||
1650  (((unsigned int ) data) & (GRPCI2DMA_BD_DATA_ALIGN-1))){
1651  return GRPCI2DMA_ERR_WRONGPTR;
1652  }
1653 
1654  /* Check indexes */
1655  int maxdata = ndata - index;
1656  if ((maxdata < 1) || (index < 0)){
1657  /* No data descriptors to use */
1658  return GRPCI2DMA_ERR_WRONGPTR;
1659  }
1660 
1661  /* Check PCI transfer size */
1662  if ( (transfer_size < 0) ||
1663  (transfer_size > MAX_DMA_TRANSFER_SIZE) ||
1664  (transfer_size%4 != 0) ) {
1665  return GRPCI2DMA_ERR_WRONGPTR;
1666  }
1667  if (transfer_size == 0){
1668  transfer_size = MAX_DMA_TRANSFER_SIZE;
1669  }
1670 
1671  /* Check total size */
1672  if ( (size <=0) || (size % 4 != 0)){
1673  return GRPCI2DMA_ERR_WRONGPTR;
1674  }
1675 
1676  /* Calculate number of data descriptors needed */
1677  int words = size/4;
1678  int blocksize = transfer_size/4;
1679  int datacnt = words/blocksize + (words%blocksize != 0? 1: 0);
1680  /* Check that we can transfer the data */
1681  if (datacnt > maxdata) {
1682  return GRPCI2DMA_ERR_TOOMANY;
1683  }

1688  uint32_t ahb_adr;

1695  ahb_adr = ahb_start + i*blocksize;
1696  /* Get current data size */
1697  if (remaining >= blocksize){
1698  datasize = blocksize - 1;
1699  remaining -= blocksize;
1700  } else {
1701  datasize = remaining -1;
1702  remaining = 0;
1703  }

1705  if (i == datacnt - 1){
1706  /* Last transfer */
1707  next = DISABLED_DESCRIPTOR;
1708  }else{
1709  next = &data[i+index+1];
1710  }
1711  /* Set Data descriptor */
1712  grpci2dma_data_bd_init(&data[i+index], pci_adr, ahb_adr, dir, endianness, datasize, next);
1713  }
1714  /* Return number of transfers used */
1715  return datacnt;
1716 }
