54 static bool altera_cyclone_v_uart_probe(
55  rtems_termios_device_context *base,
56  uint32_t uart_set_mask
57 )
58 {
59  ns16550_context *ctx = (ns16550_context *) base;
60  bool ret = true;
61  uint32_t ucr;
62  ALT_STATUS_CODE sc;
63  void* location = (void *) ctx->port;
64 
65  /* The ALT_CLK_L4_SP is required for all SoCFPGA UARTs.
66  * Check that it's enabled. */
67  if ( alt_clk_is_enabled(ALT_CLK_L4_SP) != ALT_E_TRUE ) {
68  ret = false;
69  }
70 
71  if ( ret ) {
72  sc = alt_clk_freq_get(ALT_CLK_L4_SP, &ctx->clock);
73  if ( sc != ALT_E_SUCCESS ) {
74  ret = false;
75  }
76  }
77 
78  if ( ret ) {
79  // Bring UART out of reset.
80  alt_clrbits_word(ALT_RSTMGR_PERMODRST_ADDR, uart_set_mask);
81 
82  // Verify the UCR (UART Component Version)
83  ucr = alt_read_word( ALT_UART_UCV_ADDR( location ) );
84  if ( ucr != ALT_UART_UCV_UART_COMPONENT_VER_RESET ) {
85  ret = false;
86  }
87  }
88 
89  if ( ret ) {
90  // Write SRR::UR (Shadow Reset Register :: UART Reset)
91  alt_write_word( ALT_UART_SRR_ADDR( location ), ALT_UART_SRR_UR_SET_MSK );
92 
93  // Read the MSR to work around case:119085.
94  (void)alt_read_word( ALT_UART_MSR_ADDR( location ) );
95 
96  ret = ns16550_probe( base );
97  }
98 
99  return ret;
100 }
