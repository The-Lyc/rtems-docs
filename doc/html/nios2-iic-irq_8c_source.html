44 RTEMS_INLINE_ROUTINE void __IIC_Handler(void)
45 {
46  uint32_t active;
47  uint32_t mask;
48  uint32_t vector;
49 
50  /*
51  * Obtain from the interrupt controller a bit list of pending interrupts,
52  * and then process the highest priority interrupt. This process loops,
53  * loading the active interrupt list on each pass until ipending
54  * return zero.
55  *
56  * The maximum interrupt latency for the highest priority interrupt is
57  * reduced by finding out which interrupts are pending as late as possible.
58  * Consider the case where the high priority interupt is asserted during
59  * the interrupt entry sequence for a lower priority interrupt to see why
60  * this is the case.
61  */
62 
63  active = _Nios2_Get_ctlreg_ipending();
64 
65  while (active)
66  {
67  vector = 0;
68  mask = 1;
69 
70  /*
71  * Test each bit in turn looking for an active interrupt. Once one is
72  * found call it to clear the interrupt condition.
73  */
74 
75  while (active)
76  {
77  if (active & mask)
78  {
79  if ( _ISR_Vector_table[ vector] )
80  (*_ISR_Vector_table[ vector ])(vector);
81  active &= ~mask;
82  }
83  mask <<= 1;
84  ++vector;
85  };
86 
87  active = _Nios2_Get_ctlreg_ipending();
88  }
89 
90 }
