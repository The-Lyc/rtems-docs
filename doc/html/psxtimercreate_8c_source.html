36  clockid_t clock_id,

39 )
40 {
41  POSIX_Timer_Control *ptimer;
42 
43  if ( clock_id != CLOCK_REALTIME )
44  rtems_set_errno_and_return_minus_one( EINVAL );
45 
46  if ( !timerid )
47  rtems_set_errno_and_return_minus_one( EINVAL );
48 
49  /*
50  * The data of the structure evp are checked in order to verify if they
51  * are coherent.
52  */
53 
54  if (evp != NULL) {
55  /* The structure has data */
56  if ( ( evp->sigev_notify != SIGEV_NONE ) &&
57  ( evp->sigev_notify != SIGEV_SIGNAL ) ) {
58  /* The value of the field sigev_notify is not valid */
59  rtems_set_errno_and_return_minus_one( EINVAL );
60  }
61 
62  if ( !evp->sigev_signo )
63  rtems_set_errno_and_return_minus_one( EINVAL );
64 
65  if ( !is_valid_signo(evp->sigev_signo) )
66  rtems_set_errno_and_return_minus_one( EINVAL );
67  }
68 
69  /*
70  * Allocate a timer
71  */
72  ptimer = _POSIX_Timer_Allocate();
73  if ( !ptimer ) {
74  _Objects_Allocator_unlock();
75  rtems_set_errno_and_return_minus_one( EAGAIN );
76  }
77 
78  /* The data of the created timer are stored to use them later */
79 
80  ptimer->state = POSIX_TIMER_STATE_CREATE_NEW;
81  ptimer->thread_id = _Thread_Get_executing()->Object.id;
82 
83  if ( evp != NULL ) {
84  ptimer->inf.sigev_notify = evp->sigev_notify;
85  ptimer->inf.sigev_signo = evp->sigev_signo;
86  ptimer->inf.sigev_value = evp->sigev_value;
87  }
88 
89  ptimer->overrun = 0;
90  ptimer->timer_data.it_value.tv_sec = 0;
91  ptimer->timer_data.it_value.tv_nsec = 0;
92  ptimer->timer_data.it_interval.tv_sec = 0;
93  ptimer->timer_data.it_interval.tv_nsec = 0;
94 
95  _Watchdog_Preinitialize( &ptimer->Timer, _Per_CPU_Get_snapshot() );
96  _Watchdog_Initialize( &ptimer->Timer, _POSIX_Timer_TSR );
97  _Objects_Open_u32(&_POSIX_Timer_Information, &ptimer->Object, 0);
98 
99  *timerid = ptimer->Object.id;
100  _Objects_Allocator_unlock();
101  return 0;
102 }
