81 /*
82  * This method is unique to this scheduler because it takes into
83  * account affinity as it determines the highest ready thread.
84  * Since this is used to pick a new thread to replace the victim,
85  * the highest ready thread must have affinity such that it can
86  * be executed on the victim's processor.
87  */
88 static Scheduler_Node *_Scheduler_priority_affinity_SMP_Get_highest_ready(
89  Scheduler_Context *context,
90  Scheduler_Node *victim
91 )
92 {
93  Scheduler_priority_SMP_Context *self =
94  _Scheduler_priority_SMP_Get_self( context );
95  Priority_Control index;
96  Scheduler_Node *highest = NULL;
97  Thread_Control *victim_thread;
98  uint32_t victim_cpu_index;
99  Scheduler_priority_affinity_SMP_Node *node;
100 
101  /*
102  * This is done when we need to check if reevaluations are needed.
103  */
104  if ( victim == NULL ) {
105  node = (Scheduler_priority_affinity_SMP_Node *)
106  _Scheduler_priority_Ready_queue_first(
107  &self->Bit_map,
108  &self->Ready[ 0 ]
109  );
110 
111  return &node->Base.Base.Base;
112  }
113 
114  victim_thread = _Scheduler_Node_get_owner( victim );
115  victim_cpu_index = _Per_CPU_Get_index( _Thread_Get_CPU( victim_thread ) );
116 
131  for ( index = _Priority_bit_map_Get_highest( &self->Bit_map ) ;
132  index <= PRIORITY_MAXIMUM;
133  index++ )
134  {
135  Chain_Control *chain = &self->Ready[index];
136  Chain_Node *chain_node;
137  for ( chain_node = _Chain_First( chain );
138  chain_node != _Chain_Immutable_tail( chain ) ;
139  chain_node = _Chain_Next( chain_node ) )
140  {
141  node = (Scheduler_priority_affinity_SMP_Node *) chain_node;
142 
143  /*
144  * Can this thread run on this CPU?
145  */
146  if ( _Processor_mask_Is_set( &node->Affinity, victim_cpu_index ) ) {
147  highest = &node->Base.Base.Base;
148  break;
149  }
150  }
151  if ( highest )
152  break;
153  }
154 
155  _Assert( highest != NULL );
156 
157  return highest;
158 }

192 /*
193  * This method is unique to this scheduler because it must take into
194  * account affinity as it searches for the lowest priority scheduled
195  * thread. It ignores those which cannot be replaced by the filter
196  * thread because the potential victim thread does not have affinity
197  * for that processor.
198  */
199 static Scheduler_Node * _Scheduler_priority_affinity_SMP_Get_lowest_scheduled(
200  Scheduler_Context *context,
201  Scheduler_Node *filter_base
202 )
203 {
204  Scheduler_SMP_Context *self = _Scheduler_SMP_Get_self( context );
205  Scheduler_Node *lowest_scheduled = NULL;
206  Chain_Control *scheduled = &self->Scheduled;
207  Chain_Node *chain_node;
208  Scheduler_priority_affinity_SMP_Node *filter =
209  _Scheduler_priority_affinity_SMP_Node_downcast( filter_base );
210 
211  for ( chain_node = _Chain_Last( scheduled );
212  chain_node != _Chain_Immutable_head( scheduled ) ;
213  chain_node = _Chain_Previous( chain_node ) ) {
214  Scheduler_priority_affinity_SMP_Node *node;
215  Thread_Control *thread;
216  uint32_t cpu_index;
217 
218  node = (Scheduler_priority_affinity_SMP_Node *) chain_node;
219 
220  /* cpu_index is the processor number thread is executing on */
221  thread = _Scheduler_Node_get_owner( &node->Base.Base.Base );
222  cpu_index = _Per_CPU_Get_index( _Thread_Get_CPU( thread ) );
223 
224  if ( _Processor_mask_Is_set( &filter->Affinity, cpu_index ) ) {
225  lowest_scheduled = &node->Base.Base.Base;
226  break;
227  }
228 
229  }
230 
231  return lowest_scheduled;
232 }

258 /*
259  * This method is invoked at the end of certain scheduling operations
260  * to ensure that the highest priority ready thread cannot be scheduled
261  * to execute. When we schedule with affinity, there is the possibility
262  * that we need to migrate a thread to another core to ensure that the
263  * highest priority ready threads are in fact scheduled.
264  */
265 static void _Scheduler_priority_affinity_SMP_Check_for_migrations(
266  Scheduler_Context *context
267 )
268 {
269  Scheduler_priority_SMP_Context *self;
270  Scheduler_Node *lowest_scheduled;
271  Scheduler_Node *highest_ready;
272 
273  self = _Scheduler_priority_SMP_Get_self( context );
274 
275  while (1) {
276  Priority_Control lowest_scheduled_priority;
277  Priority_Control insert_priority;
278 
279  if ( _Priority_bit_map_Is_empty( &self->Bit_map ) ) {
280  /* Nothing to do */
281  break;
282  }
283 
284  highest_ready =
285  _Scheduler_priority_affinity_SMP_Get_highest_ready( context, NULL );
286 
287  lowest_scheduled =
288  _Scheduler_priority_affinity_SMP_Get_lowest_scheduled(
289  context,
290  highest_ready
291  );
292 
293  /*
294  * If we can't find a thread to displace from the scheduled set,
295  * then we have placed all the highest priority threads possible
296  * in the scheduled set.
297  *
298  * We found the absolute highest priority thread without
299  * considering affinity. But now we have to consider that thread's
300  * affinity as we look to place it.
301  */
302 
303  if ( lowest_scheduled == NULL )
304  break;
305 
306  lowest_scheduled_priority =
307  _Scheduler_SMP_Node_priority( lowest_scheduled );
308 
309  if (
310  _Scheduler_SMP_Priority_less_equal(
311  &lowest_scheduled_priority,
312  &highest_ready->Node.Chain
313  )
314  ) {
315  break;
316  }
317 
318  /*
319  * But if we found a thread which is lower priority than one
320  * in the ready set, then we need to swap them out.
321  */
322 
323  _Scheduler_priority_SMP_Extract_from_ready( context, highest_ready );
324  insert_priority = _Scheduler_SMP_Node_priority( highest_ready );
325  insert_priority = SCHEDULER_PRIORITY_APPEND( insert_priority );
326  _Scheduler_SMP_Enqueue_to_scheduled(
327  context,
328  highest_ready,
329  insert_priority,
330  lowest_scheduled,
331  _Scheduler_SMP_Insert_scheduled,
332  _Scheduler_priority_SMP_Move_from_scheduled_to_ready,
333  _Scheduler_SMP_Allocate_processor_exact
334  );
335  }

534 bool _Scheduler_priority_affinity_SMP_Set_affinity(
535  const Scheduler_Control *scheduler,
536  Thread_Control *thread,
537  Scheduler_Node *node_base,
538  const Processor_mask *affinity
539 )
540 {
541  Scheduler_Context *context;
542  Scheduler_priority_affinity_SMP_Node *node;
543  States_Control current_state;
544  Processor_mask my_affinity;
545 
546  context = _Scheduler_Get_context( scheduler );
547  _Processor_mask_And( &my_affinity, &context->Processors, affinity );
548 
549  if ( _Processor_mask_Count( &my_affinity ) == 0 ) {
550  return false;
551  }
552 
553  node = _Scheduler_priority_affinity_SMP_Node_downcast( node_base );
554 
555  /*
556  * The old and new set are the same, there is no point in
557  * doing anything.
558  */
559  if ( _Processor_mask_Is_equal( &node->Affinity, affinity ) )
560  return true;
561 
562  current_state = thread->current_state;
563 
564  if ( _States_Is_ready( current_state ) ) {
565  _Scheduler_priority_affinity_SMP_Block( scheduler, thread, &node->Base.Base.Base );
566  }
567 
568  _Processor_mask_Assign( &node->Affinity, affinity );
569 
570  if ( _States_Is_ready( current_state ) ) {
571  /*
572  * FIXME: Do not ignore threads in need for help.
573  */
574  (void) _Scheduler_priority_affinity_SMP_Unblock( scheduler, thread, &node->Base.Base.Base );
575  }
576 
577  return true;
