50 )

52  void *p;
53  int err = 0;
54 
55  if ( size < shm_obj->size ) {
56  /* zero out if shrinking */
57  p = (void*)((uintptr_t)shm_obj->handle + (uintptr_t)size);
58  memset( p, 0, shm_obj->size - size );
