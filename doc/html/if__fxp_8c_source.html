417 int
418 rtems_fxp_attach(struct rtems_bsdnet_ifconfig *config, int attaching)
419 {
420  int error = 0;
421  struct fxp_softc *sc;
422  struct ifnet *ifp;
423  uint16_t val16;
424  uint32_t val32;
425  uint16_t data;
426  int i;
427  int s;
428  int unitNumber;
429  char *unitName;
430  u_int16_t dev_id;
431  u_int8_t interrupt;
432  int mtu;
433 
434  /*
435  * Set up some timing values
436  */
437  fxp_ticksPerSecond = rtems_clock_get_ticks_per_second();
438  DBGLVL_PRINTK(1,"fxp_attach called\n");
439 
440  /*
441  * Parse driver name
442  */
443  if ((unitNumber = rtems_bsdnet_parse_driver_name (config, &unitName)) < 0)
444  return 0;
445 
446  /*
447  * Is driver free?
448  */
449  if ((unitNumber <= 0) || (unitNumber > NFXPDRIVER)) {
450  device_printf(dev,"Bad FXP unit number.\n");
451  return 0;
452  }
453  sc = &fxp_softc[unitNumber - 1];
454  ifp = &sc->arpcom.ac_if;
455  if (ifp->if_softc != NULL) {
456  device_printf(dev,"FXP Driver already in use.\n");
457  return 0;
458  }
459 
460  memset(sc, 0, sizeof(*sc));
461 #ifdef NOTUSED
462  sc->dev = dev;
463  callout_handle_init(&sc->stat_ch);
464  mtx_init(&sc->sc_mtx, device_get_nameunit(dev), MTX_DEF | MTX_RECURSE);
465 #endif
466  s = splimp();
467 
468  /*
469  * find device on pci bus
470  */
471  { int j; int pbus, pdev, pfun;
472 
473  for (j=0; fxp_ident_table[j].devid; j++ ) {
474  i = pci_find_device( 0x8086, fxp_ident_table[j].devid,
475  unitNumber-1, &pbus, &pdev, &pfun );
476  sc->pci_bus = pbus;
477  sc->pci_dev = pdev;
478  sc->pci_fun = pfun;
479  DBGLVL_PRINTK(2,"fxp_attach: find_devid returned %d ,"
480  "pci bus %d dev %d fun %d \n",
481  i, sc->pci_bus, sc->pci_dev, sc->pci_fun);
482  if (PCIB_ERR_SUCCESS == i) {
483  if ( UNTESTED == fxp_ident_table[j].warn ) {
484  device_printf(dev,
485 "WARNING: this chip version has NOT been reported to work under RTEMS yet.\n");
486  device_printf(dev,
487 " If it works OK, report it as tested in 'c/src/libchip/network/if_fxp.c'\n");
488  }
489  break;
490  }
491  }
492  }
493 
494  /*
495  * FIXME: add search for more device types...
496  */
497  if (i != PCIB_ERR_SUCCESS) {
498  device_printf(dev, "could not find 82559ER device\n");
499  return 0;
500  }
501 
502 
503  /*
504  * Enable bus mastering. Enable memory space too, in case
505  * BIOS/Prom forgot about it.
506  */
507  fxp_pci_conf_read16(sc, PCI_COMMAND,&val16);
508  val16 |= (PCI_COMMAND_MEMORY|PCI_COMMAND_MASTER);
509  fxp_pci_conf_write16(sc, PCI_COMMAND, val16);
510  DBGLVL_PRINTK(3,"fxp_attach: PCI_COMMAND_write = 0x%x\n",val16);
511  fxp_pci_conf_read16(sc, PCI_COMMAND,&val16);
512  DBGLVL_PRINTK(4,"fxp_attach: PCI_COMMAND_read = 0x%x\n",val16);
513 
514  /*
515  * Figure out which we should try first - memory mapping or i/o mapping?
516  * We default to memory mapping. Then we accept an override from the
517  * command line. Then we check to see which one is enabled.
518  */
519 #ifdef NOTUSED
520  m1 = PCI_COMMAND_MEMORY;
521  m2 = PCI_COMMAND_IO;
522  prefer_iomap = 0;
523  if (resource_int_value(device_get_name(dev), device_get_unit(dev),
524  "prefer_iomap", &prefer_iomap) == 0 && prefer_iomap != 0) {
525  m1 = PCI_COMMAND_IO;
526  m2 = PCI_COMMAND_MEMORY;
527  }
528 
529  if (val & m1) {
530  sc->rtp = ((m1 == PCI_COMMAND_MEMORY)
531  ? SYS_RES_MEMORY : SYS_RES_IOPORT);
532  sc->rgd = ((m1 == PCI_COMMAND_MEMORY)
533  ? FXP_PCI_MMBA : FXP_PCI_IOBA);
534  sc->mem = bus_alloc_resource(dev, sc->rtp, &sc->rgd,
535  0, ~0, 1, RF_ACTIVE);
536  }
537  if (sc->mem == NULL && (val & m2)) {
538  sc->rtp = ((m2 == PCI_COMMAND_MEMORY)
539  ? SYS_RES_MEMORY : SYS_RES_IOPORT);
540  sc->rgd = ((m2 == PCI_COMMAND_MEMORY)
541  ? FXP_PCI_MMBA : FXP_PCI_IOBA);
542  sc->mem = bus_alloc_resource(dev, sc->rtp, &sc->rgd,
543  0, ~0, 1, RF_ACTIVE);
544  }
545 
546  if (!sc->mem) {
547  device_printf(dev, "could not map device registers\n");
548  error = ENXIO;
549  goto fail;
550  }
551  if (fxp_is_verbose) {
552  device_printf(dev, "using %s space register mapping\n",
553  sc->rtp == SYS_RES_MEMORY? "memory" : "I/O");
554  }
555 
556  sc->sc_st = rman_get_bustag(sc->mem);
557  sc->sc_sh = rman_get_bushandle(sc->mem);
558 
559  /*
560  * Allocate our interrupt.
561  */
562  rid = 0;
563  sc->irq = bus_alloc_resource(dev, SYS_RES_IRQ, &rid, 0, ~0, 1,
564  RF_SHAREABLE | RF_ACTIVE);
565  if (sc->irq == NULL) {
566  device_printf(dev, "could not map interrupt\n");
567  error = ENXIO;
568  goto fail;
569  }
570 
571  error = bus_setup_intr(dev, sc->irq, INTR_TYPE_NET,
572  fxp_intr, sc, &sc->ih);
573  if (error) {
574  device_printf(dev, "could not setup irq\n");
575  goto fail;
576  }
577 #endif
578 
579  /*
580  * get mapping and base address of registers
581  */
582  fxp_pci_conf_read16(sc, PCI_COMMAND,&val16);
583  DBGLVL_PRINTK(4,"fxp_attach: PCI_COMMAND_read = 0x%x\n",val16);
584  if((val16 & PCI_COMMAND_IO) != 0) {
585  sc->pci_regs_are_io = true;
586  fxp_pci_conf_read32(sc, PCI_BASE_ADDRESS_1, &val32);
587  sc->pci_regs_base = val32 & PCI_BASE_ADDRESS_IO_MASK;
588  }
589  else {
590  sc->pci_regs_are_io = false;
591  fxp_pci_conf_read32(sc, PCI_BASE_ADDRESS_0, &val32);

951 static void
952 fxp_eeprom_shiftin(struct fxp_softc *sc, int data, int length)
953 {
954  u_int16_t reg;
955  int x;
956 
957  /*
958  * Shift in data.
959  */
960  for (x = 1 << (length - 1); x; x >>= 1) {
961  if (data & x)
962  reg = FXP_EEPROM_EECS | FXP_EEPROM_EEDI;
963  else
964  reg = FXP_EEPROM_EECS;
965  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg);
966  DELAY(1);
967  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg | FXP_EEPROM_EESK);
968  DELAY(1);
969  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg);
970  DELAY(1);
971  }
972 }

974 /*
975  * Read from the serial EEPROM. Basically, you manually shift in
976  * the read opcode (one bit at a time) and then shift in the address,
977  * and then you shift out the data (all of this one bit at a time).
978  * The word size is 16 bits, so you have to provide the address for
979  * every 16 bits of data.
980  */
981 static u_int16_t
982 fxp_eeprom_getword(struct fxp_softc *sc, int offset, int autosize)
983 {
984  u_int16_t reg, data;
985  int x;
986 
987  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, FXP_EEPROM_EECS);
988  /*
989  * Shift in read opcode.
990  */
991  fxp_eeprom_shiftin(sc, FXP_EEPROM_OPC_READ, 3);
992  /*
993  * Shift in address.
994  */
995  data = 0;
996  for (x = 1 << (sc->eeprom_size - 1); x; x >>= 1) {
997  if (offset & x)
998  reg = FXP_EEPROM_EECS | FXP_EEPROM_EEDI;
999  else
1000  reg = FXP_EEPROM_EECS;
1001  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg);
1002  DELAY(1);
1003  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg | FXP_EEPROM_EESK);
1004  DELAY(1);
1005  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg);
1006  DELAY(1);
1007  reg = CSR_READ_2(sc, FXP_CSR_EEPROMCONTROL) & FXP_EEPROM_EEDO;
1008  data++;
1009  if (autosize && reg == 0) {
1010  sc->eeprom_size = data;
1011  break;
1012  }
1013  }
1014  /*
1015  * Shift out data.
1016  */
1017  data = 0;
1018  reg = FXP_EEPROM_EECS;
1019  for (x = 1 << 15; x; x >>= 1) {
1020  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg | FXP_EEPROM_EESK);
1021  DELAY(1);
1022  if (CSR_READ_2(sc, FXP_CSR_EEPROMCONTROL) & FXP_EEPROM_EEDO)
1023  data |= x;
1024  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, reg);
1025  DELAY(1);
1026  }
1027  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, 0);
1028  DELAY(1);
1029 
1030  return (data);
1031 }

1033 static void
1034 fxp_eeprom_putword(struct fxp_softc *sc, int offset, u_int16_t data)
1035 {
1036  int i;
1037 
1038  /*
1039  * Erase/write enable.
1040  */
1041  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, FXP_EEPROM_EECS);
1042  fxp_eeprom_shiftin(sc, 0x4, 3);
1043  fxp_eeprom_shiftin(sc, 0x03 << (sc->eeprom_size - 2), sc->eeprom_size);
1044  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, 0);
1045  DELAY(1);
1046  /*
1047  * Shift in write opcode, address, data.
1048  */
1049  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, FXP_EEPROM_EECS);
1050  fxp_eeprom_shiftin(sc, FXP_EEPROM_OPC_WRITE, 3);
1051  fxp_eeprom_shiftin(sc, offset, sc->eeprom_size);
1052  fxp_eeprom_shiftin(sc, data, 16);
1053  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, 0);
1054  DELAY(1);
1055  /*
1056  * Wait for EEPROM to finish up.
1057  */
1058  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, FXP_EEPROM_EECS);
1059  DELAY(1);
1060  for (i = 0; i < 1000; i++) {
1061  if (CSR_READ_2(sc, FXP_CSR_EEPROMCONTROL) & FXP_EEPROM_EEDO)
1062  break;
1063  DELAY(50);
1064  }
1065  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, 0);
1066  DELAY(1);
1067  /*
1068  * Erase/write disable.
1069  */
1070  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, FXP_EEPROM_EECS);
1071  fxp_eeprom_shiftin(sc, 0x4, 3);
1072  fxp_eeprom_shiftin(sc, 0, sc->eeprom_size);
1073  CSR_WRITE_2(sc, FXP_CSR_EEPROMCONTROL, 0);
1074  DELAY(1);
1075 }

1077 /*
1078  * From NetBSD:
1079  *
1080  * Figure out EEPROM size.
1081  *
1082  * 559's can have either 64-word or 256-word EEPROMs, the 558
1083  * datasheet only talks about 64-word EEPROMs, and the 557 datasheet
1084  * talks about the existance of 16 to 256 word EEPROMs.
1085  *
1086  * The only known sizes are 64 and 256, where the 256 version is used
1087  * by CardBus cards to store CIS information.
1088  *
1089  * The address is shifted in msb-to-lsb, and after the last
1090  * address-bit the EEPROM is supposed to output a `dummy zero' bit,
1091  * after which follows the actual data. We try to detect this zero, by
1092  * probing the data-out bit in the EEPROM control register just after
1093  * having shifted in a bit. If the bit is zero, we assume we've
1094  * shifted enough address bits. The data-out should be tri-state,
1095  * before this, which should translate to a logical one.
1096  */
1097 static void
1098 fxp_autosize_eeprom(struct fxp_softc *sc)
1099 {
1100 
1101  /* guess maximum size of 256 words */
1102  sc->eeprom_size = 8;
1103 
1104  /* autosize */
1105  (void) fxp_eeprom_getword(sc, 0, 1);
1106 }

1131 /*
1132  * Start packet transmission on the interface.
1133  */
1134 static void
1135 fxp_start(struct ifnet *ifp)
1136 {
1137  struct fxp_softc *sc = ifp->if_softc;
1138  struct fxp_cb_tx *txp;
1139 
1140  DBGLVL_PRINTK(3,"fxp_start called\n");
1141 
1142  /*
1143  * See if we need to suspend xmit until the multicast filter
1144  * has been reprogrammed (which can only be done at the head
1145  * of the command chain).
1146  */
1147  if (sc->need_mcsetup) {
1148  DBGLVL_PRINTK(3,"fxp_start need_mcsetup\n");
1149  return;
1150  }
1151 
1152  txp = NULL;
1153 
1154  /*
1155  * We're finished if there is nothing more to add to the list or if
1156  * we're all filled up with buffers to transmit.
1157  * NOTE: One TxCB is reserved to guarantee that fxp_mc_setup() can add
1158  * a NOP command when needed.
1159  */
1160  while (ifp->if_snd.ifq_head != NULL && sc->tx_queued < FXP_NTXCB - 1) {
1161  struct mbuf *m, *mb_head;
1162  int segment;
1163 
1164  /*
1165  * Grab a packet to transmit.
1166  */
1167  IF_DEQUEUE(&ifp->if_snd, mb_head);
1168 
1169  /*
1170  * Get pointer to next available tx desc.
1171  */
1172  txp = sc->cbl_last->next;
1173 
1174  /*
1175  * Go through each of the mbufs in the chain and initialize
1176  * the transmit buffer descriptors with the physical address
1177  * and size of the mbuf.
1178  */
1179 tbdinit:

1195  * mbuf chain first. Bail out if we can't get the

1209  }

1226  } else {

1228  FXP_CB_COMMAND_XMIT | FXP_CB_COMMAND_SF |
1229  FXP_CB_COMMAND_S | FXP_CB_COMMAND_I;

1231  * Set a 5 second timer just in case we don't hear
1232  * from the card again.
1233  */

1235  }

1239  * Advance the end of list forward.
1240  */
1241 
1242 #ifdef __alpha__

1244  * On platforms which can't access memory in 16-bit
1245  * granularities, we must prevent the card from DMA'ing
1246  * up the status while we update the command field.
1247  * This could cause us to overwrite the completion status.
1248  */
1249  atomic_clear_short(&sc->cbl_last->cb_command,
1250  FXP_CB_COMMAND_S);
1251 #else

1253 #endif /*__alpha__*/

1255

1258  * no other packets queued (when nothing is queued, cbl_first
1259  * sits on the last TxCB that was sent out).
1260  */
1261  if (sc->tx_queued == 0)
1262  sc->cbl_first = txp;
1263 
1264  sc->tx_queued++;
1265 
1266 #ifdef NOTUSED
1267  /*
1268  * Pass packet to bpf if there is a listener.
1269  */
1270  if (ifp->if_bpf)
1271  bpf_mtap(ifp, mb_head);
1272 #endif
1273  }
1274

1276  * We're finished. If we added to the list, issue a RESUME to get DMA
1277  * going again if suspended.
1278  */
1279  if (txp != NULL) {
1280  fxp_scb_wait(sc);
1281  fxp_scb_cmd(sc, FXP_SCB_COMMAND_CU_RESUME);
1282  }
1283

1285  * reenable interrupts
1286  */
1287  RTEMS_COMPILER_MEMORY_BARRIER();
1288  CSR_WRITE_1(sc, FXP_CSR_SCB_INTRCNTL,0);
1289  bsp_interrupt_vector_enable(sc->irq_num);
1290  RTEMS_COMPILER_MEMORY_BARRIER();

1372  (txp->cb_status & FXP_CB_STATUS_C) != 0;
1373  txp = txp->next) {

1375  m_freem(txp->mb_head);
1376  txp->mb_head = NULL;
1377  }

1388  */

1406  /*
1407  * Remove first packet from the chain.
1408  */
1409  sc->rfa_headm = m->m_next;
1410  m->m_next = NULL;
1411

1413  * Add a new buffer to the receive chain.
1414  * If this fails, the old buffer is recycled
1415  * instead.
1416  */

1418  struct ether_header *eh;
1419  int total_len;
1420

1422  (MCLBYTES - 1);
1423  if (total_len <

1425  m_freem(m);

1427  }

1433  * chip.
1434  */

1445  sizeof(struct ether_header);
1446  m->m_len -=
1447  sizeof(struct ether_header);
1448  m->m_pkthdr.len = m->m_len;

1450  }

1454  fxp_scb_wait(sc);

1464  */

1467  RTEMS_COMPILER_MEMORY_BARRIER();

1471 /*
1472  * Update packet in/out/collision statistics. The i82557 doesn't
1473  * allow you to access these counters without doing a fairly
1474  * expensive DMA to get _all_ of the statistics it maintains, so
1475  * we do this operation here only once per second. The statistics
1476  * counters in the kernel are updated from the previous dump-stats
1477  * DMA and then a new dump-stats DMA is started. The on-chip
1478  * counters are zeroed when the DMA completes. If we can't start
1479  * the DMA immediately, we don't wait - we just prepare to read
1480  * them again next time.
1481  */
1482 static void
1483 fxp_tick(void *xsc)
1484 {
1485  struct fxp_softc *sc = xsc;
1486  struct ifnet *ifp = &sc->sc_if;
1487  struct fxp_stats *sp = sc->fxp_stats;
1488  struct fxp_cb_tx *txp;
1489  int s;
1490 
1491  DBGLVL_PRINTK(4,"fxp_tick called\n");
1492 
1493  ifp->if_opackets += sp->tx_good;
1494  ifp->if_collisions += sp->tx_total_collisions;
1495  if (sp->rx_good) {
1496  ifp->if_ipackets += sp->rx_good;
1497  sc->rx_idle_secs = 0;
1498  } else {
1499  /*
1500  * Receiver's been idle for another second.
1501  */
1502  sc->rx_idle_secs++;
1503  }
1504  ifp->if_ierrors +=
1505  sp->rx_crc_errors +
1506  sp->rx_alignment_errors +
1507  sp->rx_rnr_errors +
1508  sp->rx_overrun_errors;
1509  /*
1510  * If any transmit underruns occured, bump up the transmit
1511  * threshold by another 512 bytes (64 * 8).
1512  */
1513  if (sp->tx_underruns) {
1514  ifp->if_oerrors += sp->tx_underruns;
1515  if (tx_threshold < 192)
1516  tx_threshold += 64;
1517  }
1518  s = splimp();
1519  /*
1520  * Release any xmit buffers that have completed DMA. This isn't
1521  * strictly necessary to do here, but it's advantagous for mbufs
1522  * with external storage to be released in a timely manner rather
1523  * than being defered for a potentially long time. This limits
1524  * the delay to a maximum of one second.
1525  */
1526  for (txp = sc->cbl_first; sc->tx_queued &&
1527  (txp->cb_status & FXP_CB_STATUS_C) != 0;
1528  txp = txp->next) {
1529  if (txp->mb_head != NULL) {
1530  m_freem(txp->mb_head);
1531  txp->mb_head = NULL;
1532  }
1533  sc->tx_queued--;
1534  }
1535  sc->cbl_first = txp;
1536  /*
1537  * If we haven't received any packets in FXP_MAC_RX_IDLE seconds,
1538  * then assume the receiver has locked up and attempt to clear
1539  * the condition by reprogramming the multicast filter. This is
1540  * a work-around for a bug in the 82557 where the receiver locks
1541  * up if it gets certain types of garbage in the syncronization
1542  * bits prior to the packet header. This bug is supposed to only
1543  * occur in 10Mbps mode, but has been seen to occur in 100Mbps
1544  * mode as well (perhaps due to a 10/100 speed transition).
1545  */
1546  if (sc->rx_idle_secs > FXP_MAX_RX_IDLE) {
1547  sc->rx_idle_secs = 0;
1548  fxp_mc_setup(sc);
1549  }
1550  /*
1551  * If there is no pending command, start another stats
1552  * dump. Otherwise punt for now.
1553  */
1554  if (CSR_READ_1(sc, FXP_CSR_SCB_COMMAND) == 0) {
1555  /*
1556  * Start another stats dump.
1557  */
1558  fxp_scb_cmd(sc, FXP_SCB_COMMAND_CU_DUMPRESET);
1559  } else {
1560  /*
1561  * A previous command is still waiting to be accepted.
1562  * Just zero our copy of the stats and wait for the
1563  * next timer event to update them.
1564  */
1565  sp->tx_good = 0;
1566  sp->tx_underruns = 0;
1567  sp->tx_total_collisions = 0;
1568 
1569  sp->rx_good = 0;
1570  sp->rx_crc_errors = 0;
1571  sp->rx_alignment_errors = 0;
1572  sp->rx_rnr_errors = 0;
1573  sp->rx_overrun_errors = 0;
1574  }
1575 #ifdef NOTUSED
1576  if (sc->miibus != NULL)
1577  mii_tick(device_get_softc(sc->miibus));
1578 #endif
1579  splx(s);
1580  /*
1581  * Schedule another timeout one second from now.
1582  */
1583  if (sc->stat_ch == fxp_timeout_running) {
1584  timeout(fxp_tick, sc, hz);
1585  }
1586  else if (sc->stat_ch == fxp_timeout_stop_rq) {
1587  sc->stat_ch = fxp_timeout_stopped;
1588  }
1589 }

1600  int i;
1601
