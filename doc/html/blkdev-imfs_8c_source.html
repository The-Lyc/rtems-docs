42 static ssize_t rtems_blkdev_imfs_read(
43  rtems_libio_t *iop,
44  void *buffer,
45  size_t count
46 )
47 {
48  int rv;
49  rtems_blkdev_imfs_context *ctx = IMFS_generic_get_context_by_iop(iop);
50  rtems_disk_device *dd = &ctx->dd;
51  ssize_t remaining = (ssize_t) count;
52  off_t offset = iop->offset;
53  ssize_t block_size = (ssize_t) rtems_disk_get_block_size(dd);
54  rtems_blkdev_bnum block = (rtems_blkdev_bnum) (offset / block_size);
55  ssize_t block_offset = (ssize_t) (offset % block_size);
56  char *dst = buffer;
57 
58  while (remaining > 0) {
59  rtems_bdbuf_buffer *bd;
60  rtems_status_code sc = rtems_bdbuf_read(dd, block, &bd);
61 
62  if (sc == RTEMS_SUCCESSFUL) {
63  ssize_t copy = block_size - block_offset;
64 
65  if (copy > remaining) {
66  copy = remaining;
67  }
68 
69  memcpy(dst, (char *) bd->buffer + block_offset, (size_t) copy);
70 
71  sc = rtems_bdbuf_release(bd);
72  if (sc == RTEMS_SUCCESSFUL) {
73  block_offset = 0;
74  remaining -= copy;
75  dst += copy;
76  ++block;
77  } else {
78  remaining = -1;
79  }
80  } else {
81  remaining = -1;
82  }
83  }
84 
85  if (remaining >= 0) {
86  iop->offset += count;
87  rv = (ssize_t) count;
88  } else {
89  errno = EIO;
90  rv = -1;
91  }
92 
93  return rv;
94 }
