73 static void Afe_Rx_Cb(uint32_t channel, AfeDma *pArg)
74 {
75  AfeCmd *pAfedCmd = pArg->pCurrentCommand;
76 
77  if (channel != afeDmaRxChannel)
78  return;
79 
80  /* Configure and enable interrupt on RC compare */
81  NVIC_ClearPendingIRQ(XDMAC_IRQn);
82  NVIC_DisableIRQ(XDMAC_IRQn);
83 
84  /* Release the DMA channels */
85  XDMAD_FreeChannel(pArg->pXdmad, afeDmaRxChannel);
86  SCB_InvalidateDCache_by_Addr(pAfedCmd->pRxBuff, pAfedCmd->RxSize);
87  /* Release the dataflash semaphore */
88  pArg->semaphore++;
89 
90  /* Invoke the callback associated with the current command */
91  if (pAfedCmd && pAfedCmd->callback)
92  pAfedCmd->callback(0, pAfedCmd->pArgument);
93 }

102 static uint8_t _AfeConfigureDmaChannels(AfeDma *pAfed)
103 {
104 
105  /* Driver initialize */
106  XDMAD_FreeChannel(pAfed->pXdmad, afeDmaRxChannel);
107 
108  /* Allocate a DMA channel for AFE0/1 RX. */
109  afeDmaRxChannel =
110  XDMAD_AllocateChannel(pAfed->pXdmad, pAfed->afeId, XDMAD_TRANSFER_MEMORY);
111 
112  if (afeDmaRxChannel == XDMAD_ALLOC_FAILED)
113  return AFE_ERROR;
114 
115  /* Setup callbacks for AFE0/1 RX */
116  XDMAD_SetCallback(pAfed->pXdmad, afeDmaRxChannel,
117  (XdmadTransferCallback)Afe_Rx_Cb, pAfed);
118 
119  if (XDMAD_PrepareChannel(pAfed->pXdmad, afeDmaRxChannel))
120  return AFE_ERROR;
121 
122  return AFE_OK;
123 }

132 static uint8_t _Afe_configureLinkList(Afec *pAfeHw, void *pXdmad,
133  AfeCmd *pCommand)
134 {
135  uint32_t xdmaCndc, xdmaInt;
136  sXdmadCfg xdmadRxCfg;
137  uint32_t afeId;
138 
139  if ((unsigned int)pAfeHw == (unsigned int)AFEC0) afeId = ID_AFEC0;
140 
141  if ((unsigned int)pAfeHw == (unsigned int)AFEC1) afeId = ID_AFEC1;
142 
143  /* Setup RX Link List */
144  xdmadRxCfg.mbr_ubc = XDMA_UBC_NVIEW_NDV0 |
145  XDMA_UBC_NDE_FETCH_DIS |
146  XDMA_UBC_NDEN_UPDATED |
147  pCommand->RxSize;
148 
149  xdmadRxCfg.mbr_da = (uint32_t)pCommand->pRxBuff;
150  xdmadRxCfg.mbr_sa = (uint32_t) & (pAfeHw->AFEC_LCDR);
151  xdmadRxCfg.mbr_cfg = XDMAC_CC_TYPE_PER_TRAN |
152  XDMAC_CC_MBSIZE_SINGLE |
153  XDMAC_CC_DSYNC_PER2MEM |
154  XDMAC_CC_CSIZE_CHK_1 |
155  XDMAC_CC_DWIDTH_WORD |
156  XDMAC_CC_SIF_AHB_IF1 |
157  XDMAC_CC_DIF_AHB_IF1 |
158  XDMAC_CC_SAM_FIXED_AM |
159  XDMAC_CC_DAM_INCREMENTED_AM |
160  XDMAC_CC_PERID(
161  XDMAIF_Get_ChannelNumber(afeId, XDMAD_TRANSFER_RX));
162 
163  xdmadRxCfg.mbr_bc = 0;
164  xdmadRxCfg.mbr_sus = 0;
165  xdmadRxCfg.mbr_dus = 0;
166 
167  xdmaInt = (XDMAC_CIE_BIE |
168  XDMAC_CIE_DIE |
169  XDMAC_CIE_FIE |
170  XDMAC_CIE_RBIE |
171  XDMAC_CIE_WBIE |
172  XDMAC_CIE_ROIE);
173  xdmaCndc = 0;
174 
175  if (XDMAD_ConfigureTransfer(pXdmad, afeDmaRxChannel,
176  &xdmadRxCfg, xdmaCndc, 0, xdmaInt))
177  return AFE_ERROR;
178 
179  return AFE_OK;
180 }

223 uint32_t Afe_SendData(AfeDma *pAfed, AfeCmd *pCommand)
224 {
225  Afec *pAfeHw = pAfed->pAfeHw;
226 
227  /* Try to get the dataflash semaphore */
228  if (pAfed->semaphore == 0)
229 
230  return AFE_ERROR_LOCK;
231 
232  pAfed->semaphore--;
233 
234  // Initialize the callback
235  pAfed->pCurrentCommand = pCommand;
236 
237  /* Initialize DMA controller using channel 0 for RX. */
238  if (_AfeConfigureDmaChannels(pAfed))
239  return AFE_ERROR_LOCK;
240 
241  /* Configure and enable interrupt on RC compare */
242  NVIC_ClearPendingIRQ(XDMAC_IRQn);
243  NVIC_SetPriority(XDMAC_IRQn , 1);
244  NVIC_EnableIRQ(XDMAC_IRQn);
245 
246  if (_Afe_configureLinkList(pAfeHw, pAfed->pXdmad, pCommand))
247  return AFE_ERROR_LOCK;
248 
249  AFEC_StartConversion(pAfeHw);
250 
251  /* Start DMA 0(RX) */
252  if (XDMAD_StartTransfer(pAfed->pXdmad, afeDmaRxChannel))
253  return AFE_ERROR_LOCK;
254 
255  return AFE_OK;;
256 }
