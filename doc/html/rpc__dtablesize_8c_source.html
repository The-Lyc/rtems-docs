45 /*
46  * Cache the result of getdtablesize(), so we don't have to do an
47  * expensive system call every time.
48  */
49 /*
50  * XXX In FreeBSD 2.x, you can have the maximum number of open file
51  * descriptors be greater than FD_SETSIZE (which us 256 by default).
52  *
53  * Since old programs tend to use this call to determine the first arg
54  * for select(), having this return > FD_SETSIZE is a Bad Idea(TM)!
55  */
56 int
57 _rpc_dtablesize(void)
58 {
59  static int size;
60 
61  if (size == 0) {
62  size = getdtablesize();
63  if (size > FD_SETSIZE)
64  size = FD_SETSIZE;
65  }
66  return (size);
67 }
