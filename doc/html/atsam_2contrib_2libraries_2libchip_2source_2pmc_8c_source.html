156 void PMC_EnablePeripheral(uint32_t dwId)
157 {
158  assert(dwId < 63);
159 
160  if (dwId < 32) {
161  if ((PMC->PMC_PCSR0 & ((uint32_t)1 << dwId)) == ((uint32_t)1 << dwId)) {
162  TRACE_DEBUG("PMC_EnablePeripheral: clock of peripheral" \
163  " %u is already enabled\n\r", (unsigned int)dwId);
164  } else
165  PMC->PMC_PCER0 = 1 << dwId;
166  } else {
167  dwId -= 32;
168 
169  if ((PMC->PMC_PCSR1 & ((uint32_t)1 << dwId)) == ((uint32_t)1 << dwId)) {
170  TRACE_DEBUG("PMC_EnablePeripheral: clock of peripheral" \
171  " %u is already enabled\n\r", (unsigned int)(dwId + 32));
172  } else
173  PMC->PMC_PCER1 = 1 << dwId;
174  }
175 }

257 void PMC_EnableExtOsc(void)
258 {
259  uint32_t read_MOR;
260 
261  /* Before switching MAIN OSC on external crystal : enable it and don't disable
262  * at the same time RC OSC in case of if MAIN OSC is still using RC OSC
263  */
264 
265  read_MOR = PMC->CKGR_MOR;
266  read_MOR &= ~CKGR_MOR_MOSCRCF_Msk;
267  /* reset MOSCRCF field in MOR register before select RC 12MHz */
268  read_MOR |= (CKGR_MOR_KEY_PASSWD
269  | CKGR_MOR_MOSCRCF_12_MHz
270  | CKGR_MOR_MOSCXTEN
271  | CKGR_MOR_MOSCRCEN
272  | CKGR_MOR_MOSCXTST(DEFAUTL_MAIN_OSC_COUNT));
273  /* enable external crystal - enable RC OSC */
274 
275  PMC->CKGR_MOR = read_MOR;
276 
277  while (!(PMC->PMC_SR & PMC_SR_MOSCRCS));
278 
279  /* wait end of RC oscillator stabilization */
280  while (!(PMC->PMC_SR & PMC_SR_MCKRDY));
281 
282  read_MOR |= CKGR_MOR_MOSCSEL;
283  /* select external crystal */
284 
285  PMC->CKGR_MOR = read_MOR;
286 
287  while (!(PMC->PMC_SR & PMC_SR_MOSCSELS));
288 
289  /* Wait end of Main Oscillator Selection */
290  while (!(PMC->PMC_SR & PMC_SR_MCKRDY));
291 }

310 void PMC_SelectExtOsc(void)
311 {
312  /* switch from internal RC 12 MHz to external OSC 12 MHz */
313  /* wait Main XTAL Oscillator stabilisation*/
314  if ((PMC->CKGR_MOR & CKGR_MOR_MOSCSEL) == CKGR_MOR_MOSCSEL) {
315  PMC_DisableIntRC4_8_12MHz();
316  return;
317  }
318 
319  /* enable external OSC 12 MHz */
320  PMC->CKGR_MOR |= CKGR_MOR_MOSCXTEN | CKGR_MOR_KEY_PASSWD;
321 
322  /* wait Main CLK Ready */
323  while (!(PMC->CKGR_MCFR & CKGR_MCFR_MAINFRDY));
324 
325  /* switch MAIN clock to external OSC 12 MHz*/
326  PMC->CKGR_MOR |= CKGR_MOR_MOSCSEL | CKGR_MOR_KEY_PASSWD;
327 
328  /* wait MAIN clock status change for external OSC 12 MHz selection*/
329  while (!(PMC->PMC_SR & PMC_SR_MOSCSELS));
330 
331  /* in case where MCK is running on MAIN CLK */
332  while (!(PMC->PMC_SR & PMC_SR_MCKRDY));
333 
334  PMC_DisableIntRC4_8_12MHz();
335 }

341 void PMC_SelectExtBypassOsc(void)
342 {
343  volatile uint32_t timeout;
344 
345  if ((PMC->CKGR_MOR & CKGR_MOR_MOSCXTBY) != CKGR_MOR_MOSCXTBY) {
346  PMC->CKGR_MOR = CKGR_MOR_KEY_PASSWD |
347  CKGR_MOR_MOSCRCEN |
348  CKGR_MOR_MOSCXTST(0xFF) |
349  CKGR_MOR_MOSCXTBY;
350  PMC->CKGR_MOR |= CKGR_MOR_KEY_PASSWD | CKGR_MOR_MOSCSEL;
351 
352  /* wait MAIN clock status change for external OSC 12 MHz selection*/
353  while (!(PMC->PMC_SR & PMC_SR_MOSCSELS));
354 
355  // Check if an external clock is provided
356  for (timeout = 0; timeout < 0xffff; timeout++);
357 
358  while (!(PMC->CKGR_MCFR & CKGR_MCFR_MAINFRDY));
359  }

368 void PMC_EnableIntRC4_8_12MHz(uint32_t freqSelect)
369 {
370  /* Enable Fast RC oscillator but DO NOT switch to RC now */
371  PMC->CKGR_MOR |= (CKGR_MOR_KEY_PASSWD | CKGR_MOR_MOSCRCEN);
372 
373  /* Wait the Fast RC to stabilize */
374  while (!(PMC->PMC_SR & PMC_SR_MOSCRCS));
375 
376  /* Change Fast RC oscillator frequency */
377  PMC->CKGR_MOR = (PMC->CKGR_MOR & ~CKGR_MOR_MOSCRCF_Msk) |
378  CKGR_MOR_KEY_PASSWD | freqSelect;
379 
380  /* Wait the Fast RC to stabilize */
381  while (!(PMC->PMC_SR & PMC_SR_MOSCRCS));
382 
383  /* Switch to Fast RC */
384  PMC->CKGR_MOR = (PMC->CKGR_MOR & ~CKGR_MOR_MOSCSEL) |
385  CKGR_MOR_KEY_PASSWD;
386 
387  /* wait MAIN clock status change for Fast RC oscillator */
388  while (!(PMC->PMC_SR & PMC_SR_MOSCSELS));
389 
390  /* in case where MCK is running on MAIN CLK */
391  while (!(PMC->PMC_SR & PMC_SR_MCKRDY));
392 
393 }

418 void PMC_SetPllaClock(uint32_t mul, uint32_t div)
419 {
420  if (mul != 0) {
421  /* Init PLL speed */
422  PMC->CKGR_PLLAR = CKGR_PLLAR_ONE
423  | CKGR_PLLAR_PLLACOUNT(DEFAUTL_PLLA_COUNT)
424  | CKGR_PLLAR_MULA(mul - 1)
425  | CKGR_PLLAR_DIVA(div);
426 
427  /* Wait for PLL stabilization */
428  while (!(PMC->PMC_SR & PMC_SR_LOCKA));
429  } else {
430  PMC->CKGR_PLLAR = CKGR_PLLAR_ONE; /* disable PLL A */
431  }
432 }

444 void PMC_SetMckSelection(uint32_t clockSource, uint32_t prescaler)
445 {
446  switch (clockSource) {
447  case PMC_MCKR_CSS_SLOW_CLK :
448  _PMC_SwitchMck2SlowClock();
449  _PMC_SetMckPrescaler(prescaler);
450  break;
451 
452  case PMC_MCKR_CSS_MAIN_CLK :
453  _PMC_SwitchMck2MainClock();
454  _PMC_SetMckPrescaler(prescaler);
455  break;
456 
457  case PMC_MCKR_CSS_PLLA_CLK :
458  _PMC_SetMckPrescaler(prescaler);
459  _PMC_SwitchMck2PllaClock();
460  break;
461  }
462 }

473 
474  _PMC_SwitchMck2MainClock();
475 
476  PMC->CKGR_PLLAR = PMC->CKGR_PLLAR & ~CKGR_PLLAR_MULA_Msk; /* disable PLL A */
477 
478  _PMC_SwitchMck2SlowClock();
479 
480  read_reg = PMC->CKGR_MOR;
481  read_reg = (read_reg & ~CKGR_MOR_MOSCRCEN) | CKGR_MOR_KEY_PASSWD;
482  /* disable RC OSC */
483 
484  PMC->CKGR_MOR = read_reg;
485 
486  PMC_DisableAllPeripherals(); /* disable all peripheral clocks */

496 void PMC_ConfigureMckWithPlla(uint32_t mul, uint32_t div, uint32_t prescaler)
497 {

499  _PMC_SwitchMck2MainClock();
