196 int fifo_open(
197  pipe_control_t **pipep,
198  rtems_libio_t *iop
199 )
200 {
201  pipe_control_t *pipe;
202  unsigned int prevCounter;
203  int err;
204 
205  err = pipe_new(pipep);
206  if (err)
207  return err;
208  pipe = *pipep;
209 
210  switch (LIBIO_ACCMODE(iop)) {
211  case LIBIO_FLAGS_READ:
212  pipe->readerCounter ++;
213  if (pipe->Readers ++ == 0)
214  PIPE_WAKEUPWRITERS(pipe);
215 
216  if (pipe->Writers == 0) {
217  /* Not an error */
218  if (LIBIO_NODELAY(iop))
219  break;
220 
221  prevCounter = pipe->writerCounter;
222  err = -EINTR;
223  /* Wait until a writer opens the pipe */
224  do {
225  PIPE_READWAIT(pipe);
226  } while (prevCounter == pipe->writerCounter);
227  }
228  break;
229 
230  case LIBIO_FLAGS_WRITE:
231  pipe->writerCounter ++;
232 
233  if (pipe->Writers ++ == 0)
234  PIPE_WAKEUPREADERS(pipe);
235 
236  if (pipe->Readers == 0 && LIBIO_NODELAY(iop)) {
237  PIPE_UNLOCK(pipe);
238  err = -ENXIO;
239  goto out_error;
240  }
241 
242  if (pipe->Readers == 0) {
243  prevCounter = pipe->readerCounter;
244  err = -EINTR;
245  do {
246  PIPE_WRITEWAIT(pipe);
247  } while (prevCounter == pipe->readerCounter);
248  }
249  break;
250 
251  case LIBIO_FLAGS_READ_WRITE:
252  pipe->readerCounter ++;
253  if (pipe->Readers ++ == 0)
254  PIPE_WAKEUPWRITERS(pipe);
255  pipe->writerCounter ++;
256  if (pipe->Writers ++ == 0)
257  PIPE_WAKEUPREADERS(pipe);
258  break;
259  }
260 
261  PIPE_UNLOCK(pipe);
262  return 0;
263 
264 out_error:
265  pipe_release(pipep, iop);
266  return err;
267 }

269 ssize_t pipe_read(
270  pipe_control_t *pipe,
271  void *buffer,
272  size_t count,
273  rtems_libio_t *iop
274 )
275 {
276  int chunk, chunk1, read = 0, ret = 0;
277 
278  PIPE_LOCK(pipe);
279 
280  while (PIPE_EMPTY(pipe)) {
281  /* Not an error */
282  if (pipe->Writers == 0)
283  goto out_locked;
284 
285  if (LIBIO_NODELAY(iop)) {
286  ret = -EAGAIN;
287  goto out_locked;
288  }
289 
290  /* Wait until pipe is no more empty or no writer exists */
291  pipe->waitingReaders ++;
292  PIPE_READWAIT(pipe);
293  pipe->waitingReaders --;
294  }
295 
296  /* Read chunk bytes */
297  chunk = MIN(count - read, pipe->Length);
298  chunk1 = pipe->Size - pipe->Start;
299  if (chunk > chunk1) {
300  memcpy(buffer + read, pipe->Buffer + pipe->Start, chunk1);
301  memcpy(buffer + read + chunk1, pipe->Buffer, chunk - chunk1);
302  }
303  else
304  memcpy(buffer + read, pipe->Buffer + pipe->Start, chunk);
305 
306  pipe->Start += chunk;
307  pipe->Start %= pipe->Size;
308  pipe->Length -= chunk;
309  /* For buffering optimization */
310  if (PIPE_EMPTY(pipe))
311  pipe->Start = 0;
312 
313  if (pipe->waitingWriters > 0)
314  PIPE_WAKEUPWRITERS(pipe);
315  read += chunk;
316 
317 out_locked:
318  PIPE_UNLOCK(pipe);
319 
320  if (read > 0)
321  return read;
322  return ret;
323 }

325 ssize_t pipe_write(
326  pipe_control_t *pipe,
327  const void *buffer,
328  size_t count,
329  rtems_libio_t *iop
330 )
331 {
332  int chunk, chunk1, written = 0, ret = 0;
333 
334  /* Write nothing */
335  if (count == 0)
336  return 0;
337 
338  PIPE_LOCK(pipe);
339 
340  if (pipe->Readers == 0) {
341  ret = -EPIPE;
342  goto out_locked;
343  }
344 
345  /* Write of PIPE_BUF bytes or less shall not be interleaved */
346  chunk = count <= pipe->Size ? count : 1;
347 
348  while (written < count) {
349  while (PIPE_SPACE(pipe) < chunk) {
350  if (LIBIO_NODELAY(iop)) {
351  ret = -EAGAIN;
352  goto out_locked;
353  }
354 
355  /* Wait until there is chunk bytes space or no reader exists */
356  pipe->waitingWriters ++;
357  PIPE_WRITEWAIT(pipe);
358  pipe->waitingWriters --;
359 
360  if (pipe->Readers == 0) {
361  ret = -EPIPE;
362  goto out_locked;
363  }
364  }
365 
366  chunk = MIN(count - written, PIPE_SPACE(pipe));
367  chunk1 = pipe->Size - PIPE_WSTART(pipe);
368  if (chunk > chunk1) {
369  memcpy(pipe->Buffer + PIPE_WSTART(pipe), buffer + written, chunk1);
370  memcpy(pipe->Buffer, buffer + written + chunk1, chunk - chunk1);
371  }
372  else
373  memcpy(pipe->Buffer + PIPE_WSTART(pipe), buffer + written, chunk);
374 
375  pipe->Length += chunk;
376  if (pipe->waitingReaders > 0)
377  PIPE_WAKEUPREADERS(pipe);
378  written += chunk;
379  /* Write of more than PIPE_BUF bytes can be interleaved */
380  chunk = 1;
381  }
382 
383 out_locked:
384  PIPE_UNLOCK(pipe);
385 
386 #ifdef RTEMS_POSIX_API
387  /* Signal SIGPIPE */
388  if (ret == -EPIPE)
389  kill(getpid(), SIGPIPE);
390 #endif
391 
392  if (written > 0)
393  return written;
394  return ret;
395 }
