40 void _Record_Initialize( void )
41 {
42  Record_Control *control;
43  size_t control_size;
44  uint32_t cpu_max;
45  uint32_t cpu_index;
46  unsigned int item_count;
47 
48  cpu_max = rtems_configuration_get_maximum_processors();
49  item_count = _Record_Configuration.item_count;
50  control = _Record_Configuration.controls;
51  control_size = sizeof( *control );
52  control_size += sizeof( control->Items[ 0 ] ) * item_count;
53 
54  for ( cpu_index = 0; cpu_index < cpu_max; ++cpu_index ) {
55  Per_CPU_Control *cpu;
56 
57  cpu = _Per_CPU_Get_by_index( cpu_index );
58  control->mask = item_count - 1U;
59  cpu->record = control;
60  control = (Record_Control *) ( (char *) control + control_size );
61  }
62 }

86 static void _Record_Initialize_watchdogs( void )
87 {
88  Watchdog_Interval interval;
89  uint32_t cpu_max;
90  uint32_t cpu_index;
91  sbintime_t now;
92 
93  interval = rtems_counter_frequency() / _Watchdog_Ticks_per_second;
94  interval = ( UINT32_C( 1 ) << 22 ) / interval;
95 
96  if ( interval == 0 ) {
97  interval = 1;
98  }
99 
100  _Record_Tick_interval = interval;
101 
102  cpu_max = rtems_configuration_get_maximum_processors();
103 
104  for ( cpu_index = 0; cpu_index < cpu_max; ++cpu_index ) {
105  Per_CPU_Control *cpu;
106  Record_Control *control;
107 
108  cpu = _Per_CPU_Get_by_index( cpu_index );
109  control = cpu->record;
110  _Watchdog_Preinitialize( &control->Watchdog, cpu );
111  _Watchdog_Initialize( &control->Watchdog, _Record_Watchdog );
112  _Watchdog_Per_CPU_insert_ticks(
113  &control->Watchdog,
114  cpu,
115  _Record_Tick_interval
116  );
117  }
118 
119  now = _Timecounter_Sbinuptime();
120  rtems_record_produce_2(
121  RTEMS_RECORD_UPTIME_LOW,
122  (uint32_t) ( now >> 0 ),
123  RTEMS_RECORD_UPTIME_HIGH,
124  (uint32_t) ( now >> 32 )
125  );
126 }
