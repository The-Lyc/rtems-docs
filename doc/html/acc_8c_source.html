91 extern void ACC_Configure(Acc *pAcc, uint8_t idAcc, uint8_t ucSelplus,
92  uint8_t ucSelminus, uint16_t wAc_en, uint16_t wEdge, uint16_t wInvert)
93 {
94  /* Enable peripheral clock*/
95  PMC->PMC_PCER1 = 1 << (idAcc - 32);
96 
97  /* Reset the controller */
98  pAcc->ACC_CR |= ACC_CR_SWRST;
99 
100  /* Write to the MR register */
101  ACC_CfgModeReg(pAcc,
102  ((ucSelplus << ACC_MR_SELPLUS_Pos) & ACC_MR_SELPLUS_Msk) |
103  ((ucSelminus << ACC_MR_SELMINUS_Pos) & ACC_MR_SELMINUS_Msk) |
104  ((wAc_en << 8) & ACC_MR_ACEN) |
105  ((wEdge << ACC_MR_EDGETYP_Pos) & ACC_MR_EDGETYP_Msk) |
106  ((wInvert << 12) & ACC_MR_INV));
107  /* set hysteresis and current option*/
108  pAcc->ACC_ACR = (ACC_ACR_ISEL_HISP
109  | ((0x01 << ACC_ACR_HYST_Pos) & ACC_ACR_HYST_Msk));
110 
111  /* Automatic Output Masking Period*/
112  while (pAcc->ACC_ISR & (uint32_t)ACC_ISR_MASK);
113 }

121 extern void ACC_SetComparisonPair(Acc *pAcc, uint8_t ucSelplus,
122  uint8_t ucSelminus)
123 {
124  uint32_t dwTemp;
125 
126  assert(ucSelplus < 8 && ucSelminus < 8);
127 
128  dwTemp = pAcc->ACC_MR;
129  pAcc->ACC_MR =
130  dwTemp & (uint32_t) ((~ACC_MR_SELMINUS_Msk) & (~ACC_MR_SELPLUS_Msk));
131 
132  pAcc->ACC_MR |= (((ucSelplus << ACC_MR_SELPLUS_Pos) & ACC_MR_SELPLUS_Msk) |
133  ((ucSelminus << ACC_MR_SELMINUS_Pos) & ACC_MR_SELMINUS_Msk));
134 }

141 extern uint32_t ACC_GetComparisonResult(Acc *pAcc, uint32_t dwStatus)
142 {
143  uint32_t dwTemp = pAcc->ACC_MR;
144 
145  if ((dwTemp & ACC_MR_INV) == ACC_MR_INV) {
146  if (dwStatus & ACC_ISR_SCO) {
147  return 0; /* inn>inp*/
148  } else {
149  return 1;/* inp>inn*/
150  }
151  } else {
152  if (dwStatus & ACC_ISR_SCO) {
153  return 1; /* inp>inn*/
154  } else {
155  return 0;/* inn>inp*/
156  }
