30  Thread_queue_Queue *queue,
31  Thread_Control *the_thread,
32  Thread_queue_Context *queue_context
33 )
34 {
35  POSIX_Condition_variables_Control *the_cond;
36  int mutex_error;
37 
38  the_cond = POSIX_CONDITION_VARIABLE_OF_THREAD_QUEUE_QUEUE( queue );
39 
40  mutex_error = pthread_mutex_unlock( the_cond->mutex );
41  if ( mutex_error != 0 ) {
42  /*
43  * Historically, we ignored the unlock status since the behavior
44  * is undefined by POSIX. But GNU/Linux returns EPERM in this
45  * case, so we follow their lead.
46  */
47  _Assert( mutex_error == EINVAL || mutex_error == EPERM );
48  _Thread_Continue( the_thread, STATUS_NOT_OWNER );

94 int _POSIX_Condition_variables_Wait_support(
95  pthread_cond_t *cond,
96  pthread_mutex_t *mutex,
97  const struct timespec *abstime
98 )
99 {
100  POSIX_Condition_variables_Control *the_cond;
101  unsigned long flags;
102  Thread_queue_Context queue_context;
103  int error;
104  Thread_Control *executing;
105 
106  the_cond = _POSIX_Condition_variables_Get( cond );
107  POSIX_CONDITION_VARIABLES_VALIDATE_OBJECT( the_cond, flags );
108 
109  _Thread_queue_Context_initialize( &queue_context );
110 
111  if ( abstime != NULL ) {
112  _Thread_queue_Context_set_timeout_argument( &queue_context, abstime );
113 
114  if ( _POSIX_Condition_variables_Get_clock( flags ) == CLOCK_MONOTONIC ) {
115  _Thread_queue_Context_set_enqueue_callout(
116  &queue_context,
117  _POSIX_Condition_variables_Enqueue_with_timeout_monotonic
118  );
119  } else {
120  _Thread_queue_Context_set_enqueue_callout(
121  &queue_context,
122  _POSIX_Condition_variables_Enqueue_with_timeout_realtime
123  );
124  }
125  } else {
126  _Thread_queue_Context_set_enqueue_callout(
127  &queue_context,
128  _POSIX_Condition_variables_Enqueue_no_timeout
129  );
130  }
131 
132  executing = _POSIX_Condition_variables_Acquire( the_cond, &queue_context );
133 
134  if (
135  the_cond->mutex != POSIX_CONDITION_VARIABLES_NO_MUTEX
136  && the_cond->mutex != mutex
137  ) {
138  _POSIX_Condition_variables_Release( the_cond, &queue_context );
139  return EINVAL;
140  }
141 
142  the_cond->mutex = mutex;
143 
144  _Thread_queue_Context_set_thread_state(
145  &queue_context,
146  STATES_WAITING_FOR_CONDITION_VARIABLE
147  );
148  _Thread_queue_Enqueue(
149  &the_cond->Queue.Queue,
150  POSIX_CONDITION_VARIABLES_TQ_OPERATIONS,
151  executing,
152  &queue_context
153  );
154  error = _POSIX_Get_error_after_wait( executing );
155 
156  /*
157  * If the thread is interrupted, while in the thread queue, by
158  * a POSIX signal, then pthread_cond_wait returns spuriously,
159  * according to the POSIX standard. It means that pthread_cond_wait
160  * returns a success status, except for the fact that it was not
161  * woken up a pthread_cond_signal() or a pthread_cond_broadcast().
162  */
163 
164  if ( error == EINTR ) {
165  error = 0;
166  }
167 
168  /*
169  * When we get here the dispatch disable level is 0.
170  */
171 
172  if ( error != EPERM ) {
173  int mutex_error;
174 
175  mutex_error = pthread_mutex_lock( mutex );
176  if ( mutex_error != 0 ) {
177  _Assert( mutex_error == EINVAL );
178  error = EINVAL;
179  }
180  }
181 
182  return error;
183 }
