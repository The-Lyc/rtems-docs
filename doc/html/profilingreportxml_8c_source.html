256 static void report(void *arg, const rtems_profiling_data *data)
257 {
258  context *ctx = arg;
259 
260  switch (data->header.type) {

272 int rtems_profiling_report_xml(
273  const char *name,
274  const rtems_printer *printer,
275  uint32_t indentation_level,
276  const char *indentation
277 )
278 {
279 #ifdef RTEMS_PROFILING
280  context ctx_instance = {
281  .printer = printer,
282  .indentation_level = indentation_level,
283  .indentation = indentation,
284  .retval = 0
285  };
286  context *ctx = &ctx_instance;
287  int rv;
288 
289  indent(ctx, 0);
290  rv = rtems_printf(printer, "<ProfilingReport name=\"%s\">\n", name);
291  update_retval(ctx, rv);
292 
293  rtems_profiling_iterate(report, ctx);
294 
295  indent(ctx, 0);
296  rv = rtems_printf(printer, "</ProfilingReport>\n");
297  update_retval(ctx, rv);
298 
299  return ctx->retval;
300 #else /* RTEMS_PROFILING */
301  (void) name;
302  (void) printer;
303  (void) indentation_level;
304  (void) indentation;
305 
306  return 0;
307 #endif /* RTEMS_PROFILING */
