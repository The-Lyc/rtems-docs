62 void
63 conv_c(rtems_shell_hexdump_globals* globals, PR *pr, u_char *p, size_t bufsize)
64 {
65  char buf[10];
66  char const *str;
67  wchar_t wc;
68  size_t clen, oclen;
69  int converr, pad, width;
70  char peekbuf[MB_LEN_MAX];
71 
72  if (pr->mbleft > 0) {
73  str = "**";
74  pr->mbleft--;
75  goto strpr;
76  }
77 
78  switch(*p) {
79  case '\0':
80  str = "\\0";
81  goto strpr;
82  /* case '\a': */
83  case '\007':
84  str = "\\a";
85  goto strpr;
86  case '\b':
87  str = "\\b";
88  goto strpr;
89  case '\f':
90  str = "\\f";
91  goto strpr;
92  case '\n':
93  str = "\\n";
94  goto strpr;
95  case '\r':
96  str = "\\r";
97  goto strpr;
98  case '\t':
99  str = "\\t";
100  goto strpr;
101  case '\v':
102  str = "\\v";
103  goto strpr;
104  default:
105  break;
106  }
107  /*
108  * Multibyte characters are disabled for hexdump(1) for backwards
109  * compatibility and consistency (none of its other output formats
110  * recognize them correctly).
111  */
112  converr = 0;
113  if (odmode && MB_CUR_MAX > 1) {
114  oclen = 0;
115 retry:
116  clen = mbrtowc(&wc, (char*)p, bufsize, &pr->mbstate);
117  if (clen == 0)
118  clen = 1;
119  else if (clen == (size_t)-1 || (clen == (size_t)-2 &&
120  buf == peekbuf)) {
121  memset(&pr->mbstate, 0, sizeof(pr->mbstate));
122  wc = *p;
123  clen = 1;
124  converr = 1;
125  } else if (clen == (size_t)-2) {
126  /*
127  * Incomplete character; peek ahead and see if we
128  * can complete it.
129  */
130  oclen = bufsize;
131  bufsize = peek(globals, p = (u_char*)peekbuf, MB_CUR_MAX);
132  goto retry;
133  }
134  clen += oclen;
135  } else {
136  wc = *p;
137  clen = 1;
138  }
139  if (!converr && iswprint(wc)) {
140  if (!odmode) {
141  *pr->cchar = 'c';
142  (void)printf(pr->fmt, (int)wc);
143  } else {
144  *pr->cchar = 'C';
145  assert(strcmp(pr->fmt, "%3C") == 0);
146  width = wcwidth(wc);
147  assert(width >= 0);
148  pad = 3 - width;
149  if (pad < 0)
150  pad = 0;
151  #if defined(__rtems__)
152  (void)printf("%*s%lc", pad, "", (wint_t)wc);
153  #else
154  (void)printf("%*s%lc", pad, "", wc);
155  #endif
156  pr->mbleft = clen - 1;
157  }
158  } else {
159  (void)sprintf(buf, "%03o", (int)*p);
160  str = buf;
161 strpr: *pr->cchar = 's';
162  (void)printf(pr->fmt, str);
163  }
164 }

166 void
167 conv_u(rtems_shell_hexdump_globals* globals, PR *pr, u_char *p)
168 {
169  static char const * list[] = {
170  "nul", "soh", "stx", "etx", "eot", "enq", "ack", "bel",
171  "bs", "ht", "lf", "vt", "ff", "cr", "so", "si",
172  "dle", "dcl", "dc2", "dc3", "dc4", "nak", "syn", "etb",
173  "can", "em", "sub", "esc", "fs", "gs", "rs", "us",
174  };
175 
176  /* od used nl, not lf */
177  if (*p <= 0x1f) {
178  *pr->cchar = 's';
179  if (odmode && *p == 0x0a)
180  (void)printf(pr->fmt, "nl");
181  else
182  (void)printf(pr->fmt, list[*p]);
183  } else if (*p == 0x7f) {
184  *pr->cchar = 's';
185  (void)printf(pr->fmt, "del");
186  } else if (odmode && *p == 0x20) { /* od replaced space with sp */
187  *pr->cchar = 's';
188  (void)printf(pr->fmt, " sp");
189  } else if (isprint(*p)) {
190  *pr->cchar = 'c';
191  (void)printf(pr->fmt, *p);
192  } else {
193  *pr->cchar = 'x';
194  (void)printf(pr->fmt, (int)*p);
195  }
196 }
