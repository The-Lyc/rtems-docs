17 /*
18  * The use of pages can vary. The rtems_fdisk_seg_*_page set
19  * routines use an absolute page number relative to the segment
20  * while all other page numbers are relative to the number of
21  * page descriptor pages a segment has. You need to add the
22  * number of page descriptor pages (pages_desc) to the page number
23  * when call the rtems_fdisk_seg_*_page functions.
24  *
25  * You must always show the page number as relative in any trace
26  * or error message as device-segment-page and if you have to
27  * the page number as absolute use device-segment~page. This
28  * can be seen in the page copy routine.
29  *
30  * The code is like this to avoid needing the pass the pages_desc
31  * value around. It is only used in selected places and so the
32  * extra parameter was avoided.
33  */

230 static rtems_status_code
231 rtems_fdisk_crc16_gen_factors (uint16_t pattern)
232 {
233  uint32_t b;
234 
235  rtems_fdisk_crc16_factor = malloc (sizeof (uint16_t) * 256);
236  if (!rtems_fdisk_crc16_factor)
237  return RTEMS_NO_MEMORY;
238 
239  for (b = 0; b < 256; b++)
240  {
241  uint32_t i;
242  uint16_t v = b;
243  for (i = 8; i--;)
244  v = v & 1 ? (v >> 1) ^ pattern : v >> 1;
245  rtems_fdisk_crc16_factor[b] = v & 0xffff;
246  }
247  return RTEMS_SUCCESSFUL;

1136 static void
1137 rtems_fdisk_queue_segment (rtems_flashdisk* fd, rtems_fdisk_segment_ctl* sc)
1138 {
1139 #if RTEMS_FDISK_TRACE
1140  rtems_fdisk_info (fd, " queue-seg:%02d-%03d: p=%d a=%d u=%d b=%d f=%s n=%s",
1141  sc->device, sc->segment,
1142  sc->pages, sc->pages_active, sc->pages_used, sc->pages_bad,
1143  sc->failed ? "FAILED" : "no", sc->next ? "set" : "null");
1144 #endif
1145 
1146  /*
1147  * If the segment has failed then check the failed queue and append
1148  * if not failed.
1149  */
1150  if (sc->failed)
1151  {
1152  if (!rtems_fdisk_segment_queue_present (&fd->failed, sc))
1153  rtems_fdisk_segment_queue_push_tail (&fd->failed, sc);
1154  return;
1155  }
1156 
1157  /*
1158  * Remove the queue from the available or used queue.
1159  */
1160  rtems_fdisk_segment_queue_remove (&fd->available, sc);
1161  rtems_fdisk_segment_queue_remove (&fd->used, sc);
1162 
1163  /*
1164  * Are all the pages in the segment used ?
1165  * If they are and the driver has been configured to background
1166  * erase place the segment on the used queue. If not configured
1167  * to background erase perform the erase now.
1168  *
1169  */
1170  if (rtems_fdisk_seg_pages_available (sc) == 0)
1171  {
1172  if (sc->pages_active)
1173  {
1174  /*
1175  * Keep the used queue sorted by the most number of used
1176  * pages. When we compact we want to move the pages into
1177  * a new segment and cover more than one segment.
1178  */
1179  rtems_fdisk_segment_ctl* seg = fd->used.head;
1180 
1181  while (seg)
1182  {
1183  if (sc->pages_used > seg->pages_used)
1184  break;
1185  seg = seg->next;
1186  }
1187 
1188  if (seg)
1189  rtems_fdisk_segment_queue_insert_before (&fd->used, seg, sc);
1190  else
1191  rtems_fdisk_segment_queue_push_tail (&fd->used, sc);
1192  }
1193  else
1194  {
1195  if ((fd->flags & RTEMS_FDISK_BACKGROUND_ERASE))
1196  rtems_fdisk_segment_queue_push_tail (&fd->erase, sc);
1197  else
1198  rtems_fdisk_erase_segment (fd, sc);
1199  }
1200  }
1201  else
1202  {
1203  /*
1204  * The segment has pages available so place back onto the
1205  * available list. The list is sorted from the least number
1206  * of available pages to the most. This approach means
1207  * the pages of a partially filled segment will be filled
1208  * before moving onto another emptier segment. This keeps
1209  * empty segments longer aiding compaction.
1210  *
1211  * The down side is the wear effect as a single segment
1212  * could be used more than segment. This will not be
1213  * addressed until wear support is added.
1214  *
1215  * @note Wear support can be added by having counts for
1216  * for the number of times a segment is erased. This
1217  * available list is then sorted on the least number
1218  * of available pages then empty segments are sorted
1219  * on the least number of erases the segment has.
1220  *
1221  * The erase count can be stored in specially flaged
1222  * pages and contain a counter (32bits?) and 32 bits
1223  * for each segment. When a segment is erased a
1224  * bit is cleared for that segment. When 32 erasers
1225  * has occurred the page is re-written to the flash
1226  * with all the counters updated with the number of
1227  * bits cleared and all bits set back to 1.
1228  */
1229  rtems_fdisk_segment_ctl* seg = fd->available.head;
1230 
1231  while (seg)
1232  {
1233  if (rtems_fdisk_seg_pages_available (sc) <
1234  rtems_fdisk_seg_pages_available (seg))
1235  break;
1236  seg = seg->next;
1237  }
1238 
1239  if (seg)
1240  rtems_fdisk_segment_queue_insert_before (&fd->available, seg, sc);
1241  else
1242  rtems_fdisk_segment_queue_push_tail (&fd->available, sc);
1243  }
1244 }

1246 static int
1247 rtems_fdisk_recycle_segment (rtems_flashdisk* fd,
1248  rtems_fdisk_segment_ctl* ssc,
1249  rtems_fdisk_segment_ctl* dsc,
1250  uint32_t *pages)
1251 {
1252  int ret;
1253  uint32_t spage;
1254  uint32_t used = 0;
1255  uint32_t active = 0;
1256 
1257  for (spage = 0; spage < ssc->pages; spage++)
1258  {
1259  rtems_fdisk_page_desc* spd = &ssc->page_descriptors[spage];
1260 
1261  if (rtems_fdisk_page_desc_flags_set (spd, RTEMS_FDISK_PAGE_ACTIVE) &&
1262  !rtems_fdisk_page_desc_flags_set (spd, RTEMS_FDISK_PAGE_USED))
1263  {
1264  uint32_t dst_pages;
1265  rtems_fdisk_page_desc* dpd;
1266  uint32_t dpage;
1267 
1268  dpage = rtems_fdisk_seg_next_available_page (dsc);
1269  dpd = &dsc->page_descriptors[dpage];
1270 
1271  active++;
1272 
1273  if (dpage >= dsc->pages)
1274  {
1275  rtems_fdisk_error ("recycle: %02d-%03d: " \
1276  "no page desc available: %d",
1277  dsc->device, dsc->segment,
1278  rtems_fdisk_seg_pages_available (dsc));
1279  dsc->failed = true;
1280  rtems_fdisk_queue_segment (fd, dsc);
1281  rtems_fdisk_segment_queue_push_head (&fd->used, ssc);
1282  return EIO;
1283  }
1284 
1285 #if RTEMS_FDISK_TRACE
1286  rtems_fdisk_info (fd, "recycle: %02d-%03d-%03d=>%02d-%03d-%03d",
1287  ssc->device, ssc->segment, spage,
1288  dsc->device, dsc->segment, dpage);
1289 #endif
1290  ret = rtems_fdisk_seg_copy_page (fd, ssc,
1291  spage + ssc->pages_desc,
1292  dsc,
1293  dpage + dsc->pages_desc);
1294  if (ret)
1295  {
1296  rtems_fdisk_error ("recycle: %02d-%03d-%03d=>" \
1297  "%02d-%03d-%03d: " \
1298  "copy page failed: %s (%d)",
1299  ssc->device, ssc->segment, spage,
1300  dsc->device, dsc->segment, dpage,
1301  strerror (ret), ret);
1302  rtems_fdisk_queue_segment (fd, dsc);
1303  rtems_fdisk_segment_queue_push_head (&fd->used, ssc);
1304  return ret;
1305  }
1306 
1307  *dpd = *spd;
1308 
1309  ret = rtems_fdisk_seg_write_page_desc (fd,
1310  dsc,
1311  dpage, dpd);
1312 
1313  if (ret)
1314  {
1315  rtems_fdisk_error ("recycle: %02d-%03d-%03d=>" \
1316  "%02d-%03d-%03d: copy pd failed: %s (%d)",
1317  ssc->device, ssc->segment, spage,
1318  dsc->device, dsc->segment, dpage,
1319  strerror (ret), ret);
1320  rtems_fdisk_queue_segment (fd, dsc);
1321  rtems_fdisk_segment_queue_push_head (&fd->used, ssc);
1322  return ret;
1323  }
1324 
1325  dsc->pages_active++;
1326 
1327  /*
1328  * No need to set the used bit on the source page as the
1329  * segment will be erased. Power down could be a problem.
1330  * We do the stats to make sure everything is as it should
1331  * be.
1332  */
1333 
1334  ssc->pages_active--;
1335  ssc->pages_used++;
1336 
1337  fd->blocks[spd->block].segment = dsc;
1338  fd->blocks[spd->block].page = dpage;
1339 
1340  /*
1341  * Place the segment on to the correct queue.
1342  */
1343  rtems_fdisk_queue_segment (fd, dsc);
1344 
1345  /*
1346  * Get new destination segment if necessary.
1347  */
1348  dst_pages = rtems_fdisk_seg_pages_available (dsc);
1349  if (dst_pages == 0)
1350  {
1351  dsc = rtems_fdisk_seg_most_available (&fd->available);
1352  if (!dsc)
1353  {
1354  if (ssc->pages_active == 0)
1355  {
1356  ret = rtems_fdisk_erase_segment (fd, ssc);
1357  }
1358  else
1359  {
1360  rtems_fdisk_error ("recycle: no available dst segment");
1361  ret = EIO;
1362  }
1363 
1364  return ret;
1365  }
1366  }
1367 
1368  (*pages)--;
1369  }
1370  else if (rtems_fdisk_page_desc_erased (spd))
1371  {
1372  --fd->erased_blocks;
1373  }
1374  else
1375  {
1376  used++;
1377  }
1378  }
1379 
1380 #if RTEMS_FDISK_TRACE
1381  rtems_fdisk_printf (fd, "ssc end: %d-%d: p=%ld, a=%ld, u=%ld",
1382  ssc->device, ssc->segment,
1383  pages, active, used);
1384 #endif
1385  if (ssc->pages_active != 0)
1386  {
1387  rtems_fdisk_error ("compacting: ssc pages not 0: %d",
1388  ssc->pages_active);
1389  }
1390 
1391  ret = rtems_fdisk_erase_segment (fd, ssc);
1392 
1393  return ret;
1394 }

1403 rtems_fdisk_compact (rtems_flashdisk* fd)
1404 {
1405  int ret;
1406  rtems_fdisk_segment_ctl* dsc;
1407  rtems_fdisk_segment_ctl* ssc;
1408  uint32_t compacted_segs = 0;
1409  uint32_t pages;
1410 
1411  if (rtems_fdisk_is_erased_blocks_starvation (fd))
1412  {
1413 #if RTEMS_FDISK_TRACE
1414  rtems_fdisk_printf (fd, " resolve starvation");
1415 #endif
1416 
1417  ssc = rtems_fdisk_segment_queue_pop_head (&fd->used);
1418  if (!ssc)
1419  ssc = rtems_fdisk_segment_queue_pop_head (&fd->available);
1420 
1421  if (ssc)
1422  {
1423  dsc = rtems_fdisk_seg_most_available (&fd->available);
1424  if (dsc)
1425  {
1426  ret = rtems_fdisk_recycle_segment (fd, ssc, dsc, &pages);
1427  if (ret)
1428  return ret;
1429  }
1430  else
1431  {
1432  rtems_fdisk_error ("compacting: starvation");
1433  return EIO;
1434  }
1435  }
1436  else
1437  {
1438  rtems_fdisk_error ("compacting: nothing to recycle");
1439  return EIO;
1440  }
1441  }
1442 
1443  while (fd->used.head)
1444  {
1445  uint32_t dst_pages;
1446  uint32_t segments;
1447 
1448 #if RTEMS_FDISK_TRACE
1449  rtems_fdisk_printf (fd, " compacting");
1450 #endif
1451 
1452  dsc = rtems_fdisk_seg_most_available (&fd->available);
1453 
1454  if (dsc == 0)
1455  {
1456  rtems_fdisk_error ("compacting: no available segments to compact too");
1457  return EIO;
1458  }
1459 
1460  ssc = fd->used.head;
1461  dst_pages = rtems_fdisk_seg_pages_available (dsc);
1462  segments = 0;
1463  pages = 0;
1464 
1465 #if RTEMS_FDISK_TRACE
1466  rtems_fdisk_printf (fd, " dsc:%02d-%03d: most available",
1467  dsc->device, dsc->segment);
1468 #endif
1469 
1470  /*
1471  * Count the number of segments that have active pages that fit into
1472  * the destination segment. Also limit the number of segments that
1473  * we handle during one compaction. A lower number means less aggressive
1474  * compaction or less delay when compacting but it may mean the disk
1475  * will fill.
1476  */
1477 
1478  while (ssc &&
1479  ((pages + ssc->pages_active) < dst_pages) &&
1480  ((compacted_segs + segments) < fd->compact_segs))
1481  {
1482  pages += ssc->pages_active;
1483  segments++;
1484  ssc = ssc->next;
1485  }
1486

1488  * We need a source segment and have pages to copy and
1489  * compacting one segment to another is silly. Compaction needs
1490  * to free at least one more segment.
1491  */
1492 
1493  if (!ssc || (pages == 0) || ((compacted_segs + segments) == 1))
1494  {
1495 #if RTEMS_FDISK_TRACE
1496  rtems_fdisk_printf (fd, " nothing to compact");
1497 #endif
1498  break;
1499  }
1500 
1501 #if RTEMS_FDISK_TRACE
1502  rtems_fdisk_printf (fd, " ssc scan: %d-%d: p=%ld, seg=%ld",
1503  ssc->device, ssc->segment,
1504  pages, segments);
1505 #endif
1506 
1507  rtems_fdisk_segment_queue_remove (&fd->available, dsc);
1508 
1509  /*
1510  * We now copy the pages to the new segment.
1511  */
1512 
1513  while (pages)
1514  {
1515  ssc = rtems_fdisk_segment_queue_pop_head (&fd->used);
1516 
1517  if (ssc)
1518  {
1519  ret = rtems_fdisk_recycle_segment (fd, ssc, dsc, &pages);
1520  if (ret)
1521  return ret;
1522  }
1523  }
1524 
1525  compacted_segs += segments;
1526  }
1527 
1528  return 0;
1529 }

1534 static int
1535 rtems_fdisk_recover_block_mappings (rtems_flashdisk* fd)
1536 {
1537  uint32_t device;
1538

1540  * Clear the queues.
1541  */
1542  rtems_fdisk_segment_queue_init (&fd->available);
1543  rtems_fdisk_segment_queue_init (&fd->used);
1544  rtems_fdisk_segment_queue_init (&fd->erase);
1545  rtems_fdisk_segment_queue_init (&fd->failed);
1546 
1547  /*
1548  * Clear the lock mappings.
1549  */
1550  memset (fd->blocks, 0, fd->block_count * sizeof (rtems_fdisk_block_ctl));
1551 
1552  /*
1553  * Scan each segment or each device recovering the valid pages.
1554  */
1555  fd->erased_blocks = 0;
1556  fd->starvation_threshold = 0;
1557  for (device = 0; device < fd->device_count; device++)
1558  {
1559  uint32_t segment;
1560  for (segment = 0; segment < fd->devices[device].segment_count; segment++)
1561  {
1562  rtems_fdisk_segment_ctl* sc = &fd->devices[device].segments[segment];
1563  const rtems_fdisk_segment_desc* sd = sc->descriptor;
1564  rtems_fdisk_page_desc* pd;
1565  uint32_t page;
1566  int ret;
1567 
1568 #if RTEMS_FDISK_TRACE
1569  rtems_fdisk_info (fd, "recover-block-mappings:%02d-%03d", device, segment);
1570 #endif
1571 
1572  sc->pages_desc = rtems_fdisk_page_desc_pages (sd, fd->block_size);
1573  sc->pages =
1574  rtems_fdisk_pages_in_segment (sd, fd->block_size) - sc->pages_desc;
1575  if (sc->pages > fd->starvation_threshold)
1576  fd->starvation_threshold = sc->pages;
1577 
1578  sc->pages_active = 0;
1579  sc->pages_used = 0;
1580  sc->pages_bad = 0;
1581 
1582  sc->failed = false;
1583

1585  sc->page_descriptors = malloc (sc->pages_desc * fd->block_size);
1586 
1587  if (!sc->page_descriptors)
1588  rtems_fdisk_abort ("no memory for page descriptors");
1589

1591 
1592  /*
1593  * The page descriptors are always at the start of the segment. Read
1594  * the descriptors off the device into the segment control page
1595  * descriptors.
1596  *
1597  * @todo It may be better to ask the driver to get these value
1598  * so NAND flash could be better supported.
1599  */
1600  ret = rtems_fdisk_seg_read (fd, sc, 0, (void*) pd,
1601  sc->pages_desc * fd->block_size);
1602

1604  {
1605  rtems_fdisk_error ("recover-block-mappings:%02d-%03d: " \
1606  "read page desc failed: %s (%d)",
1607  device, segment, strerror (ret), ret);
1608  return ret;
1609  }

1612  * Check each page in the segement for valid pages.
1613  * Update the stats for the segment so we know how many pages
1614  * are active and how many are used.
1615  *
1616  * If the page is active see if the block is with-in range and
1617  * if the block is a duplicate.
1618  */
1619  for (page = 0; page < sc->pages; page++, pd++)
1620  {
1621  if (rtems_fdisk_page_desc_erased (pd))
1622  {
1623  /*
1624  * Is the page erased ?
1625  */

1627  page + sc->pages_desc);
1628 
1629  if (ret == 0)
1630  {
1631  ++fd->erased_blocks;
1632  }
1633  else
1634  {
1635 #if RTEMS_FDISK_TRACE
1636  rtems_fdisk_warning (fd, "page not blank: %d-%d-%d",
1637  device, segment, page, pd->block);
1638 #endif
1639  rtems_fdisk_page_desc_set_flags (pd, RTEMS_FDISK_PAGE_USED);
1640 
1641  ret = rtems_fdisk_seg_write_page_desc (fd, sc,
1642  page, pd);
1643 
1644  if (ret)
1645  {
1646  rtems_fdisk_error ("forcing page to used failed: %d-%d-%d",
1647  device, segment, page);
1648  }
1649 
1650  sc->pages_used++;
1651  }
1652  }
1653  else
1654  {
1655  if (rtems_fdisk_page_desc_flags_set (pd, RTEMS_FDISK_PAGE_USED))
1656  {
1657  sc->pages_used++;
1658  }
