57 {

81 /*
82  * Set up the clock hardware
83  *
84  * f_pit = f_clk / 2^(preScaleCode+1) / N = 1/(us_per_tick/us_per_s)
85  *
86  * N = f_clk / 2^(preScaleCode+1) * us_per_tick / us_per_s
87  *
88  * ns_per_pit_clk = ns_per_s / (f_clk / 2^(preScaleCode+1))
89  * = ns_per_s * 2^(preScaleCode+1) / f_clk;
90  */
91 #define Clock_driver_support_initialize_hardware() \
92  do { \
93  unsigned long long N; \
94  int level; \
95  int preScaleCode = 0; \
96  N = bsp_get_CPU_clock_speed(); \
97  N *= rtems_configuration_get_microseconds_per_tick(); \
98  N /= 2*1000000; /* min_prescale * us_per_s */ \
99  while ( N > 0x10000 ) { \
100  preScaleCode++; \
101  N >>= 1; \
102  } \
103  PITC_PER_TICK = N; \
104  N = 2000000000ULL << preScaleCode; \
105  N /= bsp_get_CPU_clock_speed(); \
106  NSEC_PER_PITC = N; \
107  IDLE_COUNTER = 0; \
108  FILTERED_IDLE = 0; \
109  MAX_IDLE_COUNT = 0; \
110  bsp_allocate_interrupt(PIT3_IRQ_LEVEL, PIT3_IRQ_PRIORITY); \
111  MCF5282_INTC0_ICR58 = MCF5282_INTC_ICR_IL(PIT3_IRQ_LEVEL) | \
112  MCF5282_INTC_ICR_IP(PIT3_IRQ_PRIORITY); \
113  rtems_interrupt_disable( level ); \
114  MCF5282_INTC0_IMRH &= ~MCF5282_INTC_IMRH_INT58; \
115  MCF5282_PIT3_PCSR &= ~MCF5282_PIT_PCSR_EN; \
116  rtems_interrupt_enable( level ); \
117  MCF5282_PIT3_PCSR = MCF5282_PIT_PCSR_PRE(preScaleCode) | \
118  MCF5282_PIT_PCSR_OVW | \
119  MCF5282_PIT_PCSR_PIE | \
120  MCF5282_PIT_PCSR_RLD; \
121  MCF5282_PIT3_PMR = PITC_PER_TICK - 1; \
122  MCF5282_PIT3_PCSR = MCF5282_PIT_PCSR_PRE(preScaleCode) | \
123  MCF5282_PIT_PCSR_PIE | \
124  MCF5282_PIT_PCSR_RLD | \
125  MCF5282_PIT_PCSR_EN; \
126  rtems_timecounter_simple_install( \
127  &uC5282_tc, \
128  bsp_get_CPU_clock_speed() >> (preScaleCode + 1), \
129  PITC_PER_TICK, \
130  uC5282_tc_get_timecount \
131  ); \
132  } while (0)
