62 void GMAC_PHYMaintain(Gmac *pGmac,
63  uint8_t bPhyAddr,
64  uint8_t bRegAddr,
65  uint8_t bRW,
66  uint16_t wData)
67 {
68  /* Wait until bus idle */
69  while ((pGmac->GMAC_NSR & GMAC_NSR_IDLE) == 0);
70 
71  /* Write maintain register */
72  pGmac->GMAC_MAN = (~GMAC_MAN_WZO & GMAC_MAN_CLTTO)
73  | (GMAC_MAN_OP(bRW ? 0x2 : 0x1))
74  | GMAC_MAN_WTN(0x02)
75  | GMAC_MAN_PHYA(bPhyAddr)
76  | GMAC_MAN_REGA(bRegAddr)
77  | GMAC_MAN_DATA(wData);
78 }

99 uint8_t GMAC_SetMdcClock(Gmac *pGmac, uint32_t mck)
100 {
101  uint32_t clock_dividor;
102 #ifndef __rtems__
103  pGmac->GMAC_NCR &= ~(GMAC_NCR_RXEN | GMAC_NCR_TXEN);
104 #else /* __rtems__ */
105  assert((pGmac->GMAC_NCR & (GMAC_NCR_RXEN | GMAC_NCR_TXEN)) == 0);
106 #endif /* __rtems__ */
107 
108  if (mck <= 20000000) {
109  clock_dividor = GMAC_NCFGR_CLK_MCK_8; // MDC clock = MCK/8
110  } else if (mck <= 40000000) {
111  clock_dividor = GMAC_NCFGR_CLK_MCK_16; // MDC clock = MCK/16
112  } else if (mck <= 80000000) {
113  clock_dividor = GMAC_NCFGR_CLK_MCK_32; // MDC clock = MCK/32
114  } else if (mck <= 160000000) {
115  clock_dividor = GMAC_NCFGR_CLK_MCK_64; // MDC clock = MCK/64
116  } else if (mck <= 240000000) {
117  clock_dividor = GMAC_NCFGR_CLK_MCK_96; // MDC clock = MCK/96
118  } else {
119  TRACE_ERROR("E: No valid MDC clock.\n\r");
120  return 0;
121  }
122 
123  pGmac->GMAC_NCFGR = (pGmac->GMAC_NCFGR & (~GMAC_NCFGR_CLK_Msk)) | clock_dividor;
124 #ifndef __rtems__
125  pGmac->GMAC_NCR |= (GMAC_NCR_RXEN | GMAC_NCR_TXEN);
126 #endif /* __rtems__ */
127  return 1;
128 }

206 void GMAC_EnableRGMII(Gmac *pGmac, uint32_t duplex, uint32_t speed)
207 {
208 #ifndef __rtems__
209  pGmac->GMAC_NCR &= ~(GMAC_NCR_RXEN | GMAC_NCR_TXEN);
210 #else /* __rtems__ */
211  assert((pGmac->GMAC_NCR & (GMAC_NCR_RXEN | GMAC_NCR_TXEN)) == 0);
212 #endif /* __rtems__ */
213 
214  if (duplex == GMAC_DUPLEX_HALF)
215  pGmac->GMAC_NCFGR &= ~GMAC_NCFGR_FD;
216  else
217  pGmac->GMAC_NCFGR |= GMAC_NCFGR_FD;
218 
219 
220  if (speed == GMAC_SPEED_10M)
221  pGmac->GMAC_NCFGR &= ~GMAC_NCFGR_SPD;
222  else if (speed == GMAC_SPEED_100M)
223  pGmac->GMAC_NCFGR |= GMAC_NCFGR_SPD;
224  else
225  pGmac->GMAC_NCFGR |= GMAC_NCFGR_SPD;
226 
227  /* RGMII enable */
228  pGmac->GMAC_UR = 0;
229  pGmac->GMAC_NCFGR &= ~GMAC_NCFGR_GBE;
230 #ifndef __rtems__
231  pGmac->GMAC_NCR |= (GMAC_NCR_RXEN | GMAC_NCR_TXEN);
232 #endif /* __rtems__ */
233  return;
234 }
