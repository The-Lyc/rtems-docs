116 ofw_call(const char *service, const int nargs, const int nret, ofw_arg_t *rets,
117  ...)
118 {
119  va_list list;
120  ofw_args_t args;
121  int i;
122 
123  args.service = (ofw_arg_t) service;
124  args.nargs = nargs;
125  args.nret = nret;
126 
127  va_start(list, rets);
128  for (i = 0; i < nargs; i++)
129  args.args[i] = va_arg(list, ofw_arg_t);
130  va_end(list);
131 
132  for (i = 0; i < nret; i++)
133  args.args[i + nargs] = 0;
134 
135  (void) ofw(&args);
136 
137  for (i = 1; i < nret; i++)
138  rets[i - 1] = args.args[i + nargs];
139 
140  return args.args[nargs];
141 }

226 {
227  ofw_arg_t result[4];
228  int shift;

230  if (ofw_call("call-method", 4, 5, result, "translate", ofw_mmu,
231  virt, 0) != 0) {

233  halt();

240  shift = 32;

260 static void *ofw_claim_phys_internal(const void *phys, const unsigned int len, const unsigned int alignment)
261 {
262  /*
263  * Note that the return value check will help
264  * us to discover conflicts between OpenFirmware
265  * allocations and our use of physical memory.
266  * It is better to detect collisions here
267  * than to cope with weird errors later.
268  *
269  * So this is really not to make the loader
270  * more generic; it is here for debugging
271  * purposes.
272  */
273 
274  if (sizeof(unative_t) == 8) {
275  ofw_arg_t retaddr[2];
276  int shift = 32;
277 
278  if (ofw_call("call-method", 6, 3, retaddr, "claim",
279  ofw_memory_prop, alignment, len, ((uintptr_t) phys) >> shift,
280  ((uintptr_t) phys) & ((uint32_t) -1)) != 0) {
281  printk("Error: memory method claim() failed, halting.\n");
282  halt();
283  }
284 
285  return (void *) ((retaddr[0] << shift) | retaddr[1]);
286  } else {
287  ofw_arg_t retaddr[1];
288 
289  if (ofw_call("call-method", 5, 2, retaddr, "claim",
290  ofw_memory_prop, alignment, len, (uintptr_t) phys) != 0) {
291  printk("Error: memory method claim() failed, halting.\n");
292  halt();
293  }
294 
295  return (void *) retaddr[0];
296  }
297 }

310 {
311  uintptr_t phys_hi, phys_lo;
312 
313  if (sizeof(unative_t) == 8) {
314  int shift = 32;
315  phys_hi = (uintptr_t) phys >> shift;
316  phys_lo = (uintptr_t) phys & 0xffffffff;

320  }
321 
322  return ofw_call("call-method", 7, 1, NULL, "map", ofw_mmu, mode, size,
323  virt, phys_hi, phys_lo);

332 int ofw_memmap(memmap_t *map)
333 {
334  unsigned int ac = ofw_get_address_cells(ofw_memory) /
335  (sizeof(uintptr_t) / sizeof(uint32_t));
336  unsigned int sc = ofw_get_size_cells(ofw_memory) /
337  (sizeof(uintptr_t) / sizeof(uint32_t));
338 
339  uintptr_t buf[((ac + sc) * MEMMAP_MAX_RECORDS)];
340  int ret = ofw_get_property(ofw_memory, "reg", buf, sizeof(buf));
341  if (ret <= 0) /* ret is the number of written bytes */
342  return false;
343 
344  int pos;
345  map->total = 0;
346  map->count = 0;
347  for (pos = 0; (pos < ret / sizeof(uintptr_t)) &&
348  (map->count < MEMMAP_MAX_RECORDS); pos += ac + sc) {
349  void *start = (void *) (buf[pos + ac - 1]);
350  unsigned int size = buf[pos + ac + sc - 1];
351 
352  /*
353  * This is a hot fix of the issue which occurs on machines
354  * where there are holes in the physical memory (such as
355  * SunBlade 1500). Should we detect a hole in the physical
356  * memory, we will ignore any memory detected behind
357  * the hole and pretend the hole does not exist.
358  */
359  if ((map->count > 0) && (map->zones[map->count - 1].start +
360  map->zones[map->count - 1].size < start))
361  break;
362 
363  if (size > 0) {
364  map->zones[map->count].start = start;
365  map->zones[map->count].size = size;
366  map->count++;
367  map->total += size;
368  }
369  }
370 
371  return true;
372 }

374 static void ofw_setup_screen(phandle handle)
375 {
376  /* Check for device type */
377  char device_type[OFW_TREE_PROPERTY_MAX_VALUELEN];
378  if (ofw_get_property(handle, "device_type", device_type, OFW_TREE_PROPERTY_MAX_VALUELEN) <= 0)
379  return;
380 
381  device_type[OFW_TREE_PROPERTY_MAX_VALUELEN - 1] = '\0';
382  if (strcmp(device_type, "display") != 0)
383  return;
384 
385  /* Check for 8 bit depth */
386  uint32_t depth;
387  if (ofw_get_property(handle, "depth", &depth, sizeof(uint32_t)) <= 0)
388  depth = 0;
389 
390  /* Get device path */
391  static char path[OFW_TREE_PATH_MAX_LEN + 1];
392  size_t len = ofw_package_to_path(handle, path, OFW_TREE_PATH_MAX_LEN);
393  if (len == -1)
394  return;
395 
396  path[len] = '\0';
397 
398  /* Open the display to initialize it */
399  ihandle screen = ofw_open(path);
400  if (screen == -1)
401  return;
402 
403  if (depth == 8) {
404  /* Setup the palette so that the (inverted) 3:2:3 scheme is usable */
405  unsigned int i;
406  for (i = 0; i < 256; i++) {
407  ofw_call("call-method", 6, 1, NULL, "color!", screen,
408  255 - i, CLIP(BLUE(i) * 37), GREEN(i) * 85, CLIP(RED(i) * 37));
409  }
410  }
411 }

413 static void ofw_setup_screens_internal(phandle current)
414 {
415  while ((current != 0) && (current != -1)) {
416  ofw_setup_screen(current);
417 
418  /*
419  * Recursively process the potential child node.
420  */
421  phandle child = ofw_get_child_node(current);
422  if ((child != 0) && (child != -1))
423  ofw_setup_screens_internal(child);
424 
425  /*
426  * Iteratively process the next peer node.
427  * Note that recursion is a bad idea here.
428  * Due to the topology of the OpenFirmware device tree,
429  * the nesting of peer nodes could be to wide and the
430  * risk of overflowing the stack is too real.
431  */
432  phandle peer = ofw_get_peer_node(current);
433  if ((peer != 0) && (peer != -1)) {
434  current = peer;
435  /*
436  * Process the peer in next iteration.
437  */
438  continue;
439  }
440 
441  /*
442  * No more peers on this level.
443  */
444  break;
445  }
446 }
