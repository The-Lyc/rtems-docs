117 uint32_t MPU_CalMPURegionSize(uint32_t dwActualSizeInBytes)
118 {
119  uint32_t dwRegionSize = 32;
120  uint32_t dwReturnValue = 4;
121 
122  while (dwReturnValue < 31) {
123  if (dwActualSizeInBytes <= dwRegionSize)
124  break;
125  else
126  dwReturnValue++;
127 
128  dwRegionSize <<= 1;
129  }
130 
131  return (dwReturnValue << 1);
132 }
