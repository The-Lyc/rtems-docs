269 int b1553rt_device_init(rt_priv *pDev)
270 {
271  struct amba_dev_info *ambadev;
272  struct ambapp_core *pnpinfo;
273  union drvmgr_key_value *value;
274  unsigned int mem;
275  unsigned int sys_freq_hz;
276 
277  /* Get device information from AMBA PnP information */
278  ambadev = (struct amba_dev_info *)pDev->dev->businfo;
279  if ( ambadev == NULL ) {
280  return -1;
281  }
282  pnpinfo = &ambadev->info;
283  pDev->irqno = pnpinfo->irq;
284  pDev->regs = (struct rt_reg *)pnpinfo->apb_slv->start;
285  pDev->minor = pDev->dev->minor_drv;
286 
287 #ifdef DEBUG
288  pDev->log_i = 0;
289  memset(pDev->log,0,sizeof(pDev->log));
290  printf("LOG: 0x%x\n", &pDev->log[0]);
291  printf("LOG_I: 0x%x\n", &pDev->log_i);
292 #endif
293 
294  /* Get memory configuration from bus resources */
295  value = drvmgr_dev_key_get(pDev->dev, "dmaBaseAdr", DRVMGR_KT_POINTER);
296  if (value)
297  mem = (unsigned int)value->ptr;
298 
299  if (value && (mem & 1)) {
300  /* Remote address, address as RT looks at it. */
301 
302  /* Translate the base address into an address that the the CPU can understand */
303  pDev->memarea_base = mem & ~1;
304  drvmgr_translate_check(pDev->dev, DMAMEM_TO_CPU,
305  (void *)pDev->memarea_base_remote,
306  (void **)&pDev->memarea_base,
307  4 * 1024);
308  } else {
309  if (!value) {
310  /* Use dynamically allocated memory,
311  * 4k DMA memory + 4k for alignment
312  */
313  mem = (unsigned int)grlib_malloc(4 * 1024 * 2);
314  if ( !mem ){
315  printk("RT: Failed to allocate HW memory\n\r");
316  return -1;
317  }
318  /* align memory to 4k boundary */
319  pDev->memarea_base = (mem + 0xfff) & ~0xfff;
320  } else {
321  pDev->memarea_base = mem;
322  }
323 
324  /* Translate the base address into an address that the RT core can understand */
325  drvmgr_translate_check(pDev->dev, CPUMEM_TO_DMA,
326  (void *)pDev->memarea_base,
327  (void **)&pDev->memarea_base_remote,
328  4 * 1024);
329  }
330 
331  /* clear the used memory */
332  memset((char *)pDev->memarea_base, 0, 4 * 1024);
333 
334  /* Set base address of all descriptors */
335  pDev->memarea_base = (unsigned int)mem;
336  pDev->mem = (volatile unsigned short *)pDev->memarea_base;
337 
338  pDev->rt_event = NULL;
339 
340  /* The RT is always clocked at the same frequency as the bus
341  * If the frequency doesnt match it is defaulted to 24MHz,
342  * user can always override it.
343  */
344  pDev->cfg_freq = RT_FREQ_24MHZ;
345 
346  /* Get frequency in Hz */
347  if ( drvmgr_freq_get(pDev->dev, DEV_APB_SLV, &sys_freq_hz) == 0 ) {
348  if ( sys_freq_hz == 20000000 ) {
349  pDev->cfg_freq = RT_FREQ_20MHZ;
350  } else if ( sys_freq_hz == 16000000 ) {
351  pDev->cfg_freq = RT_FREQ_16MHZ;
352  } else if ( sys_freq_hz == 12000000 ) {
353  pDev->cfg_freq = RT_FREQ_12MHZ;
354  }
355  }
356 
357  value = drvmgr_dev_key_get(pDev->dev, "coreFreq", DRVMGR_KT_INT);
358  if ( value ) {
359  pDev->cfg_freq = value->i & RT_FREQ_MASK;
360  }
361 
362  /* RX Semaphore created with count = 0 */
363  if ( rtems_semaphore_create(rtems_build_name('R', 'T', '0', '0' + pDev->minor),
364  0,
365  RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|RTEMS_NO_PRIORITY_CEILING,
366  0,
367  &pDev->rx_sem) != RTEMS_SUCCESSFUL ) {
368  printk("RT: Failed to create rx semaphore\n");
369  return RTEMS_INTERNAL_ERROR;
370  }
371 
372  /* Device Semaphore created with count = 1 */
373  if ( rtems_semaphore_create(rtems_build_name('R', 'T', '0', '0' + pDev->minor),
374  1,
375  RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|RTEMS_NO_PRIORITY_CEILING,
376  0,
377  &pDev->dev_sem) != RTEMS_SUCCESSFUL ){
378  printk("RT: Failed to create device semaphore\n");
379  return RTEMS_INTERNAL_ERROR;
380  }
381 
382  /* Default to RT-mode */
383  rt_init(pDev);
384 
385  return 0;
386 }

445 static rtems_device_driver rt_init(rt_priv *rt)
446 {
447  rt->rx_blocking = 1;
448 
449  if ( rt->rt_event )
450  free(rt->rt_event);
451  rt->rt_event = NULL;
452 
453  rt->rt_event = grlib_malloc(EVENT_QUEUE_SIZE*sizeof(*rt->rt_event));
454 
455  if (rt->rt_event == NULL) {
456  DBG("RT driver failed to allocated memory.");
457  return RTEMS_NO_MEMORY;
458  }
459 
460  rt->ctrl_copy = rt->regs->ctrl & 0x3F00; /* Keep rtaddr and rtaddrp */
461  rt->ctrl_copy |= 0x3C0D0; /* broadcast disabled, extmdata=1, writetsw = writecmd = 1 */
462  rt->regs->ctrl = rt->ctrl_copy;
463 
464  /* Set Clock speed */
465  set_clkspd(rt, rt->cfg_freq);
466 
467  rt->regs->addr = rt->memarea_base_remote;
468  rt->regs->ipm = 0x70000; /* Enable RT RX, MEM Failure and AHB Error interrupts */
469 
470  DBG("B1553RT DMA_AREA: 0x%x\n", (unsigned int)rt->mem);
471 
472  return RTEMS_SUCCESSFUL;
473 }

684 static void b1553rt_interrupt(void *arg)
685 {
686  rt_priv *rt = arg;
687  unsigned short descriptor;
688  int signal_event=0, wake_rx_task=0;
689  unsigned int event_status=0;
690  unsigned int wc, irqv, cmd, tsw, suba, tx, miw, i;
691  unsigned int ipend;
692 
693  #define SET_ERROR_DESCRIPTOR(descriptor) (event_status = (event_status & 0x0000ffff) | descriptor<<16)
694  ipend = rt->regs->ipm;
695 
696  if (ipend == 0) {
697  /* IRQ mask has been cleared, we must have been reset */
698  /* Restore ctrl registers */
699  rt->regs->ctrl = rt->ctrl_copy;
700  rt->regs->addr = rt->memarea_base_remote;
701  rt->regs->ipm = 0x70000;
702  /* Send reset mode code event */
703  if (rt->head - rt->tail != EVENT_QUEUE_SIZE) {
704  miw = (8<<11);
705  descriptor = 64 + 32 + 8;
706  rt->rt_event[INDEX(rt->head)].miw = miw;
707  rt->rt_event[INDEX(rt->head)].time = 0;
708  rt->rt_event[INDEX(rt->head)].desc = descriptor;
709  rt->head++;
710  }
711  }
712 
713  if ( ipend & 0x1 ) {
714  /* RT IRQ */
715  if (rt->head - rt->tail != EVENT_QUEUE_SIZE) {
716 
717  irqv = rt->regs->irq;
718  cmd = irqv >> 7;
719  wc = cmd & 0x1F; /* word count / mode code */
720  suba = irqv & 0x1F; /* sub address (0-31) */
721  tx = (irqv >> 5) & 1;
722 
723  /* read status word */
724  tsw = READ_DMA(&rt->mem[tx*0x3E0+suba]);
725 
726  /* Build Message Information Word (B1553BRM-style) */
727  miw = (wc<<11) | (tsw&RT_TSW_BUS)>>4 | !(tsw&RT_TSW_OK)>>7 | (tsw&RT_TSW_ILL)>>5 |
728  (tsw&RT_TSW_PAR)>>5 | (tsw&RT_TSW_MAN)>>7;
729 
730  descriptor = (tx << 5) | suba;
731 
732  /* Mode codes */
733  if (suba == 0 || suba == 31) {
734  descriptor = 64 + (tx*32) + wc;
735  }
736 
737  /* Data received or transmitted */
738  if (descriptor < 64) {
739  wc = wc ? wc : 32; /* wc = 0 means 32 words transmitted */
740  }
741  /* RX Mode code */
742  else if (descriptor < 96) {
743  wc = (wc>>4);
744  }
745  /* TX Mode code */
746  else if (descriptor < 128) {
747  wc = (wc>>4);
748  }
749 
750  /* Copy to event queue */
751  rt->rt_event[INDEX(rt->head)].miw = miw;
752  rt->rt_event[INDEX(rt->head)].time = 0;
753 
754  for (i = 0; i < wc; i++) {
755  rt->rt_event[INDEX(rt->head)].data[i] = READ_DMA(&rt->mem[tx*0x400 + suba*32 + i]);
756  }
757  rt->rt_event[INDEX(rt->head)].desc = descriptor;
758  rt->head++;
759 
760 
761  /* Handle errors */
762  if ( tsw & RT_TSW_ILL){
763  FUNCDBG("RT: RT_ILLCMD\n\r");
764  rt->status |= RT_ILLCMD_IRQ;
765  event_status |= RT_ILLCMD_IRQ;
766  SET_ERROR_DESCRIPTOR(descriptor);
767  signal_event=1;
768  }
769 
770  if ( !(tsw & RT_TSW_OK) ) {
771  FUNCDBG("RT: RT_MERR_IRQ\n\r");
772  rt->status |= RT_MERR_IRQ;
773  event_status |= RT_MERR_IRQ;
774  SET_ERROR_DESCRIPTOR(descriptor);
775  signal_event=1;
776  }
777 
778  }
779  else {
780  /* Indicate overrun */
781  rt->rt_event[INDEX(rt->head)].desc |= 0x8000;
782  }
783  }
784 
785  if ( ipend & 0x2 ) {
786  /* Memory failure IRQ */
787  FUNCDBG("B1553RT: Memory failure\n");
788  event_status |= RT_DMAF_IRQ;
789  signal_event=1;
790  }
791 
792  if ( ipend & 0x4 ) {
793  /* AHB Error */
794  FUNCDBG("B1553RT: AHB ERROR\n");
795  event_status |= RT_DMAF_IRQ;
796  signal_event=1;
797  }
798 
799 #ifdef DEBUG
800  rt->log[rt->log_i++ % EVENT_QUEUE_SIZE] = descriptor;
801  rt->log[rt->log_i++ % EVENT_QUEUE_SIZE] = cmd;
802  rt->log[rt->log_i++ % EVENT_QUEUE_SIZE] = miw;
803  rt->log[rt->log_i++ % EVENT_QUEUE_SIZE] = tsw;
804 #endif
805 
806  wake_rx_task = 1;
807 
808  /* Wake any blocked rx thread only on receive interrupts */
809  if ( wake_rx_task ) {
810  rtems_semaphore_release(rt->rx_sem);
811  }
812 
813  /* Copy current mask to status mask */
814  if ( event_status ) {
815  if ( event_status & 0xffff0000 )
816  rt->status &= 0x0000ffff;
817  rt->status |= event_status;
818  }
819 
820  /* signal event once */
821  if ( signal_event && (rt->event_id != 0) ) {
822  rtems_event_send(rt->event_id, event_status);
823  }
824 
825 }
