247 BDIdx TaskBDAssign(TaskId taskId, void *buffer0, void *buffer1, int size, uint32 bdFlags)
248 {
249  BDIdx *bdHead;
250  TaskBD_t *bd;
251  BDIdx r = TASK_ERR_NO_ERR;
252 
253  if (TaskBDIdxTable[taskId].currBDInUse == TaskBDIdxTable[taskId].numBD) {
254  /*
255  * The buffer ring is full.
256  */
257  r = TASK_ERR_BD_RING_FULL;
258 
259  } else if ( (TaskBDIdxTable[taskId].apiConfig & API_CONFIG_BD_FLAG)
260  && ((uint32)size & (uint32)(~SDMA_DRD_MASK_LENGTH))) {
261  r = TASK_ERR_SIZE_TOO_LARGE;
262 
263  } else if ( !(TaskBDIdxTable[taskId].apiConfig & API_CONFIG_BD_FLAG)
264  && ((uint32)size & (uint32)(0xffffffff<<SDMA_BD_BIT_READY))) {
265  r = TASK_ERR_SIZE_TOO_LARGE;
266 
267  } else {
268  bdHead = &BDHead[taskId];
269 
270  /*
271  * Increase Buffer Descriptor in-use variable.
272  */
273  ++TaskBDIdxTable[taskId].currBDInUse;
274 
275  /*
276  * Get a generic TaskBD_t pointer to the BD to be assigned.
277  * Assign the buffer pointers.
278  */
279  bd = TaskBDIdxTable[taskId].BDTablePtr;
280  if (TaskBDIdxTable[taskId].numPtr == 1) {
281  bd = (TaskBD_t *)&(((TaskBD1_t *)bd)[*bdHead]);
282 
283  ((TaskBD1_t *)bd)->DataPtr[0] = (uint32)buffer0;
284  } else {
285  bd = (TaskBD_t *)&(((TaskBD2_t *)bd)[*bdHead]);
286 
287  ((TaskBD2_t *)bd)->DataPtr[0] = (uint32)buffer0;
288  ((TaskBD2_t *)bd)->DataPtr[1] = (uint32)buffer1;
289  }
290 
291 
292  if (bd->Status & SDMA_BD_MASK_READY) {
293  /*
294  * This BD is in use.
295  */
296  r = TASK_ERR_BD_BUSY;
297 
298  } else {
299 
300  /*
301  * Set status bits and length. As soon as Status is written, the
302  * BestComm may perform the transfer.
303  */
304  if (TaskBDIdxTable[taskId].apiConfig & API_CONFIG_BD_FLAG) {
305  bd->Status = ( ((uint32)SDMA_DRD_MASK_FLAGS & bdFlags)
306  | ((uint32)SDMA_DRD_MASK_LENGTH & (uint32)size)
307  | ((uint32)SDMA_BD_MASK_READY));
308  } else {
309  bd->Status = ( ((uint32)SDMA_BD_MASK_SIGN & (uint32)size)
310  | ((uint32)SDMA_BD_MASK_READY));
311  }
312 
313  /*
314  * Return the current BD index and increment.
315  */
316  r = *bdHead;
317  *bdHead = (BDIdx)((*bdHead + 1) % (BDIdx)TaskBDIdxTable[taskId].numBD);
318  }
319  }
320 
321  /*
322  * Reenable a fall-through BD tasks that might have exited.
323  */
324  if (TaskRunning[taskId]) {
325  SDMA_TASK_ENABLE(SDMA_TCR, taskId);
326  }
327  return r;
328 }

346 BDIdx TaskBDRelease(TaskId taskId)
347 {
348  BDIdx *bdTail;
349  TaskBD_t *bd;
350 
351  bdTail = &BDTail[taskId];
352 
353  if (TaskBDIdxTable[taskId].currBDInUse == 0) {
354  /*
355  * Buffer Descriptor ring is empty, Can't Release!
356  */
357  return TASK_ERR_BD_RING_EMPTY;
358  }
359 
360  /*
361  * Get a generic TaskBD_t pointer to the next BD to be released.
362  */
363  bd = TaskGetBD(taskId, *bdTail);
364 
365  /*
366  * Verify the ready bit is clear.
367  */
368  if (bd->Status & SDMA_BD_MASK_READY) {
369  return TASK_ERR_BD_BUSY;
370  }
371 
372  /*
373  * Increment the tail pointer around the ring, decrement in-use.
374  */
375  *bdTail = (BDIdx)((*bdTail + 1) % (BDIdx)TaskBDIdxTable[taskId].numBD);
376  --TaskBDIdxTable[taskId].currBDInUse;
377 
378  return *bdTail;
379 }

404 BDIdx TaskBDReset(TaskId taskId)
405 {
406  BDIdx i;
407  TaskBD_t *bd, *bdTab;
408 
409  if (TaskRunning[taskId]) {
410  return TASK_ERR_TASK_RUNNING;
411  }
412 
413  bdTab = TaskBDIdxTable[taskId].BDTablePtr;
414 
415  for (i = (BDIdx)TaskBDIdxTable[taskId].numBD - 1; i >= 0; --i) {
416 
417  if (TaskBDIdxTable[taskId].numPtr == 1) {
418  bd = (TaskBD_t *)&(((TaskBD1_t *)bdTab)[i]);
419  } else {
420  bd = (TaskBD_t *)&(((TaskBD2_t *)bdTab)[i]);
421  }
422 
423  bd->Status = 0x0;
424  }
425 
426  TaskBDIdxTable[taskId].currBDInUse = 0;
427  *TaskBDIdxTable[taskId].BDStartPtr =
428  (volatile uint32)((volatile uint32)TaskBDIdxTable[taskId].BDTablePtr
429  + MBarPhysOffsetGlobal);
430  return BDHead[taskId] = BDTail[taskId] = 0;
431 }
