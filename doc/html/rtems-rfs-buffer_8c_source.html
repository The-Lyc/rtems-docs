78 int
79 rtems_rfs_buffer_handle_request (rtems_rfs_file_system* fs,
80  rtems_rfs_buffer_handle* handle,
81  rtems_rfs_buffer_block block,
82  bool read)
83 {
84  int rc;
85 
86  /*
87  * If the handle has a buffer release it. This allows a handle to be reused
88  * without needing to close then open it again.
89  */
90  if (rtems_rfs_buffer_handle_has_block (handle))
91  {
92  /*
93  * Treat block 0 as special to handle the loading of the super block.
94  */
95  if (block && (rtems_rfs_buffer_bnum (handle) == block))
96  return 0;
97 
98  if (rtems_rfs_trace (RTEMS_RFS_TRACE_BUFFER_HANDLE_REQUEST))
99  printf ("rtems-rfs: buffer-request: handle has buffer: %" PRIu32 "\n",
100  rtems_rfs_buffer_bnum (handle));
101 
102  rc = rtems_rfs_buffer_handle_release (fs, handle);
103  if (rc > 0)
104  return rc;
105  handle->dirty = false;
106  handle->bnum = 0;
107  }
108 
109  if (rtems_rfs_trace (RTEMS_RFS_TRACE_BUFFER_HANDLE_REQUEST))
110  printf ("rtems-rfs: buffer-request: block=%" PRIu32 "\n", block);
111 
112  /*
113  * First check to see if the buffer has already been requested and is
114  * currently attached to a handle. If it is share the access. A buffer could
115  * be shared where different parts of the block have separate functions. An
116  * example is an inode block and the file system needs to handle 2 inodes in
117  * the same block at the same time.
118  */
119  if (fs->buffers_count)
120  {
121  /*
122  * Check the active buffer list for shared buffers.
123  */
124  handle->buffer = rtems_rfs_scan_chain (&fs->buffers,
125  &fs->buffers_count,
126  block);
127  if (rtems_rfs_buffer_handle_has_block (handle) &&
128  rtems_rfs_trace (RTEMS_RFS_TRACE_BUFFER_HANDLE_REQUEST))
129  printf ("rtems-rfs: buffer-request: buffer shared: refs: %d\n",
130  rtems_rfs_buffer_refs (handle) + 1);
131  }
132 
133  /*
134  * If the buffer has not been found check the local cache of released
135  * buffers. There are release and released modified lists to preserve the
136  * state.
137  */
138  if (!rtems_rfs_fs_no_local_cache (fs) &&
139  !rtems_rfs_buffer_handle_has_block (handle))
140  {
141  /*
142  * Check the local cache of released buffers.
143  */
144  if (fs->release_count)
145  handle->buffer = rtems_rfs_scan_chain (&fs->release,
146  &fs->release_count,
147  block);
148 
149  if (!rtems_rfs_buffer_handle_has_block (handle) &&
150  fs->release_modified_count)
151  {
152  handle->buffer = rtems_rfs_scan_chain (&fs->release_modified,
153  &fs->release_modified_count,
154  block);
155  /*
156  * If we found a buffer retain the dirty buffer state.
157  */
158  if (rtems_rfs_buffer_handle_has_block (handle))
159  rtems_rfs_buffer_mark_dirty (handle);
160  }
161  }
162 
163  /*
164  * If not located we request the buffer from the I/O layer.
165  */
166  if (!rtems_rfs_buffer_handle_has_block (handle))
167  {
168  rc = rtems_rfs_buffer_io_request (fs, block, read, &handle->buffer);
169 
170  if (rc > 0)
171  {
172  if (rtems_rfs_trace (RTEMS_RFS_TRACE_BUFFER_HANDLE_REQUEST))
173  printf ("rtems-rfs: buffer-request: block=%" PRIu32 ": bdbuf-%s: %d: %s\n",
174  block, read ? "read" : "get", rc, strerror (rc));
175  return rc;
176  }
177 
178  rtems_chain_set_off_chain (rtems_rfs_buffer_link(handle));
179  }
180 
181  /*
182  * Increase the reference count of the buffer.
183  */
184  rtems_rfs_buffer_refs_up (handle);
185  rtems_chain_append_unprotected (&fs->buffers,
186  rtems_rfs_buffer_link (handle));
187  fs->buffers_count++;
188 
189  handle->buffer->user = (void*) ((intptr_t) block);
190  handle->bnum = block;
191 
192  if (rtems_rfs_trace (RTEMS_RFS_TRACE_BUFFER_HANDLE_REQUEST))
193  printf ("rtems-rfs: buffer-request: block=%" PRIu32 " bdbuf-%s=%" PRIu32 " refs=%d\n",
194  block, read ? "read" : "get", handle->buffer->block,
195  handle->buffer->references);
196 
197  return 0;
198 }

200 int
201 rtems_rfs_buffer_handle_release (rtems_rfs_file_system* fs,
202  rtems_rfs_buffer_handle* handle)
203 {
204  int rc = 0;
205 
206  if (rtems_rfs_buffer_handle_has_block (handle))
207  {
208  if (rtems_rfs_trace (RTEMS_RFS_TRACE_BUFFER_HANDLE_RELEASE))
209  printf ("rtems-rfs: buffer-release: block=%" PRIu32 " %s refs=%d %s\n",
210  rtems_rfs_buffer_bnum (handle),
211  rtems_rfs_buffer_dirty (handle) ? "(dirty)" : "",
212  rtems_rfs_buffer_refs (handle),
213  rtems_rfs_buffer_refs (handle) == 0 ? "BAD REF COUNT" : "");
214 
215  if (rtems_rfs_buffer_refs (handle) > 0)
216  rtems_rfs_buffer_refs_down (handle);
217 
218  if (rtems_rfs_buffer_refs (handle) == 0)
219  {
220  rtems_chain_extract_unprotected (rtems_rfs_buffer_link (handle));
221  fs->buffers_count--;
222 
223  if (rtems_rfs_fs_no_local_cache (fs))
224  {
225  handle->buffer->user = (void*) 0;
226  rc = rtems_rfs_buffer_io_release (handle->buffer,
227  rtems_rfs_buffer_dirty (handle));
228  }
229  else
230  {
231  /*
232  * If the total number of held buffers is higher than the configured
233  * value remove a buffer from the queue with the most buffers and
234  * release. The buffers are held on the queues with the newest at the
235  * head.
236  *
237  * This code stops a large series of transactions causing all the
238  * buffers in the cache being held in queues of this file system.
239  */
240  if ((fs->release_count +
241  fs->release_modified_count) >= fs->max_held_buffers)
242  {
243  rtems_rfs_buffer* buffer;
244  bool modified;
245 
246  if (rtems_rfs_trace (RTEMS_RFS_TRACE_BUFFER_HANDLE_RELEASE))
247  printf ("rtems-rfs: buffer-release: local cache overflow:"
248  " %" PRIu32 "\n", fs->release_count + fs->release_modified_count);
249 
250  if (fs->release_count > fs->release_modified_count)
251  {
252  buffer = (rtems_rfs_buffer*)
253  rtems_chain_get_unprotected (&fs->release);
254  fs->release_count--;
255  modified = false;
256  }
257  else
258  {
259  buffer = (rtems_rfs_buffer*)
260  rtems_chain_get_unprotected (&fs->release_modified);
261  fs->release_modified_count--;
262  modified = true;
263  }
264  buffer->user = (void*) 0;
265  rc = rtems_rfs_buffer_io_release (buffer, modified);
266  }
267 
268  if (rtems_rfs_buffer_dirty (handle))
269  {
270  rtems_chain_append_unprotected (&fs->release_modified,
271  rtems_rfs_buffer_link (handle));
272  fs->release_modified_count++;
273  }
274  else
275  {
276  rtems_chain_append_unprotected (&fs->release,
277  rtems_rfs_buffer_link (handle));
278  fs->release_count++;
279  }
280  }
281  }
282  handle->buffer = NULL;
283  }
284 
285  return rc;
286 }
