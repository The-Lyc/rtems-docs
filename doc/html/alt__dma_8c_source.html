213 ALT_STATUS_CODE alt_dma_init(const ALT_DMA_CFG_t * dma_cfg)
214 {
215  // Initialize the channel information array
216  for (int i = 0; i < 8; ++i)
217  {
218  channel_info_array[i].flag = 0;
219  }
220 
221  // Update the System Manager DMA configuration items
222 
223  uint32_t dmactrl = 0;
224 
225  // Handle FPGA / CAN muxing
226  for (int i = 0; i < 4; ++i)
227  {
228  // The default is FPGA.
229  switch (dma_cfg->periph_mux[i])
230  {
231  case ALT_DMA_PERIPH_MUX_DEFAULT:
232  case ALT_DMA_PERIPH_MUX_FPGA:
233  break;
234  case ALT_DMA_PERIPH_MUX_CAN:
235  dmactrl |= (ALT_SYSMGR_DMA_CTL_CHANSEL_0_SET_MSK << i);
236  break;
237  default:
238  return ALT_E_ERROR;
239  }
240  }
241 
242  // Handle Manager security
243  // Default is Secure state.
244  switch (dma_cfg->manager_sec)
245  {
246  case ALT_DMA_SECURITY_DEFAULT:
247  case ALT_DMA_SECURITY_SECURE:
248  break;
249  case ALT_DMA_SECURITY_NONSECURE:
250  dmactrl |= ALT_SYSMGR_DMA_CTL_MGRNONSECURE_SET_MSK;
251  break;
252  default:
253  return ALT_E_ERROR;
254  }
255 
256  // Handle IRQ security
257  for (int i = 0; i < ALT_SYSMGR_DMA_CTL_IRQNONSECURE_WIDTH; ++i)
258  {
259  // Default is Secure state.
260  switch (dma_cfg->irq_sec[i])
261  {
262  case ALT_DMA_SECURITY_DEFAULT:
263  case ALT_DMA_SECURITY_SECURE:
264  break;
265  case ALT_DMA_SECURITY_NONSECURE:
266  dmactrl |= (1 << (i + ALT_SYSMGR_DMA_CTL_IRQNONSECURE_LSB));
267  break;
268  default:
269  return ALT_E_ERROR;
270  }
271  }
272 
273  alt_write_word(ALT_SYSMGR_DMA_CTL_ADDR, dmactrl);
274 
275  // Update the System Manager DMA peripheral security items
276 
277  uint32_t dmapersecurity = 0;
278 
279  for (int i = 0; i < 32; ++i)
280  {
281  // Default is Secure state.
282  switch (dma_cfg->periph_sec[i])
283  {
284  case ALT_DMA_SECURITY_DEFAULT:
285  case ALT_DMA_SECURITY_SECURE:
286  break;
287  case ALT_DMA_SECURITY_NONSECURE:
288  dmapersecurity |= (1 << i);
289  break;
290  default:
291  return ALT_E_ERROR;
292  }
293  }
294 
295  alt_write_word(ALT_SYSMGR_DMA_PERSECURITY_ADDR, dmapersecurity);
296 
297  // Take DMA out of reset.
298 
299  alt_clrbits_word(ALT_RSTMGR_PERMODRST_ADDR, ALT_RSTMGR_PERMODRST_DMA_SET_MSK);
300 
301  return ALT_E_SUCCESS;
302 }

501 ALT_STATUS_CODE alt_dma_channel_kill(ALT_DMA_CHANNEL_t channel)
502 {
503  // Validate channel
504  switch (channel)
505  {
506  case ALT_DMA_CHANNEL_0:
507  case ALT_DMA_CHANNEL_1:
508  case ALT_DMA_CHANNEL_2:
509  case ALT_DMA_CHANNEL_3:
510  case ALT_DMA_CHANNEL_4:
511  case ALT_DMA_CHANNEL_5:
512  case ALT_DMA_CHANNEL_6:
513  case ALT_DMA_CHANNEL_7:
514  break;
515  default:
516  return ALT_E_BAD_ARG;
517  }
518 
519  // Verify channel is allocated
520 
521  if (!(channel_info_array[channel].flag & ALT_DMA_CHANNEL_INFO_FLAG_ALLOCED))
522  {
523  return ALT_E_ERROR;
524  }
525 
526  // NOTE: Don't worry about the current channel state. Just issue DMAKILL
527  // instruction. The channel state cannot move from from Stopped back to
528  // Killing.
529 
530  // Configure DBGINST0 to execute DMAKILL on the requested channel thread.
531  // DMAKILL is short enough not to use DBGINST1 register.
532 
533  // For information on APB Interface, see PL330, section 2.5.1.
534  // For information on DBGINSTx, see PL330, section 3.3.20 - 3.3.21.
535  // For information on DMAKILL, see PL330, section 4.3.6.
536 
537  alt_write_word(ALT_DMA_DBGINST0_ADDR(ALT_DMASECURE_ADDR),
538  ALT_DMA_DBGINST0_INSTRUCTIONBYTE0_SET(0x1) |
539  ALT_DMA_DBGINST0_CHANNELNUMBER_SET(channel) |
540  ALT_DMA_DBGINST0_DEBUGTHREAD_SET(ALT_DMA_DBGINST0_DEBUGTHREAD_E_CHANNEL));
541 
542  // Execute the instruction held in DBGINST0
543 
544  // For information on DBGCMD, see PL330, section 3.3.19.
545 
546  alt_write_word(ALT_DMA_DBGCMD_ADDR(ALT_DMASECURE_ADDR), 0);
547 
548  // Wait for channel to move to KILLING or STOPPED state. Do not wait for
549  // the STOPPED only. If the AXI transaction hangs permanently, it can be
550  // waiting indefinately.
551 
552  ALT_STATUS_CODE status = ALT_E_SUCCESS;
553  ALT_DMA_CHANNEL_STATE_t current;
554  uint32_t i = 20000;
555 
556  while (--i)
557  {
558  status = alt_dma_channel_state_get(channel, &current);
559  if (status != ALT_E_SUCCESS)
560  {
561  break;
562  }
563  if ( (current == ALT_DMA_CHANNEL_STATE_KILLING)
564  || (current == ALT_DMA_CHANNEL_STATE_STOPPED))
565  {
566  break;
567  }
568  }
569 
570  if (i == 0)
571  {
572  status = ALT_E_TMO;
573  }
574 
575  return status;
576 }

866 ALT_STATUS_CODE alt_dma_memory_to_memory(ALT_DMA_CHANNEL_t channel,
867  ALT_DMA_PROGRAM_t * program,
868  void * dst,
869  const void * src,
870  size_t size,
871  bool send_evt,
872  ALT_DMA_EVENT_t evt)
873 {
874  ALT_STATUS_CODE status = ALT_E_SUCCESS;
875 
876  // If the size is zero, and no event is requested, just return success.
877  if ((size == 0) && (send_evt == false))
878  {
879  return status;
880  }
881 
882  if (status == ALT_E_SUCCESS)
883  {
884  status = alt_dma_program_init(program);
885  }
886 
887  if (size != 0)
888  {
889  uintptr_t udst = (uintptr_t)dst;
890  uintptr_t usrc = (uintptr_t)src;
891 
892  dprintf("DMA[M->M]: dst = %p.\n", dst);
893  dprintf("DMA[M->M]: src = %p.\n", src);
894  dprintf("DMA[M->M]: size = 0x%x.\n", size);
895 
896  // Detect if memory regions overshoots the address space.
897 
898  if (udst + size - 1 < udst)
899  {
900  return ALT_E_BAD_ARG;
901  }
902  if (usrc + size - 1 < usrc)
903  {
904  return ALT_E_BAD_ARG;
905  }
906 
907  // Detect if memory regions overlaps.
908 
909  if (udst > usrc)
910  {
911  if (usrc + size - 1 > udst)
912  {
913  return ALT_E_BAD_ARG;
914  }
915  }
916  else
917  {
918  if (udst + size - 1 > usrc)
919  {
920  return ALT_E_BAD_ARG;
921  }
922  }
923 
924  if (status == ALT_E_SUCCESS)
925  {
926  status = alt_dma_program_DMAMOV(program, ALT_DMA_PROGRAM_REG_SAR, usrc);
927  }
928  if (status == ALT_E_SUCCESS)
929  {
930  status = alt_dma_program_DMAMOV(program, ALT_DMA_PROGRAM_REG_DAR, udst);
931  }
932 
933  size_t sizeleft = size;
934 
935  //
936  // The algorithm uses the strategy described in PL330 B.3.1.
937  // It is extended for 2-byte and 1-byte unaligned cases.
938  //
939 
940  // First see how many byte(s) we need to transfer to get src to be 8 byte aligned
941  if (usrc & 0x7)
942  {
943  uint32_t aligncount = MIN(8 - (usrc & 0x7), sizeleft);
944  sizeleft -= aligncount;
945 
946  dprintf("DMA[M->M]: Total pre-alignment 1-byte burst size tranfer(s): %lu.\n", aligncount);
947 
948  // Program in the following parameters:
949  // - SS8 (Source burst size of 1-byte)
950  // - DS8 (Destination burst size of 1-byte)
951  // - SBx (Source burst length of [aligncount] transfers)
952  // - DBx (Destination burst length of [aligncount] transfers)
953  // - All other options default.
954 
955  if (status == ALT_E_SUCCESS)
956  {
957  status = alt_dma_program_DMAMOV(program, ALT_DMA_PROGRAM_REG_CCR,
958  ( ((aligncount - 1) << 4) // SB
959  | ALT_DMA_CCR_OPT_SS8
960  | ALT_DMA_CCR_OPT_SA_DEFAULT
961  | ALT_DMA_CCR_OPT_SP_DEFAULT
962  | ALT_DMA_CCR_OPT_SC_DEFAULT
963  | ((aligncount - 1) << 18) // DB
964  | ALT_DMA_CCR_OPT_DS8
965  | ALT_DMA_CCR_OPT_DA_DEFAULT
966  | ALT_DMA_CCR_OPT_DP_DEFAULT
967  | ALT_DMA_CCR_OPT_DC_DEFAULT
968  | ALT_DMA_CCR_OPT_ES_DEFAULT
969  )
970  );
971  }
972  if (status == ALT_E_SUCCESS)
973  {
974  status = alt_dma_program_DMALD(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
975  }
976  if (status == ALT_E_SUCCESS)
977  {
978  status = alt_dma_program_DMAST(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
979  }
980  }
981 
982  // This is the number of 8-byte bursts
983  uint32_t burstcount = sizeleft >> 3;
984 
985  bool correction = (burstcount != 0);
986 
987  // Update the size left to transfer
988  sizeleft &= 0x7;
989 
990  dprintf("DMA[M->M]: Total Main 8-byte burst size transfer(s): %lu.\n", burstcount);
991  dprintf("DMA[M->M]: Total Main 1-byte burst size transfer(s): %u.\n", sizeleft);
992 
993  // Determine how many 16 length bursts can be done
994 
995  if (burstcount >> 4)
996  {
997  uint32_t length16burstcount = burstcount >> 4;
998  burstcount &= 0xf;
999 
1000  dprintf("DMA[M->M]: Number of 16 burst length 8-byte transfer(s): %lu.\n", length16burstcount);
1001  dprintf("DMA[M->M]: Number of remaining 8-byte transfer(s): %lu.\n", burstcount);
1002 
1003  // Program in the following parameters:
1004  // - SS64 (Source burst size of 8-byte)
1005  // - DS64 (Destination burst size of 8-byte)
1006  // - SB16 (Source burst length of 16 transfers)
1007  // - DB16 (Destination burst length of 16 transfers)
1008  // - All other options default.
1009 
1010  if (status == ALT_E_SUCCESS)
1011  {
1012  status = alt_dma_program_DMAMOV(program, ALT_DMA_PROGRAM_REG_CCR,
1013  ( ALT_DMA_CCR_OPT_SB16
1014  | ALT_DMA_CCR_OPT_SS64
1015  | ALT_DMA_CCR_OPT_SA_DEFAULT
1016  | ALT_DMA_CCR_OPT_SP_DEFAULT
1017  | ALT_DMA_CCR_OPT_SC_DEFAULT
1018  | ALT_DMA_CCR_OPT_DB16
1019  | ALT_DMA_CCR_OPT_DS64
1020  | ALT_DMA_CCR_OPT_DA_DEFAULT
1021  | ALT_DMA_CCR_OPT_DP_DEFAULT
1022  | ALT_DMA_CCR_OPT_DC_DEFAULT
1023  | ALT_DMA_CCR_OPT_ES_DEFAULT
1024  )
1025  );
1026  }
1027 
1028  while (length16burstcount > 0)
1029  {
1030  if (status != ALT_E_SUCCESS)
1031  {
1032  break;
1033  }
1034 
1035  uint32_t loopcount = MIN(length16burstcount, 256);
1036  length16burstcount -= loopcount;
1037 
1038  dprintf("DMA[M->M]: Looping %lux 16 burst length 8-byte transfer(s).\n", loopcount);
1039 
1040  if ((status == ALT_E_SUCCESS) && (loopcount > 1))
1041  {
1042  status = alt_dma_program_DMALP(program, loopcount);
1043  }
1044  if (status == ALT_E_SUCCESS)
1045  {
1046  status = alt_dma_program_DMALD(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
1047  }
1048  if (status == ALT_E_SUCCESS)
1049  {
1050  status = alt_dma_program_DMAST(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
1051  }
1052  if ((status == ALT_E_SUCCESS) && (loopcount > 1))
1053  {
1054  status = alt_dma_program_DMALPEND(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
1055  }
1056  }
1057  }
1058 
1059  // At this point, we should have [burstcount] 8-byte transfer(s)
1060  // remaining. [burstcount] should be less than 16.
1061 
1062  // Do one more burst with a SB / DB of length [burstcount].
1063 
1064  if (burstcount)
1065  {
1066  // Program in the following parameters:
1067  // - SS64 (Source burst size of 8-byte)
1068  // - DS64 (Destination burst size of 8-byte)
1069  // - SBx (Source burst length of [burstlength] transfers)
1070  // - DBx (Destination burst length of [burstlength] transfers)
1071  // - All other options default.
1072 
1073  if (status == ALT_E_SUCCESS)
1074  {
1075  status = alt_dma_program_DMAMOV(program, ALT_DMA_PROGRAM_REG_CCR,
1076  ( ((burstcount - 1) << 4) // SB
1077  | ALT_DMA_CCR_OPT_SS64
1078  | ALT_DMA_CCR_OPT_SA_DEFAULT
1079  | ALT_DMA_CCR_OPT_SP_DEFAULT
1080  | ALT_DMA_CCR_OPT_SC_DEFAULT
1081  | ((burstcount - 1) << 18) // DB
1082  | ALT_DMA_CCR_OPT_DS64
1083  | ALT_DMA_CCR_OPT_DA_DEFAULT
1084  | ALT_DMA_CCR_OPT_DP_DEFAULT
1085  | ALT_DMA_CCR_OPT_DC_DEFAULT
1086  | ALT_DMA_CCR_OPT_ES_DEFAULT
1087  )
1088  );
1089  }
1090  if (status == ALT_E_SUCCESS)
1091  {
1092  status = alt_dma_program_DMALD(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
1093  }
1094  if (status == ALT_E_SUCCESS)
1095  {
1096  status = alt_dma_program_DMAST(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
1097  }
1098  }
1099 
1100  // This is where the last DMAMOV CCR and DMAST is done if an
1101  // alignment correction required.
1102 
1103  if ( (correction == true)
1104  && ((usrc & 0x7) != (udst & 0x7)) // If src and dst are mod-8 congruent, no correction is needed.
1105  )
1106  {
1107  if (status == ALT_E_SUCCESS)
1108  {
1109  // Determine what type of correction.
1110 
1111  // Set the source parameters to match that of the destination
1112  // parameters. This way the SAR is increment in the same fashion as
1113  // DAR. This will allow the non 8-byte transfers to copy correctly.
1114 
1115  uint32_t ccr;
1116 
1117  if ((usrc & 0x3) == (udst & 0x3))
1118  {
1119  dprintf("DMA[M->M]: Single correction 4-byte burst size tranfer.\n");
1120 
1121  // Program in the following parameters:
1122  // - SS32 (Source burst size of 4-byte)
1123  // - DS32 (Destination burst size of 4-byte)
1124  // - SB1 (Source burst length of 1 transfer)
1125  // - DB1 (Destination burst length of 1 transfer)
1126  // - All other options default.
1127 
1128  ccr = ( ALT_DMA_CCR_OPT_SB1
1129  | ALT_DMA_CCR_OPT_SS32
1130  | ALT_DMA_CCR_OPT_SA_DEFAULT
1131  | ALT_DMA_CCR_OPT_SP_DEFAULT
1132  | ALT_DMA_CCR_OPT_SC_DEFAULT
1133  | ALT_DMA_CCR_OPT_DB1
1134  | ALT_DMA_CCR_OPT_DS32
1135  | ALT_DMA_CCR_OPT_DA_DEFAULT
1136  | ALT_DMA_CCR_OPT_DP_DEFAULT
1137  | ALT_DMA_CCR_OPT_DC_DEFAULT
1138  | ALT_DMA_CCR_OPT_ES_DEFAULT
1139  );
1140  }
1141  else if ((usrc & 0x1) == (udst & 0x1))
1142  {
1143  dprintf("DMA[M->M]: Single correction 2-byte burst size tranfer.\n");
1144 
1145  // Program in the following parameters:
1146  // - SS16 (Source burst size of 2-byte)
1147  // - DS16 (Destination burst size of 2-byte)
1148  // - SB1 (Source burst length of 1 transfer)
1149  // - DB1 (Destination burst length of 1 transfer)
1150  // - All other options default.
1151

1154  | ALT_DMA_CCR_OPT_SA_DEFAULT
1155  | ALT_DMA_CCR_OPT_SP_DEFAULT
1156  | ALT_DMA_CCR_OPT_SC_DEFAULT
1157  | ALT_DMA_CCR_OPT_DB1
1158  | ALT_DMA_CCR_OPT_DS16
1159  | ALT_DMA_CCR_OPT_DA_DEFAULT
1160  | ALT_DMA_CCR_OPT_DP_DEFAULT
1161  | ALT_DMA_CCR_OPT_DC_DEFAULT
1162  | ALT_DMA_CCR_OPT_ES_DEFAULT
1163  );
1164  }
1165  else
1166  {
1167  dprintf("DMA[M->M]: Single correction 1-byte burst size tranfer.\n");
1168 
1169  // Program in the following parameters:
1170  // - SS8 (Source burst size of 1-byte)
1171  // - DS8 (Destination burst size of 1-byte)
1172  // - SB1 (Source burst length of 1 transfer)
1173  // - DB1 (Destination burst length of 1 transfer)
1174  // - All other options default.
1175 
1176  ccr = ( ALT_DMA_CCR_OPT_SB1
1177  | ALT_DMA_CCR_OPT_SS8
1178  | ALT_DMA_CCR_OPT_SA_DEFAULT
1179  | ALT_DMA_CCR_OPT_SP_DEFAULT
1180  | ALT_DMA_CCR_OPT_SC_DEFAULT
1181  | ALT_DMA_CCR_OPT_DB1
1182  | ALT_DMA_CCR_OPT_DS8
1183  | ALT_DMA_CCR_OPT_DA_DEFAULT
1184  | ALT_DMA_CCR_OPT_DP_DEFAULT
1185  | ALT_DMA_CCR_OPT_DC_DEFAULT
1186  | ALT_DMA_CCR_OPT_ES_DEFAULT
1187  );
1188  }
1189 
1190  status = alt_dma_program_DMAMOV(program, ALT_DMA_PROGRAM_REG_CCR,
1191  ccr);
1192  }
1193  if (status == ALT_E_SUCCESS)
1194  {
1195  status = alt_dma_program_DMAST(program, ALT_DMA_PROGRAM_INST_MOD_NONE);

1218  | ALT_DMA_CCR_OPT_SP_DEFAULT

1234  {
1235  status = alt_dma_program_DMAST(program, ALT_DMA_PROGRAM_INST_MOD_NONE);
1236  }
1237  }
1238  } // if (size != 0)
1239

1242  {

1244  {

1247  }
1248  }
