57 /* Wait for the current transfer to end */
58 static int gr_i2cmst_wait(gr_i2cmst_prv_t *prv_ptr, uint8_t expected_sts)
59 {
60  uint32_t tout = 0;
61  int current_sts;
62 
63  DBG("(gr_i2cmst_wait called...");
64 
65  do {
66  if (tout++ > 1000000) {
67  DBG("gr_i2cmst_wait: TIMEOUT\n");
68  return RTEMS_TIMEOUT;
69  }
70  } while (prv_ptr->reg_ptr->cmdsts & GRI2C_STS_TIP);
71 
72  current_sts = prv_ptr->reg_ptr->cmdsts & ~GRI2C_STS_IF & ~GRI2C_STS_BUSY;
73 
74  if (current_sts != expected_sts) {
75 #if defined(DEBUG)
76  if (prv_ptr->reg_ptr->cmdsts & GRI2C_STS_RXACK) {
77  DBG("Transfer NAKed..");
78  }
79  if (prv_ptr->reg_ptr->cmdsts & GRI2C_STS_AL) {
80  DBG("arbitration lost..");
81  }
82  if (prv_ptr->reg_ptr->cmdsts & GRI2C_STS_TIP) {
83  DBG("transfer still in progress, huh?..");
84  }
85  DBG("exited with IO error..)");
86 #endif
87  DBG("gr_i2cmst_wait: IO-ERROR\n");
88  return RTEMS_IO_ERROR;
89  }
90 
91  DBG("exited...)");
92 
93  return RTEMS_SUCCESSFUL;
94 }

148 static rtems_status_code gr_i2cmst_send_addr(rtems_libi2c_bus_t *bushdl,
149  uint32_t addr, int rw)
150 {
151  gr_i2cmst_prv_t *prv_ptr = (gr_i2cmst_prv_t *)bushdl;
152  uint8_t addr_byte;
153  rtems_status_code rc;
154 
155  DBG("gr_i2cmst_send_addr called, addr = 0x%x, rw = %d...",
156  addr, rw);
157 
158  /* Check if long address is needed */
159  if (addr > 0x7f) {
160  addr_byte = ((addr >> 7) & 0x06) | (rw ? 1 : 0);
161 
162  prv_ptr->reg_ptr->tdrd = addr_byte;
163  prv_ptr->reg_ptr->cmdsts = GRI2C_CMD_WR | prv_ptr->sendstart;
164  prv_ptr->sendstart = 0;
165 
166  /* Wait for transfer to complete */
167  rc = gr_i2cmst_wait(prv_ptr, GRI2C_STATUS_IDLE);
168  if (rc != RTEMS_SUCCESSFUL) {
169 
170  DBG("exited with error\n");
171 
172  return -rc;
173  }
174  }
175 
176  /* For 10-bit adresses the last byte should only be written for a
177  write operation */
178  rc = RTEMS_SUCCESSFUL;
179  if (addr <= 0x7f || rw == 0) {
180  addr_byte = (addr << 1) | (rw ? 1 : 0);
181 
182  prv_ptr->reg_ptr->tdrd = addr_byte;
183  prv_ptr->reg_ptr->cmdsts = GRI2C_CMD_WR | prv_ptr->sendstart;
184  prv_ptr->sendstart = 0;
185 
186  /* Wait for transfer to complete */
187  rc = gr_i2cmst_wait(prv_ptr, GRI2C_STATUS_IDLE);
188  if (rc != RTEMS_SUCCESSFUL) {
189  DBG("exited with error\n");
190  return -rc;
191  }
192  }
193 
194  DBG("exited\n");
195  return rc;
196 }

199 static int gr_i2cmst_read_bytes(rtems_libi2c_bus_t *bushdl,
200  unsigned char *bytes, int nbytes)
201 {
202  gr_i2cmst_prv_t *prv_ptr = (gr_i2cmst_prv_t *)bushdl;
203  unsigned char *buf = bytes;
204  rtems_status_code rc;
205  unsigned char expected_sts = GRI2C_STATUS_IDLE;
206 
207  DBG("gr_i2cmst_read_bytes called, nbytes = %d...", nbytes);
208 
209  while (nbytes-- > 0) {
210  if (nbytes == 0) {
211  /* Respond with NAK to end sequential read */
212  prv_ptr->reg_ptr->cmdsts = (GRI2C_CMD_RD | GRI2C_CMD_ACK |
213  prv_ptr->sendstart);
214  expected_sts = GRI2C_STS_RXACK;
215  } else {
216  prv_ptr->reg_ptr->cmdsts = GRI2C_CMD_RD | prv_ptr->sendstart;
217  }
218  prv_ptr->sendstart = 0;
219  /* Wait until end of transfer */
220  rc = gr_i2cmst_wait(prv_ptr, expected_sts);
221  if (rc != RTEMS_SUCCESSFUL) {
222  DBG("exited with error\n");
223  return -rc;
224  }
225  *buf++ = prv_ptr->reg_ptr->tdrd;
226  }
227 
228  DBG("exited\n");
229 
230  return buf - bytes;
231 }

245

257  }

354 /* Init stage 2 */
355 int i2cmst_init3(struct drvmgr_dev *dev)
356 {
357  gr_i2cmst_prv_t *priv;
358  char prefix[32];
359  char devName[32];
360  int rc;
361 
362  priv = (gr_i2cmst_prv_t *)dev->priv;
363 
364  /* Do initialization */
365 
366  /* Initialize i2c library */
367  rc = rtems_libi2c_initialize();
368  if (rc != 0) {
369  DBG("I2CMST: rtems_libi2c_initialize failed, exiting...\n");
370  free(dev->priv);
371  dev->priv = NULL;
372  return DRVMGR_FAIL;
373  }
374 
375  /* I/O system registered and initialized
376  * Now we take care of device initialization.
377  */
378 
379  /* Get frequency */
380  if ( drvmgr_freq_get(dev, DEV_APB_SLV, &priv->sysfreq) ) {
381  return DRVMGR_FAIL;
382  }
383  priv->sysfreq = priv->sysfreq / 1000; /* Convert to kHz */
384 
385  if ( i2cmst_device_init(priv) ) {
386  free(dev->priv);
387  dev->priv = NULL;
388  return DRVMGR_FAIL;
389  }
390 
391  /* Get Filesystem name prefix */
392  prefix[0] = '\0';
393  if ( drvmgr_get_dev_prefix(dev, prefix) ) {
394  /* Failed to get prefix, make sure of a unique FS name
395  * by using the driver minor.
396  */
397  sprintf(devName, "/dev/i2c%d", dev->minor_drv+1);
398  } else {
399  /* Got special prefix, this means we have a bus prefix
400  * And we should use our "bus minor"
401  */
402  sprintf(devName, "/dev/%si2c%d", prefix, dev->minor_bus+1);
403  }
404 
405  /* Register Bus for this Device */
406  rc = rtems_libi2c_register_bus(devName, &priv->i2clib_desc);
407  if (rc < 0) {
408  DBG("I2CMST: rtems_libi2c_register_bus(%s) failed, exiting..\n", devName);
409  free(dev->priv);
410  dev->priv = NULL;
411  return DRVMGR_FAIL;
412  }
413  priv->minor = rc;
414 
415  return DRVMGR_OK;
416 }
