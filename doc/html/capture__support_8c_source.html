80 /*
81  * Add a name.
82  */
83 static void
84 ctrace_task_name_add (rtems_id id, const rtems_name* name)
85 {
86  if (tasks.tasks == NULL)
87  {
88  tasks.size = CTRACE_TASK_BLOCK_SIZE;
89  tasks.tasks = calloc (tasks.size, sizeof (ctrace_task_name));
90  }
91  if (tasks.tasks != NULL)
92  {
93  if (rtems_object_id_get_api(id) != OBJECTS_POSIX_API)
94  {
95  size_t t;
96  for (t = 0; t < tasks.count; ++t)
97  {
98  if (tasks.tasks[t].id == id)
99  {
100  tasks.tasks[t].name = *name;
101  break;
102  }
103  }
104  if (t == tasks.count)
105  {
106  if (tasks.count >= tasks.size)
107  {
108  tasks.size += CTRACE_TASK_BLOCK_SIZE;
109  tasks.tasks = realloc (tasks.tasks,
110  tasks.size * sizeof (ctrace_task_name));
111  }
112  if (tasks.tasks != NULL)
113  {
114  tasks.tasks[tasks.count].name = *name;
115  tasks.tasks[tasks.count].id = id;
116  ++tasks.count;
117  }
118  }
119  }
120  }
121 }

123 /*
124  * Remove a task name.
125  */
126 static void
127 ctrace_task_name_remove (rtems_id id)
128 {
129  size_t t;
130  for (t = 0; t < tasks.count; ++t)
131  {
132  if (tasks.tasks[t].id == id)
133  {
134  size_t count = tasks.count - t - 1;
135  if (count != 0)
136  memmove (&tasks.tasks[t],
137  &tasks.tasks[t + 1],
138  sizeof (ctrace_task_name) * count);
139  --tasks.count;
140  break;
141  }

219  const rtems_capture_record* rec,
220  uint64_t diff,
221  const rtems_name* name)
222 {
223  uint32_t event;
224  int e;
225 
226  event = rec->events >> RTEMS_CAPTURE_EVENT_START;
227  for (e = RTEMS_CAPTURE_EVENT_START; e < RTEMS_CAPTURE_EVENT_END; e++)
228  {
229  if (event & 1)
230  {
231  fprintf(stdout,"%2i ", cpu);
232  rtems_capture_print_timestamp (rec->time);
233  fprintf (stdout, " %12" PRId32 " ", (uint32_t) diff);
234  rtems_monitor_dump_id (rec->task_id);
235  if (name != NULL)
236  {
237  fprintf (stdout, " %c%c%c%c",
238  (char) (*name >> 24) & 0xff,
239  (char) (*name >> 16) & 0xff,
240  (char) (*name >> 8) & 0xff,
241  (char) (*name >> 0) & 0xff);
242  }
243  else
244  {
245  fprintf(stdout, " ");
246  }
247  fprintf(stdout, " %3" PRId32 " %3" PRId32 " %s\n",

249  (rec->events >> RTEMS_CAPTURE_CURR_PRIORITY_EVENT) & 0xff,

252  event >>= 1;
253  }

262 void
263 rtems_capture_print_trace_records (int total, bool csv)
264 {
265  ctrace_per_cpu* per_cpu;
266  ctrace_per_cpu* cpu;
267  int cpus;
268  rtems_capture_time last_time = 0;
269  int i;
270 
271  cpus = rtems_scheduler_get_processor_maximum ();
272 
273  per_cpu = calloc (cpus, sizeof(*per_cpu));
274  if (per_cpu == NULL)
275  {
276  fprintf(stdout, "error: no memory\n");
277  return;
278  }
279 
280  while (total)
281  {
282  const rtems_capture_record* rec_out = NULL;
283  int cpu_out = -1;
284  rtems_capture_time this_time = 0;
285 
286  /* Prime the per_cpu data */
287  for (i = 0; i < cpus; i++) {
288  cpu = &per_cpu[i];
289 
290  if (cpu->read == 0)
291  {
292  rtems_status_code sc;
293  sc = rtems_capture_read (i, &cpu->read, &cpu->recs);
294  if (sc != RTEMS_SUCCESSFUL)
295  {
296  fprintf (stdout,
297  "error: trace read failed: %s\n", rtems_status_text (sc));
298  rtems_capture_flush (0);
299  free (per_cpu);
300  return;
301  }
302  /* Release the buffer if there are no records to read */
303  if (cpu->read == 0)
304  rtems_capture_release (i, 0);
305  }
306 
307  /* Read the record out from the capture buffer */
308  if (!cpu->rec_valid && (cpu->read != 0))
309  {
310  cpu->recs = rtems_capture_record_extract (cpu->recs,
311  &cpu->rec,
312  sizeof (cpu->rec));
313  cpu->rec_valid = true;
314  }
315 
316  /* Find the next record to print, the earliest recond on any core */
317  if ((cpu->rec_valid) && ((this_time == 0) || (cpu->rec.time < this_time)))
318  {
319  rec_out = &cpu->rec;
320  cpu_out = i;
321  this_time = rec_out->time;
322  }
323  }
324 
325  /* If we have read all the records abort. */
326  if (rec_out == NULL)
327  break;
328 
329  cpu = &per_cpu[cpu_out];
330 
331  /* Print the record */
332  if (csv)
333  {
334  fprintf(stdout,
335  "%03i,%08" PRIu32 ",%03" PRIu32
336  ",%03" PRIu32 ",%04" PRIx32 ",%" PRId64 "\n",
337  cpu_out,
338  (uint32_t) rec_out->task_id,
339  (rec_out->events >> RTEMS_CAPTURE_REAL_PRIORITY_EVENT) & 0xff,
340  (rec_out->events >> RTEMS_CAPTURE_CURR_PRIORITY_EVENT) & 0xff,
341  (rec_out->events >> RTEMS_CAPTURE_EVENT_START),
342  (uint64_t) rec_out->time);
343  }
344  else
345  {
346  if ((rec_out->events >> RTEMS_CAPTURE_EVENT_START) == 0)
347  {
348  rtems_capture_task_record task_rec;
349  cpu->recs = rtems_capture_record_extract (cpu->recs,
350  &task_rec,
351  sizeof (task_rec));
352  ctrace_task_name_add (rec_out->task_id, &task_rec.name);
353  rtems_capture_print_record_task (cpu_out, rec_out, &task_rec);
354  }
355  else
356  {
357  rtems_capture_time diff;
358  const rtems_name* name = NULL;
359  if (last_time != 0)
360  diff = rec_out->time - last_time;
361  else
362  diff = 0;
363  last_time = rec_out->time;
364  ctrace_task_name_find (rec_out->task_id, &name);
365  rtems_capture_print_record_capture (cpu_out, rec_out, diff, name);
366  if ((rec_out->events &
367  (RTEMS_CAPTURE_DELETED_BY_EVENT | RTEMS_CAPTURE_DELETED_EVENT)) != 0)
368  ctrace_task_name_remove (rec_out->task_id);
369  }
370  }
371 
372  /*
373  * If we have not printed all the records read increment to the next
374  * record. If we have printed all records release the records printed.
375  */
376  cpu->rec_valid = false;
377  ++cpu->printed;
378  if (cpu->printed == cpu->read)
379  {
380  rtems_capture_release (cpu_out, cpu->printed);
381  cpu->recs = NULL;
382  cpu->read = 0;
383  cpu->printed = 0;
384  }
385 
386  --total;
387  }
388 
389  /* Finished so release all the records that were printed. */
390  for (i = 0; i < cpus; i++)
391  {
392  cpu = &per_cpu[i];
393  if (cpu->read != 0)
394  {
395  rtems_capture_release (i, cpu->printed);
396  }
397  }
398 
399  free(per_cpu);
400 }
