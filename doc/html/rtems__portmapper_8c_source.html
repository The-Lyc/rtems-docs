157  caddr_t t;

159 #ifdef DEBUG
160  fprintf(stderr, "server: about do a switch\n");
161 #endif
162  switch (rqstp->rq_proc) {
163 
164  case PMAPPROC_NULL:
165  /*
166  * Null proc call
167  */
168  if ((!svc_sendreply(xprt, (xdrproc_t) xdr_void, NULL)) &&
169  debugging) {
170  abort();
171  }
172  break;
173 
174  case PMAPPROC_SET:
175  /*
176  * Set a program,version to port mapping
177  */
178  if (!svc_getargs(xprt, (xdrproc_t) xdr_pmap, (caddr_t)&reg))
179  svcerr_decode(xprt);
180  else {
181  /*
182  * check to see if already used
183  * find_service returns a hit even if
184  * the versions don't match, so check for it
185  */
186  fnd = find_service(reg.pm_prog, reg.pm_vers, reg.pm_prot);
187  if (fnd && fnd->pml_map.pm_vers == reg.pm_vers) {
188  if (fnd->pml_map.pm_port == reg.pm_port) {
189  ans = 1;
190  goto done;
191  }
192  else {
193  ans = 0;
194  goto done;
195  }
196  } else {
197  /*
198  * add to END of list
199  */
200  pml = (struct pmaplist *)
201  malloc(sizeof(struct pmaplist));
202  pml->pml_map = reg;
203  pml->pml_next = 0;
204  if (pmaplist == 0) {
205  pmaplist = pml;
206  } else {
207  for (fnd= pmaplist; fnd->pml_next != 0;
208  fnd = fnd->pml_next);
209  fnd->pml_next = pml;
210  }
211  ans = 1;
212  }
213  done:
214  if ((!svc_sendreply(xprt, (xdrproc_t) xdr_long, (caddr_t)&ans)) &&
215  debugging) {
216  fprintf(stderr, "svc_sendreply\n");
217  abort();
218  }
219  }
220  break;

223  /*
224  * Remove a program,version to port mapping.
225  */

227  svcerr_decode(xprt);

231  if ((pml->pml_map.pm_prog != reg.pm_prog) ||

239  ans = 1;

247  }

406 /*
407  * Call a remote procedure service
408  * This procedure is very quiet when things go wrong.
409  * The proc is written to support broadcast rpc. In the broadcast case,
410  * a machine should shut-up instead of complain, less the requestor be
411  * overrun with complaints at the expense of not hearing a valid reply ...
412  *
413  * This now forks so that the program & process that it calls can call
414  * back to the portmapper.
415  */
416 static void
417 callit(
418  struct svc_req *rqstp,
419  SVCXPRT *xprt )
420 {
421  struct rmtcallargs a;
422  struct pmaplist *pml;
423  u_short port;
424  struct sockaddr_in me;
425  int pid, socket = -1;
426  CLIENT *client;
427  struct authunix_parms *au = (struct authunix_parms *)rqstp->rq_clntcred;
428  struct timeval timeout;
429  char buf[ARGSIZE];
430 
431  timeout.tv_sec = 5;
432  timeout.tv_usec = 0;
433  a.rmt_args.args = buf;
434  if (!svc_getargs(xprt, (xdrproc_t) xdr_rmtcall_args, (caddr_t)&a))
435  return;
436  if ((pml = find_service(a.rmt_prog, a.rmt_vers, IPPROTO_UDP)) == NULL)
437  return;
438  /*
439  * fork a child to do the work. Parent immediately returns.
440  * Child exits upon completion.
441  */
442  if ((pid = fork()) != 0) {
443  if (debugging && (pid < 0)) {
444  fprintf(stderr, "portmap CALLIT: cannot fork.\n");
445  }
446  return;
447  }
448  port = pml->pml_map.pm_port;
449  get_myaddress(&me);
450  me.sin_port = htons(port);
451  client = clntudp_create(&me, a.rmt_prog, a.rmt_vers, timeout, &socket);
452  if (client != (CLIENT *)NULL) {
453  if (rqstp->rq_cred.oa_flavor == AUTH_UNIX) {
454  client->cl_auth = authunix_create(au->aup_machname,
455  au->aup_uid, au->aup_gid, au->aup_len, au->aup_gids);
456  }
457  a.rmt_port = (u_long)port;
458  if (clnt_call(client, a.rmt_proc, xdr_opaque_parms, &a,
459  xdr_len_opaque_parms, &a, timeout) == RPC_SUCCESS) {
460  svc_sendreply(xprt, (xdrproc_t) xdr_rmtcall_result, (caddr_t)&a);
461  }
462  AUTH_DESTROY(client->cl_auth);
463  clnt_destroy(client);
464  }
465  (void)close(socket);
466  exit(0);
467 }
