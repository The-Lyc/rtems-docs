168 /*
169  * Create the sorted table with the current and total usage.
170  */
171 static bool
172 task_usage(Thread_Control* thread, void* arg)
173 {
174  rtems_cpu_usage_data* data = (rtems_cpu_usage_data*) arg;
175  Timestamp_Control usage;
176  Timestamp_Control current = data->zero;
177  int j;
178 
179  data->stack_size += thread->Start.Initial_stack.size;
180 
181  _Thread_Get_CPU_time_used(thread, &usage);
182 
183  for (j = 0; j < data->last_task_count; j++)
184  {
185  if (thread == data->last_tasks[j])
186  {
187  _Timestamp_Subtract(&data->last_usage[j], &usage, &current);
188  break;
189  }
190  }
191 
192  /*
193  * When not using nanosecond CPU usage resolution, we have to count the
194  * number of "ticks" we gave credit for to give the user a rough guideline as
195  * to what each number means proportionally.
196  */
197  _Timestamp_Add_to(&data->total, &usage);
198  _Timestamp_Add_to(&data->current, &current);
199 
200  if (thread->is_idle)
201  {
202  _Timestamp_Add_to(&data->idle, &usage);
203  _Timestamp_Add_to(&data->current_idle, &current);
204  }
205 
206  /*
207  * Create the tasks to display sorting as we create.
208  */
209  for (j = 0; j < data->task_count; j++)
210  {
211  if (data->tasks[j])
212  {
213  int k;
214 
215  /*
216  * Sort on the current load.
217  */
218  switch (data->sort_order)
219  {
220  default:
221  data->sort_order = RTEMS_TOP_SORT_CURRENT;
222  /* drop through */
223  case RTEMS_TOP_SORT_CURRENT:
224  if (CPU_usage_Equal_to(&current, &data->zero) ||
225  CPU_usage_Less_than(&current, &data->current_usage[j]))
226  continue;
227  case RTEMS_TOP_SORT_TOTAL:
228  if (CPU_usage_Equal_to(&usage, &data->zero) ||
229  CPU_usage_Less_than(&usage, &data->usage[j]))
230  continue;
231  case RTEMS_TOP_SORT_REAL_PRI:
232  if (thread->Real_priority.priority > data->tasks[j]->Real_priority.priority)
233  continue;
234  case RTEMS_TOP_SORT_CURRENT_PRI:
235  if (
236  _Thread_Get_priority( thread )
237  > _Thread_Get_priority( data->tasks[j] )
238  ) {
239  continue;
240  }
241  case RTEMS_TOP_SORT_ID:
242  if (thread->Object.id < data->tasks[j]->Object.id)
243  continue;
244  }
245 
246  for (k = (data->task_count - 1); k >= j; k--)
247  {
248  data->tasks[k + 1] = data->tasks[k];
249  data->usage[k + 1] = data->usage[k];
250  data->current_usage[k + 1] = data->current_usage[k];
251  }
252  }
253  data->tasks[j] = thread;
254  data->usage[j] = usage;
255  data->current_usage[j] = current;
256  break;
257  }
258 
259  return false;
260 }

518 void rtems_cpu_usage_top_with_plugin(
519  const rtems_printer *printer
520 )
521 {
522  rtems_status_code sc;
523  rtems_task_priority priority;
524  rtems_name name;
525  rtems_id id;
526  rtems_cpu_usage_data data;
527  int show_lines = 25;
528 
529  memset(&data, 0, sizeof(data));
530 
531  data.thread_run = true;
532  data.single_page = true;
533  data.sort_order = RTEMS_TOP_SORT_CURRENT;
534  data.poll_rate_usecs = 3000;
535  data.show = show_lines;
536  data.printer = printer;
537 
538  sc = rtems_task_set_priority (RTEMS_SELF, RTEMS_CURRENT_PRIORITY, &priority);
539 
540  if (sc != RTEMS_SUCCESSFUL)
541  {
542  rtems_printf (printer,
543  "error: cannot obtain the current priority: %s\n", rtems_status_text (sc));
544  return;
545  }
546 
547  name = rtems_build_name('C', 'P', 'l', 't');
548 
549  sc = rtems_task_create (name, priority, 4 * 1024,
550  RTEMS_PREEMPT | RTEMS_TIMESLICE | RTEMS_NO_ASR,
551  RTEMS_FLOATING_POINT | RTEMS_LOCAL,
552  &id);
553 
554  if (sc != RTEMS_SUCCESSFUL)
555  {
556  rtems_printf (printer,
557  "error: cannot create helper thread: %s\n", rtems_status_text (sc));
558  return;
559  }
560 
561  sc = rtems_task_start (id, rtems_cpuusage_top_thread, (rtems_task_argument) &data);
562  if (sc != RTEMS_SUCCESSFUL)
563  {
564  rtems_printf (printer,
565  "error: cannot start helper thread: %s\n", rtems_status_text (sc));
566  rtems_task_delete (id);
567  return;
568  }
569 
570  while (true)
571  {
572  int c = getchar ();
573 
574  if ((c == '\r') || (c == '\n') || (c == 'q') || (c == 'Q'))
575  {
576  int loops = 50;
577 
578  data.thread_run = false;
579 
580  rtems_event_send(id, RTEMS_EVENT_1);
581 
582  while (loops && data.thread_active)
583  rtems_task_wake_after (RTEMS_MICROSECONDS_TO_TICKS (100000));
584 
585  rtems_printf (printer, "load monitoring stopped.\n");
586  return;
587  }
588  else if (c == '<')
589  {
590  if (data.sort_order == 0)
591  data.sort_order = RTEMS_TOP_SORT_MAX;
592  else
593  --data.sort_order;
594  rtems_event_send(id, RTEMS_EVENT_1);
595  }
596  else if (c == '>')
597  {
598  if (data.sort_order >= RTEMS_TOP_SORT_MAX)
599  data.sort_order = 0;
600  else
601  ++data.sort_order;
602  rtems_event_send(id, RTEMS_EVENT_1);
603  }
604  else if ((c == 's') || (c == 'S'))
605  {
606  data.single_page = !data.single_page;
607  rtems_event_send(id, RTEMS_EVENT_1);
608  }
609  else if ((c == 'a') || (c == 'A'))
610  {
611  if (data.show == 0)
612  data.show = show_lines;
613  else
614  data.show = 0;
615  rtems_event_send(id, RTEMS_EVENT_1);
616  }
617  else if (c == '+')
618  {
619  ++show_lines;
620  if (data.show != 0)
621  data.show = show_lines;
622  }
623  else if (c == '-')
624  {
625  if (show_lines > 5)
626  --show_lines;
627  if (data.show != 0)
628  data.show = show_lines;
629  }
630  else if (c == ' ')
631  {
632  rtems_event_send(id, RTEMS_EVENT_1);
633  }
634  }
635 }
