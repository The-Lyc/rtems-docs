318  // - pclk - l4_mp_clk
319  // - hclk - l4_mp_clk
320  // - ref_clk - qspi_clk
321 
322  // Check and validate the QSPI ref_clk which is connected to the HPS qspi_clk.
323  if (status == ALT_E_SUCCESS)
324  {
325  if (alt_clk_is_enabled(ALT_CLK_QSPI) != ALT_E_TRUE)
326  {
327  status = ALT_E_BAD_CLK;
328  }
329  }
330 
331  if (status == ALT_E_SUCCESS)
332  {
333  status = alt_clk_freq_get(ALT_CLK_QSPI, &qspi_clk_freq);
334  if (status == ALT_E_SUCCESS)
335  {
336  if (qspi_clk_freq > ALT_QSPI_CLK_FREQ_MAX)
337  {
338  return ALT_E_BAD_CLK;
339  }
340  }
341  }
342 
343  int qspi_clk_mhz = qspi_clk_freq / 1000000;
344 
346 
347  // Take QSPI controller out of reset.
348  alt_clrbits_word(ALT_RSTMGR_PERMODRST_ADDR, ALT_RSTMGR_PERMODRST_QSPI_SET_MSK);
349 
351 
352  // Configure the device timing
353 
354  if (status == ALT_E_SUCCESS)
355  {
356  ALT_QSPI_TIMING_CONFIG_t timing_cfg =
357  {
358  .clk_phase = (ALT_QSPI_CLK_PHASE_t)ALT_QSPI_CFG_SELCLKPHASE_RESET,
359  .clk_pol = (ALT_QSPI_CLK_POLARITY_t)ALT_QSPI_CFG_SELCLKPOL_RESET,
360  .cs_da = alt_qspi_ns_to_multiplier(ALT_QSPI_TSHSL_NS_DEF, qspi_clk_mhz),
361  .cs_dads = alt_qspi_ns_to_multiplier(ALT_QSPI_TSD2D_NS_DEF, qspi_clk_mhz),
362  .cs_eot = alt_qspi_ns_to_multiplier(ALT_QSPI_TCHSH_NS_DEF, qspi_clk_mhz),
363  .cs_sot = alt_qspi_ns_to_multiplier(ALT_QSPI_TSLCH_NS_DEF, qspi_clk_mhz),
364  .rd_datacap = 1
365  };
366 
367  dprintf("DEBUG[QSPI]: cs_da = %" PRIu32 ".\n", timing_cfg.cs_da);
368  dprintf("DEBUG[QSPI]: cs_dads = %" PRIu32 ".\n", timing_cfg.cs_dads);
369  dprintf("DEBUG[QSPI]: cs_eot = %" PRIu32 ".\n", timing_cfg.cs_eot);
370  dprintf("DEBUG[QSPI]: cs_sot = %" PRIu32 ".\n", timing_cfg.cs_sot);

373  }

382  }
383

389  }

392  // CASE 157095: Investigate using 108 MHz, and tweaking the rd_datacap param.
393 
394  if (status == ALT_E_SUCCESS)
395  {

422 ALT_STATUS_CODE alt_qspi_enable(void)
423 {
424  alt_setbits_word(ALT_QSPI_CFG_ADDR, ALT_QSPI_CFG_EN_SET_MSK);
425 
426  ALT_STATUS_CODE status = ALT_E_SUCCESS;
427 
429 
430  // Device specific configuration
431 
432 #if ALT_QSPI_PROVISION_MICRON_N25Q_SUPPORT
433  if (status == ALT_E_SUCCESS)
434  {
435  status = alt_qspi_N25Q_enable();
436  }
437 #endif
438 
439  uint32_t rdid = 0;
440 
441  // Query device capabilities
442  // This requires QSPI to be enabled.
443 
444  if (status == ALT_E_SUCCESS)
445  {
446  status = alt_qspi_device_rdid(&rdid);
447  }
448 
449  if (status == ALT_E_SUCCESS)
450  {
451  // NOTE: The size code seems to be a form of BCD (binary coded decimal).
452  // The first nibble is the 10's digit and the second nibble is the 1's
453  // digit in the number of bytes.
454 
455  // Capacity ID samples:
456  // 0x15 : 16 Mb => 2 MiB => 1 << 21 ; BCD=15
457  // 0x16 : 32 Mb => 4 MiB => 1 << 22 ; BCD=16
458  // 0x17 : 64 Mb => 8 MiB => 1 << 23 ; BCD=17
459  // 0x18 : 128 Mb => 16 MiB => 1 << 24 ; BCD=18
460  // 0x19 : 256 Mb => 32 MiB => 1 << 25 ; BCD=19
461  // 0x1a
462  // 0x1b
463  // 0x1c
464  // 0x1d
465  // 0x1e
466  // 0x1f
467  // 0x20 : 512 Mb => 64 MiB => 1 << 26 ; BCD=20
468  // 0x21 : 1024 Mb => 128 MiB => 1 << 27 ; BCD=21
469 
470  int cap_code = ALT_QSPI_STIG_RDID_CAPACITYID_GET(rdid);
471 
472  if ( ((cap_code >> 4) > 0x9) || ((cap_code & 0xf) > 0x9))
473  {
474  // If a non-valid BCD value is detected at the top or bottom nibble, chances
475  // are that the chip has a problem.
476 
477  dprintf("DEBUG[QSPI]: Invalid CapacityID encountered: 0x%02x.\n", cap_code);
478  status = ALT_E_ERROR;
479  }
480  else
481  {
482  int cap_decoded = ((cap_code >> 4) * 10) + (cap_code & 0xf);
483 
484  qspi_device_size = 1 << (cap_decoded + 6);
485 
486  dprintf("DEBUG[QSPI]: Device size = 0x%" PRIx32 ".\n", qspi_device_size);
487  }
488  }
489 
490  // Configure the device size and address bytes
491 
492  if (status == ALT_E_SUCCESS)
493  {
494  ALT_QSPI_DEV_SIZE_CONFIG_t size_cfg =
495  {
496  .block_size = ALT_QSPI_DEVSZ_BYTESPERSUBSECTOR_RESET, // 0x10 => 2^16 = 64 KiB
497  .page_size = ALT_QSPI_DEVSZ_BYTESPERDEVICEPAGE_RESET, // 0x100 => 256 B
498  .addr_size = ALT_QSPI_DEVSZ_NUMADDRBYTES_RESET, // 0x2 => 3 bytes or 0x00ffffff mask.
499  .lower_wrprot_block = 0,
500  .upper_wrprot_block = (qspi_device_size - 1) >> 16,
501  .wrprot_enable = ALT_QSPI_WRPROT_EN_RESET
502  };
503 
504  status = alt_qspi_device_size_config_set(&size_cfg);
505  }
506 
508 
509  // Configure the DMA parameters
510 
511  // This will allow DMA to work well without much intervention by users.
512 
513  if (status == ALT_E_SUCCESS)
514  {
515  status = alt_qspi_dma_config_set(4, 32);
516  }
517 
519 
520  return status;
521 }

892 static ALT_STATUS_CODE alt_qspi_replace_bank(uint32_t dst, const char * src, size_t size)
893 {
894  ALT_STATUS_CODE status = ALT_E_SUCCESS;
895 
896  //
897  // subsec_count : The total number of affected subsector(s),
898  // including partial subsector(s).
899  // subsec_addr : The aligned address of the next affected subsector,
900  // including partial subsector(s).
901  // subsec_partial_head : The number of subsector unaligned data to be
902  // written out at the start of the flash write
903  // request. This data ends at the end of the subsector
904  // or earlier depending on the number of data to be
905  // written.
906  // subsec_partial_tail : The number of subsector unaligned data to be
907  // written out at the end of the flash write request.
908  // This data starts at the start of the subsector. If
909  // only a single subsector is written (partial or
910  // full), this value will be zero.
911  //
912 
913  uint32_t subsec_count = ((dst + size - 1) >> 12) - (dst >> 12) + 1;
914  uint32_t subsec_addr = dst & ALT_QSPI_SUBSECTOR_ADDR_MSK;
915 
916  uint32_t subsec_partial_head = MIN(ALT_QSPI_SUBSECTOR_SIZE - (dst & (ALT_QSPI_SUBSECTOR_SIZE - 1)), size) & (ALT_QSPI_SUBSECTOR_SIZE - 1);
917  uint32_t subsec_partial_tail = (size - subsec_partial_head) & (ALT_QSPI_SUBSECTOR_SIZE - 1);
918 
919  dprintf("DEBUG[QSPI]: replace(): report: dst = 0x%" PRIx32 "; size = 0x%x.\n",
920  dst, size);
921  dprintf("DEBUG[QSPI]: replace(): report: subsec_count = 0x%" PRIx32 "; subsec_addr = 0x%" PRIx32 ".\n",
922  subsec_count, subsec_addr);
923  dprintf("DEBUG[QSPI]: replace(): report: partial_head = 0x%" PRIx32 "; partial_tail = 0x%" PRIx32 ".\n",
924  subsec_partial_head, subsec_partial_tail);
925 
926  // Write the first subsector, partial case.
927 
928  if (subsec_partial_head)
929  {
930  // The write request is not aligned to a subsector so we must do the
931  // Read-Modify-Write cycle to preserve the existing data at the head of
932  // the subsector not affected by the write.
933 
934  char subsec_buf[ALT_QSPI_SUBSECTOR_SIZE];
935 
936  uint32_t subsec_ofst = dst & ~ALT_QSPI_SUBSECTOR_ADDR_MSK;
937 
938  // - Read the subsector into buffer
939  // - Erase that subsector
940  // - Copy in the user data into buffer
941  // - Write out buffer to subsector
942 
943  if (status == ALT_E_SUCCESS)
944  {
945  status = alt_qspi_read_bank(subsec_buf, subsec_addr, subsec_ofst);
946  }
947  if (status == ALT_E_SUCCESS)
948  {
949  status = alt_qspi_erase_subsector_bank(subsec_addr);
950  }
951  if (status == ALT_E_SUCCESS)
952  {
953  memcpy(subsec_buf + subsec_ofst, src, subsec_partial_head);
954  status = alt_qspi_indirect_subsector_aligned_write_helper(subsec_buf, subsec_addr);
955  }
956 
957  // Do some bookkeeping on the user buffer information
958  src += subsec_partial_head;
959  size -= subsec_partial_head;
960 
961  // Do some bookkeeping on the subsector tracking
962  subsec_count--;
963  subsec_addr += ALT_QSPI_SUBSECTOR_SIZE;
964 
965  dprintf("DEBUG[QSPI]: replace(): partial head: subsec_ofst = 0x%" PRIx32 "; size left = 0x%x; status = %" PRIi32 ".\n",
966  subsec_ofst, size, status);
967  }
968 
969  // If there is a partial tail, then take 1 off the subsec_count. This way
970  // the following loop will write out all the complete subsectors. The tail
971  // will be written out afterwards.
972 
973  if (subsec_partial_tail)
974  {
975  subsec_count--;
976  }
977 
978  // Write the aligned subsectors following any partial subsectors.
979 
980  for (uint32_t i = 0; i < subsec_count; ++i)
981  {
982  // - Erase subsector
983  // - Write out buffer to subsector
984 
985  if (status == ALT_E_SUCCESS)
986  {
987  status = alt_qspi_erase_subsector_bank(subsec_addr);
988  }
989  if (status == ALT_E_SUCCESS)
990  {
991  status = alt_qspi_indirect_subsector_aligned_write_helper(src, subsec_addr);
992  }
993 
994  src += ALT_QSPI_SUBSECTOR_SIZE;
995  size -= ALT_QSPI_SUBSECTOR_SIZE;
996 
997  // Don't modify subsec_count as it's being used by the loop.
998  subsec_addr += ALT_QSPI_SUBSECTOR_SIZE;
999 
1000  dprintf("DEBUG[QSPI]: replace(): subsec aligned: size left = 0x%x, status = %" PRIi32 ".\n",
1001  size, status);
1002  }
1003 
1004  // Write the last subsector, partial case.
1005 
1006  if (subsec_partial_tail)
1007  {
1008  // The write request is not aligned to a subsector so we must do the
1009  // Read-Modify-Write cycle to preserve the existing data at the end of
1010  // the subsector not affected by the write.
1011 
1012  char subsec_buf[ALT_QSPI_SUBSECTOR_SIZE];
1013 
1014  // - Read the subsector into buffer
1015  // - Erase that subsector
1016  // - Copy in the user data into buffer
1017  // - Write out buffer to subsector
1018 
1019  if (status == ALT_E_SUCCESS)
1020  {
1021  status = alt_qspi_read_bank(subsec_buf + subsec_partial_tail,
1022  subsec_addr + subsec_partial_tail,
1023  ALT_QSPI_SUBSECTOR_SIZE - subsec_partial_tail);
1024  }
1025  if (status == ALT_E_SUCCESS)
1026  {
1027  status = alt_qspi_erase_subsector_bank(subsec_addr);
1028  }
1029  if (status == ALT_E_SUCCESS)
1030  {
1031  memcpy(subsec_buf, src, subsec_partial_tail);
1032  status = alt_qspi_indirect_subsector_aligned_write_helper(subsec_buf, subsec_addr);
1033  }
1034 
1035  src += subsec_partial_tail;
1036  size -= subsec_partial_tail;
1037 
1038  dprintf("DEBUG[QSPI]: replace(): partial tail: size left = 0x%x, status = %" PRIi32 ".\n",
1039  size, status);
1040  }
1041 
1042  return status;
1043 }
