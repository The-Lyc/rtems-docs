85 rtems_shell_main_hexdump(int argc, char *argv[])
86 {
87  rtems_shell_hexdump_globals hexdump_globals;
88  rtems_shell_hexdump_globals* globals = &hexdump_globals;
89  memset (globals, 0, sizeof (hexdump_globals));
90  vflag = FIRST;
91  ateof = 1;
92  hexdump_globals.exit_code = 1;
93  if (setjmp (hexdump_globals.exit_jmp) == 0)
94  hexdump_globals.exit_code = main_hexdump (globals, argc, argv);
95  if (curp)
96  free (curp);
97  if (savp)
98  free (savp);
99  while (fshead)
100  {
101  FS* nextfs = fshead->nextfs;
102  while (fshead->nextfu)
103  {
104  FU* nextfu = fshead->nextfu->nextfu;
105  if (fshead->nextfu->fmt)
106  free(fshead->nextfu->fmt);
107  while (fshead->nextfu->nextpr)
108  {
109  PR* nextpr = fshead->nextfu->nextpr->nextpr;
110  if (((fshead->nextfu->nextpr->flags & F_TEXT) == 0) &&
111  fshead->nextfu->nextpr->fmt)
112  free(fshead->nextfu->nextpr->fmt);
113  free(fshead->nextfu->nextpr);
114  fshead->nextfu->nextpr = nextpr;
115  }
116  free(fshead->nextfu);
117  fshead->nextfu = nextfu;
118  }
119  free(fshead);
120  fshead = nextfs;
121  }
122  if (hdstdin)
123  {
124  fclose (hdstdin);
125  free (hdstdin);
126  }
127  return hexdump_globals.exit_code;
128 }

130 int
131 main_hexdump(rtems_shell_hexdump_globals* globals, int argc, char *argv[])
132 {
133  FS *tfs;
134  char *p;
135 
136 #if RTEMS_REMOVED
137  (void)setlocale(LC_ALL, "");
138 #endif
139 
140  if (!(p = rindex(argv[0], 'o')) || strcmp(p, "od"))
141  newsyntax(globals, argc, &argv);
142  else
143  oldsyntax(globals, argc, &argv);
144 
145  /* figure out the data block size */
146  for (blocksize = 0, tfs = fshead; tfs; tfs = tfs->nextfs) {
147  tfs->bcnt = size(globals, tfs);
148  if (blocksize < tfs->bcnt)
149  blocksize = tfs->bcnt;
150  }
151  /* rewrite the rules, do syntax checking */
152  for (tfs = fshead; tfs; tfs = tfs->nextfs)
153  rewrite(globals, tfs);
154 
155  (void)next(globals, argv);
156  display(globals);
157  exit(exitval);
158  return exitval;
159 }
