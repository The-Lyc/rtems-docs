369 static rtems_status_code remember_event(
370  rtems_media_event event,
371  rtems_media_state state,
372  const char *src,
373  const char *dest
374 )
375 {
376  switch (event) {
377  case RTEMS_MEDIA_EVENT_DISK_ATTACH:
378  if (is_add_state(state)) {
379  add_disk(dest);
380  }
381  break;
382  case RTEMS_MEDIA_EVENT_PARTITION_ATTACH:
383  if (is_add_state(state)) {
384  add_partition(src, dest);
385  }
386  break;
387  case RTEMS_MEDIA_EVENT_MOUNT:
388  if (is_add_state(state)) {
389  add_mount_point(src, dest);
390  }
391  break;
392  case RTEMS_MEDIA_EVENT_UNMOUNT:
393  if (is_remove_state(state)) {
394  remove_mount_point(src);
395  }
396  break;
397  case RTEMS_MEDIA_EVENT_PARTITION_DETACH:
398  if (is_remove_state(state)) {
399  remove_partition(src);
400  }
401  break;
402  case RTEMS_MEDIA_EVENT_DISK_DETACH:
403  if (is_remove_state(state)) {
404  remove_disk(src);
405  }
406  break;
407  default:
408  break;
409  }
410 
411  return RTEMS_SUCCESSFUL;
412 }

414 static rtems_status_code process_event(
415  rtems_media_event event,
416  const char *src,
417  char **dest_ptr,
418  rtems_media_worker worker,
419  void *worker_arg
420 )
421 {
422  rtems_status_code sc = RTEMS_SUCCESSFUL;
423  rtems_media_state state = RTEMS_MEDIA_STATE_FAILED;
424  char *dest = NULL;
425 
426  sc = notify(event, RTEMS_MEDIA_STATE_INQUIRY, src, NULL);
427  if (sc == RTEMS_SUCCESSFUL) {
428  state = RTEMS_MEDIA_STATE_READY;
429  } else {
430  state = RTEMS_MEDIA_STATE_ABORTED;
431  }
432 
433  sc = (*worker)(state, src, &dest, worker_arg);
434  if (state == RTEMS_MEDIA_STATE_READY) {
435  if (sc == RTEMS_SUCCESSFUL) {
436  state = RTEMS_MEDIA_STATE_SUCCESS;
437  } else {
438  state = RTEMS_MEDIA_STATE_FAILED;
439  }
440  }
441 
442  notify(event, state, src, dest);
443  remember_event(event, state, src, dest);
444 
445  if (state == RTEMS_MEDIA_STATE_SUCCESS) {
446  sc = RTEMS_SUCCESSFUL;
447  } else if (state == RTEMS_MEDIA_STATE_ABORTED) {
448  sc = RTEMS_UNSATISFIED;
449  } else {
450  sc = RTEMS_IO_ERROR;
451  }
452 
453  if (dest_ptr != NULL && sc == RTEMS_SUCCESSFUL) {
454  *dest_ptr = dest;
455  } else {
456  free(dest);
457  }
458 
459  return sc;
460 }

462 static rtems_status_code mount_worker(
463  rtems_media_state state,
464  const char *src,
465  char **dest,
466  void *worker_arg
467 )
468 {
469  int rv = 0;
470 
471  if (state == RTEMS_MEDIA_STATE_READY) {
472  rtems_dosfs_mount_options mount_options;
473  char *mount_path = NULL;
474 
475  if (worker_arg == NULL) {
476  mount_path = rtems_media_replace_prefix(RTEMS_MEDIA_MOUNT_BASE, src);
477  } else {
478  mount_path = strdup(worker_arg);
479  }
480 
481  if (mount_path == NULL) {
482  return RTEMS_IO_ERROR;
483  }
484 
485  rv = rtems_mkdir(mount_path, S_IRWXU | S_IRWXG | S_IRWXO);
486  if (rv != 0) {
487  free(mount_path);
488 
489  return RTEMS_IO_ERROR;
490  }
491 
492  memset(&mount_options, 0, sizeof(mount_options));
493 
494  /* In case this fails, we fall back to use the default converter */
495  mount_options.converter = rtems_dosfs_create_utf8_converter("CP850");
496 
497  rv = mount(
498  src,
499  mount_path,
500  RTEMS_FILESYSTEM_TYPE_DOSFS,
501  RTEMS_FILESYSTEM_READ_WRITE,
502  &mount_options
503  );
504  if (rv != 0) {
505  rmdir(mount_path);
506  free(mount_path);
507  (*mount_options.converter->handler->destroy)(mount_options.converter);
508 
509  return RTEMS_IO_ERROR;
510  }
511 
512  *dest = mount_path;
513  }
514 
515  return RTEMS_SUCCESSFUL;
516 }
