62 void USART_SetBaudrate(Usart *pUsart,
63  uint8_t OverSamp,
64  uint32_t baudrate,
65  uint32_t masterClock)
66 {
67  unsigned int CD, FP, BaudError, ActualBaudRate;
68  /* Configure baudrate*/
69  BaudError = 10;
70  OverSamp = 0;
71 #ifdef __rtems__
72  CD = 0;
73  FP = 0;
74 #endif /* __rtems__ */
75 
76  /*Asynchronous*/
77  if ((pUsart->US_MR & US_MR_SYNC) == 0) {
78  /* 7816 mode */
79  if (((pUsart->US_MR & US_MR_USART_MODE_IS07816_T_0)
80  == US_MR_USART_MODE_IS07816_T_0)
81  || ((pUsart->US_MR & US_MR_USART_MODE_IS07816_T_1)
82  == US_MR_USART_MODE_IS07816_T_1)) {
83  /* Define the baud rate divisor register */
84  /* CD = MCK / SCK */
85  /* SCK = FIDI x BAUD = 372 x 9600 */
86  /* BOARD_MCK */
87  /* CD = MCK/(FIDI x BAUD) = 150000000 / (372x9600) = 42 */
88  CD = masterClock / (pUsart->US_FIDI * baudrate);
89  FP = 0;
90  } else {
91  while (BaudError > 5) {
92  CD = (masterClock / (baudrate * 8 * (2 - OverSamp)));
93  FP = ((masterClock / (baudrate * (2 - OverSamp))) - CD * 8);
94  ActualBaudRate = (masterClock / (CD * 8 + FP)) / (2 - OverSamp);
95  BaudError = (100 - ((baudrate * 100 / ActualBaudRate)));
96 
97  if (BaudError > 5) {
98  OverSamp++;
99 
100  if (OverSamp >= 2) {
101  TRACE_ERROR("Canont set this baudrate \n\r");
102  break;
103  }
104  }
105  }
106  }
107  }
108 
109  /*Synchronous SPI */
110  if ((pUsart->US_MR & US_MR_USART_MODE_SPI_MASTER)
111  == US_MR_USART_MODE_SPI_MASTER
112  || ((pUsart->US_MR & US_MR_SYNC) == US_MR_SYNC)) {
113  if ((pUsart->US_MR & US_MR_USCLKS_Msk) == US_MR_USCLKS_MCK) {
114  CD = masterClock / baudrate;
115  FP = ((masterClock / baudrate) - CD);
116  }
117  }
118 
119  pUsart->US_BRGR = (US_BRGR_CD(CD) | US_BRGR_FP(FP));
120 
121  /* Configure OverSamp*/
122  pUsart->US_MR |= (OverSamp << 19);
123 }

134 void USART_Configure(Usart *pUsart,
135  uint32_t mode,
136  uint32_t baudrate,
137  uint32_t masterClock)
138 {
139 
140  /* Reset and disable receiver & transmitter*/
141  pUsart->US_CR = US_CR_RSTRX | US_CR_RSTTX
142  | US_CR_RXDIS | US_CR_TXDIS | US_CR_RSTSTA;
143  pUsart->US_IDR = 0xFFFFFFFF;
144 
145  pUsart->US_MR = mode;
146  /* Configure baudrate*/
147  USART_SetBaudrate(pUsart, 0, baudrate, masterClock);
148 
149  /* Enable receiver and transmitter */
150  pUsart->US_CR = US_CR_RXEN | US_CR_TXEN;
151 
152  /* Disable buffering for printf(). */
153 #if (defined (__GNUC__) && !defined (__SAMBA__))
154  setvbuf(stdout, (char *)NULL, _IONBF, 0);
155 #endif
156 
157 }

278 void USART_Write(Usart *pUsart, uint16_t data, volatile uint32_t timeOut)
279 {
280  if (timeOut == 0) {
281  while ((pUsart->US_CSR & US_CSR_TXEMPTY) == 0);
282  } else {
283  while ((pUsart->US_CSR & US_CSR_TXEMPTY) == 0) {
284  if (timeOut == 0) {
285  TRACE_ERROR("USART_Write: Timed out.\n\r");
286  return;
287  }
288 
289  timeOut--;
290  }
291  }
292 
293  pUsart->US_THR = data;
294 }

344 void USART_PutChar(Usart *pUsart, uint8_t c)
345 {
346  /* Wait for the transmitter to be ready*/
347  while ((pUsart->US_CSR & US_CSR_TXEMPTY) == 0);
348 
349  /* Send character*/
350  pUsart->US_THR = c;
351 
352  /* Wait for the transfer to complete*/
353  while ((pUsart->US_CSR & US_CSR_TXEMPTY) == 0);
354 }
