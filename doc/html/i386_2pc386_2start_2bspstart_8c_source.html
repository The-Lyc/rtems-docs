50 static void bsp_pci_initialize_helper(void)
51 {
52  const pci_config_access_functions *pci_accessors;
53 
54  pci_accessors = pci_bios_initialize();
55  if (pci_accessors != NULL) {
56  printk("PCI bus: using PCI BIOS interface\n");
57  BSP_pci_configuration.pci_functions = pci_accessors;
58  return;
59  }
60 
61  pci_accessors = pci_io_initialize();
62  if (pci_accessors != NULL) {
63  printk("PCI bus: using PCI I/O interface\n");
64  BSP_pci_configuration.pci_functions = pci_accessors;
65  return;
66  }
67 
68  printk("PCI bus: could not initialize PCI BIOS interface\n");
