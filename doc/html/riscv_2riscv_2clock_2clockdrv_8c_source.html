58 static void riscv_clock_write_mtimecmp(
59  volatile RISCV_CLINT_timer_reg *mtimecmp,
60  uint64_t value
61 )
62 {
63 #if __riscv_xlen == 32
64  mtimecmp->val_32[0] = 0xffffffff;
65  mtimecmp->val_32[1] = (uint32_t) (value >> 32);
66  mtimecmp->val_32[0] = (uint32_t) value;
67 #elif __riscv_xlen == 64
68  mtimecmp->val_64 = value;
69 #endif
70 }

72 static uint64_t riscv_clock_read_mtime(volatile RISCV_CLINT_timer_reg *mtime)
73 {
74 #if __riscv_xlen == 32
75  uint32_t low;
76  uint32_t high_0;
77  uint32_t high_1;
78 
79  do {
80  high_0 = mtime->val_32[1];
81  low = mtime->val_32[0];
82  high_1 = mtime->val_32[1];
83  } while (high_0 != high_1);
84 
85  return (((uint64_t) high_0) << 32) | low;
86 #elif __riscv_xlen == 64
87  return mtime->val_64;
88 #endif
89 }

100 
101  riscv_clock_write_mtimecmp(&clint->mtimecmp[0], value);

130 static uint32_t riscv_clock_get_timebase_frequency(const void *fdt)
131 {
132  int node;
133  const fdt32_t *val;
134  int len=0;
135 
136  node = fdt_path_offset(fdt, "/cpus");
137 
138  val = (fdt32_t *) fdt_getprop(fdt, node, "timebase-frequency", &len);
139 
140  if (val == NULL || len < 4) {
141  int cpu0 = fdt_subnode_offset(fdt, node, "cpu@0");
142  val = (fdt32_t *) fdt_getprop(fdt, cpu0, "timebase-frequency", &len);
143 
144  if (val == NULL || len < 4) {
145  bsp_fatal(RISCV_FATAL_NO_TIMEBASE_FREQUENCY_IN_DEVICE_TREE);
146  }
147  }
148  return fdt32_to_cpu(*val);
149 }

151 static void riscv_clock_initialize(void)
152 {
153  const char *fdt;
154  riscv_timecounter *tc;
155  volatile RISCV_CLINT_regs *clint;
156  uint32_t tb_freq;
157  uint64_t us_per_tick;
158  uint32_t interval;
159 
160  fdt = bsp_fdt_get();
161  tb_freq = riscv_clock_get_timebase_frequency(fdt);
162  us_per_tick = rtems_configuration_get_microseconds_per_tick();
163  interval = (uint32_t) ((tb_freq * us_per_tick) / 1000000);
164  clint = riscv_clint;
165  tc = &riscv_clock_tc;
166 
167  tc->clint = clint;
168  tc->interval = interval;
169 
170  riscv_clock_write_mtimecmp(
171  &clint->mtimecmp[0],
172  riscv_clock_read_mtime(&clint->mtime) + interval
173  );
174 
175  /* Enable mtimer interrupts */
176  set_csr(mie, MIP_MTIP);
177 
178  /* Initialize timecounter */
179  tc->base.tc_get_timecount = riscv_clock_get_timecount;
180  tc->base.tc_counter_mask = 0xffffffff;
181  tc->base.tc_frequency = tb_freq;
182  tc->base.tc_quality = RTEMS_TIMECOUNTER_QUALITY_CLOCK_DRIVER;
183  rtems_timecounter_install(&tc->base);
184 }
