531 /********** Note About Coherency ****************/
532 
533 #ifdef SW_COHERENCY
534 #error "SW_COHERENCY not implemented"
535 /* Note: maintaining BD coherency in software is not trivial
536  * because BDs are smaller than a cache line;
537  * we cannot pad a BD to the length of a cache line because
538  * the TSEC assumes BDs layed out sequentially in memory.
539  * We cannot use zero-length BDs to pad to a full cache
540  * line either because the manual says that the length
541  * field of a TX BD must not be zero.
542  *
543  * We probably would need MMU resources to map BDs
544  * as non-cachable.
545  *
546  * Maintaining buffer coherency would be easier:
547  * - make RX buffers cache aligned so we can
548  * invalidate them w/o overlapping other data.
549  * - TX buffers may be flushed to memory. If cache
550  * lines overlap anything else (besides TX buffers)
551  * then that would only be harmful if (part of) a
552  * TX buffer would share a cache line with another
553  * type of DMA buffer that is written by another
554  * master. Note that BDs have exactly this problem;

829  * that have been sent to the cleanup_tx() callback.
830  * This routine is called from BSP_tsec_send_buf(), BSP_tsec_init_hw(),
831  * BSP_tsec_stop_hw().

836 int
837 BSP_tsec_swipe_tx(struct tsec_private *mp)
838 {
839 int rval = 0;
840 int i;
841 TSEC_BD *bd;
842 uint16_t flags;
843 void *u;
844 
845 #if DEBUG > 2
846 printf("Swipe TX entering:\n");
847 tsec_dump_tring(mp);
848 #endif
849 
850  for ( i = mp->tx_head; bd_rdbuf( (bd = &mp->tx_ring[i]) ); i = NEXT_TXI(mp, i) ) {
851 
852  flags = bd_rdfl( bd );
853  if ( (TSEC_TXBD_R & flags) ) {
854  /* nothing more to clean */
855  break;
856  }
857 
858  /* tx_ring_user[i] is only set on the last descriptor in a chain;
859  * we only count errors in the last descriptor;
860  */
861  if ( (u=mp->tx_ring_user[i]) ) {
862  mp->cleanup_txbuf(u, mp->cleanup_txbuf_arg, (flags & TSEC_TXBD_ERRS));
863  mp->tx_ring_user[i] = 0;
864  }
865 
866  bd_wrbuf( bd, 0 );
867 
868  mp->tx_avail++;
869 
870  rval++;
871  }
872  mp->tx_head = i;
873 
874 #if DEBUG > 2
875 tsec_dump_tring(mp);
876 printf("Swipe TX leaving\n");
877 #endif
878 
879  return rval;
880 }

908 
909  /* Do graceful stop (if not in stop condition already) */
910  if ( ! (TSEC_DMACTRL_GTS & fec_rd(b, TSEC_DMACTRL)) ) {
911  /* Make sure GTSC is clear */
912  fec_wr(b, TSEC_IEVENT, TSEC_IEVENT_GTSC);

923

930  /* wait */;
931  }

934  fec_clr(b, TSEC_MACCFG1, TSEC_MACCFG1_SOFT_RESET);

937  fec_wr (b, TSEC_IEVENT, TSEC_IEVENT_ALL);
938 }

998 /*
999  * Setup an interface.
1000  * Allocates resources for descriptor rings and sets up the driver software structure.
1001  *
1002  * Arguments:
1003  * unit:
1004  * interface # (1..2). The interface must not be attached to BSD already.
1005  *
1006  * driver_tid:
1007  * ISR posts RTEMS event # ('unit' - 1) to task with ID 'driver_tid' and disables interrupts
1008  * from this interface.
1009  *
1010  * void (*cleanup_txbuf)(void *user_buf, void *cleanup_txbuf_arg, int error_on_tx_occurred):
1011  * Pointer to user-supplied callback to release a buffer that had been sent
1012  * by BSP_tsec_send_buf() earlier. The callback is passed 'cleanup_txbuf_arg'
1013  * and a flag indicating whether the send had been successful.
1014  * The driver no longer accesses 'user_buf' after invoking this callback.
1015  * CONTEXT: This callback is executed either by BSP_tsec_swipe_tx() or
1016  * BSP_tsec_send_buf(), BSP_tsec_init_hw(), BSP_tsec_stop_hw() (the latter
1017  * ones calling BSP_tsec_swipe_tx()).
1018  * void *cleanup_txbuf_arg:
1019  * Closure argument that is passed on to 'cleanup_txbuf()' callback;
1020  *
1021  * void *(*alloc_rxbuf)(int *p_size, uintptr_t *p_data_addr),
1022  * Pointer to user-supplied callback to allocate a buffer for subsequent
1023  * insertion into the RX ring by the driver.
1024  * RETURNS: opaque handle to the buffer (which may be a more complex object
1025  * such as an 'mbuf'). The handle is not used by the driver directly
1026  * but passed back to the 'consume_rxbuf()' callback.
1027  * Size of the available data area and pointer to buffer's data area
1028  * in '*psize' and '*p_data_area', respectively.
1029  * If no buffer is available, this routine should return NULL in which
1030  * case the driver drops the last packet and re-uses the last buffer
1031  * instead of handing it out to 'consume_rxbuf()'.
1032  * CONTEXT: Called when initializing the RX ring (BSP_tsec_init_hw()) or when
1033  * swiping it (BSP_tsec_swipe_rx()).
1034  *
1035  *
1036  * void (*consume_rxbuf)(void *user_buf, void *consume_rxbuf_arg, int len);
1037  * Pointer to user-supplied callback to pass a received buffer back to
1038  * the user. The driver no longer accesses the buffer after invoking this
1039  * callback (with 'len'>0, see below). 'user_buf' is the buffer handle
1040  * previously generated by 'alloc_rxbuf()'.
1041  * The callback is passed 'cleanup_rxbuf_arg' and a 'len'
1042  * argument giving the number of bytes that were received.
1043  * 'len' may be <=0 in which case the 'user_buf' argument is NULL.
1044  * 'len' == 0 means that the last 'alloc_rxbuf()' had failed,
1045  * 'len' < 0 indicates a receiver error. In both cases, the last packet
1046  * was dropped/missed and the last buffer will be re-used by the driver.
1047  * NOTE: the data are 'prefixed' with two bytes, i.e., the ethernet packet header
1048  * is stored at offset 2 in the buffer's data area. Also, the FCS (4 bytes)
1049  * is appended. 'len' accounts for both.
1050  * CONTEXT: Called from BSP_tsec_swipe_rx().
1051  * void *cleanup_rxbuf_arg:
1052  * Closure argument that is passed on to 'consume_rxbuf()' callback;
1053  *
1054  * rx_ring_size, tx_ring_size:
1055  * How many big to make the RX and TX descriptor rings. Note that the sizes
1056  * may be 0 in which case a reasonable default will be used.
1057  * If either ring size is < 0 then the RX or TX will be disabled.
1058  * Note that it is illegal in this case to use BSP_tsec_swipe_rx() or
1059  * BSP_tsec_swipe_tx(), respectively.
1060  *
1061  * irq_mask:
1062  * Interrupts to enable. OR of flags from above.
1063  *
1064  */

1493 static void
1494 hash_prog(struct tsec_private *mp, uint32_t tble, const uint8_t *enaddr, int accept)
1495 {
1496 uint8_t s;
1497 uint32_t reg, bit;
1498 
1499  s = ether_crc32_le(enaddr, ETHER_ADDR_LEN);
1500 
1501  /* bit-reverse */
1502  s = ((s&0x0f) << 4) | ((s&0xf0) >> 4);
1503  s = ((s&0x33) << 2) | ((s&0xcc) >> 2);
1504  s = ((s&0x55) << 1) | ((s&0xaa) >> 1);
1505 
1506  reg = tble + ((s >> (5-2)) & ~3);
1507  bit = 1 << (31 - (s & 31));
1508 
1509  if ( accept ) {
1510  if ( 0 == mp->mc_refcnt[s]++ )
1511  fec_set( mp->base, reg, bit );
1512  } else {
1513  if ( mp->mc_refcnt[s] > 0 && 0 == --mp->mc_refcnt[s] )
1514  fec_clr( mp->base, reg, bit );
1515  }
1516 }

1759 int
1760 BSP_tsec_send_buf(struct tsec_private *mp, void *m_head, void *data_p, int len)
1761 {

1764 register unsigned l,d,t;
1765 register struct mbuf *m1;
1766 int nmbs;
1767 int ismbuf = (len <= 0);
1768 
1769 #if DEBUG > 2
1770  printf("send entering...\n");
1771  tsec_dump_tring(mp);
1772 #endif
1773 /* Only way to get here is when we discover that the mbuf chain
1774  * is too long for the tx ring
1775  */
1776 startover:
1777 
1778  rval = 0;
1779 
1780 #ifdef PARANOIA
1781  assert(m_head);
1782 #endif
1783 
1784  /* if no descriptor is available; try to wipe the queue */
1785  if ( (mp->tx_avail < 1) && TSEC_CLEAN_ON_SEND(mp)<=0 ) {
1786  return -1;
1787  }
1788 
1789  t = mp->tx_tail;
1790 
1791 #ifdef PARANOIA
1792  assert( ! bd_rdbuf( &mp->tx_ring[t] ) );
1793  assert( ! mp->tx_ring_user[t] );
1794 #endif
1795 
1796  if ( ! (m1 = m_head) )
1797  return 0;
1798 
1799  if ( ismbuf ) {
1800  /* find first mbuf with actual data */
1801  while ( 0 == m1->m_len ) {
1802  if ( ! (m1 = m1->m_next) ) {
1803  /* end reached and still no data to send ?? */
1804  m_freem(m_head);
1805  return 0;
1806  }
1807  }
1808  }
1809 
1810  /* Don't use the first descriptor yet because BSP_tsec_swipe_tx()
1811  * needs bd->buf == NULL as a marker. Hence, we
1812  * start with the second mbuf and fill the first descriptor
1813  * last.
1814  */
1815 
1816  l = t;
1817  d = NEXT_TXI(mp,t);
1818 
1819  mp->tx_avail--;
1820 
1821  nmbs = 1;
1822  if ( ismbuf ) {
1823  register struct mbuf *m;
1824  for ( m=m1->m_next; m; m=m->m_next ) {
1825  if ( 0 == m->m_len )
1826  continue; /* skip empty mbufs */
1827 
1828  nmbs++;
1829 
1830  if ( mp->tx_avail < 1 && TSEC_CLEAN_ON_SEND(mp)<=0 ) {
1831  /* not enough descriptors; cleanup...
1832  * the first slot was never used, so we start
1833  * at mp->d_tx_h->next;
1834  */
1835  for ( l = NEXT_TXI(mp, t); l!=d; l=NEXT_TXI(mp, l) ) {
1836  bd = & mp->tx_ring[l];
1837 #ifdef PARANOIA
1838  assert( mp->tx_ring_user[l] == 0 );
1839 #endif
1840  bd_wrbuf( bd, 0 );
1841  bd_clrfl( bd, TSEC_TXBD_R | TSEC_TXBD_L );
1842 
1843  mp->tx_avail++;
1844  }
1845  mp->tx_avail++;
1846  if ( nmbs > TX_AVAILABLE_RING_SIZE(mp) ) {
1847  /* this chain will never fit into the ring */
1848  if ( nmbs > mp->stats.maxchain )
1849  mp->stats.maxchain = nmbs;
1850  mp->stats.repack++;
1851  if ( ! (m_head = repackage_chain(m_head)) ) {
1852  /* no cluster available */
1853  mp->stats.odrops++;
1854 #ifdef PARANOIA
1855  printf("send return 0\n");
1856  tsec_dump_tring(mp);
1857 #endif
1858  return 0;
1859  }
1860 #ifdef PARANOIA
1861  printf("repackaged; start over\n");
1862 #endif
1863  goto startover;
1864  }
1865 #ifdef PARANOIA
1866  printf("send return -1\n");
1867  tsec_dump_tring(mp);
1868 #endif
1869  return -1;
1870  }
1871 
1872  mp->tx_avail--;
1873 
1874 #ifdef PARANOIA
1875  assert( ! mp->tx_ring_user[d] );
1876  if ( d == t ) {
1877  tsec_dump_tring(mp);
1878  printf("l %i, d %i, t %i, nmbs %i\n", l,d,t, nmbs);
1879  } else
1880  assert( d != t );
1881  assert( ! bd_rdbuf( &mp->tx_ring[d] ) );
1882 #endif
1883 
1884  /* fill this slot */
1885  rval += tsec_assign_desc( &mp->tx_ring[d], mtod(m, uint32_t), m->m_len, TSEC_TXBD_R);
1886 
1887  FLUSH_BUF(mtod(m, uint32_t), m->m_len);
1888 
1889  l = d;
1890  d = NEXT_TXI(mp, d);
1891  }
1892 
1893  /* fill first slot - don't release to DMA yet */
1894  rval += tsec_assign_desc( &mp->tx_ring[t], mtod(m1, uint32_t), m1->m_len, 0);
1895 
1896 
1897  FLUSH_BUF(mtod(m1, uint32_t), m1->m_len);
1898 
1899  } else {
1900  /* fill first slot with raw buffer - don't release to DMA yet */
1901  rval += tsec_assign_desc( &mp->tx_ring[t], (uint32_t)data_p, len, 0);
1902 
1903  FLUSH_BUF( (uint32_t)data_p, len);
1904  }
1905 
1906  /* tag last slot; this covers the case where 1st==last */
1907  bd_setfl( &mp->tx_ring[l], TSEC_TXBD_L );
1908 
1909  /* mbuf goes into last desc */
1910  mp->tx_ring_user[l] = m_head;
1911 
1912  membarrier();
1913 
1914  /* turn over the whole chain by flipping ownership of the first desc */
1915  bd_setfl( &mp->tx_ring[t], TSEC_TXBD_R );
1916 
1917  membarrier();
1918 
1919 #if DEBUG > 2
1920  printf("send return (l=%i, t=%i, d=%i) %i\n", l, t, d, rval);
1921  tsec_dump_tring(mp);
1922 #endif
1923 
1924  /* notify the device */
1925  fec_wr( mp->base, TSEC_TSTAT, TSEC_TSTAT_THLT );
1926 
1927  /* Update softc */
1928  mp->stats.packet++;
1929  if ( nmbs > mp->stats.maxchain )
1930  mp->stats.maxchain = nmbs;
1931 
1932  /* remember new tail */
1933  mp->tx_tail = d;
1934 
1935  return rval; /* #bytes sent */
1936 }

1938 /*
1939  * Retrieve all received buffers from the RX ring, replacing them
1940  * by fresh ones (obtained from the alloc_rxbuf() callback). The
1941  * received buffers are passed to consume_rxbuf().
1942  *
1943  * RETURNS: number of buffers processed.
1944  */
1945 int
1946 BSP_tsec_swipe_rx(struct tsec_private *mp)
1947 {
1948 int rval = 0, err;
1949 unsigned i;
1950 uint16_t flags;
1951 TSEC_BD *bd;
1952 void *newbuf;
1953 int sz;
1954 uint16_t len;
1955 uintptr_t baddr;
1956 
1957  i = mp->rx_tail;
1958  bd = mp->rx_ring + i;
1959  flags = bd_rdfl( bd );
1960 
1961  while ( ! (TSEC_RXBD_E & flags) ) {
1962 
1963  /* err is not valid if not qualified by TSEC_RXBD_L */
1964  if ( ( err = (TSEC_RXBD_ERROR & flags) ) ) {
1965  /* make sure error is < 0 */
1966  err |= 0xffff0000;
1967  /* pass 'L' flag out so they can verify... */
1968  err |= (flags & TSEC_RXBD_L);
1969  }
1970 
1971 #ifdef PARANOIA
1972  assert( flags & TSEC_RXBD_L );
1973  assert( mp->rx_ring_user[i] );
1974 #endif
1975 
1976  if ( err || !(newbuf = mp->alloc_rxbuf(&sz, &baddr)) ) {
1977  /* drop packet and recycle buffer */
1978  newbuf = mp->rx_ring_user[i];
1979  mp->consume_rxbuf( 0, mp->consume_rxbuf_arg, err );
1980  } else {
1981  len = ld_be16( &bd->len );
1982 
1983 #ifdef PARANOIA
1984  assert( 0 == (baddr & (RX_BUF_ALIGNMENT-1)) );
1985  assert( len > 0 );
1986 #endif
1987 
1988  mp->consume_rxbuf( mp->rx_ring_user[i], mp->consume_rxbuf_arg, len );
1989 
1990  mp->rx_ring_user[i] = newbuf;
1991  st_be16( &bd->len, sz );
1992  bd_wrbuf( bd, baddr );
1993  }
1994 
1995  RTEMS_COMPILER_MEMORY_BARRIER();
1996 
1997  bd_wrfl( &mp->rx_ring[i], (flags & ~TSEC_RXBD_ERROR) | TSEC_RXBD_E );
1998 
1999  rval++;
2000 
2001  i = NEXT_RXI( mp, i );
2002  bd = mp->rx_ring + i;
2003  flags = bd_rdfl( bd );
2004  }
2005 
2006  fec_wr( mp->base, TSEC_RSTAT, TSEC_RSTAT_QHLT );
2007 
2008  mp->rx_tail = i;
2009  return rval;
2010 }
