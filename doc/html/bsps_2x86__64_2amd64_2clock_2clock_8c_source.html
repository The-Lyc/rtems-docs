54 /*
55  * When the CPUID instruction is executed with a source operand of 1 in the EAX
56  * register, bit 9 of the CPUID feature flags returned in the EDX register
57  * indicates the presence (set) or absence (clear) of a local APIC.
58  */
59 bool has_apic_support()
60 {
61  uint32_t eax, ebx, ecx, edx;
62  cpuid(1, &eax, &ebx, &ecx, &edx);
63  return (edx >> 9) & 1;
64 }

66 /*
67  * Initializes the APIC by hardware and software enabling it, and sets up the
68  * amd64_apic_base pointer that can be used as a 32-bit addressable array to
69  * access APIC registers.
70  */
71 void apic_initialize(void)
72 {
73  if ( !has_apic_support() ) {
74  printf("warning: cpuid claims no APIC support - trying anyway.\n");
75  }
76 
77  /*
78  * The APIC base address is a 36-bit physical address.
79  * We have identity-paging setup at the moment, which makes this simpler, but
80  * that's something to note since the variables below use virtual addresses.
81  *
82  * Bits 0-11 (inclusive) are 0, making the address page (4KiB) aligned.
83  * Bits 12-35 (inclusive) of the MSR point to the rest of the address.
84  */
85  uint64_t apic_base_msr = rdmsr(APIC_BASE_MSR);
86  amd64_apic_base = (uint32_t*) apic_base_msr;
87  amd64_apic_base = (uint32_t*) ((uintptr_t) amd64_apic_base & 0x0ffffff000);
88 
89  /* Hardware enable the APIC just to be sure */
90  wrmsr(
91  APIC_BASE_MSR,
92  apic_base_msr | APIC_BASE_MSR_ENABLE,
93  apic_base_msr >> 32
94  );
95 
96  DBG_PRINTF("APIC is at 0x%" PRIxPTR "\n", (uintptr_t) amd64_apic_base);
97  DBG_PRINTF(
98  "APIC ID at *0x%" PRIxPTR "=0x%" PRIx32 "\n",
99  (uintptr_t) &amd64_apic_base[APIC_REGISTER_APICID],
100  amd64_apic_base[APIC_REGISTER_APICID]
101  );
102 
103  DBG_PRINTF(
104  "APIC spurious vector register *0x%" PRIxPTR "=0x%" PRIx32 "\n",
105  (uintptr_t) &amd64_apic_base[APIC_REGISTER_SPURIOUS],
106  amd64_apic_base[APIC_REGISTER_SPURIOUS]
107  );
108 
109  /*
110  * Software enable the APIC by mapping spurious vector and setting enable bit.
111  */
112  uintptr_t old;
113  amd64_install_raw_interrupt(
114  BSP_VECTOR_SPURIOUS,
115  (uintptr_t) apic_spurious_handler,
116  &old
117  );
118  amd64_apic_base[APIC_REGISTER_SPURIOUS] =
119  APIC_SPURIOUS_ENABLE | BSP_VECTOR_SPURIOUS;
120 
121  DBG_PRINTF(
122  "APIC spurious vector register *0x%" PRIxPTR "=0x%" PRIx32 "\n",
123  (uintptr_t) &amd64_apic_base[APIC_REGISTER_SPURIOUS],
124  amd64_apic_base[APIC_REGISTER_SPURIOUS]
125  );
126 
127  /*
128  * The PIC may send spurious IRQs even when disabled, and without remapping
129  * IRQ7 would look like an exception.
130  */
131  pic_remap(PIC1_REMAP_DEST, PIC2_REMAP_DEST);
132  pic_disable();
133 }

153 uint32_t apic_timer_calibrate(void)
154 {
155  /* Configure APIC timer in one-shot mode to prepare for calibration */
156  amd64_apic_base[APIC_REGISTER_LVT_TIMER] = BSP_VECTOR_APIC_TIMER;
157  amd64_apic_base[APIC_REGISTER_TIMER_DIV] = APIC_TIMER_SELECT_DIVIDER;
158 
159  /* Enable the channel 2 timer gate and disable the speaker output */
160  uint8_t chan2_value = (inport_byte(PIT_PORT_CHAN2_GATE) | PIT_CHAN2_TIMER_BIT)
161  & ~PIT_CHAN2_SPEAKER_BIT;
162  outport_byte(PIT_PORT_CHAN2_GATE, chan2_value);
163 
164  /* Initialize PIT in one-shot mode on Channel 2 */
165  outport_byte(
166  PIT_PORT_MCR,
167  PIT_SELECT_CHAN2 | PIT_SELECT_ACCESS_LOHI |
168  PIT_SELECT_ONE_SHOT_MODE | PIT_SELECT_BINARY_MODE
169  );
170 
171  /*
172  * Disable interrupts while we calibrate for 2 reasons:
173  * - Writing values to the PIT should be atomic (for now, this is okay
174  * because we're the only ones ever touching the PIT ports, but an
175  * interrupt resetting the PIT could mess calibration up).
176  * - We need to make sure that no interrupts throw our synchronization of
177  * the APIC timer off.
178  */
179  amd64_disable_interrupts();
180 
181  /* Set PIT reload value */
182  uint32_t pit_ticks = PIT_CALIBRATE_TICKS;
183  uint8_t low_tick = pit_ticks & 0xff;
184  uint8_t high_tick = (pit_ticks >> 8) & 0xff;
185 
186  outport_byte(PIT_PORT_CHAN2, low_tick);
187  stub_io_wait();
188  outport_byte(PIT_PORT_CHAN2, high_tick);
189 
190  /* Start APIC timer's countdown */
191  const uint32_t apic_calibrate_init_count = 0xffffffff;
192 
193  /* Restart PIT by disabling the gated input and then re-enabling it */
194  chan2_value &= ~PIT_CHAN2_TIMER_BIT;
195  outport_byte(PIT_PORT_CHAN2_GATE, chan2_value);
196  chan2_value |= PIT_CHAN2_TIMER_BIT;
197  outport_byte(PIT_PORT_CHAN2_GATE, chan2_value);
198  amd64_apic_base[APIC_REGISTER_TIMER_INITCNT] = apic_calibrate_init_count;
199 
200  while ( pit_ticks <= PIT_CALIBRATE_TICKS ) {
201  /* Send latch command to read multi-byte value atomically */
202  outport_byte(PIT_PORT_MCR, PIT_SELECT_CHAN2);
203  pit_ticks = inport_byte(PIT_PORT_CHAN2);
204  pit_ticks |= inport_byte(PIT_PORT_CHAN2) << 8;
205  }
206  uint32_t apic_currcnt = amd64_apic_base[APIC_REGISTER_TIMER_CURRCNT];
207 
208  DBG_PRINTF("PIT stopped at 0x%" PRIx32 "\n", pit_ticks);
209 
210  /* Stop APIC timer to calculate ticks to time ratio */
211  amd64_apic_base[APIC_REGISTER_LVT_TIMER] = APIC_DISABLE;
212 
213  /* Get counts passed since we started counting */
214  uint32_t apic_ticks_per_sec = apic_calibrate_init_count - apic_currcnt;
215 
216  DBG_PRINTF(
217  "APIC ticks passed in 1/%d of a second: 0x%" PRIx32 "\n",
218  PIT_CALIBRATE_DIVIDER,
219  apic_ticks_per_sec
220  );
221 
222  /* We ran the PIT for a fraction of a second */
223  apic_ticks_per_sec = apic_ticks_per_sec * PIT_CALIBRATE_DIVIDER;
224 
225  /* Re-enable interrupts now that calibration is complete */
226  amd64_enable_interrupts();
227 
228  /* Confirm that the APIC timer never hit 0 and IRQ'd during calibration */
229  assert(Clock_driver_ticks == 0);
230  assert(apic_ticks_per_sec != 0 &&
231  apic_ticks_per_sec != apic_calibrate_init_count);
232 
233  DBG_PRINTF(
234  "CPU frequency: 0x%" PRIu64 "\nAPIC ticks/sec: 0x%" PRIu32 "\n",
235  /* Multiply to undo effects of divider */
236  (uint64_t) apic_ticks_per_sec * APIC_TIMER_DIVIDE_VALUE,
237  apic_ticks_per_sec
238  );
239 
240  return apic_ticks_per_sec;
241 }

243 void apic_timer_initialize(uint64_t desired_freq_hz)
244 {
245  uint32_t apic_ticks_per_sec = 0;
246  uint32_t apic_tick_collections[APIC_TIMER_NUM_CALIBRATIONS] = {0};
247  uint64_t apic_tick_total = 0;
248  for (uint32_t i = 0; i < APIC_TIMER_NUM_CALIBRATIONS; i++) {
249  apic_tick_collections[i] = apic_timer_calibrate();
250  apic_tick_total += apic_tick_collections[i];
251  }
252  apic_ticks_per_sec = apic_tick_total / APIC_TIMER_NUM_CALIBRATIONS;
253 
254  /*
255  * The APIC timer counter is decremented at the speed of the CPU bus
256  * frequency (and we use a frequency divider).
257  *
258  * This means:
259  * apic_ticks_per_sec = (cpu_bus_frequency / timer_divide_value)
260  *
261  * Therefore:
262  * reload_value = apic_ticks_per_sec / desired_freq_hz
263  */
264  uint32_t apic_timer_reload_value = apic_ticks_per_sec / desired_freq_hz;
265 
266  amd64_apic_base[APIC_REGISTER_LVT_TIMER] = BSP_VECTOR_APIC_TIMER | APIC_SELECT_TMR_PERIODIC;
267  amd64_apic_base[APIC_REGISTER_TIMER_DIV] = APIC_TIMER_SELECT_DIVIDER;
268  amd64_apic_base[APIC_REGISTER_TIMER_INITCNT] = apic_timer_reload_value;
269 }

271 void amd64_clock_driver_initialize(void)
272 {
273  uint64_t us_per_tick = rtems_configuration_get_microseconds_per_tick();
274  uint64_t irq_ticks_per_sec = 1000000 / us_per_tick;
275  DBG_PRINTF(
276  "us_per_tick = %d\nDesired frequency = %d irqs/sec\n",
277  us_per_tick,
278  irq_ticks_per_sec
279  );
280 
281  /* Setup and enable the APIC itself */
282  apic_initialize();
283  /* Setup and initialize the APIC timer */
284  apic_timer_initialize(irq_ticks_per_sec);
285 
286  amd64_clock_tc.tc_get_timecount = amd64_clock_get_timecount;
287  amd64_clock_tc.tc_counter_mask = 0xffffffff;
288  amd64_clock_tc.tc_frequency = irq_ticks_per_sec;
289  amd64_clock_tc.tc_quality = RTEMS_TIMECOUNTER_QUALITY_CLOCK_DRIVER;
290  rtems_timecounter_install(&amd64_clock_tc);
291 }
