50 /* On SMP use map table above relative to SMP Boot CPU (normally CPU0) */
51 static inline int bsp_irq_cpu(int irq)
52 {
53  /* protect from bad user configuration, default to boot cpu */
54  if (rtems_configuration_get_maximum_processors() <= GRLIB_irq_to_cpu[irq])
55  return GRLIB_Cpu_Index;

121 void bsp_interrupt_set_affinity(
122  rtems_vector_number vector,
123  const Processor_mask *affinity
124 )
125 {
126  uint32_t unmasked = 0;
127  uint32_t cpu_count = rtems_scheduler_get_processor_maximum();
128  uint32_t cpu_index;
129 
130  for (cpu_index = 0; cpu_index < cpu_count; ++cpu_index) {
131  if (_Processor_mask_Is_set(affinity, cpu_index)) {
132  GRLIB_Cpu_Unmask_interrupt(vector, cpu_index);
133  ++unmasked;
134  }
135  }
136 
137  if (unmasked > 1) {
138  GRLIB_Enable_interrupt_broadcast(vector);
139  } else {
140  GRLIB_Disable_interrupt_broadcast(vector);
141  }
142 }
