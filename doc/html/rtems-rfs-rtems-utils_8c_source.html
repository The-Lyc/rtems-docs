32 rtems_rfs_rtems_set_handlers (rtems_filesystem_location_info_t* loc,
33  rtems_rfs_inode_handle* inode)
34 {
35  uint16_t mode = rtems_rfs_inode_get_mode (inode);
36  loc->handlers = NULL;
37  if (RTEMS_RFS_S_ISDIR (mode))
38  loc->handlers = rtems_rfs_rtems_handlers (dir);
39  else if (RTEMS_RFS_S_ISCHR (mode) || RTEMS_RFS_S_ISBLK(mode))
40  loc->handlers = rtems_rfs_rtems_handlers (device);
41  else if (RTEMS_RFS_S_ISLNK (mode))
42  loc->handlers = rtems_rfs_rtems_handlers (link);
43  else if (RTEMS_RFS_S_ISREG (mode))
44  loc->handlers = rtems_rfs_rtems_handlers (file);
45  else
46  {
47  printf ("rtems-rfs: mode type unknown: %04x\n", mode);
48  return false;
49  }
50  return true;
51 }
