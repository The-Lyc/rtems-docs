135 /*
136  * bsp_spurious_initialize
137  *
138  * Install the spurious handler for most traps. Note that set_vector()
139  * will unmask the corresponding asynchronous interrupt, so the initial
140  * interrupt mask is restored after the handlers are installed.
141  */
142 
143 void bsp_spurious_initialize()
144 {
145  uint32_t trap;
146  uint32_t level;
147  uint32_t mask;
148 
149  level = sparc_disable_interrupts();
150  mask = LEON_REG.Interrupt_Mask;
151 
152  for ( trap=0 ; trap<256 ; trap++ ) {
153 
154  /*
155  * Skip window overflow, underflow, and flush as well as software
156  * trap 0,9,10 which we will use as a shutdown, IRQ disable, IRQ enable.
157  * Also avoid trap 0x70 - 0x7f which cannot happen and where some of the
158  * space is used to pass parameters to the program.
159  */
160 
161  if (( trap == 5 || trap == 6 ) ||
162 #if defined(SPARC_USE_LAZY_FP_SWITCH)
163  ( trap == 4 ) ||
164 #endif
165  (( trap >= 0x11 ) && ( trap <= 0x1f )) ||
166  (( trap >= 0x70 ) && ( trap <= 0x83 )) ||
167  ( trap == 0x80 + SPARC_SWTRAP_IRQDIS ) ||
168 #if defined(SPARC_USE_SYNCHRONOUS_FP_SWITCH)
169  ( trap == 0x80 + SPARC_SWTRAP_IRQDIS_FP ) ||
170 #endif
171  ( trap == 0x80 + SPARC_SWTRAP_IRQEN ))
172  continue;
173 
174  set_vector(
175  (rtems_isr_entry) bsp_spurious_handler,
176  SPARC_SYNCHRONOUS_TRAP( trap ),
177  1
178  );
179  }
180 
181  LEON_REG.Interrupt_Mask = mask;
182  sparc_enable_interrupts(level);
183 
184 }
