166 static int
167 parity(u_char c)
168 {
169  int i;
170 
171  i = c ^ (c >> 1) ^ (c >> 2) ^ (c >> 3) ^
172  (c >> 4) ^ (c >> 5) ^ (c >> 6) ^ (c >> 7);
173  return (i & 1);
174 }

176 static void
177 setup(rtems_shell_dd_globals* globals)
178 {
179  u_int cnt;
180  struct timeval tv;
181 
182  if (in.name == NULL) {
183  in.name = "stdin";
184  in.fd = STDIN_FILENO;
185  } else {
186  in.fd = open(in.name, O_RDONLY, 0);
187  if (in.fd == -1)
188  err(exit_jump, 1, "%s", in.name);
189  }
190 
191  getfdtype(globals, &in);
192 
193  if (files_cnt > 1 && !(in.flags & ISTAPE))
194  errx(exit_jump, 1, "files is not supported for non-tape devices");
195 
196  if (out.name == NULL) {
197  /* No way to check for read access here. */
198  out.fd = STDOUT_FILENO;
199  out.name = "stdout";
200  } else {
201 #define OFLAGS \
202  (O_CREAT | (ddflags & (C_SEEK | C_NOTRUNC) ? 0 : O_TRUNC))
203  out.fd = open(out.name, O_RDWR | OFLAGS, DD_DEFFILEMODE);
204  /*
205  * May not have read access, so try again with write only.
206  * Without read we may have a problem if output also does
207  * not support seeks.
208  */
209  if (out.fd == -1) {
210  out.fd = open(out.name, O_WRONLY | OFLAGS, DD_DEFFILEMODE);
211  out.flags |= NOREAD;
212  }
213  if (out.fd == -1)
214  err(exit_jump, 1, "%s", out.name);
215  }
216 
217  getfdtype(globals, &out);
218 
219  /*
220  * Allocate space for the input and output buffers. If not doing
221  * record oriented I/O, only need a single buffer.
222  */
223  if (!(ddflags & (C_BLOCK | C_UNBLOCK))) {
224  if ((in.db = malloc(out.dbsz + in.dbsz - 1)) == NULL)
225  err(exit_jump, 1, "input buffer");
226  out.db = in.db;
227  } else if ((in.db = malloc(MAX(in.dbsz, cbsz) + cbsz)) == NULL ||
228  (out.db = malloc(out.dbsz + cbsz)) == NULL)
229  err(exit_jump, 1, "output buffer");
230  in.dbp = in.db;
231  out.dbp = out.db;
232 
233  /* Position the input/output streams. */
234  if (in.offset)
235  pos_in(globals);
236  if (out.offset)
237  pos_out(globals);
238 
239  /*
240  * Truncate the output file. If it fails on a type of output file
241  * that it should _not_ fail on, error out.
242  */
243  if ((ddflags & (C_OF | C_SEEK | C_NOTRUNC)) == (C_OF | C_SEEK) &&
244  out.flags & ISTRUNC)
245  if (ftruncate(out.fd, out.offset * out.dbsz) == -1)
246  err(exit_jump, 1, "truncating %s", out.name);
247 
248  if (ddflags & (C_LCASE | C_UCASE | C_ASCII | C_EBCDIC | C_PARITY)) {
249  if (ctab != NULL) {
250  for (cnt = 0; cnt <= 0377; ++cnt)
251  casetab[cnt] = ctab[cnt];
252  } else {
253  for (cnt = 0; cnt <= 0377; ++cnt)
254  casetab[cnt] = cnt;
255  }
256  if ((ddflags & C_PARITY) && !(ddflags & C_ASCII)) {
257  /*
258  * If the input is not EBCDIC, and we do parity
259  * processing, strip input parity.
260  */
261  for (cnt = 200; cnt <= 0377; ++cnt)
262  casetab[cnt] = casetab[cnt & 0x7f];
263  }
264  if (ddflags & C_LCASE) {
265  for (cnt = 0; cnt <= 0377; ++cnt)
266  casetab[cnt] = tolower(casetab[cnt]);
267  } else if (ddflags & C_UCASE) {
268  for (cnt = 0; cnt <= 0377; ++cnt)
269  casetab[cnt] = toupper(casetab[cnt]);
270  }
271  if ((ddflags & C_PARITY)) {
272  /*
273  * This should strictly speaking be a no-op, but I
274  * wonder what funny LANG settings could get us.
275  */
276  for (cnt = 0; cnt <= 0377; ++cnt)
277  casetab[cnt] = casetab[cnt] & 0x7f;
278  }
279  if ((ddflags & C_PARSET)) {
280  for (cnt = 0; cnt <= 0377; ++cnt)
281  casetab[cnt] = casetab[cnt] | 0x80;
282  }
283  if ((ddflags & C_PAREVEN)) {
284  for (cnt = 0; cnt <= 0377; ++cnt)
285  if (parity(casetab[cnt]))
286  casetab[cnt] = casetab[cnt] | 0x80;
287  }
288  if ((ddflags & C_PARODD)) {
289  for (cnt = 0; cnt <= 0377; ++cnt)
290  if (!parity(casetab[cnt]))
291  casetab[cnt] = casetab[cnt] | 0x80;
292  }
293 
294  ctab = casetab;
295  }
296 
297  (void)gettimeofday(&tv, (struct timezone *)NULL);
298  st.start = tv.tv_sec + tv.tv_usec * 1e-6;
299 }

301 static void
302 getfdtype(rtems_shell_dd_globals* globals, IO *io)
303 {
304  struct stat sb;
305 #if RTEMS_REMOVED
306  int type;
307 #endif
308 
309  if (fstat(io->fd, &sb) == -1)
310  err(exit_jump, 1, "%s", io->name);
311  if (S_ISREG(sb.st_mode))
312  io->flags |= ISTRUNC;
313 #if RTEMS_REMOVED
314  if (S_ISCHR(sb.st_mode) || S_ISBLK(sb.st_mode)) {
315  if (ioctl(io->fd, FIODTYPE, &type) == -1) {
316  err(exit_jump, 1, "%s", io->name);
317  } else {
318  if (type & D_TAPE)
319  io->flags |= ISTAPE;
320  else if (type & (D_DISK | D_MEM))
321  io->flags |= ISSEEK;
322  if (S_ISCHR(sb.st_mode) && (type & D_TAPE) == 0)
323  io->flags |= ISCHR;
324  }
325  return;
326  }
327 #else
328  io->flags |= ISSEEK;
329 #endif
330  errno = 0;
331  if (lseek(io->fd, (off_t)0, SEEK_CUR) == -1 && errno == ESPIPE)
332  io->flags |= ISPIPE;
333  else
334  io->flags |= ISSEEK;
335 }

337 static void
338 dd_in(rtems_shell_dd_globals* globals)
339 {
340  ssize_t n;
341 
342  for (;;) {
343  switch (cpy_cnt) {
344  case -1: /* count=0 was specified */
345  return;
346  case 0:
347  break;
348  default:
349  if (st.in_full + st.in_part >= (uintmax_t)cpy_cnt)
350  return;
351  break;
352  }
353 
354  /*
355  * Zero the buffer first if sync; if doing block operations,
356  * use spaces.
357  */
358  if (ddflags & C_SYNC) {
359  if (ddflags & C_FILL)
360  memset(in.dbp, fill_char, in.dbsz);
361  else if (ddflags & (C_BLOCK | C_UNBLOCK))
362  memset(in.dbp, ' ', in.dbsz);
363  else
364  memset(in.dbp, 0, in.dbsz);
365  }
366 
367  n = read(in.fd, in.dbp, in.dbsz);
368  if (n == 0) {
369  in.dbrcnt = 0;
370  return;
371  }
372 
373  /* Read error. */
374  if (n == -1) {
375  /*
376  * If noerror not specified, die. POSIX requires that
377  * the warning message be followed by an I/O display.
378  */
379  if (!(ddflags & C_NOERROR))
380  err(exit_jump, 1, "%s", in.name);
381  warn("%s", in.name);
382  summary(globals);
383 
384  /*
385  * If it's a seekable file descriptor, seek past the
386  * error. If your OS doesn't do the right thing for
387  * raw disks this section should be modified to re-read
388  * in sector size chunks.
389  */
390  if (in.flags & ISSEEK &&
391  lseek(in.fd, (off_t)in.dbsz, SEEK_CUR))
392  warn("%s", in.name);
393 
394  /* If sync not specified, omit block and continue. */
395  if (!(ddflags & C_SYNC))
396  continue;
397 
398  /* Read errors count as full blocks. */
399  in.dbcnt += in.dbrcnt = in.dbsz;
400  ++st.in_full;
401 
402  /* Handle full input blocks. */
403  } else if ((size_t)n == in.dbsz) {
404  in.dbcnt += in.dbrcnt = n;
405  ++st.in_full;
406 
407  /* Handle partial input blocks. */
408  } else {
409  /* If sync, use the entire block. */
410  if (ddflags & C_SYNC)
411  in.dbcnt += in.dbrcnt = in.dbsz;
412  else
413  in.dbcnt += in.dbrcnt = n;
414  ++st.in_part;
415  }
416 
417  /*
418  * POSIX states that if bs is set and no other conversions
419  * than noerror, notrunc or sync are specified, the block
420  * is output without buffering as it is read.
421  */
422  if (ddflags & C_BS) {
423  out.dbcnt = in.dbcnt;
424  dd_out(globals, 1);
425  in.dbcnt = 0;
426  continue;
427  }
428 
429  if (ddflags & C_SWAB) {
430  if ((n = in.dbrcnt) & 1) {
431  ++st.swab;
432  --n;
433  }
434  swab(in.dbp, in.dbp, (size_t)n);
435  }
436 
437  in.dbp += in.dbrcnt;
438  (*cfunc)(globals);
439  }
440 }

446 static void
447 dd_close(rtems_shell_dd_globals* globals)
448 {
449  if (cfunc == def)
450  def_close(globals);
451  else if (cfunc == block)
452  block_close(globals);
453  else if (cfunc == unblock)
454  unblock_close(globals);
455  if (ddflags & C_OSYNC && out.dbcnt && out.dbcnt < out.dbsz) {
456  if (ddflags & C_FILL)
457  memset(out.dbp, fill_char, out.dbsz - out.dbcnt);
458  else if (ddflags & (C_BLOCK | C_UNBLOCK))
459  memset(out.dbp, ' ', out.dbsz - out.dbcnt);
460  else
461  memset(out.dbp, 0, out.dbsz - out.dbcnt);
462  out.dbcnt = out.dbsz;
463  }
464  if (out.dbcnt || pending)
465  dd_out(globals, 1);
466 }

468 void
469 dd_out(rtems_shell_dd_globals* globals, int force)
470 {
471  u_char *outp;
472  size_t cnt, i, n;
473  ssize_t nw;
474  static int warned;
475  int sparse;
476 
477  /*
478  * Write one or more blocks out. The common case is writing a full
479  * output block in a single write; increment the full block stats.
480  * Otherwise, we're into partial block writes. If a partial write,
481  * and it's a character device, just warn. If a tape device, quit.
482  *
483  * The partial writes represent two cases. 1: Where the input block
484  * was less than expected so the output block was less than expected.
485  * 2: Where the input block was the right size but we were forced to
486  * write the block in multiple chunks. The original versions of dd(1)
487  * never wrote a block in more than a single write, so the latter case
488  * never happened.
489  *
490  * One special case is if we're forced to do the write -- in that case
491  * we play games with the buffer size, and it's usually a partial write.
492  */
493  outp = out.db;
494  for (n = force ? out.dbcnt : out.dbsz;; n = out.dbsz) {
495  for (cnt = n;; cnt -= nw) {
496  sparse = 0;
497  if (ddflags & C_SPARSE) {
498  sparse = 1; /* Is buffer sparse? */
499  for (i = 0; i < cnt; i++)
500  if (outp[i] != 0) {
501  sparse = 0;
502  break;
503  }
504  }
505  if (sparse && !force) {
506  pending += cnt;
507  nw = cnt;
508  } else {
509  if (pending != 0) {
510  if (force)
511  pending--;
512  if (lseek(out.fd, pending, SEEK_CUR) ==
513  -1)
514  err(exit_jump, 2, "%s: seek error creating sparse file",
515  out.name);
516  if (force)
517  write(out.fd, outp, 1);
518  pending = 0;
519  }
520  if (cnt)
521  nw = write(out.fd, outp, cnt);
522  else
523  return;
524  }
525 
526  if (nw <= 0) {
527  if (nw == 0)
528  errx(exit_jump, 1, "%s: end of device", out.name);
529  if (errno != EINTR)
530  err(exit_jump, 1, "%s", out.name);
531  nw = 0;
532  }
533  outp += nw;
534  st.bytes += nw;
535  if ((size_t)nw == n) {
536  if (n != out.dbsz)
537  ++st.out_part;
538  else
539  ++st.out_full;
540  break;
541  }
542  ++st.out_part;
543  if ((size_t)nw == cnt)
544  break;
545  if (out.flags & ISTAPE)
546  errx(exit_jump, 1, "%s: short write on tape device",
547  out.name);
548  if (out.flags & ISCHR && !warned) {
549  warned = 1;
550  warnx("%s: short write on character device",
551  out.name);
552  }
553  }
554  if ((out.dbcnt -= n) < out.dbsz)
555  break;
556  }
557 
558  /* Reassemble the output block. */
559  if (out.dbcnt)
560  (void)memmove(out.db, out.dbp - out.dbcnt, out.dbcnt);
561  out.dbp = out.db + out.dbcnt;
562 }
