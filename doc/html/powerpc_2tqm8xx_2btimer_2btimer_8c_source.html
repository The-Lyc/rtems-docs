56 void
57 benchmark_timer_initialize (void)
58 {
59  /*
60  * Reset timers 1 and 2
61  */
62  m8xx.tgcr &= ~0x00FF;
63  m8xx.tcn1 = 0;
64  m8xx.tcn2 = 0;
65  m8xx.ter1 = 0xFFFF;
66  m8xx.ter2 = 0xFFFF;
67 
68  /*
69  * Cascade timers 1 and 2
70  */
71  m8xx.tgcr |= M8xx_TGCR_CAS2;
72 
73  /*
74  * Configure timers 1 and 2 to a single 32-bit, BUS_clock timer.
75  */
76  m8xx.tmr2 = (0 << 8) | 0x2;
77  m8xx.tmr1 = 0;
78 
79  /*
80  * Start the timers
81  */
82  m8xx.tgcr |= 0x0011;
83 }
