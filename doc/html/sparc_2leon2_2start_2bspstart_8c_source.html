37 /*
38  * set_snooping
39  *
40  * Read the data cache configuration register to determine if
41  * bus snooping is available. This is needed for some drivers so
42  * that they can select the most efficient copy routines.
43  */
44 static inline int set_snooping(void)
45 {
46  unsigned int tmp = *(unsigned int *)0x80000014; /* Cache control register */
47  return ((tmp>>23) & 1); /* Data cache snooping enabled */
48 }
