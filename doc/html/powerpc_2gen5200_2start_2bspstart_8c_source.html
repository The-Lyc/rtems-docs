120 void bsp_start(void)
121 {
122  /*
123  * Get CPU identification dynamically. Note that the get_ppc_cpu_type()
124  * function store the result in global variables so that it can be used
125  * later...
126  */
127  get_ppc_cpu_type();
128  get_ppc_cpu_revision();
129 
130  #if defined(HAS_UBOOT) && defined(SHOW_MORE_INIT_SETTINGS)
131  {
132  void dumpUBootBDInfo( bd_t * );
133  dumpUBootBDInfo( &bsp_uboot_board_info );
134  }
135  #endif
136 
137  cpu_init();
138 
139  if(get_ppc_cpu_revision() >= 0x2014) {
140  /* Special settings for MPC5200B (B variant) */
141  uint32_t xlb_cfg = mpc5200.config;
142 
143  /* XXX: The Freescale documentation for BSDIS seems to be wrong */
144  xlb_cfg |= XLB_CFG_BSDIS;
145 
146  xlb_cfg &= ~XLB_CFG_PLDIS;
147 
148  mpc5200.config = xlb_cfg;
149  }
150 
151  bsp_time_base_frequency = XLB_CLOCK / 4;
152  bsp_clicks_per_usec = (XLB_CLOCK/4000000);
153 
154  /* Initialize exception handler */
155  ppc_exc_cache_wb_check = 0;
156  ppc_exc_initialize();
157  ppc_exc_set_handler(ASM_ALIGN_VECTOR, ppc_exc_alignment_handler);
158 
159  /* Initialize interrupt support */
160  bsp_interrupt_initialize();
161 
162  /*
163  * If the BSP was built with IRQ benchmarking enabled,
164  * then intialize it.
165  */
166  #if (BENCHMARK_IRQ_PROCESSING == 1)
167  BSP_IRQ_Benchmarking_Reset();
168  #endif
169 
170  #ifdef SHOW_MORE_INIT_SETTINGS
171  printk("Exit from bspstart\n");
172  #endif
173 }
