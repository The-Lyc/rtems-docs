129 static rtems_status_code
130 rtems_nvdisk_crc16_gen_factors (uint16_t pattern)
131 {
132  uint32_t b;
133 
134  rtems_nvdisk_crc16_factor = malloc (sizeof (uint16_t) * 256);
135  if (!rtems_nvdisk_crc16_factor)
136  return RTEMS_NO_MEMORY;
137 
138  for (b = 0; b < 256; b++)
139  {
140  uint32_t i;
141  uint16_t v = b;
142  for (i = 8; i--;)
143  v = v & 1 ? (v >> 1) ^ pattern : v >> 1;
144  rtems_nvdisk_crc16_factor[b] = v & 0xffff;
145  }
146  return RTEMS_SUCCESSFUL;
147 }

468 {

476

479

481 
482 #if RTEMS_NVDISK_TRACE

484  block, dc->device, page, crc);
485 #endif
486 
487  ret = rtems_nvdisk_read_checksum (nvd, dc->device, page, &crc);
488 
489  if (ret)
490  return ret;
491 
492  if (crc == 0xffff)
493  {
494 #if RTEMS_NVDISK_TRACE
495  rtems_nvdisk_warning (nvd, "read-block: crc not set: %d", block);
496 #endif
497  memset (buffer, 0, nvd->block_size);
498  return 0;
499  }
500 
501  ret = rtems_nvdisk_read_page (nvd, dc->device, page + dc->pages_desc, buffer);
502 
503  if (ret)
504  return ret;
505 
506  cs = rtems_nvdisk_page_checksum (buffer, nvd->block_size);
507 
508  if (cs != crc)

510  rtems_nvdisk_error ("read-block: crc failure: %d: buffer:%04x page:%04x",

513  }

515  return 0;

678 {
679  rtems_nvdisk* nvd = rtems_disk_get_driver_data (dd);
680  rtems_blkdev_request* r = argp;
681 
682  if (nvd->device_count == 0)
683  {

685  return -1;
686  }
687 
688  errno = 0;
689 
690  rtems_mutex_lock (&nvd->lock);
691 
692  switch (req)

694  case RTEMS_BLKIO_REQUEST:
695  switch (r->req)

697  case RTEMS_BLKDEV_REQ_READ:
698  errno = rtems_nvdisk_read (nvd, r);
699  break;
700 
701  case RTEMS_BLKDEV_REQ_WRITE:
702  errno = rtems_nvdisk_write (nvd, r);
703  break;
704 
705  default:
706  errno = EINVAL;
707  break;
708  }
709  break;
710

718

720  rtems_blkdev_ioctl (dd, req, argp);
721  break;
722  }
723 
724  rtems_mutex_unlock (&nvd->lock);
725 
726  return errno == 0 ? 0 : -1;
