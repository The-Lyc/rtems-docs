68 static void Init_PLL (void)

70  unsigned int n;
71 
72  /* Configure PLL registers */
73  *((uint16_t*)PLL_LOCKCNT) = 0x1000;
74  *((uint16_t*)PLL_DIV) = PLL_CSEL|PLL_SSEL;
75  *((uint16_t*)PLL_CTL) = PLL_MSEL|PLL_DF;
76 
77  /* Commands to set PLL values */
78  __asm__ ("cli r0;");
79  __asm__ ("idle;");
80  __asm__ ("sti r0;");
81 
82  /* Delay for PLL stabilization */
83  for (n=0; n<200; n++) {}
84 }
