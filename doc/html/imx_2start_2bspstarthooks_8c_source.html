48  node = fdt_path_offset_namelen(
49  fdt,
50  memory_path,

52  );
53 
54  if (node >= 0) {
55  int len;
56  const void *val;
57 
58  val = fdt_getprop(fdt, node, "reg", &len);
59  if (len == 8) {
60  uint32_t begin;

64  size = fdt32_to_cpu(((fdt32_t *) val)[1]);

73  }
74  }
