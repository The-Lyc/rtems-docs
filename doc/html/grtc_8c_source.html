518 /* Reads as much as possible but not more than 'max' bytes from the TC receive buffer.
519  * Number of bytes put into 'buf' is returned.
520  */
521 static int grtc_hw_read_try(struct grtc_priv *pDev, char *buf, int max)
522 {
523  struct grtc_regs *regs = pDev->regs;
524  unsigned int rp, wp, asr, bufmax, rrp, rwp;
525  unsigned int upper, lower;
526  unsigned int count, cnt, left;
527 
528  FUNCDBG();
529 
530  if ( max < 1 )
531  return 0;
532 
533  rp = READ_REG(&regs->rp);
534  asr = READ_REG(&regs->asr);
535  bufmax = (asr & GRTC_ASR_RXLEN) >> GRTC_ASR_RXLEN_BIT;
536  bufmax = (bufmax+1) << 10; /* Convert from 1kbyte blocks into bytes */
537  wp = READ_REG(&regs->wp);
538 
539  /* Relative rp and wp */
540  rrp = rp - (asr & GRTC_ASR_BUFST);
541  rwp = wp - (asr & GRTC_ASR_BUFST);
542 
543  lower = grtc_hw_data_avail_lower(rrp,rwp,bufmax);
544  upper = grtc_hw_data_avail_upper(rrp,rwp,bufmax);
545 
546  DBG("grtc_hw_read_try: AVAIL: Lower: %d, Upper: %d\n",lower,upper);
547  DBG("grtc_hw_read_try: rp: 0x%x, rrp: 0x%x, wp: 0x%x, rwp: 0x%x, bufmax: %d\n, start: 0x%x\n",
548  rp,rrp,wp,rwp,bufmax,pDev->buffer);
549 
550  if ( (upper+lower) == 0 )
551  return 0;
552 
553  /* Count bytes will be read */
554  count = (upper+lower) > max ? max : (upper+lower);
555  left = count;
556 
557  /* Read from upper part of data buffer */
558  if ( upper > 0 ){
559  if ( left < upper ){
560  cnt = left;
561  }else{
562  cnt = upper; /* Read all upper data available */
563  }
564  DBG("grtc_hw_read_try: COPYING %d from upper\n",cnt);
565  /* Convert from Remote address (RP) into CPU Local address */
566  memcpy(buf, (void *)((rp - (unsigned int)pDev->buf_remote) + (unsigned int)pDev->buf), cnt);
567  buf += cnt;
568  left -= cnt;
569  }
570 
571  /* Read from lower part of data buffer */
572  if ( left > 0 ){
573  if ( left < lower ){
574  cnt = left;
575  }else{
576  cnt = lower; /* Read all lower data available */
577  }
578  DBG("grtc_hw_read_try: COPYING %d from lower\n",cnt);
579  memcpy(buf, (void *)pDev->buf, cnt);
580  buf += cnt;
581  left -= cnt;
582  }
583 
584  /* Update hardware RP pointer to tell hardware about new space available */
585  if ( (rp+count) >= ((asr&GRTC_ASR_BUFST)+bufmax) ){
586  regs->rp = (rp+count-bufmax);

625 {

627  unsigned int tmp;
628 
629  if ( !pDev->buf || (((unsigned int)pDev->buf & ~GRTC_ASR_BUFST) != 0) ||

634  }

637

639  pDev->overrun_condition = 0;

643 #endif

652  }

655  * In case of a previous overrun condition it could be in incorrect
656  * state (where rtems_semaphore_flush was used).
657  */
658  rtems_semaphore_obtain(pDev->sem_rx, RTEMS_NO_WAIT, 0);
659

678  * 1. Let hardware know about our DMA area (size and location)
679  * 2. Set DMA read/write posistions to zero.

694

728  * the timeout expires.

730 static int grtc_wait_data(struct grtc_priv *pDev, int count, rtems_interval timeout)
731 {
732  int avail;
733  int ret;
734  SPIN_IRQFLAGS(irqflags);
735 
736  FUNCDBG();
737 
738  if ( count < 1 )
739  return 0;
740 
741  SPIN_LOCK_IRQ(&pDev->devlock, irqflags);
742 
743  /* Enable interrupts when receiving CLTUs, Also clear old pending CLTUs store

747  pDev->regs->imr = READ_REG(&pDev->regs->imr) | GRTC_INT_CS;

760  * RTEMS_TIMEOUT = timeout expired, probably not enough data available

763  * RTEMS_OBJECT_WAS_DELETED, RTEMS_INVALID_ID = driver error.

765  SPIN_LOCK_IRQ(&pDev->devlock, irqflags);
766  }else{

768  }
769

771  pDev->regs->imr = READ_REG(&pDev->regs->imr) & ~GRTC_INT_CS;
772 
773  SPIN_UNLOCK_IRQ(&pDev->devlock, irqflags);
774

1236  * every data byte, in the DMA area, one control byte is stripped.
1237  */
1238 static int grtc_hw_copy(struct grtc_priv *pDev, unsigned char *buf, int max, int partial)
1239 {
1240  struct grtc_regs *regs = pDev->regs;
1241  unsigned int rp, wp, asr, bufmax, rrp, rwp;
1242  unsigned int upper, lower;
1243  unsigned int count, cnt, left;
1244  int ret, tot, tmp;
1245 
1246  FUNCDBG();
1247 
1248  if ( max < 1 )
1249  return 0;
1250 
1251  rp = READ_REG(&regs->rp);
1252  asr = READ_REG(&regs->asr);
1253  bufmax = (asr & GRTC_ASR_RXLEN) >> GRTC_ASR_RXLEN_BIT;
1254  bufmax = (bufmax+1) << 10; /* Convert from 1kbyte blocks into bytes */
1255  wp = READ_REG(&regs->wp);
1256 
1257  /* Relative rp and wp */
1258  rrp = rp - (asr & GRTC_ASR_BUFST);
1259  rwp = wp - (asr & GRTC_ASR_BUFST);
1260 
1261  lower = grtc_hw_data_avail_lower(rrp,rwp,bufmax) >> 1;
1262  upper = grtc_hw_data_avail_upper(rrp,rwp,bufmax) >> 1;
1263 
1264  DBG("grtc_hw_copy: AVAIL: Lower: %d, Upper: %d\n",lower,upper);
1265  DBG("grtc_hw_copy: rp: 0x%x, rrp: 0x%x, wp: 0x%x, rwp: 0x%x, bufmax: %d\n, start: 0x%x\n",
1266  rp,rrp,wp,rwp,bufmax,pDev->buf_remote);
1267 
1268  if ( (upper+lower) == 0 || (!partial && ((upper+lower)<max) ) )
1269  return 0;
1270 
1271  /* Count bytes will be read */
1272  count = (upper+lower) > max ? max : (upper+lower);
1273  left = count;
1274  tot = 0;
1275 
1276  /* Read from upper part of data buffer */
1277  if ( upper > 0 ){
1278  if ( left < upper ){
1279  cnt = left;
1280  }else{
1281  cnt = upper; /* Read all upper data available */
1282  }
1283  DBG("grtc_hw_copy: COPYING %d from upper\n",cnt);
1284  if ( (tot=grtc_copy((unsigned short *)((rp-(unsigned int)pDev->buf_remote)+(unsigned int)pDev->buf), buf, cnt)) != cnt ) {
1285  /* Failed to copy due to an receive error */
1286  DBG("grtc_hw_copy(upper): not all in DMA buffer (%d)\n",tot);
1287  count = tot;
1288  ret = -1;
1289  goto out;
1290  }
1291  buf += cnt;
1292  left -= cnt;
1293  }
1294 
1295  /* Read from lower part of data buffer */
1296  if ( left > 0 ){
1297  if ( left < lower ){
1298  cnt = left;
1299  }else{
1300  cnt = lower; /* Read all lower data available */
1301  }
1302  DBG("grtc_hw_copy: COPYING %d from lower\n",cnt);
1303  if ( (tmp=grtc_copy((unsigned short *)pDev->buf, buf, cnt)) != cnt ) {
1304  /* Failed to copy due to an receive error */
1305  DBG("grtc_hw_copy(lower): not all in DMA buffer (%d)\n",tot);
1306  count = tot+tmp;
1307  ret = -1;
1308  goto out;
1309  }
1310  buf += cnt;
1311  left -= cnt;
1312  }
1313  ret = count;
1314 
1315 out:
1316  count = count*2;
1317  /* Update hardware RP pointer to tell hardware about new space available */
1318  if ( (rp+count) >= ((asr&GRTC_ASR_BUFST)+bufmax) ){
1319  regs->rp = (rp+count-bufmax);
1320  } else {
1321  regs->rp = rp+count;
1322  }
1323 
1324  return ret;
1325 }

1340  *
1341  * Return Values

1343  * 1 - more to process, no free frames
1344  * 2 - more to process, frame received
1345  * negative - more to process, frame dropped
1346  */
1347 static int process_dma(struct grtc_priv *pDev)
1348 {
1349  int ret, err;
1350  int left, total_len;
1351  unsigned char *dst;
1352  struct grtc_frame *frm;
1353 
1354  switch( pDev->frame_state ) {
1355  case FRM_STATE_NONE:
1356  DBG2("FRAME_STATE_NONE\n");
1357 
1358  /* Find Start of next frame by searching for 0x01 */
1359  ret = grtc_hw_find_frm(pDev);
1360  if ( ret != 0 ) {
1361  /* Frame start not found */
1362  return 0;
1363  }
1364 
1365  /* Start of frame found, Try to copy header */
1366  pDev->frm = NULL;
1367  pDev->frame_state = FRM_STATE_HDR;
1368 
1369  case FRM_STATE_HDR:
1370  DBG2("FRAME_STATE_HDR\n");
1371 
1372  /* Wait for all of header to be in place by setting partial to 0 */
1373  ret = grtc_hw_copy(pDev, (unsigned char *)pDev->hdr, 5, 0);
1374  if ( ret < 0 ) {
1375  /* Error copying header, restart scanning for new frame */
1376  DEBUG_ERR_LOG(pDev,1);
1377  pDev->stats.err++;
1378  pDev->stats.err_hdr++;
1379  DBG("FRAME_STATE_HDR: copying failed %d\n",ret);
1380  pDev->frame_state = FRM_STATE_NONE;
1381  return -1;
1382  } else if ( ret != 5 ) {
1383  DBG("FRAME_STATE_HDR: no header (%d)\n",ret);
1384  /* Not all bytes available, come back later */
1385  return 0;
1386  }
1387 
1388  /* The complete header has been copied, parse it */
1389  pDev->frmlen = (((unsigned short *)pDev->hdr)[1] & 0x3ff)+1;
1390  if ( pDev->frmlen < 5 ) {
1391  /* Error: frame length is not correct */
1392  pDev->stats.err++;
1393  pDev->stats.err_hdr++;
1394  DBG("FRAME_STATE_HDR: frame length error: %d\n", pDev->frmlen);
1395  pDev->frame_state = FRM_STATE_NONE;
1396  return -1;
1397  }
1398  pDev->frame_state = FRM_STATE_ALLOC;
1399 
1400  case FRM_STATE_ALLOC:
1401  DBG2("FRAME_STATE_ALLOC\n");
1402  /* Header has been read, allocate a frame to put payload and header into */
1403 
1404  /* Allocate Frame matching Frame length */
1405  err = 0;
1406  frm = grtc_pool_get_frm(pDev,pDev->frmlen,&err);
1407  if ( !frm ) {
1408  /* Couldn't find frame */
1409  DEBUG_ERR_LOG(pDev,2);
1410  pDev->stats.dropped++;
1411  DBG2("No free frames\n");
1412  if ( err == 0 ){
1413  /* Frame length exist in pool configuration, but no
1414  * frames are available for that frame length.
1415  */
1416  DEBUG_ERR_LOG(pDev,3);
1417  pDev->stats.dropped_no_buf++;
1418  return 1;
1419  } else {
1420  /* Frame length of incoming frame is larger than the
1421  * frame length in any of the configured frame pools.
1422  *
1423  * This may be because of an corrupt header. We simply
1424  * scan for the end of frame marker in the DMA buffer
1425  * so we can drop the frame.
1426  */
1427  DEBUG_ERR_LOG(pDev,4);
1428  pDev->stats.dropped_too_long++;
1429  pDev->frame_state = FRM_STATE_NONE;
1430  return -2;
1431  }
1432  }
1433  frm->len = 5; /* Only header currenlty in frame */
1434 
1435  /* Copy Frame Header into frame structure */
1436  ((unsigned char*)&frm->hdr)[0] = ((unsigned char*)pDev->hdr)[0];
1437  ((unsigned char*)&frm->hdr)[1] = ((unsigned char*)pDev->hdr)[1];
1438  ((unsigned char*)&frm->hdr)[2] = ((unsigned char*)pDev->hdr)[2];
1439  ((unsigned char*)&frm->hdr)[3] = ((unsigned char*)pDev->hdr)[3];
1440  ((unsigned char*)&frm->hdr)[4] = ((unsigned char*)pDev->hdr)[4];
1441 
1442  /* Calc Total and Filler byte count in frame */
1443  total_len = pDev->frmlen / 7;
1444  total_len = total_len * 7;
1445  if ( pDev->frmlen != total_len )
1446  total_len += 7;
1447 
1448  pDev->filler = total_len - pDev->frmlen;
1449 
1450  pDev->frame_state = FRM_STATE_PAYLOAD;
1451  pDev->frm = frm;
1452 
1453  case FRM_STATE_PAYLOAD:
1454  DBG2("FRAME_STATE_PAYLOAD\n");
1455  /* Parts of payload and the complete header has been read */
1456  frm = pDev->frm;
1457 
1458  dst = (unsigned char *)&frm->data[frm->len-5];
1459  left = pDev->frmlen-frm->len;
1460 
1461  ret = grtc_hw_copy(pDev,dst,left,1);
1462  if ( ret < 0 ) {
1463  DEBUG_ERR_LOG(pDev,5);
1464  /* Error copying header, restart scanning for new frame */
1465  pDev->frame_state = FRM_STATE_NONE;
1466  frm->next = NULL;
1467  grtc_pool_add_frms(frm);
1468  pDev->frm = NULL;
1469  pDev->stats.err++;
1470  pDev->stats.err_payload++;
1471  return -1;
1472  } else if ( ret != left ) {
1473  /* Not all bytes available, come back later */
1474  frm->len += ret;
1475  return 0;
1476  }
1477  frm->len += ret;
1478  pDev->frame_state = FRM_STATE_FILLER;
1479 
1480  case FRM_STATE_FILLER:
1481  DBG2("FRAME_STATE_FILLER\n");
1482  /* check filler data */
1483  frm = pDev->frm;
1484 
1485  ret = grtc_hw_check_ending(pDev,pDev->filler);
1486  if ( ret != 0 ) {
1487  /* Error in frame, drop frame */
1488  DEBUG_ERR_LOG(pDev,6);
1489  pDev->frame_state = FRM_STATE_NONE;
1490  frm->next = NULL;
1491  grtc_pool_add_frms(frm);
1492  pDev->frm = NULL;
1493  pDev->stats.err++;
1494  pDev->stats.err_ending++;
1495  return -1;
1496  }
1497 
1498  /* A complete frame received, put it into received frame queue */
1499  if ( pDev->ready.head ) {
1500  /* Queue not empty */
1501  pDev->ready.tail->next = frm;
1502  } else {
1503  /* Queue empty */
1504  pDev->ready.head = frm;
1505  }
1506  pDev->ready.tail = frm;
1507  frm->next = NULL;
1508  pDev->ready.cnt++;
1509  pDev->stats.frames_recv++;
1510 
1511  pDev->frame_state = FRM_STATE_NONE;
1512  frm->next = NULL;
1513  return 2;
1514 
1515 #if 0
1516  case FRM_STATE_DROP:
1517  DBG2("FRAME_STATE_DROP\n");
1518  break;
1519 #endif
1520 
1521  default:
1522  printk("GRTC: internal error\n");
1523  pDev->frame_state = FRM_STATE_NONE;
1524  break;
1525  }
1526 
1527  return 0;
1528 }
