156 STATIC int spwtdp_init2(struct drvmgr_dev *dev)
157 {
158  int status;
159  struct spwtdp_priv *priv;
160 
161  DBG("SPWTDP[%d] on bus %s\n", dev->minor_drv, dev->parent->dev->name);
162 
163  if (spwtdp_count >= SPWTDP_MAX)
164  return DRVMGR_ENORES;
165 
166  priv = dev->priv;
167  if (priv == NULL)
168  return DRVMGR_NOMEM;
169 
170  /* Register device */
171  priv->dev = dev;
172  priv->index = spwtdp_count;
173  priv_tab[priv->index] = priv;
174  snprintf(priv->devname, DEVNAME_LEN, "spwtdp%01u", priv->index);
175  spwtdp_count++;
176 
177  /* Initialize Semaphore */
178  if (rtems_semaphore_create(
179  rtems_build_name('S', 'T', 'P', '0' + priv->index), 1,
180  RTEMS_FIFO | RTEMS_SIMPLE_BINARY_SEMAPHORE | \
181  RTEMS_NO_INHERIT_PRIORITY | RTEMS_LOCAL | \
182  RTEMS_NO_PRIORITY_CEILING, 0, &priv->sem) != RTEMS_SUCCESSFUL) {
183  priv->sem = RTEMS_ID_NONE;
184  return DRVMGR_FAIL;
185  }
186 
187  /* Initialize SPWTDP Hardware */
188  status = spwtdp_init(priv);
189  if (status) {
190  printk("Failed to initialize spwtdp driver %d\n", status);
191  return -1;
192  }
193 
194  return DRVMGR_OK;

197 /* Hardware Reset of SPWTDP */
198 STATIC int spwtdp_hw_reset(struct spwtdp_priv *priv)
199 {
200  int i = 1000;
201  SPIN_IRQFLAGS(irqflags);
202 
203  SPIN_LOCK_IRQ(&priv->devlock, irqflags);
204 
205  /* Clear interrupts */
206  REG_WRITE(&priv->regs->ists, SPWTDP_IRQ_WCLEAR);
207 
208  /* Reset the SPWTDP core */
209  REG_WRITE(&priv->regs->conf[0], CONF0_RS);
210 
211  /* Wait for reset */
212  while ((REG_READ(&priv->regs->conf[0]) & CONF0_RS) && (i > 0)) {
213  i--;
214  }
215 
216  SPIN_UNLOCK_IRQ(&priv->devlock, irqflags);
217 
218  return ((i > 0)? SPWTDP_ERR_OK : SPWTDP_ERR_ERROR);
219 }

667 STATIC void spwtdp_isr(void *arg)
668 {
669  struct spwtdp_priv *priv = arg;
670  unsigned int ists = REG_READ(&priv->regs->ists);
671  SPIN_ISR_IRQFLAGS(irqflags);
672 
673  /* Return if the SPWTDP didn't generate the IRQ */
674  if (ists == 0)
675  return;
676 
677  SPIN_LOCK(&priv->devlock, irqflags);
678  REG_WRITE(&priv->regs->ists, ists); /* clear handled interrupt events */
679  SPIN_UNLOCK(&priv->devlock, irqflags);

951 int spwtdp_precision_get(void *spwtdp, uint8_t *fine, uint8_t *coarse)
952 {
953  struct spwtdp_priv *priv = (struct spwtdp_priv *)spwtdp;
954  int coarse_precision, fine_precision;
955 
956  if (priv == NULL) {
957  /* SPWTDP not initialized */
958  return SPWTDP_ERR_NOINIT;
959  }
960 
961  if (priv->open == 0)
962  return SPWTDP_ERR_EINVAL;
963 
964  unsigned int preamble = REG_READ(&priv->regs->dat_ctrl);
965 
966  if (preamble & 0x80) {
967  DBG("Pfield second extension set: unknown format");
968  return SPWTDP_ERR_ERROR;
969  }
970  if (!((preamble & 0x7000) == 0x2000 || (preamble & 0x7000) == 0x1000)) {
971  DBG(" PField indicates not unsegmented code: unknown format");
972  return SPWTDP_ERR_ERROR;
973  }
974  /*
975  * coarse_precision = 32;
976  * fine_precision = 24;
977  */
978  coarse_precision = ((preamble >> 10) & 0x3) + 1;
979  if (preamble & 0x80)
980  coarse_precision += (preamble >> 5) & 0x3;
981  fine_precision = (preamble >> 8) & 0x3;
982  if (preamble & 0x80)
983  fine_precision += (preamble >> 2) & 0x7;
984  if (coarse!=NULL)
985  *coarse = coarse_precision;
986  if (fine!=NULL)
987  *fine = fine_precision;
988 
989  return SPWTDP_ERR_OK;
990 }
