46  rtems_termios_device_context *base,
47  const char *buf,
48  size_t n
49 )
50 {
51  fe310_uart_context * ctx = (fe310_uart_context*) base;
52  size_t i;
53 
54  ctx->regs->div = riscv_get_core_frequency() / 115200 - 1;
55  ctx->regs->txctrl |= 1;
56  ctx->regs->rxctrl |= 1;
57 
58  for (i = 0; i < n; ++i) {
59  while ((ctx->regs->txdata & TXRXREADY) != 0) {
60  /* Wait */
61  }
62 
63  ctx->regs->txdata = buf[i];
64  }
65 }
