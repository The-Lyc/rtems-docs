31  * Allocate chain of free clusters from Files Allocation Table
32  *
33  * PARAMETERS:
34  * fs_info - FS info
35  * chain - the number of the first allocated cluster (first cluster
36  * in the chain)
37  * count - count of clusters to allocate (chain length)
38  *
39  * RETURNS:
40  * RC_OK on success, or error code if error occured (errno set
41  * appropriately)
42  *
43  *
44  */
45 int
46 fat_scan_fat_for_free_clusters(
47  fat_fs_info_t *fs_info,
48  uint32_t *chain,
49  uint32_t count,
50  uint32_t *cls_added,
51  uint32_t *last_cl,
52  bool zero_fill
53  )
54 {
55  int rc = RC_OK;
56  uint32_t cl4find = 2;
57  uint32_t save_cln = FAT_UNDEFINED_VALUE;
58  uint32_t data_cls_val = fs_info->vol.data_cls + 2;
59  uint32_t i = 2;
60 
61  if (fs_info->vol.next_cl - 2 < fs_info->vol.data_cls)
62  cl4find = fs_info->vol.next_cl;
63 
64  *cls_added = 0;
65 
66  /*
67  * fs_info->vol.data_cls is exactly the count of data clusters
68  * starting at cluster 2, so the maximum valid cluster number is
69  * (fs_info->vol.data_cls + 1)
70  */
71  while (*cls_added != count && i < data_cls_val)
72  {
73  uint32_t next_cln = 0;
74 
75  rc = fat_get_fat_cluster(fs_info, cl4find, &next_cln);
76  if ( rc != RC_OK )
77  {
78  if (*cls_added != 0)
79  fat_free_fat_clusters_chain(fs_info, (*chain));
80  return rc;
81  }
82 
83  if (next_cln == FAT_GENFAT_FREE)
84  {
85  /*
86  * We are enforced to process allocation of the first free cluster
87  * by separate 'if' statement because otherwise undo function
88  * wouldn't work properly
89  */
90  if (*cls_added == 0)
91  {
92  *chain = cl4find;
93  rc = fat_set_fat_cluster(fs_info, cl4find, FAT_GENFAT_EOC);
94  if ( rc != RC_OK )
95  {
96  /*
97  * this is the first cluster we tried to allocate so no
98  * cleanup activity needed
99  */
100  return rc;
101  }
102  }
103  else
104  {
105  /* set EOC value to new allocated cluster */
106  rc = fat_set_fat_cluster(fs_info, cl4find, FAT_GENFAT_EOC);
107  if ( rc != RC_OK )
108  {
109  /* cleanup activity */
110  fat_free_fat_clusters_chain(fs_info, (*chain));
111  return rc;
112  }
113 
114  rc = fat_set_fat_cluster(fs_info, save_cln, cl4find);
115  if ( rc != RC_OK )
116  goto cleanup;
117  }
118 
119  if (zero_fill)
120  {
121  ssize_t bytes_written =
122  fat_cluster_set(fs_info, cl4find, 0, fs_info->vol.bpc, 0);
123 
124  if (fs_info->vol.bpc != bytes_written)
125  {
126  rc = -1;
127  goto cleanup;
128  }
129  }
130 
131  save_cln = cl4find;
132  (*cls_added)++;
133  }
134  i++;
135  cl4find++;
136  if (cl4find >= data_cls_val)
137  cl4find = 2;
138  }
139 
140  *last_cl = save_cln;
141  fs_info->vol.next_cl = save_cln;
142 
143  if (fs_info->vol.free_cls != FAT_UNDEFINED_VALUE)
144  fs_info->vol.free_cls -= (*cls_added);
145 
146  fat_buf_release(fs_info);
147 
148  return RC_OK;
149 
150 cleanup:
151 
152  /* cleanup activity */
153  fat_free_fat_clusters_chain(fs_info, (*chain));
154 
155  /*
156  * Trying to save last allocated cluster for future use
157  *
158  * NOTE: Deliberately ignoring return value.
159  */
160  (void) fat_set_fat_cluster(fs_info, cl4find, FAT_GENFAT_FREE);
161  fat_buf_release(fs_info);
162  return rc;
163 }

165 /* fat_free_fat_clusters_chain --
166  * Free chain of clusters in Files Allocation Table.
167  *
168  * PARAMETERS:
169  * fs_info - FS info
170  * chain - number of the first cluster in the chain
171  *
172  * RETURNS:
173  * RC_OK on success, or -1 if error occured (errno set appropriately)
174  */
175 int
176 fat_free_fat_clusters_chain(
177  fat_fs_info_t *fs_info,
178  uint32_t chain
179  )
180 {
181  int rc = RC_OK, rc1 = RC_OK;
182  uint32_t cur_cln = chain;
183  uint32_t next_cln = 0;
184  uint32_t freed_cls_cnt = 0;
185 
186  while ((cur_cln & fs_info->vol.mask) < fs_info->vol.eoc_val)
187  {
188  rc = fat_get_fat_cluster(fs_info, cur_cln, &next_cln);
189  if ( rc != RC_OK )
190  {
191  if(fs_info->vol.free_cls != FAT_UNDEFINED_VALUE)
192  fs_info->vol.free_cls += freed_cls_cnt;
193 
194  fat_buf_release(fs_info);
195  return rc;
196  }
197 
198  rc = fat_set_fat_cluster(fs_info, cur_cln, FAT_GENFAT_FREE);
199  if ( rc != RC_OK )
200  rc1 = rc;
201 
202  freed_cls_cnt++;
203  cur_cln = next_cln;
204  }
205 
206  fs_info->vol.next_cl = chain;
207  if (fs_info->vol.free_cls != FAT_UNDEFINED_VALUE)
208  fs_info->vol.free_cls += freed_cls_cnt;
209 
210  fat_buf_release(fs_info);
211  if (rc1 != RC_OK)
212  return rc1;
213 
214  return RC_OK;
215 }

217 /* fat_get_fat_cluster --
218  * Fetches the contents of the cluster (link to next cluster in the chain)
219  * from Files Allocation Table.
220  *
221  * PARAMETERS:
222  * fs_info - FS info
223  * cln - number of cluster to fetch the contents from
224  * ret_val - contents of the cluster 'cln' (link to next cluster in
225  * the chain)
226  *
227  * RETURNS:
228  * RC_OK on success, or -1 if error occured
229  * and errno set appropriately
230  */
231 int
232 fat_get_fat_cluster(
233  fat_fs_info_t *fs_info,
234  uint32_t cln,
235  uint32_t *ret_val
236  )
237 {
238  int rc = RC_OK;
239  uint8_t *sec_buf;
240  uint32_t sec = 0;
241  uint32_t ofs = 0;
242 
243  /* sanity check */
244  if ( (cln < 2) || (cln > (fs_info->vol.data_cls + 1)) )
245  rtems_set_errno_and_return_minus_one(EIO);
246 
247  sec = (FAT_FAT_OFFSET(fs_info->vol.type, cln) >> fs_info->vol.sec_log2) +
248  fs_info->vol.afat_loc;
249  ofs = FAT_FAT_OFFSET(fs_info->vol.type, cln) & (fs_info->vol.bps - 1);
250 
251  rc = fat_buf_access(fs_info, sec, FAT_OP_TYPE_READ, &sec_buf);
252  if (rc != RC_OK)
253  return rc;
254 
255  switch ( fs_info->vol.type )
256  {
257  case FAT_FAT12:
258  /*
259  * we are enforced in complex computations for FAT12 to escape CPU
260  * align problems for some architectures
261  */
262  *ret_val = (*(sec_buf + ofs));
263  if ( ofs == (fs_info->vol.bps - 1) )
264  {
265  rc = fat_buf_access(fs_info, sec + 1, FAT_OP_TYPE_READ,
266  &sec_buf);
267  if (rc != RC_OK)
268  return rc;
269 
270  *ret_val |= *sec_buf << 8;
271  }
272  else
273  {
274  *ret_val |= *(sec_buf + ofs + 1) << 8;
275  }
276 
277  if ( FAT_CLUSTER_IS_ODD(cln) )
278  *ret_val = (*ret_val) >> FAT12_SHIFT;
279  else
280  *ret_val = (*ret_val) & FAT_FAT12_MASK;
281  break;
282 
283  case FAT_FAT16:
284  *ret_val = *((uint16_t *)(sec_buf + ofs));
285  *ret_val = CF_LE_W(*ret_val);
286  break;
287 
288  case FAT_FAT32:
289  *ret_val = *((uint32_t *)(sec_buf + ofs));
290  *ret_val = CF_LE_L(*ret_val);
291  break;
292 
293  default:
294  rtems_set_errno_and_return_minus_one(EIO);
295  break;
296  }
297 
298  return RC_OK;
299 }
