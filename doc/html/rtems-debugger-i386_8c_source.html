125 /*
126  * A function to get a segment register.
127  */
128 static inline uint32_t
129 get_seg_reg(size_t reg)
130 {
131  int v = 0;
132  switch (reg) {
133  case REG_CS:
134  GET_REG(CS, v);
135  break;
136  case REG_SS:
137  GET_REG(SS, v);
138  break;
139  case REG_DS:
140  GET_REG(DS, v);
141  break;
142  case REG_ES:
143  GET_REG(ES, v);
144  break;
145  case REG_FS:
146  GET_REG(FS, v);
147  break;
148  case REG_GS:
149  GET_REG(GS, v);
150  break;
151  }
152  return v & 0xffff;
153 }

205 static void
206 target_exception(CPU_Exception_frame* frame)
207 {
208  target_printk("[} frame = %08" PRIx32 " sig=%d (%" PRIx32 ")\n",
209  (uint32_t) frame,
210  rtems_debugger_target_exception_to_signal(frame),
211  frame->idtIndex);
212  target_printk("[} EAX = %" PRIx32 " EBX = %" PRIx32 \
213  " ECX = %" PRIx32 " EDX = %" PRIx32 "\n",
214  frame->eax, frame->ebx, frame->ecx, frame->edx);
215  target_printk("[} ESI = %" PRIx32 " EDI = %" PRIx32 \
216  " EBP = %" PRIx32 " ESP = %" PRIx32 "\n",
217  frame->esi, frame->edi, frame->ebp, frame->esp0);
218  target_printk("[} EIP = %" PRIx32"\n", frame->eip);
219 
220  frame->eflags &= ~EFLAGS_TRAP;
221 
222  switch (rtems_debugger_target_exception(frame)) {
223  case rtems_debugger_target_exc_consumed:
224  default:
225  break;
226  case rtems_debugger_target_exc_step:
227  frame->eflags |= EFLAGS_TRAP;
228  break;
229  case rtems_debugger_target_exc_cascade:
230  orig_currentExcHandler(frame);
231  break;
232  }
233 }

287 int
288 rtems_debugger_target_read_regs(rtems_debugger_thread* thread)
289 {
290  if (!rtems_debugger_thread_flag(thread,
291  RTEMS_DEBUGGER_THREAD_FLAG_REG_VALID)) {
292  size_t i;
293 
294  for (i = 0; i < rtems_debugger_target_reg_num(); ++i) {
295  if (rtems_debugger_is_int_reg(i))
296  rtems_debugger_set_int_reg(thread, i, 0xdeaddead);
297  }
298 
299  if (thread->frame) {
300  CPU_Exception_frame* frame = thread->frame;
301 
302  rtems_debugger_set_int_reg(thread, REG_EAX, frame->eax);
303  rtems_debugger_set_int_reg(thread, REG_ECX, frame->ecx);
304  rtems_debugger_set_int_reg(thread, REG_EDX, frame->edx);
305  rtems_debugger_set_int_reg(thread, REG_EBX, frame->ebx);
306  rtems_debugger_set_int_reg(thread, REG_ESP, frame->esp0);
307  rtems_debugger_set_int_reg(thread, REG_EBP, frame->ebp);
308  rtems_debugger_set_int_reg(thread, REG_ESI, frame->esi);
309  rtems_debugger_set_int_reg(thread, REG_EDI, frame->edi);
310  rtems_debugger_set_int_reg(thread, REG_EIP, frame->eip);
311  rtems_debugger_set_int_reg(thread, REG_EFLAGS, frame->eflags);
312  rtems_debugger_set_int_reg(thread, REG_CS, frame->cs);
313 
314  /*
315  * Get the signal from the frame.
316  */
317  thread->signal = rtems_debugger_target_exception_to_signal(frame);
318  }
319  else {
320  rtems_debugger_set_int_reg(thread, REG_EBX, thread->tcb->Registers.ebx);
321  rtems_debugger_set_int_reg(thread, REG_ESI, thread->tcb->Registers.esi);
322  rtems_debugger_set_int_reg(thread, REG_EDI, thread->tcb->Registers.edi);
323  rtems_debugger_set_int_reg(thread, REG_EFLAGS, thread->tcb->Registers.eflags);
324  rtems_debugger_set_int_reg(thread, REG_ESP, (intptr_t) thread->tcb->Registers.esp);
325  rtems_debugger_set_int_reg(thread, REG_EBP, (intptr_t) thread->tcb->Registers.ebp);
326  rtems_debugger_set_int_reg(thread, REG_EIP, *((DB_UINT*) thread->tcb->Registers.esp));
327  rtems_debugger_set_int_reg(thread, REG_EAX, (intptr_t) thread);
328 
329  rtems_debugger_set_int_reg(thread, REG_CS, get_seg_reg(REG_CS));
330 
331  /*
332  * Blocked threads have no signal.
333  */
334  thread->signal = 0;
335  }
336 
337  rtems_debugger_set_int_reg(thread, REG_SS, get_seg_reg(REG_SS));
338  rtems_debugger_set_int_reg(thread, REG_DS, get_seg_reg(REG_DS));
339  rtems_debugger_set_int_reg(thread, REG_ES, get_seg_reg(REG_ES));
340  rtems_debugger_set_int_reg(thread, REG_FS, get_seg_reg(REG_FS));
341  rtems_debugger_set_int_reg(thread, REG_GS, get_seg_reg(REG_GS));
342 
343  thread->flags |= RTEMS_DEBUGGER_THREAD_FLAG_REG_VALID;
344  thread->flags &= ~RTEMS_DEBUGGER_THREAD_FLAG_REG_DIRTY;
345  }
346 
347  return 0;
348 }

350 int
351 rtems_debugger_target_write_regs(rtems_debugger_thread* thread)
352 {
353  if (rtems_debugger_thread_flag(thread,
354  RTEMS_DEBUGGER_THREAD_FLAG_REG_DIRTY)) {
355  /*
356  * Only write to debugger controlled threads. Do not touch the registers
357  * for threads blocked in the context switcher.
358  */
359  if (rtems_debugger_thread_flag(thread,
360  RTEMS_DEBUGGER_THREAD_FLAG_EXCEPTION)) {
361  CPU_Exception_frame* frame = thread->frame;
362  frame->eax = rtems_debugger_get_int_reg(thread, REG_EAX);
363  frame->ecx = rtems_debugger_get_int_reg(thread, REG_ECX);
364  frame->edx = rtems_debugger_get_int_reg(thread, REG_EDX);
365  frame->ebx = rtems_debugger_get_int_reg(thread, REG_EBX);
366  frame->esp0 = rtems_debugger_get_int_reg(thread, REG_ESP);
367  frame->ebp = rtems_debugger_get_int_reg(thread, REG_EBP);
368  frame->esi = rtems_debugger_get_int_reg(thread, REG_ESI);
369  frame->edi = rtems_debugger_get_int_reg(thread, REG_EDI);
370  frame->eip = rtems_debugger_get_int_reg(thread, REG_EIP);
371  frame->eflags = rtems_debugger_get_int_reg(thread, REG_EFLAGS);
372  frame->cs = rtems_debugger_get_int_reg(thread, REG_CS);
373  }
374  thread->flags &= ~RTEMS_DEBUGGER_THREAD_FLAG_REG_DIRTY;
375  }
376  return 0;
377 }

413 int
414 rtems_debugger_target_thread_stepping(rtems_debugger_thread* thread)
415 {
416  if (rtems_debugger_thread_flag(thread,
417  (RTEMS_DEBUGGER_THREAD_FLAG_STEP |
418  RTEMS_DEBUGGER_THREAD_FLAG_STEPPING))) {
419  CPU_Exception_frame* frame = thread->frame;
420  /*
421  * Single step instructions with interrupts masked to avoid stepping into
422  * an interrupt handler.
423  */
424  if ((frame->eflags & EFLAGS_INTR_ENABLE) == 0)
425  thread->flags |= RTEMS_DEBUGGER_THREAD_FLAG_INTS_DISABLED;
426  else
427  frame->eflags &= ~EFLAGS_INTR_ENABLE;
428  frame->eflags |= EFLAGS_TRAP;
429  }
430  return 0;
431 }

433 int
434 rtems_debugger_target_exception_to_signal(CPU_Exception_frame* frame)
435 {
436  int sig = RTEMS_DEBUGGER_SIGNAL_HUP;
437  switch (frame->idtIndex) {
438  case 1: /* debug exception */
439  case 3: /* breakpoint int3 */
440  sig = RTEMS_DEBUGGER_SIGNAL_TRAP;
441  break;
442  case 4: /* int overflow */
443  case 5: /* out-of-bounds */
444  sig = RTEMS_DEBUGGER_SIGNAL_URG;
445  break;
446  case 6: /* invalid opcode */
447  sig = RTEMS_DEBUGGER_SIGNAL_ILL;
448  break;
449  case 8: /* double fault */
450  case 16: /* fp error */
451  sig = RTEMS_DEBUGGER_SIGNAL_EMT;
452  break;
453  case 0: /* divide by zero */
454  case 7: /* FPU not avail. */
455  sig = RTEMS_DEBUGGER_SIGNAL_FPE;
456  break;
457  case 9: /* i387 seg overr. */
458  case 10: /* Invalid TSS */
459  case 11: /* seg. not pres. */
460  case 12: /* stack except. */
461  case 13: /* general prot. */
462  case 14: /* page fault */
463  case 17: /* alignment check */
464  sig = RTEMS_DEBUGGER_SIGNAL_SEGV;
465  break;
466  case 2: /* NMI */
467  case 18: /* machine check */
468  sig = RTEMS_DEBUGGER_SIGNAL_BUS;
469  break;
470  default:
471  break;
472  }
473  return sig;
474 }
