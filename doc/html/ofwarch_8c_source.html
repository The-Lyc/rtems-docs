84 static int wake_cpus_in_node(phandle child, uint64_t current_mid,
85  uintptr_t physmem_start)
86 {
87  int cpus;
88 
89  for (cpus = 0; (child != 0) && (child != -1);
90  child = ofw_get_peer_node(child), cpus++) {
91  char type_name[OFW_TREE_PROPERTY_MAX_VALUELEN];
92 
93  if (ofw_get_property(child, "device_type", type_name,
94  OFW_TREE_PROPERTY_MAX_VALUELEN) > 0) {
95  type_name[OFW_TREE_PROPERTY_MAX_VALUELEN - 1] = 0;
96  if (strcmp(type_name, "cpu") == 0) {
97  uint32_t mid;
98 
99  /*
100  * "upa-portid" for US, "portid" for US-III,
101  * "cpuid" for US-IV
102  */
103  if ((ofw_get_property(child, "upa-portid", &mid, sizeof(mid)) <= 0)
104  && (ofw_get_property(child, "portid", &mid, sizeof(mid)) <= 0)
105  && (ofw_get_property(child, "cpuid", &mid, sizeof(mid)) <= 0))
106  continue;
107 
108  if (current_mid != mid) {
109  /*
110  * Start secondary processor.
111  */
112  (void) ofw_call("SUNW,start-cpu", 3, 1,
113  NULL, child, KERNEL_VIRTUAL_ADDRESS,
114  physmem_start | AP_PROCESSOR);
115  }
116  }
117  }
118  }
119 
120  return cpus;
121 }

126 int ofw_cpu(uint16_t mid_mask, uintptr_t physmem_start)
127 {
128  /* Get the current CPU MID */
129  uint64_t current_mid;
130 
131  asm volatile (
132  "ldxa [%1] %2, %0\n"
133  : "=r" (current_mid)
134  : "r" (0), "i" (ASI_ICBUS_CONFIG)
135  );
136 
137  current_mid >>= ICBUS_CONFIG_MID_SHIFT;
138  current_mid &= mid_mask;
139 
140  /* Wake up the CPUs */
141 
142  phandle cpus_parent = ofw_find_device("/ssm@0,0");
143  if ((cpus_parent == 0) || (cpus_parent == -1))
144  cpus_parent = ofw_find_device("/");
145 
146  phandle node = ofw_get_child_node(cpus_parent);
147  int cpus = wake_cpus_in_node(node, current_mid, physmem_start);
148  while ((node != 0) && (node != -1)) {
149  char name[OFW_TREE_PROPERTY_MAX_VALUELEN];
150 
151  if (ofw_get_property(node, "name", name,
152  OFW_TREE_PROPERTY_MAX_VALUELEN) > 0) {
153  name[OFW_TREE_PROPERTY_MAX_VALUELEN - 1] = 0;
154  if (strcmp(name, "cmp") == 0) {
155  phandle subnode = ofw_get_child_node(node);
156  cpus += wake_cpus_in_node(subnode,
157  current_mid, physmem_start);
158  }
159  }
160  node = ofw_get_peer_node(node);
161  }
162 
163  return cpus;
164 }
