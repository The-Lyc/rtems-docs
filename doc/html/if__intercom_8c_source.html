75 static void if_intercom_service(intercom_packet *packet, void *arg)
76 {
77  rtems_bsdnet_semaphore_obtain();
78 
79  if_intercom_control *self = arg;
80  struct ifnet *ifp = &self->arpcom.ac_if;
81  struct mbuf *m = new_mbuf(ifp, true);
82 
83  memcpy(mtod(m, void *), packet->data, packet->size);
84 
85  /* Ethernet header */
86  struct ether_header *eh = mtod(m, struct ether_header *);
87 
88  /* Discard Ethernet header and CRC */
89  int sz = (int) packet->size - ETHER_HDR_LEN;
90 
91  /* Update mbuf */
92  m->m_len = sz;
93  m->m_pkthdr.len = sz;
94  m->m_data = mtod(m, char *) + ETHER_HDR_LEN;
95 
96  /* Hand over */
97  ether_input(ifp, eh, m);
98 
99  ++self->received_frames;
100  qoriq_intercom_free_packet(packet);

102  rtems_bsdnet_semaphore_release();
103 }

117 static struct mbuf *get_next_fragment(struct ifnet *ifp, struct mbuf *m)
118 {
119  struct mbuf *n = NULL;
120  int size = 0;
121 
122  while (true) {
123  if (m == NULL) {
124  /* Dequeue first fragment of the next frame */
125  IF_DEQUEUE(&ifp->if_snd, m);
126 
127  /* Empty queue? */
128  if (m == NULL) {
129  return m;
130  }
131  }
132 
133  /* Get fragment size */
134  size = m->m_len;
135 
136  if (size > 0) {
137  /* Now we have a not empty fragment */
138  break;
139  } else {
140  /* Discard empty fragments */
141  m = m_free(m);
142  }
143  }
144 
145  /* Discard empty successive fragments */
146  n = m->m_next;
147  while (n != NULL && n->m_len <= 0) {
148  n = m_free(n);
149  }
150  m->m_next = n;
151 
152  return m;
153 }

155 static void if_intercom_start(struct ifnet *ifp)
156 {
157  if_intercom_control *self = ifp->if_softc;
158  int destination_core = self->destination_core;
159  intercom_packet *packet = self->packet;
160  size_t size = packet->size;
161  char *data = packet->data + size;
162  struct mbuf *m = NULL;
163 
164  while ((m = get_next_fragment(ifp, m)) != NULL) {
165  size_t fragment_size = (size_t) m->m_len;
166  size_t new_size = size + fragment_size;
167 
168  assert(new_size <= 2048);
169 
170  memcpy(data, mtod(m, void *), fragment_size);
171  data += fragment_size;
172  size = new_size;
173 
174  m = m_free(m);
175 
176  /* Last fragment of frame ? */
177  if (m == NULL) {
178  packet->size = size;
179  qoriq_intercom_send_packet(destination_core, packet);
180  ++self->transmitted_frames;
181  packet = allocate_packet();
182  data = packet->data;
183  size = 0;
184  }
185  }
186 
187  packet->size = size;
188  self->packet = packet;
189  ifp->if_flags &= ~IFF_OACTIVE;
190 }
