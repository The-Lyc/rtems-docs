168 static int ptyPollRead(rtems_termios_device_context *base)
169 { /* Characters written to the client side*/
170  rtems_pty_context *pty = (rtems_pty_context *)base;
171  unsigned char value;
172  unsigned int omod;
173  int count;
174  int result;
175 
176  count=read(pty->socket,&value,sizeof(value));
177  if (count<0)
178  return -1;
179 
180  if (count<1) {
181  /* Unfortunately, there is no way of passing an EOF
182  * condition through the termios driver. Hence, we
183  * resort to an ugly hack. Setting cindex>ccount
184  * causes the termios driver to return a read count
185  * of '0' which is what we want here. We leave
186  * 'errno' untouched.
187  */
188  pty->ttyp->cindex=pty->ttyp->ccount+1;
189  return pty->ttyp->termios.c_cc[VEOF];
190  };
191 
192  omod=pty->iac_mode;
193  pty->iac_mode=0;
194  switch(omod & 0xff) {
195  case IAC_ESC:
196  switch(value) {
197  case IAC_ESC :
198  /* in case this is an ESC ESC sequence in SB mode */
199  pty->iac_mode = omod>>8;
200  return IAC_ESC;
201  case IAC_DONT:
202  case IAC_DO :
203  case IAC_WONT:
204  case IAC_WILL:
205  pty->iac_mode=value;
206  return -1;
207  case IAC_SB :
208 #if DEBUG & DEBUG_DETAIL
209  printk("SB\n");
210 #endif
211  pty->iac_mode=value;
212  pty->sb_ind=0;
213  return -100;
214  case IAC_GA :
215  return -1;
216  case IAC_EL :
217  return 0x03; /* Ctrl-C*/
218  case IAC_EC :
219  return '\b';
220  case IAC_AYT :
221  write(pty->socket,IAC_AYT_RSP,strlen(IAC_AYT_RSP));
222  return -1;
223  case IAC_AO :
224  return -1;
225  case IAC_IP :
226  write(pty->socket,IAC_IP_RSP,strlen(IAC_IP_RSP));
227  return -1;
228  case IAC_BRK :
229  write(pty->socket,IAC_BRK_RSP,strlen(IAC_BRK_RSP));
230  return -1;
231  case IAC_DMARK:
232  return -2;
233  case IAC_NOP :
234  return -1;
235  case IAC_SE :
236 #if DEBUG & DEBUG_DETAIL
237  {
238  int i;
239  printk("SE");
240  for (i=0; i<pty->sb_ind; i++)
241  printk(" %02x",pty->sb_buf[i]);
242  printk("\n");
243  }
244 #endif
245  handleSB(pty);
246  return -101;
247  case IAC_EOR :
248  return -102;
249  default :
250  return -1;
251  };
252  break;
253 
254  case IAC_SB:
255  pty->iac_mode=omod;
256  if (IAC_ESC==value) {
257  pty->iac_mode=(omod<<8)|value;
258  } else {
259  if (pty->sb_ind < SB_MAX)
260  pty->sb_buf[pty->sb_ind++]=value;
261  }
262  return -1;
263 
264  case IAC_WILL:
265  if (value==34){
266  send_iac(pty,IAC_DONT, 34); /*LINEMODE*/
267  send_iac(pty,IAC_DO , 1); /*ECHO */
268  } else if (value==31) {
269  send_iac(pty,IAC_DO , 31); /*NAWS */
270 #if DEBUG & DEBUG_DETAIL
271  printk("replied DO NAWS\n");
272 #endif
273  } else {
274  send_iac(pty,IAC_DONT,value);
275  }
276  return -1;
277  case IAC_DONT:
278  return -1;
279  case IAC_DO :
280  if (value==3) {
281  send_iac(pty,IAC_WILL, 3); /* GO AHEAD*/
282  } else if (value==1) {
283  send_iac(pty,IAC_WILL, 1); /* ECHO */
284  } else {
285  send_iac(pty,IAC_WONT,value);
286  };
287  return -1;
288  case IAC_WONT:
289  if (value==1) {
290  send_iac(pty,IAC_WILL, 1);
291  } else { /* ECHO */
292  send_iac(pty,IAC_WONT,value);
293  }
294  return -1;
295  default:
296  if (value==IAC_ESC) {
297  pty->iac_mode=value;
298  return -1;
299  } else {
300  result=value;
301  if ( 0
302  /* map CRLF to CR for symmetry */
303  || ((value=='\n') && pty->last_cr)
304  /* map telnet CRNUL to CR down here */
305  || ((value==0) && pty->last_cr)
306  ) result=-1;
307  pty->last_cr=(value=='\r');
308  return result;
309  };
310  };
311  /* should never get here but keep compiler happy */
312  return -1;
313 }

368  ioctl_command_t request, void *buffer)
369 {

373  switch (request) {

377 #if DEBUG & DEBUG_WH

379  "ioctl(TIOCGWINSZ), returning %ix%i\n",
380  wp->ws_col,

382 #endif
383  break;
384  case TIOCSWINSZ:
385 #if DEBUG & DEBUG_WH
386  fprintf(stderr,
387  "ioctl(TIOCGWINSZ), setting %ix%i\n",
388  wp->ws_col,
389  wp->ws_row);
390 #endif
391 
392  p->height = wp->ws_row;
393  p->width = wp->ws_col;
394  break;
395  default:
396  rtems_set_errno_and_return_minus_one(EINVAL);
