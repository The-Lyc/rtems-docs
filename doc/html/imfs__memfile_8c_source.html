136 /*
137  * IMFS_memfile_extend
138  *
139  * This routine insures that the in-memory file is of the length
140  * specified. If necessary, it will allocate memory blocks to
141  * extend the file.
142  */
143 static int IMFS_memfile_extend(
144  IMFS_memfile_t *memfile,
145  bool zero_fill,
146  off_t new_length
147 )
148 {
149  unsigned int block;
150  unsigned int new_blocks;
151  unsigned int old_blocks;
152  unsigned int offset;
153 
154  /*
155  * Perform internal consistency checks
156  */
157  IMFS_assert( memfile );
158 
159  /*
160  * Verify new file size is supported
161  */
162  if ( new_length >= IMFS_MEMFILE_MAXIMUM_SIZE )
163  rtems_set_errno_and_return_minus_one( EFBIG );
164 
165  /*
166  * Verify new file size is actually larger than current size
167  */
168  if ( new_length <= memfile->File.size )
169  return 0;
170 
171  /*
172  * Calculate the number of range of blocks to allocate
173  */
174  new_blocks = new_length / IMFS_MEMFILE_BYTES_PER_BLOCK;
175  old_blocks = memfile->File.size / IMFS_MEMFILE_BYTES_PER_BLOCK;
176  offset = memfile->File.size - old_blocks * IMFS_MEMFILE_BYTES_PER_BLOCK;
177 
178  /*
179  * Now allocate each of those blocks.
180  */
181  for ( block=old_blocks ; block<=new_blocks ; block++ ) {
182  if ( !IMFS_memfile_addblock( memfile, block ) ) {
183  if ( zero_fill ) {
184  size_t count = IMFS_MEMFILE_BYTES_PER_BLOCK - offset;
185  block_p *block_ptr =
186  IMFS_memfile_get_block_pointer( memfile, block, 0 );
187 
188  memset( &(*block_ptr) [offset], 0, count);
189  offset = 0;
190  }
191  } else {
192  for ( ; block>=old_blocks ; block-- ) {
193  IMFS_memfile_remove_block( memfile, block );
194  }
195  rtems_set_errno_and_return_minus_one( ENOSPC );
196  }
197  }
198 
199  /*
200  * Set the new length of the file.
201  */
202  memfile->File.size = new_length;
203 
204  IMFS_mtime_ctime_update( &memfile->File.Node );
205 
206  return 0;
207 }

310 /*
311  * IMFS_memfile_destroy
312  *
313  * This routine frees all memory associated with an in memory file.
314  *
315  * NOTE: This is an exceptionally conservative implementation.
316  * It will check EVERY pointer which is non-NULL and insure
317  * any child non-NULL pointers are freed. Optimistically, all that
318  * is necessary is to scan until a NULL pointer is found. There
319  * should be no allocated data past that point.
320  *
321  * In experimentation on the powerpc simulator, it was noted
322  * that using blocks which held 128 slots versus 16 slots made
323  * a significant difference in the performance of this routine.
324  *
325  * Regardless until the IMFS implementation is proven, it
326  * is better to stick to simple, easy to understand algorithms.
327  */
328 static void IMFS_memfile_destroy(
329  IMFS_jnode_t *the_jnode
330 )
331 {
332  IMFS_memfile_t *memfile;
333  int i;
334  int j;
335  unsigned int to_free;
336  block_p *p;
337 
338  memfile = (IMFS_memfile_t *) the_jnode;
339 
340  /*
341  * Perform internal consistency checks
342  */
343  IMFS_assert( memfile );
344 
345  /*
346  * Eventually this could be set smarter at each call to
347  * memfile_free_blocks_in_table to greatly speed this up.
348  */
349  to_free = IMFS_MEMFILE_BLOCK_SLOTS;
350 
351  /*
352  * Now start freeing blocks in this order:
353  * + indirect
354  * + doubly indirect
355  * + triply indirect
356  */
357 
358  if ( memfile->indirect ) {
359  memfile_free_blocks_in_table( &memfile->indirect, to_free );
360  }
361 
362  if ( memfile->doubly_indirect ) {
363  for ( i=0 ; i<IMFS_MEMFILE_BLOCK_SLOTS ; i++ ) {
364  if ( memfile->doubly_indirect[i] ) {
365  memfile_free_blocks_in_table(
366  (block_p **)&memfile->doubly_indirect[i], to_free );
367  }
368  }
369  memfile_free_blocks_in_table( &memfile->doubly_indirect, to_free );
370 
371  }
372 
373  if ( memfile->triply_indirect ) {
374  for ( i=0 ; i<IMFS_MEMFILE_BLOCK_SLOTS ; i++ ) {
375  p = (block_p *) memfile->triply_indirect[i];
376  if ( !p ) /* ensure we have a valid pointer */
377  break;
378  for ( j=0 ; j<IMFS_MEMFILE_BLOCK_SLOTS ; j++ ) {
379  if ( p[j] ) {
380  memfile_free_blocks_in_table( (block_p **)&p[j], to_free);
381  }
382  }
383  memfile_free_blocks_in_table(
384  (block_p **)&memfile->triply_indirect[i], to_free );
385  }
386  memfile_free_blocks_in_table(
387  (block_p **)&memfile->triply_indirect, to_free );
388  }
389 
390  IMFS_node_destroy_default( the_jnode );
391 }

393 /*
394  * IMFS_memfile_read
395  *
396  * This routine read from memory file pointed to by the_jnode into
397  * the specified data buffer specified by destination. The file
398  * is NOT extended. An offset greater than the length of the file
399  * is considered an error. Read from an offset for more bytes than
400  * are between the offset and the end of the file will result in
401  * reading the data between offset and the end of the file (truncated
402  * read).
403  */
404 static ssize_t IMFS_memfile_read(
405  IMFS_file_t *file,
406  off_t start,
407  unsigned char *destination,
408  unsigned int length
409 )
410 {
411  block_p *block_ptr;
412  unsigned int block;
413  unsigned int my_length;
414  unsigned int to_copy = 0;
415  unsigned int last_byte;
416  unsigned int copied;
417  unsigned int start_offset;
418  unsigned char *dest;
419 
420  dest = destination;
421 
422  /*
423  * Perform internal consistency checks
424  */
425  IMFS_assert( file );
426  IMFS_assert( dest );
427 
428  /*
429  * Linear files (as created from a tar file are easier to handle
430  * than block files).
431  */
432  my_length = length;
433 
434  /*
435  * If the last byte we are supposed to read is past the end of this
436  * in memory file, then shorten the length to read.
437  */
438  last_byte = start + length;
439  if ( last_byte > file->Memfile.File.size )
440  my_length = file->Memfile.File.size - start;
441 
442  copied = 0;
443 
444  /*
445  * Three phases to the read:
446  * + possibly the last part of one block
447  * + all of zero of more blocks
448  * + possibly the first part of one block
449  */
450 
451  /*
452  * Phase 1: possibly the last part of one block
453  */
454  start_offset = start % IMFS_MEMFILE_BYTES_PER_BLOCK;
455  block = start / IMFS_MEMFILE_BYTES_PER_BLOCK;
456  if ( start_offset ) {
457  to_copy = IMFS_MEMFILE_BYTES_PER_BLOCK - start_offset;
458  if ( to_copy > my_length )
459  to_copy = my_length;
460  block_ptr = IMFS_memfile_get_block_pointer( &file->Memfile, block, 0 );
461  if ( !block_ptr )
462  return copied;
463  memcpy( dest, &(*block_ptr)[ start_offset ], to_copy );
464  dest += to_copy;
465  block++;
466  my_length -= to_copy;
467  copied += to_copy;
468  }
469 
470  /*
471  * Phase 2: all of zero of more blocks
472  */
473  to_copy = IMFS_MEMFILE_BYTES_PER_BLOCK;
474  while ( my_length >= IMFS_MEMFILE_BYTES_PER_BLOCK ) {
475  block_ptr = IMFS_memfile_get_block_pointer( &file->Memfile, block, 0 );
476  if ( !block_ptr )
477  return copied;
478  memcpy( dest, &(*block_ptr)[ 0 ], to_copy );
479  dest += to_copy;
480  block++;
481  my_length -= to_copy;
482  copied += to_copy;
483  }
484 
485  /*
486  * Phase 3: possibly the first part of one block
487  */
488  IMFS_assert( my_length < IMFS_MEMFILE_BYTES_PER_BLOCK );
489 
490  if ( my_length ) {
491  block_ptr = IMFS_memfile_get_block_pointer( &file->Memfile, block, 0 );
492  if ( !block_ptr )
493  return copied;
494  memcpy( dest, &(*block_ptr)[ 0 ], my_length );
495  copied += my_length;
496  }
497 
498  IMFS_update_atime( &file->Node );
499 
500  return copied;
501 }
