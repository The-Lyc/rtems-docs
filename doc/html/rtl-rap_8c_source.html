221 static bool
222 rtems_rtl_rap_relocate (rtems_rtl_rap* rap, rtems_rtl_obj* obj)
223 {
224  #define SYMNAME_BUFFER_SIZE (1024)
225  char* symname_buffer = NULL;
226  int section;
227 
228  if (rtems_rtl_trace (RTEMS_RTL_TRACE_RELOC))
229  printf ("rtl: relocation\n");
230 
231  symname_buffer = malloc (SYMNAME_BUFFER_SIZE);
232  if (!symname_buffer)
233  {
234  rtems_rtl_set_error (ENOMEM, "no memory for local symbol name buffer");
235  return false;
236  }
237 
238  for (section = 0; section < RTEMS_RTL_RAP_SECS; ++section)
239  {
240  rtems_rtl_obj_sect* targetsect;
241  uint32_t header = 0;
242  int relocs;
243  bool is_rela;
244  int r;
245 
246  targetsect = rtems_rtl_obj_find_section (obj, rap_sections[section].name);
247 
248  if (!targetsect)
249  continue;
250 
251  if (!rtems_rtl_rap_read_uint32 (rap->decomp, &header))
252  {
253  free (symname_buffer);
254  return false;
255  }
256 
257  /*
258  * Bit 31 of the header indicates if the relocations for this section
259  * have a valid addend field.
260  */
261 
262  is_rela = (header & (1 << 31)) != 0 ? true : false;
263  relocs = header & ~(1 << 31);
264 
265  if (relocs && rtems_rtl_trace (RTEMS_RTL_TRACE_RELOC))
266  printf ("rtl: relocation: %s: header: %08" PRIx32 " relocs: %d %s\n",
267  rap_sections[section].name,
268  header, relocs, is_rela ? "rela" : "rel");
269 
270  for (r = 0; r < relocs; ++r)
271  {
272  uint32_t info = 0;
273  uint32_t offset = 0;
274  uint32_t addend = 0;
275  Elf_Word type;
276  const char* symname = NULL;
277  uint32_t symname_size;
278  Elf_Word symtype = 0;
279  Elf_Word symvalue = 0;
280 
281  if (!rtems_rtl_rap_read_uint32 (rap->decomp, &info))
282  {
283  free (symname_buffer);
284  return false;
285  }
286 
287  if (!rtems_rtl_rap_read_uint32 (rap->decomp, &offset))
288  {
289  free (symname_buffer);
290  return false;
291  }
292 
293  /*
294  * The types are:
295  *
296  * 0 Section symbol offset in addend.
297  * 1 Symbol appended to the relocation record.
298  * 2 Symbol is in the strtabl.
299  *
300  * If type 2 bits 30:8 is the offset in the strtab. If type 1 the bits
301  * are the size of the string. The lower 8 bits of the info field if the
302  * ELF relocation type field.
303  */
304 
305  if (((info & (1 << 31)) == 0) || is_rela)
306  {
307  if (!rtems_rtl_rap_read_uint32 (rap->decomp, &addend))
308  {
309  free (symname_buffer);
310  return false;
311  }
312  }
313 
314  if (rtems_rtl_trace (RTEMS_RTL_TRACE_RELOC))
315  printf (" %2d: info=%08" PRIx32 " offset=%" PRIu32
316  " addend=%" PRIu32 "\n",
317  r, info, offset, addend);
318 
319  type = info & 0xff;
320 
321  if ((info & (1 << 31)) == 0)
322  {
323  rtems_rtl_obj_sect* symsect;
324 
325  symsect = rtems_rtl_obj_find_section_by_index (obj, info >> 8);
326  if (!symsect)
327  {
328  free (symname_buffer);
329  rtems_rtl_set_error (EINVAL, "symsect not found: %d", info >> 8);
330  return false;
331  }
332 
333  symvalue = (Elf_Addr) symsect->base + addend;
334  }
335  else if (rtems_rtl_elf_rel_resolve_sym (type))
336  {
337  rtems_rtl_obj_sym* symbol;
338 
339  symname_size = (info & ~(3 << 30)) >> 8;
340 
341  if ((info & (1 << 30)) != 0)
342  {
343  symname = rap->strtab + symname_size;
344  }
345  else
346  {
347  if (symname_size > (SYMNAME_BUFFER_SIZE - 1))
348  {
349  free (symname_buffer);
350  rtems_rtl_set_error (EINVAL, "reloc symbol too big");
351  return false;
352  }
353 
354  if (!rtems_rtl_obj_comp_read (rap->decomp, symname_buffer, symname_size))
355  {
356  free (symname_buffer);
357  return false;
358  }
359 
360  symname_buffer[symname_size] = '\0';
361  symname = symname_buffer;
362  }
363 
364  symbol = rtems_rtl_symbol_obj_find (obj, symname);
365 
366  if (!symbol)
367  {
368  rtems_rtl_set_error (EINVAL, "global symbol not found: %s", symname);
369  free (symname_buffer);
370  return false;
371  }
372 
373  symvalue = (Elf_Addr) symbol->value;
374  }
375 
376  if (is_rela)
377  {
378  Elf_Rela rela;
379 
380  rela.r_offset = offset;
381  rela.r_info = type;
382 
383  if ((info & (1 << 31)) == 0)
384  rela.r_addend = 0;
385  else rela.r_addend = addend;
386 
387  if (rtems_rtl_trace (RTEMS_RTL_TRACE_RELOC))
388  printf (" %2d: rela: type:%-2d off:%" PRIu32 " addend:%d"
389  " symname=%s symtype=%ju symvalue=0x%08jx\n",
390  r, (int) type, offset, (int) addend,
391  symname, (uintmax_t) symtype, (uintmax_t) symvalue);
392 
393  if (rtems_rtl_elf_relocate_rela (obj,
394  &rela,
395  targetsect,
396  symname,
397  symtype,
398  symvalue) == rtems_rtl_elf_rel_failure)
399  {
400  free (symname_buffer);
401  return false;
402  }
403  }
404  else
405  {
406  Elf_Rel rel;
407 
408  rel.r_offset = offset;
409  rel.r_info = type;
410 
411  if (rtems_rtl_trace (RTEMS_RTL_TRACE_RELOC))
412  printf (" %2d: rel: type:%-2d off:%" PRIu32
413  " symname=%s symtype=%ju symvalue=0x%08jx\n",
414  r, (int) type, offset,
415  symname, (uintmax_t) symtype, (uintmax_t) symvalue);
416 
417  if (rtems_rtl_elf_relocate_rel (obj,
418  &rel,
419  targetsect,
420  symname,
421  symtype,
422  symvalue) == rtems_rtl_elf_rel_failure)
423  {
424  free (symname_buffer);
425  return false;
426  }
427  }
428  }
429  }
430 
431  free (symname_buffer);
432 
433  return true;
434 }

443 static bool
444 rtems_rtl_rap_load_linkmap (rtems_rtl_rap* rap, rtems_rtl_obj* obj)
445 {
446  void* detail;
447  struct link_map* tmp1;
448  section_detail* tmp2;
449  uint32_t obj_detail_size;
450  uint32_t pos = 0;
451  int i;
452  int j;
453 
454  obj_detail_size = sizeof (struct link_map) * obj->obj_num;
455 
456  for (i = 0; i < obj->obj_num; ++i)
457  {
458  obj_detail_size += (obj->sec_num[i] * sizeof (section_detail));
459  }
460 
461  detail = rtems_rtl_alloc_new (RTEMS_RTL_ALLOC_OBJECT,
462  obj_detail_size + rap->strtable_size, true);
463 
464  if (!detail)
465  {
466  rap->strtable_size = 0;
467  rtems_rtl_set_error (ENOMEM, "no memory for obj global syms");
468  return false;
469  }
470 
471  rap->strtable = detail + obj_detail_size;
472 
473  /*
474  * Read the obj names and section names
475  */
476  if (!rtems_rtl_obj_comp_read (rap->decomp,
477  rap->strtable,
478  rap->strtable_size))
479  {
480  rtems_rtl_alloc_del (RTEMS_RTL_ALLOC_OBJECT, detail);
481  return false;
482  }
483 
484  obj->linkmap = (struct link_map*) detail;
485 
486  if (rtems_rtl_trace (RTEMS_RTL_TRACE_DETAIL))
487  {
488  if (rap->rpathlen > 0)
489  printf ("File rpath:\n");
490  }
491 
492  while (pos < rap->rpathlen)
493  {
494  if (rtems_rtl_trace (RTEMS_RTL_TRACE_DETAIL))
495  {
496  printf (" %s\n", rap->strtable + pos);
497  }
498  pos = pos + strlen (rap->strtable + pos) + 1;
499  }
500 
501  if (rap->rpathlen > 0)
502  pos = rap->rpathlen;
503 
504  for (i = 0; i < obj->obj_num; ++i)
505  {
506  tmp1 = obj->linkmap + i;
507  tmp1->name = rap->strtable + pos;
508  tmp1->sec_num = obj->sec_num[i];
509  tmp1->rpathlen = rap->rpathlen;
510  tmp1->rpath = (char*) rap->strtable;
511  pos += strlen (tmp1->name) + 1;
512 
513  if (!i)
514  {
515  tmp1->l_next = NULL;
516  tmp1->l_prev = NULL;
517  }
518  else
519  {
520  (tmp1 - 1)->l_next = tmp1;
521  tmp1->l_prev = tmp1 - 1;
522  tmp1->l_next = NULL;
523  }
524  }
525 
526  tmp2 = (section_detail*) (obj->linkmap + obj->obj_num);
527 
528  for (i = 0; i < obj->obj_num; ++i)
529  {
530  if (rtems_rtl_trace (RTEMS_RTL_TRACE_DETAIL))
531  {
532  printf ("File %d: %s\n", i, (obj->linkmap + i)->name);
533  printf ("Section: %d sections\n", (unsigned int) obj->sec_num[i]);
534  }
535 
536  obj->linkmap[i].sec_detail = tmp2;
537 
538  for (j = 0; j < obj->sec_num[i]; ++j)
539  {
540  uint32_t name;
541  uint32_t rap_id;
542  uint32_t offset;
543  uint32_t size;
544 
545  if (!rtems_rtl_rap_read_uint32 (rap->decomp, &name) ||
546  !rtems_rtl_rap_read_uint32 (rap->decomp, &offset) ||
547  !rtems_rtl_rap_read_uint32 (rap->decomp, &size))
548  {
549  rtems_rtl_alloc_del (RTEMS_RTL_ALLOC_SYMBOL, obj->linkmap);
550  obj->linkmap = NULL;
551  return false;
552  }
553 
554  rap_id = offset >> 28;
555  offset = offset & 0xfffffff;
556 
557  tmp2->name = rap->strtable + name;
558  tmp2->offset = offset;
559  tmp2->rap_id = rap_id;
560  tmp2->size = size;
561  pos += strlen (tmp2->name) + 1;
562 
563  if (rtems_rtl_trace (RTEMS_RTL_TRACE_DETAIL))
564  {
565  printf ("name:%16s offset:0x%08x rap_id:%d size:0x%x\n",
566  tmp2->name, (unsigned int) tmp2->offset,
567  (unsigned int) tmp2->rap_id, (unsigned int) tmp2->size);
568  }
569 
570  tmp2 += 1;
571  }
572  }
573  return true;
574 }

576 static bool
577 rtems_rtl_rap_load_symbols (rtems_rtl_rap* rap, rtems_rtl_obj* obj)
578 {
579  rtems_rtl_obj_sym* gsym;
580  int sym;
581 
582  obj->global_size =
583  rap->symbols * sizeof (rtems_rtl_obj_sym) + rap->strtab_size;
584 
585  obj->global_table = rtems_rtl_alloc_new (RTEMS_RTL_ALLOC_SYMBOL,
586  obj->global_size, true);
587  if (!obj->global_table)
588  {
589  obj->global_size = 0;
590  rtems_rtl_set_error (ENOMEM, "no memory for obj global syms");
591  return false;
592  }
593 
594  obj->global_syms = rap->symbols;
595 
596  rap->strtab = (((char*) obj->global_table) +
597  (rap->symbols * sizeof (rtems_rtl_obj_sym)));
598 
599  if (!rtems_rtl_obj_comp_read (rap->decomp, rap->strtab, rap->strtab_size))
600  {
601  rtems_rtl_alloc_del (RTEMS_RTL_ALLOC_SYMBOL, obj->global_table);
602  return false;
603  }
604 
605  for (sym = 0, gsym = obj->global_table; sym < rap->symbols; ++sym)
606  {
607  rtems_rtl_obj_sect* symsect;
608  uint32_t data;
609  uint32_t name;
610  uint32_t value;
611 
612  if (!rtems_rtl_rap_read_uint32 (rap->decomp, &data) ||
613  !rtems_rtl_rap_read_uint32 (rap->decomp, &name) ||
614  !rtems_rtl_rap_read_uint32 (rap->decomp, &value))
615  {
616  rtems_rtl_alloc_del (RTEMS_RTL_ALLOC_SYMBOL, obj->global_table);
617  obj->global_table = NULL;
618  obj->global_syms = 0;
619  obj->global_size = 0;
620  return false;
621  }
622 
623  if (rtems_rtl_trace (RTEMS_RTL_TRACE_SYMBOL))
624  printf ("rtl: sym:load: data=0x%08" PRIx32 " name=0x%08" PRIx32
625  " value=0x%08" PRIx32 "\n",
626  data, name, value);
627 
628  /*
629  * If there is a globally exported symbol already present and this
630  * symbol is not weak raise an error. If the symbol is weak and present
631  * globally ignore this symbol and use the global one and if it is not
632  * present take this symbol global or weak. We accept the first weak
633  * symbol we find and make it globally exported.
634  */
635  if (rtems_rtl_symbol_global_find (rap->strtab + name) &&
636  (ELF_ST_BIND (data & 0xffff) != STB_WEAK))
637  {
638  rtems_rtl_set_error (EINVAL,
639  "duplicate global symbol: %s", rap->strtab + name);
640  rtems_rtl_alloc_del (RTEMS_RTL_ALLOC_SYMBOL, obj->global_table);
641  obj->global_table = NULL;
642  obj->global_syms = 0;
643  obj->global_size = 0;
644  return false;
645  }
646 
647  symsect = rtems_rtl_obj_find_section_by_index (obj, data >> 16);
648  if (!symsect)
649  {
650  rtems_rtl_alloc_del (RTEMS_RTL_ALLOC_SYMBOL, obj->global_table);
651  obj->global_table = NULL;
652  obj->global_syms = 0;
653  obj->global_size = 0;
654  rtems_rtl_set_error (EINVAL, "section index not found: %" PRIu32, data >> 16);
655  return false;
656  }
657 
658  rtems_chain_set_off_chain (&gsym->node);
659  gsym->name = rap->strtab + name;
660  gsym->value = (uint8_t*) (value + symsect->base);
661  gsym->data = data & 0xffff;
662 
663  if (rtems_rtl_trace (RTEMS_RTL_TRACE_SYMBOL))
664  printf ("rtl: sym:add:%-2d name:%-20s bind:%-2d type:%-2d val:%8p sect:%d\n",
665  sym, gsym->name,
666  (int) ELF_ST_BIND (data & 0xffff),
667  (int) ELF_ST_TYPE (data & 0xffff),
668  gsym->value, (int) (data >> 16));
669 
670  ++gsym;
671  }
672 
673  if (obj->global_syms)
674  rtems_rtl_symbol_obj_add (obj);
675 
676  return true;
677 }

679 static bool
680 rtems_rtl_rap_parse_header (uint8_t* rhdr,
681  size_t* rhdr_len,
682  uint32_t* length,
683  uint32_t* version,
684  uint32_t* compression,
685  uint32_t* checksum)
686 {
687  char* sptr = (char*) rhdr;
688  char* eptr;
689 
690  *rhdr_len = 0;
691 
692  /*
693  * "RAP," = 4 bytes, total 4
694  */
695 
696  if ((rhdr[0] != 'R') || (rhdr[1] != 'A') || (rhdr[2] != 'P') || (rhdr[3] != ','))
697  return false;
698 
699  sptr = sptr + 4;
700 
701  /*
702  * "00000000," = 9 bytes, total 13
703  */
704 
705  *length = strtoul (sptr, &eptr, 10);
706 
707  if (*eptr != ',')
708  return false;
709 
710  sptr = eptr + 1;
711 
712  /*
713  * "0000," = 5 bytes, total 18
714  */
715 
716  *version = strtoul (sptr, &eptr, 10);
717 
718  if (*eptr != ',')
719  return false;
720 
721  sptr = eptr + 1;
722 
723  /*
724  * "NONE," and "LZ77," = 5 bytes, total 23
725  */
726 
727  if ((sptr[0] == 'N') &&
728  (sptr[1] == 'O') &&
729  (sptr[2] == 'N') &&
730  (sptr[3] == 'E'))
731  {
732  *compression = RTEMS_RTL_COMP_NONE;
733  eptr = sptr + 4;
734  }
735  else if ((sptr[0] == 'L') &&
736  (sptr[1] == 'Z') &&
737  (sptr[2] == '7') &&
738  (sptr[3] == '7'))
739  {
740  *compression = RTEMS_RTL_COMP_LZ77;
741  eptr = sptr + 4;
742  }
743  else
744  return false;
745 
746  if (*eptr != ',')
747  return false;
748 
749  sptr = eptr + 1;
