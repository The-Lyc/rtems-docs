56 /* Translate termios' tcflag_t into sci settings */
57 static int _sci_set_cflags(
58  struct scidev_t *sci_dev,
59  tcflag_t c_cflag,
60  speed_t spd )
61 {
62  uint8_t smr ;
63  uint8_t brr ;
64 
65  if ( spd )
66  {
67  if ( _sci_get_brparms( spd, &smr, &brr ) != 0 )
68  return -1 ;
69  }
70 
71  if ( c_cflag & CSIZE )
72  {
73  if ( c_cflag & CS8 )
74  smr &= ~SCI_SEVEN_BIT_DATA;
75  else if ( c_cflag & CS7 )
76  smr |= SCI_SEVEN_BIT_DATA;
77  else
78  return -1 ;
79  }
80 
81  if ( c_cflag & CSTOPB )
82  smr |= SCI_STOP_BITS_2;
83  else
84  smr &= ~SCI_STOP_BITS_2;
85 
86  if ( c_cflag & PARENB )
87  smr |= SCI_PARITY_ON ;
88  else
89  smr &= ~SCI_PARITY_ON ;
90 
91  if ( c_cflag & PARODD )
92  smr |= SCI_ODD_PARITY ;
93  else
94  smr &= ~SCI_ODD_PARITY;
95 
96  write8( smr, sci_dev->addr + SCI_SMR );
97  write8( brr, sci_dev->addr + SCI_BRR );
98 
99  return 0 ;
100 }

102 static void _sci_init(
103  rtems_device_minor_number minor )
104 {
105  uint16_t temp16 ;
106 
107  /* Pin function controller initialisation for asynchronous mode */
108  if( minor == 0)
109  {
110  temp16 = read16( PFC_PBCR1);
111  temp16 &= ~( PB8MD | PB9MD );
112  temp16 |= (PB_TXD0 | PB_RXD0);
113  write16( temp16, PFC_PBCR1);
114  }
115  else
116  {
117  temp16 = read16( PFC_PBCR1);
118  temp16 &= ~( PB10MD | PB11MD);
119  temp16 |= (PB_TXD1 | PB_RXD1);
120  write16( temp16, PFC_PBCR1);
121  }
122 
123  /* disable sck-pin */
124  if( minor == 0)
125  {
126  temp16 = read16( PFC_PBCR1);
127  temp16 &= ~(PB12MD);
128  write16( temp16, PFC_PBCR1);
129  }
130  else
131  {
132  temp16 = read16( PFC_PBCR1);
133  temp16 &= ~(PB13MD);
134  write16( temp16, PFC_PBCR1);
135  }
136 }

138 static void _sci_tx_polled(
139  int minor,
140  const char buf )
141 {
142  struct scidev_t *scidev = &sci_device[minor] ;
143  int8_t ssr ;
144 
145  while ( !inb((scidev->addr + SCI_SSR) & SCI_TDRE ))
146  ;
147  write8(buf,scidev->addr+SCI_TDR);
148 
149  ssr = inb(scidev->addr+SCI_SSR);
150  ssr &= ~SCI_TDRE ;
151  write8(ssr,scidev->addr+SCI_SSR);
152 }

154 static int _sci_rx_polled (
155  int minor)
156 {
157  struct scidev_t *scidev = &sci_device[minor] ;
158 
159  unsigned char c;
160  char ssr ;
161  ssr = read8(scidev->addr + SCI_SSR) ;
162 
163  if (ssr & (SCI_PER | SCI_FER | SCI_ORER))
164  write8(ssr & ~(SCI_PER | SCI_FER | SCI_ORER), scidev->addr+SCI_SSR);
165 
166  if ( !(ssr & SCI_RDRF) )
167  return -1;
168 
169  c = read8(scidev->addr + SCI_RDR) ;
170 
171  write8(ssr & ~SCI_RDRF,scidev->addr + SCI_SSR);
172  return c;
173 }

200  }
201 
202  /* default hardware setup */
203

212 rtems_device_driver sh_sci_open(
213  rtems_device_major_number major,
214  rtems_device_minor_number minor,
215  void * arg )
216 {
217  uint8_t temp8;
218 
219  /* check for valid minor number */
220  if(( minor > ( SCI_MINOR_DEVICES -1 )) || ( minor < 0 ))
221  {
222  return RTEMS_INVALID_NUMBER;
223  }
224 
225  /* device already opened */
226  if ( sci_device[minor].opened > 0 )
227  {
228  sci_device[minor].opened++ ;
229  return RTEMS_SUCCESSFUL ;
230  }
231 
232  _sci_init( minor );
233 
234  if (minor == 0) {
235  temp8 = read8(sci_device[minor].addr + SCI_SCR);
236  temp8 &= ~(SCI_TE | SCI_RE) ;
237  write8(temp8, sci_device[minor].addr + SCI_SCR); /* Clear SCR */
238  _sci_set_cflags( &sci_device[minor], sci_device[minor].cflags, sci_device[minor].spd );
239 
240 /* FIXME: Should be one bit delay */
241  CPU_delay(50000); /* microseconds */
242 
243  temp8 |= SCI_RE | SCI_TE;
244  write8(temp8, sci_device[minor].addr + SCI_SCR); /* Enable clock output */
245  } else {
246  temp8 = read8(sci_device[minor].addr + SCI_SCR);
247  temp8 &= ~(SCI_TE | SCI_RE) ;
248  write8(temp8, sci_device[minor].addr + SCI_SCR); /* Clear SCR */
249  _sci_set_cflags( &sci_device[minor], sci_device[minor].cflags, sci_device[minor].spd );
250 
251 /* FIXME: Should be one bit delay */
252  CPU_delay(50000); /* microseconds */
253 
254  temp8 |= SCI_RE | SCI_TE;
255  write8(temp8, sci_device[minor].addr + SCI_SCR); /* Enable clock output */
256  }
257 
258  sci_device[minor].opened++ ;
259 
260  return RTEMS_SUCCESSFUL ;
261 }

288  rtems_device_major_number major,
289  rtems_device_minor_number minor,
290  void * arg
291 )
292 {
293  int count = 0;
294 
295  rtems_libio_rw_args_t *rw_args = (rtems_libio_rw_args_t *) arg;
296  char * buffer = rw_args->buffer;
297  int maximum = rw_args->count;
298 
299  for (count = 0; count < maximum; count++) {
300  buffer[ count ] = _sci_rx_polled(minor);
301  if (buffer[ count ] == '\n' || buffer[ count ] == '\r') {
302  buffer[ count++ ] = '\n';
303  break;
304  }
305  }
306 
307  rw_args->bytes_moved = count;
308  return (count >= 0) ? RTEMS_SUCCESSFUL : RTEMS_UNSATISFIED;
