83 /*
84  * Convert the unix version of time to dos's idea of time to be used in
85  * file timestamps. The passed in unix time is assumed to be in GMT.
86  */
87 void
88 msdos_date_unix2dos(unsigned int t, uint16_t *ddp,
89  uint16_t *dtp)
90 {
91  uint32_t days;
92  uint32_t inc;
93  uint32_t year;
94  uint32_t month;
95  uint16_t *months;
96 
97  /*
98  * If the time from the last conversion is the same as now, then
99  * skip the computations and use the saved result.
100  */
101  if (lasttime != t) {
102  lasttime = t;
103  lastdtime = (((t % 60) >> 1) << MSDOS_DT_2SECONDS_SHIFT)
104  + (((t / 60) % 60) << MSDOS_DT_MINUTES_SHIFT)
105  + (((t / 3600) % 24) << MSDOS_DT_HOURS_SHIFT);
106 
107  /*
108  * If the number of days since 1970 is the same as the last
109  * time we did the computation then skip all this leap year
110  * and month stuff.
111  */
112  days = t / (SECONDSPERDAY);
113  if (days != lastday) {
114  lastday = days;
115  for (year = 1970;; year++) {
116  inc = year & 0x03 ? 365 : 366;
117  if (days < inc)
118  break;
119  days -= inc;
120  }
121  months = year & 0x03 ? regyear : leapyear;
122  for (month = 0; month < 12; month++) {
123  if (days < months[month])
124  break;
125  days -= months[month];
126  }
127  lastddate = ((days + 1) << MSDOS_DD_DAY_SHIFT)
128  + ((month + 1) << MSDOS_DD_MONTH_SHIFT);
129  /*
130  * Remember dos's idea of time is relative to 1980.
131  * unix's is relative to 1970. If somehow we get a
132  * time before 1980 then don't give totally crazy
133  * results.
134  */
135  if (year > 1980)
136  lastddate += (year - 1980) <<
137  MSDOS_DD_YEAR_SHIFT;
138  }
139  }
140  *dtp = lastdtime;
141  *ddp = lastddate;
142 }
143 
144 /*
145  * The number of days between Jan 1, 1970 and Jan 1, 1980. In that
146  * interval there were 8 regular years and 2 leap years.
147  */
148 /* #define DAYSTO1980 ((8 * 365) + (2 * 366)) */
149 #define DAYSTO1980 ((uint32_t) 3652)
150 
151 static uint16_t lastdosdate;
152 static uint32_t lastseconds;

154 /*
155  * Convert from dos' idea of time to unix'. This will probably only be
156  * called from the stat(), and fstat() system calls and so probably need
157  * not be too efficient.
158  */
159 unsigned int
160 msdos_date_dos2unix(unsigned int dd, unsigned int dt)
161 {
162  uint32_t seconds;
163  uint32_t m, month;
164  uint32_t y, year;
165  uint32_t days;
166  uint16_t *months;
167 
168  seconds = 2 * ((dt & MSDOS_DT_2SECONDS_MASK) >> MSDOS_DT_2SECONDS_SHIFT)
169  + ((dt & MSDOS_DT_MINUTES_MASK) >> MSDOS_DT_MINUTES_SHIFT) * 60
170  + ((dt & MSDOS_DT_HOURS_MASK) >> MSDOS_DT_HOURS_SHIFT) * 3600;
171  /*
172  * If the year, month, and day from the last conversion are the
173  * same then use the saved value.
174  */
175  if (lastdosdate != dd) {
176  lastdosdate = dd;
177  days = 0;
178  year = (dd & MSDOS_DD_YEAR_MASK) >> MSDOS_DD_YEAR_SHIFT;
179  for (y = 0; y < year; y++)
180  days += y & 0x03 ? 365 : 366;
181  months = year & 0x03 ? regyear : leapyear;
182  /*
183  * Prevent going from 0 to 0xffffffff in the following
184  * loop.
185  */
186  month = (dd & MSDOS_DD_MONTH_MASK) >> MSDOS_DD_MONTH_SHIFT;
187  if (month == 0) {
188  month = 1;
189  }
190  for (m = 0; m < month - 1; m++)
191  days += months[m];
192  days += ((dd & MSDOS_DD_DAY_MASK) >> MSDOS_DD_DAY_SHIFT) - 1;
193  lastseconds = (days + DAYSTO1980) * SECONDSPERDAY;
194  }
195  return seconds + lastseconds;
196 }

301 static ssize_t
302 msdos_filename_process_dot_names (const uint8_t *src_name,
303  const size_t src_size,
304  uint8_t *dest_name,
305  const size_t dest_size)
306 {
307  ssize_t returned_size = 0;
308  int eno = 0;
309  /*
310  * The filenames "." and ".." are handled specially, since they
311  * don't follow dos filename rules.
312  */
313  if ( src_name[0] == UTF8_FULL_STOP
314  && src_size == UTF8_FULL_STOP_SIZE) {
315  if (dest_size >= UTF8_FULL_STOP_SIZE) {
316  dest_name[0] = UTF8_FULL_STOP;
317  returned_size = UTF8_FULL_STOP_SIZE;
318  }
319  else
320  eno = ENAMETOOLONG;
321  }
322  else if ( eno == 0
323  && src_name[0] == UTF8_FULL_STOP
324  && src_name[1] == UTF8_FULL_STOP
325  && src_size == ( 2 * UTF8_FULL_STOP_SIZE ) ) {
326  if (dest_size >= 2 * UTF8_FULL_STOP_SIZE) {
327  dest_name[0] = UTF8_FULL_STOP;
328  dest_name[1] = UTF8_FULL_STOP;
329  returned_size = 2 * UTF8_FULL_STOP_SIZE;
330  }
331  else
332  eno = ENAMETOOLONG;
333  }
334 
335  if (eno != 0) {
336  errno = eno;
337  returned_size = -1;
338  }
339 
340  return returned_size;
341 }
