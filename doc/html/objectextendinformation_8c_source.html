32 Objects_Maximum _Objects_Extend_information(
33  Objects_Information *information
34 )
35 {
36  Objects_Control *the_object;
37  uint32_t block_count;
38  uint32_t block;
39  uint32_t index_base;
40  uint32_t index_end;
41  uint32_t index;
42  Objects_Maximum extend_count;
43  Objects_Maximum old_maximum;
44  uint32_t new_maximum;
45  size_t object_block_size;
46  Objects_Control *new_object_block;
47  bool do_extend;
48  Objects_Id api_class_and_node;
49 
50  _Assert(
51  _Objects_Allocator_is_owner()
52  || !_System_state_Is_up( _System_state_Get() )
53  );
54  _Assert( _Objects_Is_auto_extend( information ) );
55 
56  extend_count = _Objects_Extend_size( information );
57  old_maximum = _Objects_Get_maximum_index( information );
58  new_maximum = (uint32_t) old_maximum + extend_count;
59  api_class_and_node = information->maximum_id & ~OBJECTS_INDEX_MASK;
60 
61  /*
62  * Search for a free block of indexes. If we do NOT need to allocate or
63  * extend the block table, then we will change do_extend.
64  */
65  do_extend = true;
66  index_base = extend_count;
67  block = 1;
68 
69  if ( information->object_blocks == NULL ) {
70  block_count = 1;
71  } else {
72  block_count = old_maximum / extend_count;
73 
74  for ( ; block < block_count; block++ ) {
75  if ( information->object_blocks[ block ] == NULL ) {
76  do_extend = false;
77  break;
78  } else
79  index_base += extend_count;
80  }
81  }
82 
83  index_end = index_base + extend_count;
84 
85  /*
86  * We need to limit the number of objects to the maximum number
87  * representable in the index portion of the object Id. In the
88  * case of 16-bit Ids, this is only 256 object instances.
89  */
90  if ( new_maximum > OBJECTS_ID_FINAL_INDEX ) {
91  return 0;
92  }
93 
94  /*
95  * Allocate the name table, and the objects and if it fails either return or
96  * generate a fatal error depending on auto-extending being active.
97  */
98  object_block_size = extend_count * information->object_size;
99  new_object_block = _Workspace_Allocate( object_block_size );
100  if ( new_object_block == NULL ) {
101  return 0;
102  }
103 
104  /*
105  * Do we need to grow the tables?
106  */
107  if ( do_extend ) {
108  ISR_lock_Context lock_context;
109  Objects_Control **object_blocks;
110  Objects_Control **local_table;
111  Objects_Maximum *inactive_per_block;
112  void *old_tables;
113  size_t table_size;
114  uintptr_t object_blocks_size;
115  uintptr_t local_table_size;
116 
117  /*
118  * Growing the tables means allocating a new area, doing a copy and
119  * updating the information table.
120  *
121  * If the maximum is minimum we do not have a table to copy. First
122  * time through.
123  *
124  * The allocation has:
125  *
126  * Objects_Control *object_blocks[ block_count ];
127  * Objects_Control *local_table[ maximum ];
128  * Objects_Maximum inactive_count[ block_count ];
129  *
130  * This is the order in memory. Watch changing the order. See the memcpy
131  * below.
132  */
133 
134  /*
135  * Up the block count and maximum.
136  */
137  block_count++;
138 
139  /*
140  * Allocate the tables and break it up.
141  */
142  object_blocks_size = block_count * sizeof( *object_blocks );
143  local_table_size = new_maximum * sizeof( *local_table );
144  table_size = object_blocks_size
145  + local_table_size
146  + block_count * sizeof( *inactive_per_block );
147  object_blocks = _Workspace_Allocate( table_size );
148  if ( object_blocks == NULL ) {
149  _Workspace_Free( new_object_block );
150  return 0;
151  }
152 
153  /*
154  * Break the block into the various sections.
155  */
156  local_table = _Addresses_Add_offset(
157  object_blocks,
158  object_blocks_size
159  );
160  inactive_per_block = _Addresses_Add_offset(
161  local_table,
162  local_table_size
163  );
164 
165  /*
166  * Take the block count down. Saves all the (block_count - 1)
167  * in the copies.
168  */
169  block_count--;
170 
171  if ( old_maximum > extend_count ) {
172  /*
173  * Copy each section of the table over. This has to be performed as
174  * separate parts as size of each block has changed.
175  */
176  memcpy(
177  object_blocks,
178  information->object_blocks,
179  block_count * sizeof( *object_blocks )
180  );
181  memcpy(
182  inactive_per_block,
183  information->inactive_per_block,
184  block_count * sizeof( *inactive_per_block )
185  );
186  } else {
187  object_blocks[ 0 ] = NULL;
188  inactive_per_block[ 0 ] = 0;
189  }
190 
191  memcpy(
192  local_table,
193  information->local_table,
194  old_maximum * sizeof( *local_table )
195  );
196 
197  /*
198  * Initialise the new entries in the table.
199  */
200  for ( index = index_base ; index < index_end ; ++index ) {
201  local_table[ index ] = NULL;
202  }
203 
204  /* FIXME: https://devel.rtems.org/ticket/2280 */
205  _ISR_lock_ISR_disable( &lock_context );
206 
207  old_tables = information->object_blocks;
208 
209  information->object_blocks = object_blocks;
210  information->inactive_per_block = inactive_per_block;
211  information->local_table = local_table;
212  information->maximum_id = api_class_and_node
213  | (new_maximum << OBJECTS_INDEX_START_BIT);
214 
215  _ISR_lock_ISR_enable( &lock_context );
216 
217  _Workspace_Free( old_tables );
218 
219  block_count++;
220  }
221 
222  /*
223  * Assign the new object block to the object block table.
224  */
225  information->object_blocks[ block ] = new_object_block;
226  information->inactive_per_block[ block ] = information->objects_per_block;
227  information->inactive += information->objects_per_block;
228 
229  /*
230  * Append to inactive chain.
231  */
232  the_object = new_object_block;
233  for ( index = index_base ; index < index_end ; ++index ) {
234  the_object->id = api_class_and_node
235  | ( ( index + OBJECTS_INDEX_MINIMUM ) << OBJECTS_INDEX_START_BIT );
236 
237  _Chain_Initialize_node( &the_object->Node );
238  _Chain_Append_unprotected( &information->Inactive, &the_object->Node );
239 
240  the_object = _Addresses_Add_offset( the_object, information->object_size );
241  }
242 
243  return block;
244 }
