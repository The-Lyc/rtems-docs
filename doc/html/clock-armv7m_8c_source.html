80 static void _ARMV7M_Clock_initialize_early(void)
81 {
82  volatile ARMV7M_Systick *systick;
83  uint32_t us_per_tick;
84  uint64_t freq;
85  uint32_t interval;
86 
87  systick = _ARMV7M_Systick;
88  us_per_tick = rtems_configuration_get_microseconds_per_tick();
89  freq = _ARMV7M_Clock_frequency();
90 
91  interval = (uint32_t) ((freq * us_per_tick) / 1000000);
92 
93  systick->rvr = interval;
94  systick->cvr = 0;
95  systick->csr = ARMV7M_SYSTICK_CSR_ENABLE | ARMV7M_SYSTICK_CSR_CLKSOURCE;
96 }
