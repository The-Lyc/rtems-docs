180 | Purpose: |

184 \*-------------------------------------------------------------------------*/

186  const msdos_format_request_param_t *rqdata,
187  int fd, /* file descriptor index */
188  uint32_t start_sector, /* sector number to fill to */
189  uint32_t sector_cnt, /* number of sectors to fill to */
190  const char fill_byte /* byte to fill into sectors */
191  )
192 /*-------------------------------------------------------------------------*\
193 | Return Value: |
194 | 0, if success, -1 and errno if failed |
195 \*=========================================================================*/
196 {
197  int ret_val = 0;
198  uint32_t total_sectors = sector_cnt;
199  int last_percent = -1;
200 
201  /*
202  * fill buffer
203  */
204  if (ret_val == 0) {
205  memset(fmt_params->sec,fill_byte,fmt_params->bytes_per_sector);
206  }
207 
208  msdos_format_printf (rqdata, MSDOS_FMT_INFO_LEVEL_DETAIL,
209  "Filling : ");
210  /*
211  * write to consecutive sectors
212  */
213  while ((ret_val == 0) &&
214  (sector_cnt > 0)) {
215  int percent = (sector_cnt * 100) / total_sectors;
216  if (percent != last_percent) {
217  if ((percent & 1) == 0)
218  msdos_format_printf (rqdata, MSDOS_FMT_INFO_LEVEL_DETAIL, ".");
219  last_percent = percent;
220  }
221  ret_val = msdos_format_write_sec(fd,start_sector,
222  fmt_params->bytes_per_sector,
223  fmt_params->sec);
224  start_sector++;
225  sector_cnt--;
226  }
227 
228  msdos_format_printf (rqdata, MSDOS_FMT_INFO_LEVEL_DETAIL, "\n");
229 
230  if (ret_val)
231  msdos_format_printf (rqdata, MSDOS_FMT_INFO_LEVEL_INFO,
232  "filling error on sector: %d\n", start_sector);
233 
234  return ret_val;

273 | Function: |
274 \*-------------------------------------------------------------------------*/
275 static int msdos_format_eval_sectors_per_cluster
276 (
277 /*-------------------------------------------------------------------------*\
278 | Purpose: |
279 | function to check/adjust sectors_per_cluster to legal values |
280 +---------------------------------------------------------------------------+
281 | Input Parameters: |
282 \*-------------------------------------------------------------------------*/
283  int fattype, /* type code of FAT (FAT_FAT12 ...) */
284  uint32_t bytes_per_sector, /* byte count per sector (512) */
285  const uint32_t total_sector_cnt, /* total number of secters per volume */
286  const uint32_t rsvd_sector_cnt, /* number of reserved sectors */
287  const uint32_t root_dir_sector_cnt,/* number of sectors for the root dir */
288  uint8_t fat_num, /* number of fat copies */
289  uint32_t sectors_per_cluster, /* sectors per cluster (requested) */
290  const bool skip_alignment, /* true for no cluster alignment */
291  uint32_t *sectors_per_cluster_adj, /* ret: sec per cluster (granted) */
292  uint32_t *sectors_per_fat_ptr, /* ret: sectors needed for one FAT */
293  uint32_t *data_cluster_cnt
294  )
295 /*-------------------------------------------------------------------------*\
296 | Return Value: |
297 | 0, if success, -1 and errno if failed |
298 \*=========================================================================*/
299 {
300 
301  bool finished = false;
302  uint32_t fatdata_cluster_cnt;
303  uint32_t fat_capacity;
304  uint32_t sectors_per_fat;
305  uint32_t fatdata_sect_cnt;
306  uint32_t fat_sectors_cnt;
307  /*
308  * ensure, that maximum cluster size (64KiB) is not exceeded
309  */
310  while (MS_BYTES_PER_CLUSTER_LIMIT / bytes_per_sector < sectors_per_cluster) {
311  sectors_per_cluster /= 2;
312  }
313 
314  do {
315  /*
316  * compute number of data clusters for current data:
317  * - compute cluster count for data AND fat
318  * - compute storage size for FAT
319  * - subtract from total cluster count
320  */
321  fatdata_sect_cnt = total_sector_cnt
322  - loc_align_object (rsvd_sector_cnt, sectors_per_cluster, skip_alignment);
323  if (fattype == FAT_FAT12) {
324  fatdata_sect_cnt = fatdata_sect_cnt
325  - loc_align_object (root_dir_sector_cnt, sectors_per_cluster, skip_alignment);
326  fatdata_cluster_cnt = fatdata_sect_cnt/sectors_per_cluster;
327  fat_capacity = fatdata_cluster_cnt * 3 / 2;
328  }
329  else if (fattype == FAT_FAT16) {
330  fatdata_sect_cnt = fatdata_sect_cnt
331  - loc_align_object (root_dir_sector_cnt, sectors_per_cluster, skip_alignment);
332  fatdata_cluster_cnt = fatdata_sect_cnt/sectors_per_cluster;
333  fat_capacity = fatdata_cluster_cnt * 2;
334  }
335  else { /* FAT32 */
336  fatdata_cluster_cnt = fatdata_sect_cnt/sectors_per_cluster;
337  fat_capacity = fatdata_cluster_cnt * 4;
338  }
339 
340  sectors_per_fat = ((fat_capacity
341  + (bytes_per_sector - 1))
342  / bytes_per_sector);
343 
344  fat_sectors_cnt = loc_align_object (sectors_per_fat * fat_num,
345  sectors_per_cluster,
346  skip_alignment);
347 
348  *data_cluster_cnt = (fatdata_cluster_cnt -
349  ((fat_sectors_cnt
350  + (sectors_per_cluster - 1))
351  / sectors_per_cluster));
352  /*
353  * data cluster count too big? Then make clusters bigger
354  */
355  if (((fattype == FAT_FAT12) && (*data_cluster_cnt > FAT_FAT12_MAX_CLN)) ||
356  ((fattype == FAT_FAT16) && (*data_cluster_cnt > FAT_FAT16_MAX_CLN))) {
357  sectors_per_cluster *= 2;
358  }
359  else {
360  finished = true;
361  }
362  /*
363  * when maximum cluster size is exceeded, we have invalid data, abort...
364  */
365  if (fattype == FAT_FAT12) {
366  if (MS_BYTES_PER_CLUSTER_LIMIT_FAT12 < (sectors_per_cluster * bytes_per_sector)) {
367  finished = true;
368  }
369  } else if ((sectors_per_cluster * bytes_per_sector)
370  > MS_BYTES_PER_CLUSTER_LIMIT) {
371  finished = true;
372  }
373  } while (!finished);
374 
375  *sectors_per_cluster_adj = sectors_per_cluster;
376  *sectors_per_fat_ptr = fat_sectors_cnt / fat_num;
377 
378  return 0;
379 }

381 static uint8_t
382 msdos_get_fat_type( const uint32_t bytes_per_sector,
383  const uint32_t sectors_per_cluster,
384  const uint32_t number_of_clusters )
385 {
386  uint32_t ms_sectors_per_cluster_limit_FAT12 =
387  ( MS_BYTES_PER_CLUSTER_LIMIT_FAT12 +1 ) / bytes_per_sector;
388  uint32_t ms_sectors_per_cluster_limit_FAT16 =
389  ( 0x8000 +1 ) / bytes_per_sector;
390  uint8_t fattype = FAT_FAT32;
391 
392  if ( number_of_clusters < FAT_FAT12_MAX_CLN
393  && sectors_per_cluster <= ms_sectors_per_cluster_limit_FAT12 ) {
394  fattype = FAT_FAT12;
395  }
396  else if ( number_of_clusters < FAT_FAT16_MAX_CLN
397  && sectors_per_cluster <= ms_sectors_per_cluster_limit_FAT16 ) {
398  fattype = FAT_FAT16;
399  }
400 
401  return fattype;
402 }

404 static int
405 msdos_set_sectors_per_cluster_from_request(
406  const msdos_format_request_param_t *rqdata,
407  msdos_format_param_t *fmt_params )
408 {
409  int ret_val = -1;
410  uint32_t onebit;
411 
412  if ( rqdata != NULL && rqdata->sectors_per_cluster > 0 ) {
413  fmt_params->sectors_per_cluster = rqdata->sectors_per_cluster;
414  }
415  /*
416  * check sectors per cluster.
417  * must be power of 2
418  * must be smaller than or equal to 128
419  * sectors_per_cluster*bytes_per_sector must not be bigger than 64K
420  */
421  for ( onebit = 128; onebit >= 1; onebit = onebit >> 1 ) {
422  if ( fmt_params->sectors_per_cluster >= onebit ) {
423  fmt_params->sectors_per_cluster = onebit;
424  if ( fmt_params->sectors_per_cluster
425  <= MS_BYTES_PER_CLUSTER_LIMIT / fmt_params->bytes_per_sector ) {
426  /* value is small enough so this value is ok */
427  onebit = 1;
428  ret_val = 0;
429  }
430  }
431  }
432 
433  if (ret_val != 0) {
434  errno = EINVAL;
435  }
436 
437  return ret_val;
438 }

440 static void
441 msdos_set_default_sectors_per_cluster_for_fattype(
442  msdos_format_param_t *fmt_params,
443  const uint64_t total_size )
444 {
445  if ( fmt_params->fattype == FAT_FAT12
446  || fmt_params->fattype == FAT_FAT16 ) {
447  /* start trying with small clusters */
448  fmt_params->sectors_per_cluster = 2;
449  }
450  else {
451  uint32_t gigs = ( total_size + ONE_GB ) / ONE_GB;
452  int b;
453  /* scale with the size of disk... */
454  for ( b = 31; b > 0; b-- ) {
455  if ( (gigs & ( 1 << b) ) != 0 )
456  break;
457  }
458  fmt_params->sectors_per_cluster = 1 << b;
459  }
460 }

462 /*=========================================================================*\
463 | Function: |
464 \*-------------------------------------------------------------------------*/
465 static int msdos_format_determine_fmt_params
466 (
467 /*-------------------------------------------------------------------------*\
468 | Purpose: |
469 | determine parameters for formatting |
470 +---------------------------------------------------------------------------+
471 | Input Parameters: |
472 \*-------------------------------------------------------------------------*/
473  int fd, /* disk file descriptor */
474  const msdos_format_request_param_t *rqdata, /* requested fmt parameters */
475  msdos_format_param_t *fmt_params/* computed fmt parameters */
476  )
477 /*-------------------------------------------------------------------------*\
478 | Return Value: |
479 | 0, if success, -1 and errno if failed |
480 \*=========================================================================*/
481 {
482  int ret_val = 0;
483  uint32_t sectors_per_cluster_adj = 0;
484  uint64_t total_size = 0;
485  uint32_t data_clusters_cnt;
486  uint8_t iteration_cnt = 0;
487  uint8_t fat_type = UINT8_MAX;
488 
489  memset(fmt_params,0,sizeof(*fmt_params));
490 
491  /*
492  * this one is fixed in this implementation.
493  * At least one thing we don't have to magically guess...
494  */
495  if (ret_val == 0) {
496  ret_val = rtems_disk_fd_get_media_block_size(fd, &fmt_params->bytes_per_sector);
497  }
498  if (ret_val == 0) {
499  ret_val = rtems_disk_fd_get_block_count(fd, &fmt_params->totl_sector_cnt);
500  }
501  if (ret_val == 0) {
502  total_size = (uint64_t)fmt_params->bytes_per_sector * fmt_params->totl_sector_cnt;
503  msdos_format_printf (rqdata, MSDOS_FMT_INFO_LEVEL_DETAIL,
504  "bytes per sector: %" PRIu32 "\ntotal sectors: %" PRIu32 "\ntotal size: %" PRIu64 "\n",
505  fmt_params->bytes_per_sector, fmt_params->totl_sector_cnt, total_size);
506  }
507 
508  /*
509  * determine number of FATs
510  */
511  if (ret_val == 0) {
512  if ((rqdata == NULL) || (rqdata->fat_num == 0)) {
513  fmt_params->fat_num = 2;
514  }
515  else if (rqdata->fat_num <= 6) {
516  fmt_params->fat_num = rqdata->fat_num;
517  }
518  else {
519  errno = EINVAL;
520  ret_val = -1;
521  }
522  }
523 
524  if (ret_val == 0)
525  msdos_format_printf (rqdata, MSDOS_FMT_INFO_LEVEL_DETAIL,
526  "number of fats: %d\n", fmt_params->fat_num);
527 
528  /*
529  * Now we get sort of a loop when determining things:
530  * The FAT type (FAT12/16/32) is determined ONLY from the
531  * data cluster count:
532  * Disks with data cluster count < 4085 are FAT12.
533  * Disks with data cluster count < 65525 are FAT16.
534  * The rest is FAT32 (no FAT128 available yet :-)
535  *
536  * The number of data clusters is the
537  * total capacity
538  * minus reserved sectors
539  * minus root directory ares
540  * minus storage needed for the FAT (and its copy/copies).
541  *
542  * The last item once again depends on the FAT type and the cluster count.
543  *
544  * So here is what we do in this formatter:
545  * - If a FAT type is requested from the caller, we try to modify
546  * the cluster size, until the data cluster count is in range
547  * - If no FAT type is given, we estimate a useful FAT type from
548  * the disk capacity and then adapt the cluster size
549  */
550 
551  /*
552  * determine characteristic values:
553  * - number of sectors
554  * - number of reserved sectors
555  * - number of used sectors
556  * - sectors per cluster
557  */
558  /*
559  * determine FAT type and sectors per cluster
560  * depends on
561  */
562  if (ret_val == 0) {
563  fmt_params->sectors_per_cluster = 1;
564  /*
565  * limiting values for disk size, fat type, sectors per cluster
566  * NOTE: maximum sect_per_clust is arbitrarily chosen with values that
567  * are a compromise concerning capacity and efficency
568  */
569  uint32_t fat12_sect_per_clust = 8;
570  uint32_t fat16_sect_per_clust = 32;
571 
572  if (rqdata != NULL && rqdata->sectors_per_cluster != 0) {
573  fat12_sect_per_clust = rqdata->sectors_per_cluster;
574  fat16_sect_per_clust = rqdata->sectors_per_cluster;
575  }
576 
577  if (fmt_params->totl_sector_cnt < FAT_FAT12_MAX_CLN * fat12_sect_per_clust) {
578  fmt_params->fattype = FAT_FAT12;
579  /* start trying with small clusters */
580  fmt_params->sectors_per_cluster = 2;
581  }
582  else if (fmt_params->totl_sector_cnt < FAT_FAT16_MAX_CLN * fat16_sect_per_clust) {
583  fmt_params->fattype = FAT_FAT16;
584  /* start trying with small clusters */
585  fmt_params->sectors_per_cluster = 2;
586  }
587  else {
588  uint32_t gigs = (total_size + ONE_GB) / ONE_GB;
589  int b;
590  fmt_params->fattype = FAT_FAT32;
591  /* scale with the size of disk... */
592  for (b = 31; b > 0; b--)
593  if ((gigs & (1 << b)) != 0)
594  break;
595  fmt_params->sectors_per_cluster = 1 << b;
596  }
597 
598  ret_val = msdos_set_sectors_per_cluster_from_request( rqdata, fmt_params );
599 
600  /* For now we will estimate the number of data clusters to the total number
601  * of clusters */
602  if (ret_val == 0) {
603  data_clusters_cnt =
604  fmt_params->totl_sector_cnt / fmt_params->sectors_per_cluster;
605  }
606 
607  while( ret_val == 0
608  && fmt_params->fattype != fat_type
609  && fmt_params->totl_sector_cnt > 0 ) {
610  /*
611  * Skip aligning structures or d align them
612  */
613  if (ret_val == 0 && rqdata != NULL)
614  fmt_params->skip_alignment = rqdata->skip_alignment;
615 
616  if (ret_val == 0) {
617  msdos_format_printf (rqdata, MSDOS_FMT_INFO_LEVEL_DETAIL,
618  "sectors per cluster: %d\n", fmt_params->sectors_per_cluster);
619 
620  if (fmt_params->fattype == FAT_FAT32) {
621  /* recommended: for FAT32, always set reserved sector count to 32 */
622  fmt_params->rsvd_sector_cnt = 32;
623  /* for FAT32, always set files per root directory 0 */
624  fmt_params->files_per_root_dir = 0;
625  /* location of copy of MBR */
626  fmt_params->mbr_copy_sec = 6;
627  /* location of fsinfo sector */
628  fmt_params->fsinfo_sec = 1;
629 
630  }
631  else {
632  /* recommended: for FAT12/FAT16, always set reserved sector count to 1 */
633  fmt_params->rsvd_sector_cnt = 1;
634  /* recommended: for FAT16, set files per root directory to 512 */
635  /* for FAT12/FAT16, set files per root directory */
636  /* must fill up an even count of sectors */
637  if ((rqdata != NULL) &&
638  (rqdata->files_per_root_dir > 0)) {
639  fmt_params->files_per_root_dir = rqdata->files_per_root_dir;
640  }
641  else {
642  if (fmt_params->fattype == FAT_FAT16) {
643  fmt_params->files_per_root_dir = 512;
644  }
645  else {
646  fmt_params->files_per_root_dir = 64;
647  }
648  }
649  fmt_params->files_per_root_dir = (fmt_params->files_per_root_dir +
650  (2*fmt_params->bytes_per_sector/
651  FAT_DIRENTRY_SIZE-1));
652  fmt_params->files_per_root_dir -= (fmt_params->files_per_root_dir %
653  (2*fmt_params->bytes_per_sector
654  /FAT_DIRENTRY_SIZE));
655 
656  }
657  fmt_params->root_dir_sectors =
658  (((fmt_params->files_per_root_dir * FAT_DIRENTRY_SIZE)
659  + fmt_params->bytes_per_sector - 1)
660  / fmt_params->bytes_per_sector);
661  }
662  if (ret_val == 0) {
663  /*
664  * check values to get legal arrangement of FAT type and cluster count
665  */
666 
667  ret_val = msdos_format_eval_sectors_per_cluster(fmt_params->fattype,
668  fmt_params->bytes_per_sector,
669  fmt_params->totl_sector_cnt,
670  fmt_params->rsvd_sector_cnt,
671  fmt_params->root_dir_sectors,
672  fmt_params->fat_num,
673  fmt_params->sectors_per_cluster,
674  fmt_params->skip_alignment,
675  &sectors_per_cluster_adj,
