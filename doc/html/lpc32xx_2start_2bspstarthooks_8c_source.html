141 BSP_START_TEXT_SECTION bool lpc32xx_start_pll_setup(
142  uint32_t hclkpll_ctrl,
143  uint32_t hclkdiv_ctrl,
144  bool force
145 )
146 {
147  uint32_t pwr_ctrl = LPC32XX_PWR_CTRL;
148  bool settings_ok =
149  ((LPC32XX_HCLKPLL_CTRL ^ hclkpll_ctrl) & BSP_MSK32(1, 16)) == 0
150  && ((LPC32XX_HCLKDIV_CTRL ^ hclkdiv_ctrl) & BSP_MSK32(0, 8)) == 0;
151 
152  if ((pwr_ctrl & PWR_NORMAL_RUN_MODE) == 0 || (!settings_ok && force)) {
153  /* Disable HCLK PLL output */
154  LPC32XX_PWR_CTRL = pwr_ctrl & ~PWR_NORMAL_RUN_MODE;
155 
156  /* Configure HCLK PLL */
157  LPC32XX_HCLKPLL_CTRL = hclkpll_ctrl;
158  while ((LPC32XX_HCLKPLL_CTRL & HCLK_PLL_LOCK) == 0) {
159  /* Wait */
160  }
161 
162  /* Setup HCLK divider */
163  LPC32XX_HCLKDIV_CTRL = hclkdiv_ctrl;
164 
165  /* Enable HCLK PLL output */
166  LPC32XX_PWR_CTRL = pwr_ctrl | PWR_NORMAL_RUN_MODE;
167  }
168 
169  return settings_ok;
170 }
