85 /* Function: ASCS_calc_clkreg
86  Arguments: sysfreq: System clock frequency in kHz
87  etrfreq: ETR frequency in Hz
88  Return values: Value of core's CLK-register
89  Description: Calculates value of core's CLK-register. See
90  GRASCS IP core documentation for details.

92 static int ASCS_calc_clkreg(int sysfreq, int etrfreq) {
93 
94  if(cfg->caps->usconf)
95  return 1000000/etrfreq;
96  else
97  return sysfreq*1000/etrfreq;

100 /* Function: ASCS_get_sysfreq
101  Arguments: -
102  Return values: System clock frequency in kHz, -1 if failed
103  Description: Uses AMBA plug and play to lookup system frequency
104 */
105 static int ASCS_get_sysfreq(void) {
106 
107  struct ambapp_apb_info gpt;
108  struct gptimer_regs *tregs;
109  int tmp;
110 
111  if(ambapp_find_apbslv(&ambapp_plb, VENDOR_GAISLER, GAISLER_GPTIMER, &gpt) == 1) {
112  tregs = (struct gptimer_regs *) gpt.start;
113  tmp = (tregs->scaler_reload + 1)*1000;
114  DBG("ASCS_get_sysfreq: Detected system frequency %i kHz\n",tmp);
115  if((tmp < GRASCS_MIN_SFREQ) || (tmp > GRASCS_MAX_SFREQ)) {
116  DBG("ASCS_get_sysfreq: System frequency is invalid for ASCS core\n");
117  return -1;
118  }
119  else
120  return (tregs->scaler_reload + 1)*1000;
121  }
122  DBG("ASCS_get_sysfreq: Failed to detect system frequency\n");
123  return -1;
124 }

126 /* Function: ASCS_irqhandler
127  Arguments: v: not used
128  Return values: -
129  Description: Determines the source of the interrupt, clears the
130  appropriate bits in the core's STS register and releases
131  the associated semaphore
132 */
133 static rtems_isr ASCS_irqhandler(void *v) {
134 
135  if(cfg->regs->sts & GRASCS_STS_TCDONE) {
136  /* Clear TC done bit */
137  cfg->regs->sts |= GRASCS_STS_TCDONE;
138 
139  if(--cfg->tcwords == 0)
140  /* No more TCs to perform right now */
141  rtems_semaphore_release(cfg->tcsem2);
142  else {
143  /* Block not sent yet, start next TC */
144  if(cfg->caps->dbits == 8) {
145  cfg->tcptr++;
146  cfg->regs->tcd = *((unsigned char*)cfg->tcptr);
147  }
148  else if(cfg->caps->dbits == 16) {
149  cfg->tcptr += 2;
150  cfg->regs->tcd = *((unsigned short int*)cfg->tcptr);
151  }
152  else {
153  cfg->tcptr += 4;
154  cfg->regs->tcd = *((unsigned int*)cfg->tcptr);
155  }
156  }
157  }
158 
159  if(cfg->regs->sts & GRASCS_STS_TMDONE) {
160  /* Clear TM done bit */
161  cfg->regs->sts |= GRASCS_STS_TMDONE;
162 
163  /* Store received data */
164  if(cfg->caps->dbits == 8) {
165  *((unsigned char*)cfg->tmptr) = (unsigned char)(cfg->regs->tmd & 0xFF);
166  cfg->tmptr++;
167  }
168  else if(cfg->caps->dbits == 16) {
169  *((unsigned short int*)cfg->tmptr) = (unsigned short int)(cfg->regs->tmd & 0xFFFF);
170  cfg->tmptr += 2;
171  }
172  else {
173  *((unsigned int*)cfg->tmptr) = cfg->regs->tmd;
174  cfg->tmptr += 4;
175  }
176 
177  if(--cfg->tmwords == 0)
178  /* No more TMs to perform right now */
179  rtems_semaphore_release(cfg->tmsem2);
180  else
181  /* Block not received yet, start next TM */
182  cfg->regs->cmd |= GRASCS_CMD_SENDTM;
183  }
184 }

190 /* Function: ASCS_init
191  Arguments: -
192  Return values: 0 if successful, -1 if unsuccessful
193  Description: Initializes the ASCS core
194 */
195 int ASCS_init(void) {
196 
197  int base, irq, tmp;
198 
199  DBG("ASCS_init: Starting initialization of ASCS core\n");
200 
201  /* Allocate memory for config, status and capability struct */
202  if((cfg = grlib_malloc(sizeof(*cfg))) == NULL) {
203  DBG("ASCS_init: Could not allocate memory for cfg struc\n");
204  return -1;
205  }
206 
207  if((cfg->caps = grlib_calloc(1,sizeof(*cfg->caps))) == NULL) {
208  DBG("ASCS_init: Could not allocate memory for caps struc\n");
209  goto init_error1;
210  }
211 
212  /* Create semaphores for blocking ASCS_TC/TM functions */
213  if(rtems_semaphore_create(rtems_build_name('A','S','C','0'),1,
214  (RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|
215  RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|
216  RTEMS_NO_PRIORITY_CEILING), 0,
217  &cfg->tcsem1) != RTEMS_SUCCESSFUL) {
218  DBG("ASCS_init: Failed to create semaphore ASC0\n");
219  goto init_error2;
220  }
221  if(rtems_semaphore_create(rtems_build_name('A','S','C','1'),1,
222  (RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|
223  RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|
224  RTEMS_NO_PRIORITY_CEILING), 0,
225  &cfg->tmsem1) != RTEMS_SUCCESSFUL) {
226  DBG("ASCS_init: Failed to create semaphore ASC1\n");
227  goto init_error2;
228  }
229  /* Create semaphores for waiting on ASCS_TC/TM interrupt */
230  if(rtems_semaphore_create(rtems_build_name('A','S','C','2'),0,
231  (RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|
232  RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|
233  RTEMS_NO_PRIORITY_CEILING), 0,
234  &cfg->tcsem2) != RTEMS_SUCCESSFUL) {
235  DBG("ASCS_init: Failed to create semaphore ASC2\n");
236  goto init_error2;
237  }
238  if(rtems_semaphore_create(rtems_build_name('A','S','C','3'),0,
239  (RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|
240  RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|
241  RTEMS_NO_PRIORITY_CEILING), 0,
242  &cfg->tmsem2) != RTEMS_SUCCESSFUL) {
243  DBG("ASCS_init: Failed to create semaphore ASC3\n");
244  goto init_error2;
245  }
246 
247  /* Set pointer to core registers */
248  if(ASCS_get_addr(&base, &irq) == -1)
249  goto init_error2;
250 
251  cfg->regs = (GRASCS_regs*)base;
252 
253  /* Read core capabilities */
254  tmp = cfg->regs->sts;
255  cfg->caps->dbits = ((tmp >> GRASCS_STS_DBITS_BITS) & 0x1F) + 1;
256  cfg->caps->nslaves = ((tmp >> GRASCS_STS_NSLAVES_BITS) & 0xF) + 1;
257  cfg->caps->tmconf = (tmp >> GRASCS_STS_TMCONF_BITS) & 0x1;
258  cfg->caps->usconf = (tmp >> GRASCS_STS_USCONF_BITS) & 0x1;
259 
260  /* Reset and configure core */
261  cfg->running = 0;
262  cfg->regs->cmd |= GRASCS_CMD_RESET;
263  if((tmp = ASCS_get_sysfreq()) == -1)
264  goto init_error2;
265  cfg->caps->clkfreq = tmp;
266  while(ASCS_iface_status())
267  ;
268  cfg->regs->clk = ASCS_calc_clkreg(tmp, GRASCS_DEFAULT_ETRFREQ);
269  cfg->regs->cmd = GRASCS_CMD_US1C;
270  cfg->regs->cmd |= (tmp/1000 << GRASCS_CMD_US1_BITS) | GRASCS_CMD_US1C |
271  GRASCS_CMD_TCDONE | GRASCS_CMD_TMDONE;
272 
273  /* Register interrupt routine */
274  rtems_interrupt_handler_install(irq, "grascs",
275  RTEMS_INTERRUPT_SHARED,
276  ASCS_irqhandler, NULL);
277 
278  return 0;
279 
280  init_error2:
281  free(cfg->caps);
282  init_error1:
283  free(cfg);
284  return -1;
285 }

287 /* Function: ASCS_input_select
288  Arguments: slave: The number of the slave that is active,
289  numbered from 0-15
290  Return values: 0 if successful, -GRASCS_ERROR_CAPFAULT if slave value
291  is negative or too big, -GRASCS_ERROR_TRANSACTIVE if
292  a TM is active.
293  Description: Sets the slave_sel bits in the core's CMD register.
294  they are used to choose which slave the core listens
295  to when performing a TM. The bits can't be set
296  during a TM, and the function will in such a case fail.
297 */
298 int ASCS_input_select(int slave) {
299 
300  if((slave < 0) || (slave > cfg->caps->nslaves)) {
301  /* Slave number is negative or too big */
302  DBG("ASCS_input_select: Wrong slave number\n");
303  return -GRASCS_ERROR_CAPFAULT;
304  }
305 
306  if(rtems_semaphore_obtain(cfg->tmsem1,RTEMS_NO_WAIT,RTEMS_NO_TIMEOUT) !=
307  RTEMS_SUCCESSFUL) {
308  /* Can't change active slave during a TM */
309  DBG("ASCS_input_select: Transaction active\n");
310  return -GRASCS_ERROR_TRANSACTIVE;
311  }
312 
313  cfg->regs->cmd = ((cfg->regs->cmd &= ~GRASCS_CMD_SLAVESEL) |
314  (slave << GRASCS_CMD_SLAVESEL_BITS));
315 
316  rtems_semaphore_release(cfg->tmsem1);
317  return 0;
318 }

320 /* Function: ASCS_etr_select
321  Arguments: src: The source of the ETR signal, valid values are
322  0-GRASCS_MAX_TMS (0 = internal source, 1-GRASCS_MAX_TMS =
323  external time markers 1-GRASCS_MAX_TMS).
324  freq: ETR frequency in Hz. Valid values are
325  GRASCS_MIN_ETRFREQ-GRASCS_MAX_ETRFREQ
326  Return values: 0 if successful, -GRASCS_ERROR_CAPFAULT if src or freq values
327  are invalid, -GRASCS_ERROR_STARTSTOP if synchronization interface
328  isn't stopped.
329  Description: Changes the source for the ETR signal. The frequency of source signal
330  is assumed to be the same as the frequency of the freq input
331 */
332 int ASCS_etr_select(int etr, int freq) {
333 
334  if((etr < 0) || (etr > GRASCS_MAX_TMS) || ((cfg->caps->tmconf == 0) && (etr > 0)) ||
335  (freq < GRASCS_MIN_ETRFREQ) || (freq > GRASCS_MAX_ETRFREQ)) {
336  /* ETR source value or frequency is invalid */
337  DBG("ASCS_etr_select: Wrong etr src number or wrong frequency\n");
338  return -GRASCS_ERROR_CAPFAULT;
339  }
340 
341  if(cfg->regs->sts & GRASCS_STS_ERUNNING) {
342  /* Synchronization interface is running */
343  DBG("ASCS_etr_select: Synch interface is running\n");
344  return -GRASCS_ERROR_STARTSTOP;
345  }
346 
347  cfg->regs->clk = ASCS_calc_clkreg(cfg->caps->clkfreq,freq);
348  cfg->regs->cmd = ((cfg->regs->cmd &= ~GRASCS_CMD_ETRCTRL) |
349  (etr << GRASCS_CMD_ETRCTRL_BITS));
350 
351  return 0;
352 }

366 /* Function: ASCS_stop
367  Arguments: -
368  Return values: -
369  Description: Disables the serial interface. This function will
370  block until possible calls to TC_send(_block) and
371  TM_recv(_block) has returned in order to be sure
372  that started transactions will be performed.
373 */
374 void ASCS_stop(void) {
375 
376  /* Set internal status to stopped */
377  cfg->running = 0;
378 
379  /* Obtain semaphores to avoid possible situation where a
380  TC_send(_block) or TM_recv(_block) is aborted and driver is
381  waiting forever for an interrupt */
382  rtems_semaphore_obtain(cfg->tcsem1,RTEMS_WAIT,RTEMS_NO_TIMEOUT);
383  rtems_semaphore_obtain(cfg->tmsem1,RTEMS_WAIT,RTEMS_NO_TIMEOUT);
384 
385  /* Change actual register value */
386  cfg->regs->cmd &= ~GRASCS_CMD_STARTSTOP;
387 
388  /* Release the semaphores */
389  rtems_semaphore_release(cfg->tcsem1);
390  rtems_semaphore_release(cfg->tmsem1);
391 }

393 /* Function: ASCS_iface_status
394  Arguments: -
395  Return values: 0 if both serial interface and synch interface is stopped,
396  1 if serial interface is running buth synch interface is
397  stopped, 2 if serial interface is stopped but synch interface
398  is running, 3 if both serial and synch interface is running
399  Description: Reads the core's STS register and reports the status of the
400  serial and synch interfaces
401 */
402 int ASCS_iface_status(void) {
403 
404  return ((cfg->regs->sts & 0x3) & (0x2 | cfg->running));
405 }

407 /* Function: ASCS_TC_send
408  Arguments: word: Pointer to a word that should be sent
409  Return values: 0 on success
410  -GRASCS_ERROR_STARTSTOP if serial interface is stopped,
411  -GRASCS_ERROR_TRANSACTIVE if another TC is in progress.
412  Description: Start a TC and sends the data that word points to.
413 */
414 int ASCS_TC_send(int *word) {
415 
416  int retval;
417 
418  if(rtems_semaphore_obtain(cfg->tcsem1,RTEMS_NO_WAIT,RTEMS_NO_TIMEOUT) !=
419  RTEMS_SUCCESSFUL) {
420  /* Can't start a TC_send if another TC_send of TC_send_block is
421  in progress */
422  DBG("ASCS_TC_send: Could not obtain semaphore, transcation probably in progress\n");
423  return -GRASCS_ERROR_TRANSACTIVE;
424  }
425 
426  if(!cfg->running) {
427  /* Can't start a TC if serial interface isn't started */
428  DBG("ASCS_TC_send: Serial interface is not started\n");
429  retval = -GRASCS_ERROR_STARTSTOP;
430  }
431  else {
432  /* Start the transfer */
433  cfg->tcwords = 1;
434  if(cfg->caps->dbits == 8)
435  cfg->regs->tcd = *((unsigned char*)word);
436  else if(cfg->caps->dbits == 16)
437  cfg->regs->tcd = *((unsigned short int*)((int)word & ~1));
438  else
439  cfg->regs->tcd = *((unsigned int*)((int)word & ~3));
440 
441  /* Wait until transfer is complete */
442  rtems_semaphore_obtain(cfg->tcsem2,RTEMS_WAIT,RTEMS_NO_TIMEOUT);
443  retval = 0;
444  }
445 
446  rtems_semaphore_release(cfg->tcsem1);
447 
448  return retval;
449 }

451 /* Function: ASCS_TC_send_block
452  Arguments: block: Pointer to the start of a datablock that
453  should be sent.
454  ntrans: Number of transfers needed to transfer
455  the block.
456  Return values: 0 if successfull, -GRASCS_ERROR_STARTSTOP if TC
457  couldn't be started because serial interface is
458  stopped, -GRASCS_ERROR_TRANSACTIVE if TC couldn't
459  be started because another TC isn't done yet.
460  Description: Starts ntrans TCs and sends the data that starts at the
461  address that block points to. The size of each
462  transaction will vary depending on whether the core is
463  configured for 8, 16, or 32 bits data transfers.
464 */
465 int ASCS_TC_send_block(int *block, int ntrans) {
466 
467  int retval;
468 
469  if(rtems_semaphore_obtain(cfg->tcsem1,RTEMS_NO_WAIT,RTEMS_NO_TIMEOUT) !=
470  RTEMS_SUCCESSFUL) {
471  /* Can't start a TC_send_block if another TC_send of TC_send_block is
472  in progress */
473  DBG("ASCS_TC_send_block: Could not obtain semaphore, transcation probably in progress\n");
474  return -GRASCS_ERROR_TRANSACTIVE;
475  }
476 
477  if(!cfg->running) {
478  /* Can't start a TC if serial interface isn't started */
479  DBG("ASCS_TC_send_block: Serial interface is not started\n");
480  retval = -GRASCS_ERROR_STARTSTOP;
481  }
482  else {
483  /* Start the first transfer */
484  cfg->tcwords = ntrans;
485  if(cfg->caps->dbits == 8) {
486  cfg->tcptr = (int)block;
487  cfg->regs->tcd = *((unsigned char*)cfg->tcptr);
488  }
489  else if(cfg->caps->dbits == 16) {
490  cfg->tcptr = (int)block & ~1;
491  cfg->regs->tcd = *((unsigned short int*)cfg->tcptr);
492  }
493  else {
494  cfg->tcptr = (int)block & ~3;
495  cfg->regs->tcd = *((unsigned int*)cfg->tcptr);
496  }
497 
498  /* Wait until all transfers are complete */
499  rtems_semaphore_obtain(cfg->tcsem2,RTEMS_WAIT,RTEMS_NO_TIMEOUT);
500  retval = 0;
501  }
502 
503  rtems_semaphore_release(cfg->tcsem1);
504 
505  return retval;
506 }
