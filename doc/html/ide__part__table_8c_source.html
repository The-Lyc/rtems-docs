29 /*
30  * get_sector --
31  * gets sector from the disk
32  *
33  * PARAMETERS:
34  * fd - file descriptor
35  * sector_num - number of sector to read
36  * sector - returned pointer to pointer to allocated
37  * sector_data_t structure
38  *
39  * RETURNS:
40  * RTEMS_SUCCESSFUL, if success;
41  * RTEMS_NO_MEMORY, if canot allocate memory for sector data;
42  * other error codes returned by rtems_bdbuf_read().
43  *
44  * NOTES:
45  * get_sector() operates with device via bdbuf library,
46  * and does not support devices with sector size other than 512 bytes
47  */
48 static rtems_status_code
49 get_sector(int fd,
50  uint32_t sector_num,
51  rtems_sector_data_t **sector)
52 {
53  rtems_sector_data_t *s;
54  ssize_t n;
55  off_t off;
56  off_t new_off;
57 
58  if (sector == NULL)
59  {
60  return RTEMS_INTERNAL_ERROR;
61  }
62 
63  off = sector_num * RTEMS_IDE_SECTOR_SIZE;
64  new_off = lseek(fd, off, SEEK_SET);
65  if (new_off != off) {
66  return RTEMS_IO_ERROR;
67  }
68 
69  s = (rtems_sector_data_t *) malloc(sizeof(rtems_sector_data_t) + RTEMS_IDE_SECTOR_SIZE);
70  if (s == NULL)
71  {
72  return RTEMS_NO_MEMORY;
73  }
74 
75  n = read(fd, s->data, RTEMS_IDE_SECTOR_SIZE);
76  if (n != RTEMS_IDE_SECTOR_SIZE)
77  {
78  free(s);
79  return RTEMS_IO_ERROR;
80  }
81 
82  s->sector_num = sector_num;
83 
84  *sector = s;
85 
86  return RTEMS_SUCCESSFUL;
87 }

150 /*
151  * data_to_part_desc --
152  * parses raw partition table sector data
153  * to partition description structure
154  *
155  * PARAMETERS:
156  * data - raw partition table sector data
157  * new_part_desc - pointer to returned partition description structure
158  *
159  * RETURNS:
160  * RTEMS_SUCCESSFUL, if success;
161  * RTEMS_NO_MEMOTY, if cannot allocate memory for part_desc_t strucure;
162  * RTEMS_INTERNAL_ERROR, if other error occurs.
163  */
164 static rtems_status_code
165 data_to_part_desc(uint8_t *data, rtems_part_desc_t **new_part_desc)
166 {
167  rtems_part_desc_t *part_desc;
168  uint32_t temp;
169 
170  if (new_part_desc == NULL)
171  {
172  return RTEMS_INTERNAL_ERROR;
173  }
174 
175  *new_part_desc = NULL;
176 
177  if ((part_desc = calloc(1, sizeof(rtems_part_desc_t))) == NULL)
178  {
179  return RTEMS_NO_MEMORY;
180  }
181 
182  part_desc->bootable = *(data + RTEMS_IDE_PARTITION_BOOTABLE_OFFSET);
183  part_desc->sys_type = *(data + RTEMS_IDE_PARTITION_SYS_TYPE_OFFSET);
184 
185  /* read the offset start position and partition size in sectors */
186 
187  /* due to incorrect data alignment one have to align data first */
188  memcpy(&temp, data + RTEMS_IDE_PARTITION_START_OFFSET, sizeof(uint32_t));
189  part_desc->start = LE_TO_CPU_U32(temp);
190 
191  memcpy(&temp, data + RTEMS_IDE_PARTITION_SIZE_OFFSET, sizeof(uint32_t));
192  part_desc->size = LE_TO_CPU_U32(temp);
193 
194  /*
195  * use partitions that are
196  * - extended
197  * or
198  * - FAT type and non-zero
199  */
200  if (is_extended(part_desc->sys_type) ||
201  ((is_fat_partition(part_desc->sys_type)) && (part_desc->size != 0))) {
202  *new_part_desc = part_desc;
203  }
204  else {
205  /* empty partition */
206  free(part_desc);
207  }
208  return RTEMS_SUCCESSFUL;
209 }

212 /*
213  * read_extended_partition --
214  * recursively reads extended partition sector from the device
215  * and constructs the partition table tree
216  *
217  * PARAMETERS:
218  * fd - file descriptor
219  * start - start sector of primary extended partition, used for
220  * calculation of absolute partition sector address
221  * ext_part - description of extended partition to process
222  *
223  * RETURNS:
224  * RTEMS_SUCCESSFUL if success,
225  * RTEMS_NO_MEMOTY if cannot allocate memory for part_desc_t strucure,
226  * RTEMS_INTERNAL_ERROR if other error occurs.
227  */
228 static rtems_status_code
229 read_extended_partition(int fd, uint32_t start, rtems_part_desc_t *ext_part)
230 {
231  int i;
232  rtems_sector_data_t *sector = NULL;
233  uint32_t here;
234  uint8_t *data;
235  rtems_part_desc_t *new_part_desc;
236  rtems_status_code rc;
237 
238  if ((ext_part == NULL) || (ext_part->disk_desc == NULL))
239  {
240  return RTEMS_INTERNAL_ERROR;
241  }
242 
243  /* get start sector of current extended partition */
244  here = ext_part->start;
245 
246  /* get first extended partition sector */
247 
248  rc = get_sector(fd, here, &sector);
249  if (rc != RTEMS_SUCCESSFUL)
250  {
251  if (sector)
252  free(sector);
253  return rc;
254  }
255 
256  if (!msdos_signature_check(sector))
257  {
258  free(sector);
259  return RTEMS_INTERNAL_ERROR;
260  }
261 
262  /* read and process up to 4 logical partition descriptors */
263 
264  data = sector->data + RTEMS_IDE_PARTITION_TABLE_OFFSET;
265 
266  for (i = 0; i < RTEMS_IDE_PARTITION_MAX_SUB_PARTITION_NUMBER; i++)
267  {
268  /* if data_to_part_desc fails skip this partition
269  * and parse the next one
270  */
271  rc = data_to_part_desc(data, &new_part_desc);
272  if (rc != RTEMS_SUCCESSFUL)
273  {
274  free(sector);
275  return rc;
276  }
277 
278  if (new_part_desc == NULL)
279  {
280  data += RTEMS_IDE_PARTITION_DESCRIPTOR_SIZE;
281  continue;
282  }
283 
284  ext_part->sub_part[i] = new_part_desc;
285  new_part_desc->ext_part = ext_part;
286  new_part_desc->disk_desc = ext_part->disk_desc;
287 
288  if (is_extended(new_part_desc->sys_type))
289  {
290  new_part_desc->log_id = EMPTY_PARTITION;
291  new_part_desc->start += start;
292  read_extended_partition(fd, start, new_part_desc);
293  }
294  else
295  {
296  rtems_disk_desc_t *disk_desc = new_part_desc->disk_desc;
297  disk_desc->partitions[disk_desc->last_log_id] = new_part_desc;
298  new_part_desc->log_id = ++disk_desc->last_log_id;
299  new_part_desc->start += here;
300  new_part_desc->end = new_part_desc->start + new_part_desc->size - 1;
301  }
302  data += RTEMS_IDE_PARTITION_DESCRIPTOR_SIZE;
303  }
304 
305  free(sector);
306 
307  return RTEMS_SUCCESSFUL;
308 }

311 /*
312  * read_mbr --
313  * reads Master Boot Record (sector 0) of physical device and
314  * constructs disk description structure
315  *
316  * PARAMETERS:
317  * disk_desc - returned disc description structure
318  *
319  * RETURNS:
320  * RTEMS_SUCCESSFUL if success,
321  * RTEMS_INTERNAL_ERROR otherwise
322  */
323 static rtems_status_code
324 read_mbr(int fd, rtems_disk_desc_t *disk_desc)
325 {
326  int part_num;
327  rtems_sector_data_t *sector = NULL;
328  rtems_part_desc_t *part_desc;
329  uint8_t *data;
330  rtems_status_code rc;
331 
332  /* get MBR sector */
333  rc = get_sector(fd, 0, &sector);
334  if (rc != RTEMS_SUCCESSFUL)
335  {
336  if (sector)
337  free(sector);
338  return rc;
339  }
340 
341  /* check if the partition table structure is MS-DOS style */
342  if (!msdos_signature_check(sector))
343  {
344  free(sector);
345  return RTEMS_INTERNAL_ERROR;
346  }
347 
348  /* read and process 4 primary partition descriptors */
349 
350  data = sector->data + RTEMS_IDE_PARTITION_TABLE_OFFSET;
351 
352  for (part_num = 0;
353  part_num < RTEMS_IDE_PARTITION_MAX_SUB_PARTITION_NUMBER;
354  part_num++)
355  {
356  rc = data_to_part_desc(data, &part_desc);
357  if (rc != RTEMS_SUCCESSFUL)
358  {
359  free(sector);
360  return rc;
361  }
362 
363  if (part_desc != NULL)
364  {
365  part_desc->log_id = part_num + 1;
366  part_desc->disk_desc = disk_desc;
367  part_desc->end = part_desc->start + part_desc->size - 1;
368  disk_desc->partitions[part_num] = part_desc;
369  }
370  else
371  {
372  disk_desc->partitions[part_num] = NULL;
373  }
374 
375  data += RTEMS_IDE_PARTITION_DESCRIPTOR_SIZE;
376  }
377 
378  free(sector);
379 
380  disk_desc->last_log_id = RTEMS_IDE_PARTITION_MAX_SUB_PARTITION_NUMBER;
381 
382  /* There cannot be more than one extended partition,
383  but we are to process each primary partition */
384  for (part_num = 0;
385  part_num < RTEMS_IDE_PARTITION_MAX_SUB_PARTITION_NUMBER;
386  part_num++)
387  {
388  part_desc = disk_desc->partitions[part_num];
389  if (part_desc != NULL && is_extended(part_desc->sys_type))
390  {
391  read_extended_partition(fd, part_desc->start, part_desc);
392  free(part_desc);
393  disk_desc->partitions[part_num] = NULL;
394  }
395  }
396 
397  return RTEMS_SUCCESSFUL;
398 }

460  * and creates disk description structure
461  *
462  * PARAMETERS:
463  * dev_name - path to physical device in /dev filesystem
464  * disk_desc - returned disc description structure
465  *
466  * RETURNS:
467  * RTEMS_SUCCESSFUL if success,

470 static rtems_status_code
471 partition_table_get(const char *dev_name, rtems_disk_desc_t *disk_desc)
472 {
473  struct stat dev_stat;
474  rtems_status_code rc;
475  int fd;
476 
477  fd = open(dev_name, O_RDONLY);
478  if (fd < 0)
479  {
480  return RTEMS_INTERNAL_ERROR;
481  }
482 
483  rc = fstat(fd, &dev_stat);
484  if (rc != RTEMS_SUCCESSFUL)
485  {

488  }
489

492  RTEMS_IDE_SECTOR_SIZE;
493 
494  rc = read_mbr(fd, disk_desc);
495 
496  close(fd);
497 
498  return rc;
499 }
