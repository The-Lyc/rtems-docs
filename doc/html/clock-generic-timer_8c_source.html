27 /*
28  * Clock driver using the ARMv7-AR Generic Timer. The BSP must provide the
29  * following function via <bsp.h>:
30  *
31  * void arm_generic_timer_get_config(uint32_t *frequency, uint32_t *irq);
32  *
33  * The BSP may optionally define ARM_GENERIC_TIMER_USE_VIRTUAL in <bsp.h> to
34  * use the virtual timer instead of the physical timer.
35  */

84 static void arm_gt_clock_at_tick(void)
85 {
86  uint64_t cval;
87  uint32_t interval;
88 
89  interval = arm_gt_clock_instance.interval;
90  cval = arm_gt_clock_get_compare_value();
91  cval += interval;
92  arm_gt_clock_set_compare_value(cval);
93 #ifdef ARM_GENERIC_TIMER_UNMASK_AT_TICK
94  arm_gt_clock_set_control(0x1);
95 #endif /* ARM_GENERIC_TIMER_UNMASK_AT_TICK */
96 }

143 static void arm_gt_clock_initialize(void)
144 {
145  uint64_t frequency;
146  uint64_t us_per_tick;
147  uint32_t interval;
148  uint64_t cval;
149  struct timecounter *tc;
150 
151  tc = &arm_gt_clock_instance.tc;
152  frequency = tc->tc_frequency;
153  us_per_tick = rtems_configuration_get_microseconds_per_tick();
154  interval = (uint32_t) ((frequency * us_per_tick) / 1000000);
155  cval = arm_gt_clock_get_count();
156  cval += interval;
157  arm_gt_clock_instance.interval = interval;
158 
159  arm_gt_clock_gt_init(cval);
160  arm_gt_clock_secondary_initialization(cval);
161 
162  tc->tc_get_timecount = arm_gt_clock_get_timecount;
163  tc->tc_counter_mask = 0xffffffff;
164  tc->tc_quality = RTEMS_TIMECOUNTER_QUALITY_CLOCK_DRIVER;
165  rtems_timecounter_install(tc);
166 }
