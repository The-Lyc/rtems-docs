106 rtems_rfs_rtems_eval_token(

108  void *arg,
109  const char *token,

111 )
112 {

116  bool access_ok = rtems_rfs_rtems_eval_perms (ctx, RTEMS_FS_PERMS_EXEC, inode);
117 
118  if (access_ok) {
119  if (rtems_filesystem_is_current_directory (token, tokenlen)) {
120  rtems_filesystem_eval_path_clear_token (ctx);
121  } else {
122  rtems_filesystem_location_info_t *currentloc =
123  rtems_filesystem_eval_path_get_currentloc( ctx );
124  rtems_rfs_file_system* fs = rtems_rfs_rtems_pathloc_dev (currentloc);
125  rtems_rfs_ino entry_ino;
126  uint32_t entry_doff;

129  inode,
130  token,
131  tokenlen,
132  &entry_ino,
133  &entry_doff
134  );
135 
136  if (rc == 0) {
137  rc = rtems_rfs_inode_close (fs, inode);
138  if (rc == 0) {
139  rc = rtems_rfs_inode_open (fs, entry_ino, inode, true);
140  }
141 
142  if (rc != 0) {
143  /*
144  * This prevents the rtems_rfs_inode_close() from doing something in
145  * rtems_rfs_rtems_eval_path().
146  */
147  memset (inode, 0, sizeof(*inode));

159

161

163  rtems_rfs_rtems_follow_link (ctx, fs, entry_ino);
164  } else {

166  if (rc == 0) {
167  rtems_rfs_rtems_set_pathloc_ino (currentloc, entry_ino);
168  rtems_rfs_rtems_set_pathloc_doff (currentloc, entry_doff);
169 
170  if (!terminal) {
171  status = RTEMS_FILESYSTEM_EVAL_PATH_GENERIC_CONTINUE;
172  }
173  } else {
174  rtems_filesystem_eval_path_error (
175  ctx,
176  rtems_rfs_rtems_error ("eval_path: set handlers", rc)
177  );
178  }
179  }
180  }
181  }
182  }
183

185 }

749 int
750 rtems_rfs_rtems_initialise (rtems_filesystem_mount_table_entry_t* mt_entry,
751  const void* data)
752 {
753  rtems_rfs_rtems_private* rtems;
754  rtems_rfs_file_system* fs;
755  uint32_t flags = 0;
756  uint32_t max_held_buffers = RTEMS_RFS_FS_MAX_HELD_BUFFERS;
757  const char* options = data;
758  int rc;
759 
760  /*
761  * Parse the options the user specifiies.
762  */
763  while (options)
764  {
765  printf ("options=%s\n", options);
766  if (strncmp (options, "hold-bitmaps",
767  sizeof ("hold-bitmaps") - 1) == 0)
768  flags |= RTEMS_RFS_FS_BITMAPS_HOLD;
769  else if (strncmp (options, "no-local-cache",
770  sizeof ("no-local-cache") - 1) == 0)
771  flags |= RTEMS_RFS_FS_NO_LOCAL_CACHE;
772  else if (strncmp (options, "max-held-bufs",
773  sizeof ("max-held-bufs") - 1) == 0)
774  {
775  max_held_buffers = strtoul (options + sizeof ("max-held-bufs"), 0, 0);
776  }
777  else
778  return rtems_rfs_rtems_error ("initialise: invalid option", EINVAL);
779 
780  options = strchr (options, ',');
781  if (options)
782  {
783  ++options;
784  if (*options == '\0')
785  options = NULL;
786  }
787  }
788 
789  rtems = malloc (sizeof (rtems_rfs_rtems_private));
790  if (!rtems)
791  return rtems_rfs_rtems_error ("initialise: local data", ENOMEM);
792 
793  memset (rtems, 0, sizeof (rtems_rfs_rtems_private));
794 
795  rc = rtems_rfs_mutex_create (&rtems->access);
796  if (rc > 0)
797  {
798  free (rtems);
799  return rtems_rfs_rtems_error ("initialise: cannot create mutex", rc);
800  }
801 
802  rc = rtems_rfs_mutex_lock (&rtems->access);
803  if (rc > 0)
804  {
805  rtems_rfs_mutex_destroy (&rtems->access);
806  free (rtems);
807  return rtems_rfs_rtems_error ("initialise: cannot lock access mutex", rc);
808  }
809 
810  rc = rtems_rfs_fs_open (mt_entry->dev, rtems, flags, max_held_buffers, &fs);
811  if (rc)
812  {
813  rtems_rfs_mutex_unlock (&rtems->access);
814  rtems_rfs_mutex_destroy (&rtems->access);
815  free (rtems);
816  return rtems_rfs_rtems_error ("initialise: open", errno);
817  }
818 
819  mt_entry->fs_info = fs;
820  mt_entry->ops = &rtems_rfs_ops;
821  mt_entry->mt_fs_root->location.node_access = (void*) RTEMS_RFS_ROOT_INO;
822  mt_entry->mt_fs_root->location.handlers = &rtems_rfs_rtems_dir_handlers;
823 
824  rtems_rfs_rtems_unlock (fs);
825 
826  return 0;
827 }
