64 static int ppc_clock_exception_handler(
65  BSP_Exception_frame *frame,
66  unsigned number
67 )
68 {
69  uint32_t delta = ppc_clock_decrementer_value;
70  uint32_t next = ppc_clock_next_time_base;
71  uint32_t dec = 0;
72  uint32_t now = 0;
73  uint32_t msr = 0;
74 
75  do {
76  /* Increment clock ticks */
77  Clock_driver_ticks += 1;
78 
79  /* Enable external exceptions */
80  msr = ppc_external_exceptions_enable();
81 
82  /* Call clock ticker */
83  ppc_clock_tick();
84 
85  /* Restore machine state */
86  ppc_external_exceptions_disable( msr);
87 
88  /* Next time base */
89  next += delta;
90 
91  /* Current time */
92  now = ppc_time_base();
93 
94  /* New decrementer value */
95  dec = next - now;
96  } while (dec > delta);
97 
98  /* Set decrementer */
99  ppc_set_decrementer_register( dec);
100 
101  /* Expected next time base */
102  ppc_clock_next_time_base = next;
103 
104  return 0;
105 }

170 void _Clock_Initialize( void )
171 {
172  uint64_t frequency = bsp_time_base_frequency;
173  uint64_t us_per_tick = rtems_configuration_get_microseconds_per_tick();
174  uint32_t interval = (uint32_t) ((frequency * us_per_tick) / 1000000);
175 
176  /*
177  * Set default ticker.
178  */
179  ppc_clock_tick = rtems_timecounter_tick;
180 
181  if (ppc_cpu_is_bookE() != PPC_BOOKE_405) {
182  /* Decrementer value */
183  ppc_clock_decrementer_value = interval - 1;
184 
185  /* Check decrementer value */
186  if (ppc_clock_decrementer_value == 0) {
187  ppc_clock_decrementer_value = PPC_CLOCK_DECREMENTER_MAX;
188  RTEMS_SYSLOG_ERROR( "decrementer value would be zero, will be set to maximum value instead\n");
189  }
190  if (ppc_cpu_is_bookE()) {
191  /* Set decrementer auto-reload value */
192  PPC_SET_SPECIAL_PURPOSE_REGISTER( BOOKE_DECAR, ppc_clock_decrementer_value);
193 
194  /* Install exception handler */
195  ppc_exc_set_handler( ASM_BOOKE_DEC_VECTOR, ppc_clock_exception_handler_booke);
196 
197  /* Enable decrementer and auto-reload */
198  PPC_SET_SPECIAL_PURPOSE_REGISTER_BITS( BOOKE_TCR, BOOKE_TCR_DIE | BOOKE_TCR_ARE);
199  } else {
200  /* Here the decrementer value is actually the interval */
201  ++ppc_clock_decrementer_value;
202 
203  /* Initialize next time base */
204  ppc_clock_next_time_base = ppc_time_base() + ppc_clock_decrementer_value;
205 
206  /* Install exception handler */
207  ppc_exc_set_handler( ASM_DEC_VECTOR, ppc_clock_exception_handler_first);
208  }
209 
210  /* Set the decrementer value */
211  ppc_set_decrementer_register( ppc_clock_decrementer_value);
212  } else {
213  /* PIT interval value */
214  ppc_clock_decrementer_value = interval;
215 
216  /* Install exception handler */
217  ppc_exc_set_handler(ASM_BOOKE_DEC_VECTOR, ppc_clock_exception_handler_ppc405);
218 
219  /* Enable PIT and auto-reload */
220  PPC_SET_SPECIAL_PURPOSE_REGISTER_BITS(PPC405_TCR, BOOKE_TCR_DIE | BOOKE_TCR_ARE);
221 
222  /* Set PIT auto-reload and initial value */
223  PPC_SET_SPECIAL_PURPOSE_REGISTER(PPC405_PIT, interval);
224  }
225 
226  /* Install timecounter */
227  ppc_tc.tc_get_timecount = ppc_get_timecount;
228  ppc_tc.tc_counter_mask = 0xffffffff;
229  ppc_tc.tc_frequency = frequency;
230  ppc_tc.tc_quality = RTEMS_TIMECOUNTER_QUALITY_CLOCK_DRIVER;
231  rtems_timecounter_install(&ppc_tc);
232 }
