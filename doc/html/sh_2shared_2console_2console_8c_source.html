43  void *arg
44 )
45 {
46  rtems_device_driver status;
47  struct stat st;
48  int rv;
49 
50  rv = stat( BSP_CONSOLE_DEVNAME, &st );
51  if ( rv != 0 )
52  rtems_fatal_error_occurred(rv);
53 
54  status = rtems_io_register_name(
55  "/dev/console",
56  rtems_filesystem_dev_major_t( st.st_rdev ),
57  rtems_filesystem_dev_minor_t( st.st_rdev )
58  );
59  if (status != RTEMS_SUCCESSFUL)
60  rtems_fatal_error_occurred(status);
61 
62  return RTEMS_SUCCESSFUL;
63 }
