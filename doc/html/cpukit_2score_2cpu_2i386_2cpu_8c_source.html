59 void _CPU_Initialize(void)
60 {
61 #if CPU_HARDWARE_FP
62  register uint16_t fp_status __asm__ ("ax");
63  register Context_Control_fp *fp_context;
64 #endif
65 
66  /*
67  * The following code saves a NULL i387 context which is given
68  * to each task at start and restart time. The following code
69  * is based upon that provided in the i386 Programmer's
70  * Manual and should work on any coprocessor greater than
71  * the i80287.
72  *
73  * NOTE: The NO WAIT form of the coprocessor instructions
74  * MUST be used in case there is not a coprocessor
75  * to wait for.
76  */
77 
78 #if CPU_HARDWARE_FP
79  fp_status = 0xa5a5;
80  __asm__ volatile( "fninit" );
81  __asm__ volatile( "fnstsw %0" : "=a" (fp_status) : "0" (fp_status) );
82 
83  if ( fp_status == 0 ) {
84 
85  fp_context = &_CPU_Null_fp_context;
86 
87 #ifdef __SSE__
88  asm volatile( "fstcw %0":"=m"(fp_context->fpucw) );
89 #else
90  __asm__ volatile( "fsave (%0)" : "=r" (fp_context)
91  : "0" (fp_context)
92  );
93 #endif
94  }
95 #endif
96 
97 #ifdef __SSE__
98 
99  __asm__ volatile("stmxcsr %0":"=m"(fp_context->mxcsr));
100 
101  /* The BSP must enable the SSE extensions (early).
102  * If any SSE instruction was already attempted
103  * then that crashed the system.
104  * As a courtesy, we double-check here but it
105  * may be too late (which is also why we don't
106  * enable SSE here).
107  */
108  {
109  uint32_t cr4;
110  __asm__ __volatile__("mov %%cr4, %0":"=r"(cr4));
111  if ( 0x600 != (cr4 & 0x600) ) {
112  printk("PANIC: RTEMS was compiled for SSE but BSP did not enable it"
113  "(CR4: 0%" PRIu32 ")\n", cr4);
114  while ( 1 ) {
115  __asm__ __volatile__("hlt");
116  }
117  }
118  }
119 #endif
120 }

122 /*
123  * Stack alignment note:
124  *
125  * We want the stack to look to the '_entry_point' routine
126  * like an ordinary stack frame as if '_entry_point' was
127  * called from C-code.
128  * Note that '_entry_point' is jumped-to by the 'ret'
129  * instruction returning from _CPU_Context_switch() or
130  * _CPU_Context_restore() thus popping the _entry_point
131  * from the stack.
132  * However, _entry_point expects a frame to look like this:
133  *
134  * args [_Thread_Handler expects no args, however]
135  * ------ (alignment boundary)
136  * SP-> return_addr return here when _entry_point returns which (never happens)
137  *
138  *
139  * Hence we must initialize the stack as follows
140  *
141  * [arg1 ]: n/a
142  * [arg0 (aligned)]: n/a
143  * [ret. addr ]: NULL
144  * SP-> [jump-target ]: _entry_point
145  *
146  * When Context_switch returns it pops the _entry_point from
147  * the stack which then finds a standard layout.
148  */

150 void _CPU_Context_Initialize(
151  Context_Control *the_context,
152  void *_stack_base,
153  size_t _size,
154  uint32_t _isr,
155  void (*_entry_point)( void ),
156  bool is_fp,
157  void *tls_area
158 )
159 {
160  uint32_t _stack;
161  uint32_t tcb;
162 
163  (void) is_fp; /* avoid warning for being unused */
164 
165  if ( _isr ) {
166  the_context->eflags = CPU_EFLAGS_INTERRUPTS_OFF;
167  } else {
168  the_context->eflags = CPU_EFLAGS_INTERRUPTS_ON;
169  }
170 
171  _stack = ((uint32_t)(_stack_base)) + (_size);
172  _stack &= ~ (CPU_STACK_ALIGNMENT - 1);
173  _stack -= 2*sizeof(void *); /* see above for why we need to do this */
174  *((void (**)(void))(_stack)) = (_entry_point);
175  the_context->ebp = (void *) 0;
176  the_context->esp = (void *) _stack;
177 
178  if ( tls_area != NULL ) {
179  tcb = (uint32_t) _TLS_TCB_after_TLS_block_initialize( tls_area );
180  } else {
181  tcb = 0;
182  }
183 
184  the_context->gs.limit_15_0 = 0xffff;
185  the_context->gs.base_address_15_0 = (tcb >> 0) & 0xffff;
186  the_context->gs.type = 0x2;
187  the_context->gs.descriptor_type = 0x1;
188  the_context->gs.limit_19_16 = 0xf;
189  the_context->gs.present = 0x1;
190  the_context->gs.operation_size = 0x1;
191  the_context->gs.granularity = 0x1;
192  the_context->gs.base_address_23_16 = (tcb >> 16) & 0xff;
193  the_context->gs.base_address_31_24 = (tcb >> 24) & 0xff;
194 }
