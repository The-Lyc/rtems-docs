59 static void start_thread_if_necessary(uint32_t cpu_index_self)
60 {
61 #if QORIQ_THREAD_COUNT > 1
62  uint32_t i;
63 
64  for (i = 1; i < QORIQ_THREAD_COUNT; ++i) {
65  uint32_t cpu_index_next = cpu_index_self + i;
66 
67  if (
68  is_started_by_u_boot(cpu_index_self)
69  && cpu_index_next < rtems_configuration_get_maximum_processors()
70  && _SMP_Should_start_processor(cpu_index_next)
71  ) {
72  /* Thread Initial Next Instruction Address (INIA) */
73  PPC_SET_THREAD_MGMT_REGISTER(321, (uintptr_t) _start_thread);
74 
75  /* Thread Initial Machine State (IMSR) */
76  PPC_SET_THREAD_MGMT_REGISTER(289, QORIQ_INITIAL_MSR);
77 
78  /* Thread Enable Set (TENS) */
79  PPC_SET_SPECIAL_PURPOSE_REGISTER(FSL_EIS_TENS, 1U << i);
80  }
81  }
82 #endif
83 }

125 static uint32_t discover_processors(void)
126 {
127  const void *fdt = bsp_fdt_get();
128  int cpus = fdt_path_offset(fdt, "/cpus");
129  int node = fdt_first_subnode(fdt, cpus);
130  uint32_t cpu = 0;
131 
132  while (node >= 0 && cpu < RTEMS_ARRAY_SIZE(qoriq_start_spin_table_addr)) {
133  int len;
134  fdt64_t *addr_fdt = (fdt64_t *)
135  fdt_getprop(fdt, node, "cpu-release-addr", &len);
136 
137  if (addr_fdt != NULL) {
138  uintptr_t addr = (uintptr_t) fdt64_to_cpu(*addr_fdt);
139 
140  qoriq_start_spin_table_addr[cpu] = (qoriq_start_spin_table *) addr;
141  }
142 
143  ++cpu;
144  node = fdt_next_subnode(fdt, node);
145  }
146 
147  return cpu * QORIQ_THREAD_COUNT;
148 }

164 static bool release_processor(
165  qoriq_start_spin_table *spin_table,
166  uint32_t cpu_index
167 )
168 {
169  bool spin_table_present = (spin_table != NULL);
170 
171  if (spin_table_present) {
172  const Per_CPU_Control *cpu = _Per_CPU_Get_by_index(cpu_index);
173 
174  spin_table->pir = cpu_index;
175  spin_table->r3 = (uintptr_t) cpu->interrupt_stack_high;
176  rtems_cache_flush_multiple_data_lines(spin_table, sizeof(*spin_table));
177  ppc_synchronize_data();
178  spin_table->addr = (uintptr_t) _start_secondary_processor;
179  rtems_cache_flush_multiple_data_lines(spin_table, sizeof(*spin_table));
180  }
181 
182  return spin_table_present;
183 }

194 bool _CPU_SMP_Start_processor(uint32_t cpu_index)
195 {
196 #if QORIQ_THREAD_COUNT > 1
197  if (is_started_by_u_boot(cpu_index)) {
198  qoriq_start_spin_table *spin_table = get_spin_table(cpu_index);
199 
200  return release_processor(spin_table, cpu_index);
201  } else {
202  return _SMP_Should_start_processor(cpu_index - 1);
203  }
204 #else
205  qoriq_start_spin_table *spin_table = get_spin_table(cpu_index);
206 
207  return release_processor(spin_table, cpu_index);
208 #endif
209 }
