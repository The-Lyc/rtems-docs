116 // Extract bits V[Begin:End], where range is inclusive, and Begin must be < 63.
117 static uint32_t extractBits(uint64_t v, uint32_t begin, uint32_t end) {
118  return (v & ((1ULL << (begin + 1)) - 1)) >> end;
119 }

121 static int64_t SignExtend64(uint64_t val, unsigned bits) {
122  return (int64_t )(((int64_t) (val << (64 - bits))) >> (64 - bits));

149 static rtems_rtl_elf_rel_status
150 rtems_rtl_elf_reloc_rela (rtems_rtl_obj* obj,
151  const Elf_Rela* rela,
152  const rtems_rtl_obj_sect* sect,
153  const char* symname,
154  const Elf_Byte syminfo,
155  const Elf_Word symvalue,
156  const bool parsing) {
157  Elf_Word *where;
158  Elf_Word tmp;
159  Elf_Word addend = (Elf_Word) rela->r_addend;
160  Elf_Word local = 0;
161 
162  char bits = (sizeof(Elf_Word) * 8);
163  where = (Elf_Addr *)(sect->base + rela->r_offset);
164 
165  // Final PCREL value
166  Elf_Word pcrel_val = symvalue - ((Elf_Word) where);
167 
168  if (syminfo == STT_SECTION) {
169  local = 1;
170  return rtems_rtl_elf_rel_no_error;
171  }
172 
173  if (parsing) {
174  return rtems_rtl_elf_rel_no_error;
175  }
176 
177  switch (ELF_R_TYPE(rela->r_info)) {
178  case R_TYPE(NONE):
179  break;
180 
181  case R_TYPE(RVC_BRANCH): {
182  uint16_t insn = ((*where) & 0xFFFF) & 0xE383;
183  uint16_t imm8 = extractBits(pcrel_val, 8, 8) << 12;
184  uint16_t imm4_3 = extractBits(pcrel_val, 4, 3) << 10;
185  uint16_t imm7_6 = extractBits(pcrel_val, 7, 6) << 5;
186  uint16_t imm2_1 = extractBits(pcrel_val, 2, 1) << 3;
187  uint16_t imm5 = extractBits(pcrel_val, 5, 5) << 2;
188  insn |= imm8 | imm4_3 | imm7_6 | imm2_1 | imm5;
189 
190  write16le(where, insn);
191  }
192  break;
193 
194  case R_TYPE(RVC_JUMP): {
195  uint16_t insn = ((*where) & 0xFFFF) & 0xE003;
196  uint16_t imm11 = extractBits(pcrel_val, 11, 11) << 12;
197  uint16_t imm4 = extractBits(pcrel_val, 4, 4) << 11;
198  uint16_t imm9_8 = extractBits(pcrel_val, 9, 8) << 9;
199  uint16_t imm10 = extractBits(pcrel_val, 10, 10) << 8;
200  uint16_t imm6 = extractBits(pcrel_val, 6, 6) << 7;
201  uint16_t imm7 = extractBits(pcrel_val, 7, 7) << 6;
202  uint16_t imm3_1 = extractBits(pcrel_val, 3, 1) << 3;
203  uint16_t imm5 = extractBits(pcrel_val, 5, 5) << 2;
204  insn |= imm11 | imm4 | imm9_8 | imm10 | imm6 | imm7 | imm3_1 | imm5;
205 
206  write16le(where, insn);
207  }
208  break;
209 
210  case R_TYPE(RVC_LUI): {
211  int64_t imm = SignExtend64(symvalue + 0x800, bits) >> 12;
212  if (imm == 0) { // `c.lui rd, 0` is illegal, convert to `c.li rd, 0`
213  write16le(where, (read16le(where) & 0x0F83) | 0x4000);
214  } else {
215  uint16_t imm17 = extractBits(symvalue + 0x800, 17, 17) << 12;
216  uint16_t imm16_12 = extractBits(symvalue + 0x800, 16, 12) << 2;
217  write16le(where, (read16le(where) & 0xEF83) | imm17 | imm16_12);
218  }
219  }
220  break;
221 
222  case R_TYPE(JAL): {
223  uint32_t insn = read32le(where) & 0xFFF;
224  uint32_t imm20 = extractBits(pcrel_val, 20, 20) << 31;
225  uint32_t imm10_1 = extractBits(pcrel_val, 10, 1) << 21;
226  uint32_t imm11 = extractBits(pcrel_val, 11, 11) << 20;
227  uint32_t imm19_12 = extractBits(pcrel_val, 19, 12) << 12;
228  insn |= imm20 | imm10_1 | imm11 | imm19_12;
229 
230  write32le(where, insn);
231  }
232  break;
233 
234  case R_TYPE(BRANCH): {
235 
236  uint32_t insn = read32le(where) & 0x1FFF07F;
237  uint32_t imm12 = extractBits(pcrel_val, 12, 12) << 31;
238  uint32_t imm10_5 = extractBits(pcrel_val, 10, 5) << 25;
239  uint32_t imm4_1 = extractBits(pcrel_val, 4, 1) << 8;
240  uint32_t imm11 = extractBits(pcrel_val, 11, 11) << 7;
241  insn |= imm12 | imm10_5 | imm4_1 | imm11;
242 
243  write32le(where, insn);
244  }
245  break;
246 
247  case R_TYPE(64):
248  write64le(where, symvalue);
249  break;
250  case R_TYPE(32):
251  write32le(where, symvalue);
252  break;
253 
254  case R_TYPE(SET6):
255  *((uint8_t *) where) = (*where & 0xc0) | (symvalue & 0x3f);
256  break;
257  case R_TYPE(SET8):
258  *((uint8_t *) where) = symvalue;
259  break;
260  case R_TYPE(SET16):
261  write16le(where, symvalue);
262  break;
263  case R_TYPE(SET32):
264  write32le(where, symvalue);
265  break;
266 
267  case R_TYPE(ADD8):
268  *((uint8_t *) where) = *((uint8_t *) where) + symvalue;
269  break;
270  case R_TYPE(ADD16):
271  write16le(where, read16le(where) + symvalue);
272  break;
273  case R_TYPE(ADD32):
274  write32le(where, read32le(where) + symvalue);
275  break;
276  case R_TYPE(ADD64):
277  write64le(where, read64le(where) + symvalue);
278  break;
279 
280  case R_TYPE(SUB6):
281  *((uint8_t *) where) = (*where & 0xc0) | (((*where & 0x3f) - symvalue) & 0x3f);
282  break;
283  case R_TYPE(SUB8):
284  *((uint8_t *) where) = *((uint8_t *) where) - symvalue;
285  break;
286  case R_TYPE(SUB16):
287  write16le(where, read16le(where) - symvalue);
288  break;
289  case R_TYPE(SUB32):
290  write32le(where, read32le(where) - symvalue);
291  break;
292  case R_TYPE(SUB64):
293  write64le(where, read64le(where) - symvalue);
294  break;
295 
296  case R_TYPE(32_PCREL): {
297  write32le(where, pcrel_val);
298 
299  if (rtems_rtl_trace (RTEMS_RTL_TRACE_RELOC))
300  printf ("rtl: R_RISCV_32_PCREL %p @ %p in %s\n",
301  (void *) * (where), where, rtems_rtl_obj_oname (obj));
302 
303  }
304  break;
305 
306  case R_TYPE(PCREL_HI20): {
307  int64_t hi = SignExtend64(pcrel_val + 0x800, bits); //pcrel_val + 0x800;
308  write32le(where, (read32le(where) & 0xFFF) | (hi & 0xFFFFF000));
309 
310  }
311  break;
312 
313  case R_TYPE(GOT_HI20):
314  case R_TYPE(HI20): {
315 
316  uint64_t hi = symvalue + 0x800;
317  write32le(where, (read32le(where) & 0xFFF) | (hi & 0xFFFFF000));
318  }
319  break;
320 
321  case R_TYPE(PCREL_LO12_I): {
322  uint64_t hi = (pcrel_val + 0x800) >> 12;
323  uint64_t lo = pcrel_val - (hi << 12);
324  write32le(where, (read32le(where) & 0xFFFFF) | ((lo & 0xFFF) << 20));
325  }
326  break;
327 
328  case R_TYPE(LO12_I): {
329 
330  uint64_t hi = (symvalue + 0x800) >> 12;
331  uint64_t lo = symvalue - (hi << 12);
332  write32le(where, (read32le(where) & 0xFFFFF) | ((lo & 0xFFF) << 20));
333 
334  }
335  break;
336 
337  case R_TYPE(PCREL_LO12_S): {
338  uint64_t hi = (pcrel_val + 0x800) >> 12;
339  uint64_t lo = pcrel_val - (hi << 12);
340  uint32_t imm11_5 = extractBits(lo, 11, 5) << 25;
341  uint32_t imm4_0 = extractBits(lo, 4, 0) << 7;
342  write32le(where, (read32le(where) & 0x1FFF07F) | imm11_5 | imm4_0);
343 
344  }
345  break;
346 
347  case R_TYPE(LO12_S): {
348  uint64_t hi = (symvalue + 0x800) >> 12;
349  uint64_t lo = symvalue - (hi << 12);
350  uint32_t imm11_5 = extractBits(lo, 11, 5) << 25;
351  uint32_t imm4_0 = extractBits(lo, 4, 0) << 7;
352  write32le(where, (read32le(where) & 0x1FFF07F) | imm11_5 | imm4_0);
353  }
354  break;
355 
356  case R_TYPE(CALL_PLT):
357  case R_TYPE(CALL): {
358  int64_t hi = SignExtend64(pcrel_val + 0x800, bits);
359  write32le(where, (read32le(where) & 0xFFF) | (hi & 0xFFFFF000));
360  int64_t hi20 = SignExtend64(pcrel_val + 0x800, bits);
361  int64_t lo = pcrel_val - (hi20 << 12);
362  write32le(((char *) where) + 4, (read32le(((char *) where) + 4) & 0xFFFFF) | ((lo & 0xFFF) << 20));
363  }
364  break;
365 
366  default:
367  rtems_rtl_set_error (EINVAL,
368  "%s: Unsupported relocation type %ld "
369  "in non-PLT relocations",
370  sect->name, (uint32_t) ELF_R_TYPE(rela->r_info));
371  return rtems_rtl_elf_rel_failure;
372  }
373 
374  return rtems_rtl_elf_rel_no_error;
375 }
