45 static uint32_t get_config_max( const Objects_Information *info )
46 {
47  if ( _Objects_Is_auto_extend( info ) ) {
48  return info->objects_per_block | RTEMS_UNLIMITED_OBJECTS;
49  }
50 
51  return _Objects_Get_maximum_index( info );
52 }
