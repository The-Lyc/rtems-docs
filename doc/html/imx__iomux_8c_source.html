29 /*
30  * Pin mux and pad control driver for imx5 and imx6.
31  *
32  * This driver implements the fdt_pinctrl interface for configuring the gpio and
33  * peripheral pins based on fdt configuration data.
34  *
35  * When the driver attaches, it walks the entire fdt tree and automatically
36  * configures the pins for each device which has a pinctrl-0 property and whose
37  * status is "okay". In addition it implements the fdt_pinctrl_configure()
38  * method which any other driver can call at any time to reconfigure its pins.
39  *
40  * The nature of the fsl,pins property in fdt data makes this driver's job very
41  * easy. Instead of representing each pin and pad configuration using symbolic
42  * properties such as pullup-enable="true" and so on, the data simply contains
43  * the addresses of the registers that control the pins, and the raw values to
44  * store in those registers.
45  *
46  * The imx5 and imx6 SoCs also have a small number of "general purpose
47  * registers" in the iomuxc device which are used to control an assortment
48  * of completely unrelated aspects of SoC behavior. This driver provides other
49  * drivers with direct access to those registers via simple accessor functions.
50  */

112 /* Return true if there is no status or status is "okay" or "ok". */
113 static bool
114 imx_fdt_node_status_okay(const void *fdt, int node)
115 {
116  const void *status;
117  int len;
118 
119  status = fdt_getprop(fdt, node, "status", &len);
120  if ((status == NULL) ||
121  (strncmp(status, "okay", MIN(5, len)) == 0) ||
122  (strncmp(status, "ok", MIN(3, len)) == 0)) {
123  return true;
124  }
125 
126  return false;
127 }

129 /*
130  * Walk through all subnodes and handle "pinctrl-0" if the node is enabled. This
131  * does roughly the same as FreeBSDs fdt_pinctrl_configure_tree but without the
132  * whole open firmware (OF*) functions.
133  */
134 static void
135 imx_pinctrl_configure_children(const void *fdt, int parent)
136 {
137  int node;
138  const uint32_t *phandle;
139  int len;
140 
141  fdt_for_each_subnode(node, fdt, parent) {
142  if (imx_fdt_node_status_okay(fdt, node)) {
143  imx_pinctrl_configure_children(fdt, node);
144  phandle = fdt_getprop(fdt, node, "pinctrl-0", &len);
145  if (phandle != NULL && len == sizeof(*phandle)) {
146  imx_iomux_configure_pins(fdt,
147  fdt32_to_cpu(*phandle));
148  }
149  }
150  }
151 }

179 imx_iomux_getencprop_alloc(const char *fdt, int node, const char *name,
180  size_t elsz, void **buf)
181 {
182  int len;
183  const uint32_t *val;
184  int i;
185  uint32_t *cell;
186 
187  val = fdt_getprop(fdt, node, "fsl,pins", &len);
188  if (val == NULL || len < 0 || len % elsz != 0) {
189  return (-1);
190  }
191 
192  cell = malloc((size_t)len);
193  *buf = cell;
194  if (cell == NULL) {
195  return (-1);
196  }
197 
198  for (i = 0; i < len / 4; ++i) {
199  cell[i] = fdt32_to_cpu(val[i]);
200  }
201 
202  return (len / (int)elsz);
203 }

249 static void
250 iomux_configure_input(struct iomux_softc *sc, uint32_t reg, uint32_t val)
251 {
252  u_int select, mask, shift, width;
253 
254  /* If register and value are zero, there is nothing to configure. */
255  if (reg == 0 && val == 0)
256  return;
257 
258  /*
259  * If the config value has 0xff in the high byte it is encoded:
260  * 31 23 15 7 0
261  * | 0xff | shift | width | select |
262  * We need to mask out the old select value and OR in the new, using a
263  * mask of the given width and shifting the values up by shift.
264  */
265  if ((val & 0xff000000) == 0xff000000) {
266  select = val & 0x000000ff;
267  width = (val & 0x0000ff00) >> 8;
268  shift = (val & 0x00ff0000) >> 16;
269  mask = ((1u << width) - 1) << shift;
270  val = (RD4(sc, reg) & ~mask) | (select << shift);
271  }
272  WR4(sc, reg, val);
273 }

275 #ifndef __rtems__
276 static int
277 iomux_configure_pins(device_t dev, phandle_t cfgxref)
278 #else /* __rtems__ */
279 int imx_iomux_configure_pins(const void *fdt, uint32_t cfgxref)
280 #endif /* __rtems__ */
281 {
282  struct iomux_softc *sc;
283  struct pincfg *cfgtuples, *cfg;
284  phandle_t cfgnode;
285  int i, ntuples;
286  uint32_t sion;
287 
288 #ifndef __rtems__
289  sc = device_get_softc(dev);
290 #else /* __rtems__ */
291  sc = iomux_sc;
292 #endif /* __rtems__ */
293  cfgnode = OF_node_from_xref(cfgxref);
294  ntuples = OF_getencprop_alloc(cfgnode, "fsl,pins", sizeof(*cfgtuples),
295  (void **)&cfgtuples);
296  if (ntuples < 0)
297  return (ENOENT);
298  if (ntuples == 0)
299  return (0); /* Empty property is not an error. */
300  for (i = 0, cfg = cfgtuples; i < ntuples; i++, cfg++) {
301  sion = (cfg->padconf_val & PADCONF_SION) ? PADMUX_SION : 0;
302  WR4(sc, cfg->mux_reg, cfg->mux_val | sion);
303  iomux_configure_input(sc, cfg->input_reg, cfg->input_val);
304  if ((cfg->padconf_val & PADCONF_NONE) == 0)
305  WR4(sc, cfg->padconf_reg, cfg->padconf_val);
306 #ifndef __rtems__
307  if (bootverbose) {
308  char name[32];
309  OF_getprop(cfgnode, "name", &name, sizeof(name));
310  printf("%16s: muxreg 0x%04x muxval 0x%02x "
311  "inpreg 0x%04x inpval 0x%02x "
312  "padreg 0x%04x padval 0x%08x\n",
313  name, cfg->mux_reg, cfg->mux_val | sion,
314  cfg->input_reg, cfg->input_val,
315  cfg->padconf_reg, cfg->padconf_val);
316  }
317 #endif /* __rtems__ */
318  }
319  OF_prop_free(cfgtuples);
320  return (0);
321 }

346 static int
347 iomux_attach(device_t dev)
348 {
349  struct iomux_softc * sc;
350  int rid;
351 
352  sc = device_get_softc(dev);
353  sc->dev = dev;
354 
355  switch (imx_soc_type()) {
356  case IMXSOC_51:
357  sc->last_gpregaddr = 1 * sizeof(uint32_t);
358  break;
359  case IMXSOC_53:
360  sc->last_gpregaddr = 2 * sizeof(uint32_t);
361  break;
362  case IMXSOC_6DL:
363  case IMXSOC_6S:
364  case IMXSOC_6SL:
365  case IMXSOC_6Q:
366  sc->last_gpregaddr = 13 * sizeof(uint32_t);
367  break;
368  case IMXSOC_6UL:
369  sc->last_gpregaddr = 14 * sizeof(uint32_t);
370  break;
371  default:
372  device_printf(dev, "Unknown SoC type\n");
373  return (ENXIO);
374  }
375 
376  rid = 0;
377  sc->mem_res = bus_alloc_resource_any(dev, SYS_RES_MEMORY, &rid,
378  RF_ACTIVE);
379  if (sc->mem_res == NULL) {
380  device_printf(dev, "Cannot allocate memory resources\n");
381  return (ENXIO);
382  }
383 
384  iomux_sc = sc;
385 
386  /*
387  * Register as a pinctrl device, and call the convenience function that
388  * walks the entire device tree invoking FDT_PINCTRL_CONFIGURE() on any
389  * pinctrl-0 property cells whose xref phandle refers to a configuration
390  * that is a child node of our node in the tree.
391  *
392  * The pinctrl bindings documentation specifically mentions that the
393  * pinctrl device itself may have a pinctrl-0 property which contains
394  * static configuration to be applied at device init time. The tree
395  * walk will automatically handle this for us when it passes through our
396  * node in the tree.
397  */
398  fdt_pinctrl_register(dev, "fsl,pins");
399  fdt_pinctrl_configure_tree(dev);
400 
401  return (0);
402 }
