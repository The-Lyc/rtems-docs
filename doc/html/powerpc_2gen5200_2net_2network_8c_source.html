250 /*
251  * Function: MPC5200_eth_addr_filter_set
252  *
253  * Description: Set individual address filter for unicast address and
254  * set physical address registers.
255  *
256  * Returns: void
257  *
258  * Notes:
259  *
260  */
261 static void mpc5200_eth_addr_filter_set(mpc5200_fec_context *self) {
262  unsigned char *mac;
263  unsigned char currByte; /* byte for which to compute the CRC */
264  int byte; /* loop - counter */
265  int bit; /* loop - counter */
266  unsigned long crc = 0xffffffff; /* initial value */
267 
268  /*
269  * Get the mac address of ethernet controller
270  */
271  mac = (unsigned char *)(&self->arpcom.ac_enaddr);
272 
273  /*
274  * The algorithm used is the following:
275  * we loop on each of the six bytes of the provided address,
276  * and we compute the CRC by left-shifting the previous
277  * value by one position, so that each bit in the current
278  * byte of the address may contribute the calculation. If
279  * the latter and the MSB in the CRC are different, then
280  * the CRC value so computed is also ex-ored with the
281  * "polynomium generator". The current byte of the address
282  * is also shifted right by one bit at each iteration.
283  * This is because the CRC generatore in hardware is implemented
284  * as a shift-register with as many ex-ores as the radixes
285  * in the polynomium. This suggests that we represent the
286  * polynomiumm itself as a 32-bit constant.
287  */
288  for(byte = 0; byte < 6; byte++)
289  {
290 
291  currByte = mac[byte];
292 
293  for(bit = 0; bit < 8; bit++)
294  {
295 
296  if((currByte & 0x01) ^ (crc & 0x01))
297  {
298 
299  crc >>= 1;
300  crc = crc ^ 0xedb88320;
301 
302  }
303  else
304  {
305 
306  crc >>= 1;
307 
308  }
309 
310  currByte >>= 1;
311 
312  }
313 
314  }
315 
316  crc = crc >> 26;
317 
318  /*
319  * Set individual hash table register
320  */
321  if(crc >= 32)
322  {
323 
324  mpc5200.iaddr1 = (1 << (crc - 32));
325  mpc5200.iaddr2 = 0;
326 
327  }
328  else
329  {
330 
331  mpc5200.iaddr1 = 0;
332  mpc5200.iaddr2 = (1 << crc);
333 
334  }
335 
336  /*
337  * Set physical address
338  */
339  mpc5200.paddr1 = (mac[0] << 24) + (mac[1] << 16) + (mac[2] << 8) + mac[3];
340  mpc5200.paddr2 = (mac[4] << 24) + (mac[5] << 16) + 0x8808;
341 
342  }

344 static int mpc5200_eth_mii_transfer(
345  int phyAddr,
346  unsigned regAddr,
347  uint32_t data
348 )
349 {
350  int timeout = 0xffff;
351 
352  mpc5200.ievent = FEC_INTR_MII;
353 
354  mpc5200.mii_data = MPC5200_FEC_MII_DATA_ST
355  | MPC5200_FEC_MII_DATA_TA
356  | (phyAddr << MPC5200_FEC_MII_DATA_PA_SHIFT)
357  | (regAddr << MPC5200_FEC_MII_DATA_RA_SHIFT)
358  | data;
359 
360  while (timeout > 0 && (mpc5200.ievent & FEC_INTR_MII) == 0) {
361  --timeout;
362  }
363 
364  return timeout > 0 ? 0 : -1;
365 }

505 {
506  mpc5200_fec_context *self = (mpc5200_fec_context *) handle;
507  volatile uint32_t ievent;
508 
509  ievent = mpc5200.ievent;
510 
511  mpc5200.ievent = ievent;
512  /*
513  * check errors, update statistics
514  */
515  if (ievent & FEC_INTR_LATE_COL) {
516  self->txLateCollision++;
517  }
518  if (ievent & FEC_INTR_COL_RETRY) {
519  self->txRetryLimit++;
520  }
521  if (ievent & FEC_INTR_XFIFO_UN) {
522  self->txUnderrun++;
523  }
524  if (ievent & FEC_INTR_XFIFO_ERR) {
525  self->txFIFOError++;
526  }
527  if (ievent & FEC_INTR_RFIFO_ERR) {
528  self->rxFIFOError++;
529  }
530  /*
531  * fatal error ocurred?
532  */
533  if (ievent & (FEC_INTR_XFIFO_ERR | FEC_INTR_RFIFO_ERR)) {
534  mpc5200.imask &= ~(FEC_INTR_XFERREN | FEC_INTR_RFERREN);
535  mpc5200_fec_request_restart(self);
536  }

565 /*
566  * Function: mpc5200_fec_initialize_hardware
567  *
568  * Description: Configure the MPC5200 FEC registers and enable the
569  * SmartComm tasks. This function "turns on" the driver.
570  *
571  * Returns: void
572  *
573  * Notes:
574  *
575  */
576 static void mpc5200_fec_initialize_hardware(mpc5200_fec_context *self)
577 {
578  /* We want at most 2.5MHz */
579  uint32_t div = 2 * 2500000;
580  uint32_t mii_speed = (IPB_CLOCK + div - 1) / div;
581 
582  /*
583  * Reset mpc5200 FEC
584  */
585  mpc5200_fec_reset(self);
586  mpc5200_fec_init_mib(self);
587 
588  /*
589  * Clear FEC-Lite interrupt event register (IEVENT)
590  */
591  mpc5200.ievent = FEC_INTR_CLEAR_ALL;
592 
593  /*
594  * Set interrupt mask register
595  */
596  mpc5200.imask = FEC_INTR_MASK_USED;
597  /*
598  * Set FEC-Lite receive control register (R_CNTRL)
599  * frame length=1518, MII mode for 18-wire-transceiver
600  */
601  mpc5200.r_cntrl = ((ETHER_MAX_LEN << FEC_RCNTRL_MAX_FL_SHIFT)
602  | FEC_RCNTRL_FCE
603  | FEC_RCNTRL_MII_MODE);
604 
605  /*
606  * Set FEC-Lite transmit control register (X_CNTRL)
607  * full-duplex, heartbeat disabled
608  */
609  mpc5200.x_cntrl = FEC_XCNTRL_FDEN;
610 
611 
612 
613  /*
614  * Set MII_SPEED = IPB clock / (2 * mii_speed))
615  * and do not drop the Preamble.
616  */
617  mpc5200.mii_speed = mii_speed << 1;
618 
619  /*
620  * Set Opcode/Pause Duration Register
621  */
622  mpc5200.op_pause = 0x00010020;
623 
624  /*
625  * Set Rx FIFO alarm and granularity value
626  */
627  mpc5200.rfifo_cntrl = (FEC_FIFO_CNTRL_FRAME
628  | (0x7 << FEC_FIFO_CNTRL_GR_SHIFT));
629  mpc5200.rfifo_alarm = 0x0000030c;
630 
631  /*
632  * Set Tx FIFO granularity value
633  */
634  mpc5200.tfifo_cntrl = (FEC_FIFO_CNTRL_FRAME
635  | (0x7 << FEC_FIFO_CNTRL_GR_SHIFT));
636 
637  /*
638  * Set transmit fifo watermark register (X_WMRK), default = 64
639  */
640  mpc5200.tfifo_alarm = 0x00000100; /* 256 bytes */
641  mpc5200.x_wmrk = FEC_XWMRK_256; /* 256 bytes */
642 
643  /*
644  * Set individual address filter for unicast address
645  * and set physical address registers.
646  */
647  mpc5200_eth_addr_filter_set(self);
648 
649  /*
650  * Set multicast address filter
651  */
652  mpc5200.gaddr1 = 0x00000000;
653  mpc5200.gaddr2 = 0x00000000;
654 
655  /*
