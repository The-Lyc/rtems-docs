214  unsigned sum = 0;

223 /*
224  * APIC ICR write and status check function.
225  */
226 int
227 send_ipi(unsigned int dst, unsigned int v)
228 {
229  int to, send_status, apicid;
230 
231  apicid = imps_cpu_apic_map[dst];
232 
233  IMPS_LAPIC_WRITE(LAPIC_ICR+0x10, (apicid << 24));
234  IMPS_LAPIC_WRITE(LAPIC_ICR, v);
235 
236  /* Wait for send to finish */
237  to = 0;
238  do {
239  UDELAY(100);
240  send_status = IMPS_LAPIC_READ(LAPIC_ICR) & LAPIC_ICR_STATUS_PEND;
241  } while (send_status && (to++ < 1000));
242 
243  return (to < 1000);
244 }

246 /*
247  * Primary function for booting individual CPUs.
248  *
249  * This must be modified to perform whatever OS-specific initialization
250  * that is required.
251  */
252 static int
253 boot_cpu(imps_processor *proc)
254 {
255  int apicid = proc->apic_id, success = 1;
256  int cpuid;
257  unsigned bootaddr;
258  unsigned bios_reset_vector = PHYS_TO_VIRTUAL(BIOS_RESET_VECTOR);
259 
260  cpuid = imps_apic_cpu_map[apicid];
261  /*
262  * Copy boot code for secondary CPUs here. Find it in between
263  * "patch_code_start" and "patch_code_end" symbols. The other CPUs
264  * will start there in 16-bit real mode under the 1MB boundary.
265  * "patch_code_start" should be placed at a 4K-aligned address
266  * under the 1MB boundary.
267  */
268 
269  volatile uint32_t *reset;
270 
271  bootaddr = (512-64)*1024;
272  reset= (volatile uint32_t *)bootaddr;
273 
274  memcpy(
275  (char *) bootaddr,
276  _binary_appstart_bin_start,
277  (size_t)_binary_appstart_bin_size
278  );
279 
280  /* Pass start function, stack region and gdtdescr to AP
281  * see startAP.S for location */
282  reset[1] = (uint32_t)secondary_cpu_initialize;
283  reset[2] = (uint32_t)_Per_CPU_Get_by_index(cpuid)->interrupt_stack_high;
284  memcpy(
285  (char*) &reset[3],
286  &gdtdesc,
287  6);
288  /*
289  * Generic CPU startup sequence starts here.
290  */
291 
292  /* set BIOS reset vector */
293  CMOS_WRITE_BYTE(CMOS_RESET_CODE, CMOS_RESET_JUMP);
294  *((volatile unsigned *) bios_reset_vector) = ((bootaddr & 0xFF000) << 12);
295 
296  /* clear the APIC error register */
297  IMPS_LAPIC_WRITE(LAPIC_ESR, 0);
298  IMPS_LAPIC_READ(LAPIC_ESR);
299 
300  /* assert INIT IPI */
301  send_ipi(
302  cpuid,
303  LAPIC_ICR_TM_LEVEL | LAPIC_ICR_LEVELASSERT | LAPIC_ICR_DM_INIT
304  );
305  UDELAY(10000);
306 
307  /* de-assert INIT IPI */
308  send_ipi(cpuid, LAPIC_ICR_TM_LEVEL | LAPIC_ICR_DM_INIT);
309 
310  UDELAY(10000);
311 
312  /*
313  * Send Startup IPIs if not an old pre-integrated APIC.
314  */
315 
316  if (proc->apic_ver >= APIC_VER_NEW) {
317  int i;
318  for (i = 1; i <= 2; i++) {
319  send_ipi(cpuid, LAPIC_ICR_DM_SIPI | ((bootaddr >> 12) & 0xFF));
320  UDELAY(1000);
321  }
322  }
323 
324  /*
325  * Wait until AP is in protected mode before starting the next AP
326  */
327  while (reset[2] != 0);
328 
329  /*
330  * Generic CPU startup sequence ends here, the rest is cleanup.
331  */
332 
333  /* clear the APIC error register */
334  IMPS_LAPIC_WRITE(LAPIC_ESR, 0);
335  IMPS_LAPIC_READ(LAPIC_ESR);
336 
337  /* clean up BIOS reset vector */
338  CMOS_WRITE_BYTE(CMOS_RESET_CODE, 0);
339  *((volatile unsigned *) bios_reset_vector) = 0;
340 
341  return success;
342 }

405 imps_read_config_table(unsigned start, int count)
406 {
407  while (count-- > 0) {
408  switch (*((unsigned char *)start)) {
409  case IMPS_BCT_PROCESSOR:
410  if ( imps_num_cpus < rtems_configuration_get_maximum_processors() ) {
411  if (_SMP_Should_start_processor((uint32_t) imps_num_cpus)) {
412  add_processor((imps_processor *)start);
413  }
414  } else
415  imps_num_cpus++;
416  start += 12; /* 20 total */
417  break;
418  case IMPS_BCT_BUS:
419  add_bus((imps_bus *)start);
420  break;
421  case IMPS_BCT_IOAPIC:
422  add_ioapic((imps_ioapic *)start);
423  break;
424 #if 0 /* XXXXX uncomment this if "add_io_interrupt" is implemented */
425  case IMPS_BCT_IO_INTERRUPT:
426  add_io_interrupt((imps_interrupt *)start);
427  break;
428 #endif
429 #if 0 /* XXXXX uncomment this if "add_local_interrupt" is implemented */
430  case IMPS_BCT_LOCAL_INTERRUPT:
431  add_local_interupt((imps_interrupt *)start);
432  break;
433 #endif
434  default:
435  break;
436  }
437  start += 8;
438  }
439  if ( imps_num_cpus > rtems_configuration_get_maximum_processors() ) {
440  printk(
441  "WARNING!! Found more CPUs (%d) than configured for (%d)!!\n",
442  imps_num_cpus - 1,
443  rtems_configuration_get_maximum_processors()
444  );
445  imps_num_cpus = rtems_configuration_get_maximum_processors();
446  return;
447  }

450 static int
451 imps_bad_bios(imps_fps *fps_ptr)
452 {
453  int sum;
454  imps_cth *local_cth_ptr
455  = (imps_cth *) PHYS_TO_VIRTUAL(fps_ptr->cth_ptr);
456 
457  if (fps_ptr->feature_info[0] > IMPS_FPS_DEFAULT_MAX) {
458  printk(" Invalid MP System Configuration type %d\n",
459  fps_ptr->feature_info[0]);
460  return 1;
461  }
462 
463  if (fps_ptr->cth_ptr) {
464  sum = get_checksum((unsigned)local_cth_ptr,
465  local_cth_ptr->base_length);
466  if (local_cth_ptr->sig != IMPS_CTH_SIGNATURE || sum) {
467  printk(
468  " Bad MP Config Table sig 0x%x and/or checksum 0x%x\n",
469  (unsigned)(fps_ptr->cth_ptr),
470  sum
471  );
472  return 1;
473  }
474  if (local_cth_ptr->spec_rev != fps_ptr->spec_rev) {
475  printk(
476  " Bad MP Config Table sub-revision # %d\n",
477  local_cth_ptr->spec_rev
478  );
479  return 1;
480  }
481  if (local_cth_ptr->extended_length) {
482  sum = (get_checksum(((unsigned)local_cth_ptr)
483  + local_cth_ptr->base_length,
484  local_cth_ptr->extended_length)
485  + local_cth_ptr->extended_checksum) & 0xFF;
486  if (sum) {
487  printk(" Bad Extended MP Config Table checksum 0x%x\n", sum);
488  return 1;
489  }
490  }
491  } else if (!fps_ptr->feature_info[0]) {
492  printk(" Missing configuration information\n");
493  return 1;
494  }
495 
496  return 0;
497 }

499 static void
500 imps_read_bios(imps_fps *fps_ptr)
501 {
502  int apicid;
503  unsigned cth_start, cth_count;
504  imps_cth *local_cth_ptr
505  = (imps_cth *)PHYS_TO_VIRTUAL(fps_ptr->cth_ptr);
506  char *str_ptr;
507 
508  printk("Intel MultiProcessor Spec 1.%d BIOS support detected\n",
509  fps_ptr->spec_rev);
510 
511  /*
512  * Do all checking of errors which would definitely
513  * lead to failure of the SMP boot here.
514  */
515  if (imps_bad_bios(fps_ptr)) {
516  printk(" Disabling MPS support\n");
517  return;
518  }
519 
520  if (fps_ptr->feature_info[1] & IMPS_FPS_IMCRP_BIT) {
521  str_ptr = "IMCR and PIC";
522  } else {
523  str_ptr = "Virtual Wire";
524  }
525  if (fps_ptr->cth_ptr) {
526  imps_lapic_addr = local_cth_ptr->lapic_addr;
527  } else {
528  imps_lapic_addr = LAPIC_ADDR_DEFAULT;
529  }
530  printk(" APIC config: \"%s mode\" Local APIC address: 0x%x\n",
531  str_ptr, imps_lapic_addr);
532  if (imps_lapic_addr != (READ_MSR_LO(0x1b) & 0xFFFFF000)) {
533  printk("Inconsistent Local APIC address, Disabling SMP support\n");
534  return;
535  }
536  imps_lapic_addr = PHYS_TO_VIRTUAL(imps_lapic_addr);
537 
538  /*
539  * Setup primary CPU.
540  */
541  apicid = IMPS_LAPIC_READ(LAPIC_SPIV);
542  IMPS_LAPIC_WRITE(LAPIC_SPIV, apicid|LAPIC_SPIV_ENABLE_APIC);
543  apicid = APIC_ID(IMPS_LAPIC_READ(LAPIC_ID));
544  imps_cpu_apic_map[0] = apicid;
545  imps_apic_cpu_map[apicid] = 0;
546 
547  if (fps_ptr->cth_ptr) {
548  char str1[16], str2[16];
549  memcpy(str1, local_cth_ptr->oem_id, 8);
550  str1[8] = 0;
551  memcpy(str2, local_cth_ptr->prod_id, 12);
552  str2[12] = 0;
553  printk(" OEM id: %s Product id: %s\n", str1, str2);
554  cth_start = ((unsigned) local_cth_ptr) + sizeof(imps_cth);
555  cth_count = local_cth_ptr->entry_count;
556  } else {
557  *((volatile unsigned *) IOAPIC_ADDR_DEFAULT) = IOAPIC_ID;
558  defconfig.ioapic.id
559  = APIC_ID(*((volatile unsigned *)
560  (IOAPIC_ADDR_DEFAULT+IOAPIC_RW)));
561  *((volatile unsigned *) IOAPIC_ADDR_DEFAULT) = IOAPIC_VER;
562  defconfig.ioapic.ver
563  = APIC_VERSION(*((volatile unsigned *)
564  (IOAPIC_ADDR_DEFAULT+IOAPIC_RW)));
565  defconfig.proc[apicid].flags
566  = IMPS_FLAG_ENABLED|IMPS_CPUFLAG_BOOT;
567  defconfig.proc[!apicid].flags = IMPS_FLAG_ENABLED;
568  imps_num_cpus = 2;
569  if (fps_ptr->feature_info[0] == 1
570  || fps_ptr->feature_info[0] == 5) {
571  memcpy(defconfig.bus[0].bus_type, "ISA ", 6);
572  }
573  if (fps_ptr->feature_info[0] == 4
574  || fps_ptr->feature_info[0] == 7) {
575  memcpy(defconfig.bus[0].bus_type, "MCA ", 6);
576  }
577  if (fps_ptr->feature_info[0] > 4) {
578  defconfig.proc[0].apic_ver = 0x10;
579  defconfig.proc[1].apic_ver = 0x10;
580  defconfig.bus[1].type = IMPS_BCT_BUS;
581  }
582  if (fps_ptr->feature_info[0] == 2) {
583  defconfig.intin[2].type = 255;
584  defconfig.intin[13].type = 255;
585  }
586  if (fps_ptr->feature_info[0] == 7) {
587  defconfig.intin[0].type = 255;
588  }
589  cth_start = (unsigned) &defconfig;
590  cth_count = DEF_ENTRIES;
591  }
592  imps_read_config_table(cth_start, cth_count);
593 
594  /* %%%%% ESB read extended entries here */
595 
596  imps_enabled = 1;
597 }

599 /*
600  * Given a region to check, this actually looks for the "MP Floating
601  * Pointer Structure". The return value indicates if the correct
602  * signature and checksum for a floating pointer structure of the
603  * appropriate spec revision was found. If so, then do not search
604  * further.
605  *
606  * NOTE: The memory scan will always be in the bottom 1 MB.
607  *
608  * This function presumes that "start" will always be aligned to a 16-bit
609  * boundary.
610  *
611  * Function finished.
612  */
613 static int
614 imps_scan(unsigned start, unsigned length)
615 {
616  printk("Scanning from 0x%x for %d bytes\n", start, length);
617 
618  while (length > 0) {
619  imps_fps *fps_ptr = (imps_fps *) PHYS_TO_VIRTUAL(start);
620 
621  if (fps_ptr->sig == IMPS_FPS_SIGNATURE
622  && fps_ptr->length == 1
623  && (fps_ptr->spec_rev == 1 || fps_ptr->spec_rev == 4)
624  && !get_checksum(start, 16)) {
625  printk("Found MP Floating Structure Pointer at %x\n", start);

680 /*
681  * This is the primary function for probing for MPS compatible hardware
682  * and BIOS information. Call this during the early stages of OS startup,
683  * before memory can be messed up.
684  *
685  * The probe looks for the "MP Floating Pointer Structure" at locations
686  * listed at the top of page 4-2 of the spec.
687  *
688  * Environment requirements from the OS to run:
689  *
690  * (1) : A non-linear virtual to physical memory mapping is probably OK,
691  * as (I think) the structures all fall within page boundaries,
692  * but a linear mapping is recommended. Currently assumes that
693  * the mapping will remain identical over time (which should be
694  * OK since it only accesses memory which shouldn't be munged
695  * by the OS anyway).
696  * (2) : The OS only consumes memory which the BIOS says is OK to use,
697  * and not any of the BIOS standard areas (the areas 0x400 to
698  * 0x600, the EBDA, 0xE0000 to 0xFFFFF, and unreported physical
699  * RAM). Sometimes a small amount of physical RAM is not
700  * reported by the BIOS, to be used to store MPS and other
701  * information.
702  * (3) : It must be possible to read the CMOS.
703  * (4) : There must be between 512K and 640K of lower memory (this is a
704  * sanity check).
705  *
706  * Function finished.
707  */
708 int
709 imps_probe(void)
710 {
711  /*
712  * Determine possible address of the EBDA
713  */
714  unsigned ebda_addr = *((unsigned short *)
715  PHYS_TO_VIRTUAL(EBDA_SEG_ADDR)) << 4;
716 
717  /*
718  * Determine amount of installed lower memory (not *available*
719  * lower memory).
720  *
721  * NOTE: This should work reliably as long as we verify the
722  * machine is at least a system that could possibly have
723  * MPS compatibility to begin with.
724  */
725  unsigned mem_lower = ((CMOS_READ_BYTE(CMOS_BASE_MEMORY+1) << 8)
726  | CMOS_READ_BYTE(CMOS_BASE_MEMORY)) << 10;
727 
728 #ifdef IMPS_DEBUG
729  imps_enabled = 0;
730  imps_num_cpus = 1;
731 #endif
732 
733  /*
734  * Sanity check : if this isn't reasonable, it is almost impossibly
735  * unlikely to be an MPS compatible machine, so return failure.
736  */
737  if (mem_lower < 512*1024 || mem_lower > 640*1024) {
738  return 0;
739  }
740 
741  if (ebda_addr > mem_lower - 1024
742  || ebda_addr + *((unsigned char *) PHYS_TO_VIRTUAL(ebda_addr))
743  * 1024 > mem_lower) {
744  ebda_addr = 0;
745  }
746 
747  if (((ebda_addr && imps_scan(ebda_addr, 1024))
748  || (!ebda_addr && imps_scan(mem_lower - 1024, 1024))
749  || imps_scan(0xF0000, 0x10000)) && imps_enabled) {
750  return imps_num_cpus;
751  }
752 
753  /*
754  * If no BIOS info on MPS hardware is found, then return failure.
755  */
756 
757  return 0;
758 }
