108 /* ata_io_data_request --
109  * Form read/write request for an ATA device and enqueue it to
110  * IDE controller.
111  *
112  * PARAMETERS:
113  * device - device identifier
114  * req - read/write request from block device driver
115  *
116  * RETURNS:
117  * RTEMS_SUCCESSFUL on success, or error code if
118  * error occured
119  */
120 static rtems_status_code
121 ata_io_data_request(ata_ide_dev_t *ata_dev, rtems_blkdev_request *req)
122 {
123  ata_req_t *areq; /* ATA request */
124  rtems_device_minor_number ctrl_minor;
125  uint8_t dev;
126 
127  /* get controller which serves the ATA device */
128  ctrl_minor = ata_dev->ctrl_minor;
129 
130  /* get ATA device identifier (0 or 1) */
131  dev = ata_dev->device;
132 
133  areq = malloc(sizeof(ata_req_t));
134  if (areq == NULL)
135  {
136  rtems_blkdev_request_done(req, RTEMS_NO_MEMORY);
137  return RTEMS_SUCCESSFUL;
138  }
139 
140  areq->breq = req;
141  areq->cnt = req->bufnum;
142  areq->cbuf = 0;
143  areq->pos = 0;
144 
145  /* set up registers masks */
146  areq->regs.to_write = ATA_REGISTERS_POSITION;
147  areq->regs.to_read = ATA_REGISTERS_VALUE(IDE_REGISTER_STATUS);
148 
149  /* choose device on the controller for which the command will be issued */
150  areq->regs.regs[IDE_REGISTER_DEVICE_HEAD] =
151  (dev << IDE_REGISTER_DEVICE_HEAD_DEV_POS);
152 
153  /* Find ATA command and its type */
154  if (ATA_DEV_INFO(ctrl_minor, dev).mode_active & ATA_MODES_DMA)
155  {
156  /* XXX: never has been tested */
157  areq->type = ATA_COMMAND_TYPE_DMA;
158  if (req->req == RTEMS_BLKDEV_REQ_READ)
159  areq->regs.regs[IDE_REGISTER_COMMAND] = ATA_COMMAND_READ_DMA;
160  else
161  areq->regs.regs[IDE_REGISTER_COMMAND] = ATA_COMMAND_WRITE_DMA;
162  }
163  else
164  {
165  if (req->req == RTEMS_BLKDEV_REQ_READ)
166  {
167  areq->type = ATA_COMMAND_TYPE_PIO_IN;
168  areq->regs.regs[IDE_REGISTER_COMMAND] = ATA_COMMAND_READ_SECTORS;
169 #if ATA_DEBUG
170  ata_printf("ata_io_data_request: type: READ: %lu, %lu cmd:%02x\n",
171  req->bufs[0].block, req->bufnum,
172  areq->regs.regs[IDE_REGISTER_COMMAND]);
173 #endif
174  }
175  else
176  {
177  areq->type = ATA_COMMAND_TYPE_PIO_OUT;
178  areq->regs.regs[IDE_REGISTER_COMMAND] = ATA_COMMAND_WRITE_SECTORS;
179 #if ATA_DEBUG
180  ata_printf("ata_io_data_request: type: WRITE: %lu, %lu cmd:%02x\n",
181  req->bufs[0].block, req->bufnum,
182  areq->regs.regs[IDE_REGISTER_COMMAND]);
183 #endif
184  }
185  }
186 
187  /*
188  * Fill position registers
189  */
190  if (ATA_DEV_INFO(ctrl_minor, dev).lba_avaible)
191  {
192  uint32_t start = req->bufs[0].block;
193  areq->regs.regs[IDE_REGISTER_LBA0] = (uint8_t)start;
194  areq->regs.regs[IDE_REGISTER_LBA1] = (uint8_t)(start >> 8);
195  areq->regs.regs[IDE_REGISTER_LBA2] = (uint8_t)(start >> 16);
196  /* Set as the head register write above */
197  areq->regs.regs[IDE_REGISTER_LBA3] |= (uint8_t) (start >> 24);
198  areq->regs.regs[IDE_REGISTER_LBA3] |= IDE_REGISTER_LBA3_L;
199  }
200  else
201  {
202  uint32_t count = req->bufs[0].block;
203 
204  areq->regs.regs[IDE_REGISTER_SECTOR_NUMBER] =
205  (count % ATA_DEV_INFO(ctrl_minor, dev).sectors) + 1;
206 
207  /* now count = number of tracks: */
208  count /= ATA_DEV_INFO(ctrl_minor, dev).sectors;
209  areq->regs.regs[IDE_REGISTER_DEVICE_HEAD] |=
210  (count / ATA_DEV_INFO(ctrl_minor, dev).cylinders);
211 
212  /* now count = number of cylinders */
213  count %= ATA_DEV_INFO(ctrl_minor, dev).cylinders;
214  areq->regs.regs[IDE_REGISTER_CYLINDER_LOW] = (uint8_t)count;
215  areq->regs.regs[IDE_REGISTER_CYLINDER_HIGH] = (uint8_t)(count >> 8);
216  areq->regs.regs[IDE_REGISTER_DEVICE_HEAD] &=
217  ~IDE_REGISTER_DEVICE_HEAD_L;
218  }
219 
220  /*
221  * Fill sector count register. We have a number of buffers (bufnum) which
222  * can be of a specific length (bufs[0].length / ATA_SECTOR_SIZE).
223  */
224  areq->regs.regs[IDE_REGISTER_SECTOR_COUNT] =
225  areq->breq->bufnum * (areq->breq->bufs[0].length / ATA_SECTOR_SIZE);
226 
227  /* add request to the queue of awaiting requests to the controller */
228  ata_add_to_controller_queue(ctrl_minor, areq);
229 
230  return RTEMS_SUCCESSFUL;
231 }

233 /* ata_non_data_request --
234  * Form and serve request of NON DATA type for an ATA device.
235  * Processing of NON DATA request is SYNChronous operation.
236  *
237  * PARAMETERS:
238  * device - device identifier
239  * cmd - command
240  * argp - arguments for command
241  *
242  * RETURNS:
243  * RTEMS_SUCCESSFUL on success, or error code if
244  * error occured
245  */
246 static rtems_status_code
247 ata_non_data_request(ata_ide_dev_t *ata_dev, uint32_t cmd, void *argp)
248 {
249  rtems_status_code rc;
250  ata_req_t *areq; /* ATA request */
251  rtems_device_minor_number ctrl_minor;
252  uint8_t dev;
253  ata_queue_msg_t msg;
254 
255  /* get controller which serves the ATA device */
256  ctrl_minor = ata_dev->ctrl_minor;
257 
258  /* get ATA device identifier (0 or 1) */
259  dev = ata_dev->device;
260 
261  /* form the request */
262  areq = malloc(sizeof(ata_req_t));
263  if (areq == NULL)
264  {
265  return RTEMS_NO_MEMORY;
266  }
267  memset(areq, 0, sizeof(ata_req_t));
268 
269  areq->type = ATA_COMMAND_TYPE_NON_DATA;
270  areq->regs.to_write = ATA_REGISTERS_VALUE(IDE_REGISTER_COMMAND);
271  areq->regs.regs[IDE_REGISTER_DEVICE_HEAD] |=
272  (dev << IDE_REGISTER_DEVICE_HEAD_DEV_POS);
273  areq->breq = NULL;
274  areq->regs.to_read = ATA_REGISTERS_VALUE(IDE_REGISTER_ERROR);
275 
276  /*
277  * depending on command fill command register and additional registers
278  * which are needed for command execution
279  */
280  switch(cmd)
281  {
282  case ATAIO_SET_MULTIPLE_MODE:
283  areq->regs.regs[IDE_REGISTER_COMMAND] =
284  ATA_COMMAND_SET_MULTIPLE_MODE;
285  areq->regs.to_write |=
286  ATA_REGISTERS_VALUE(IDE_REGISTER_SECTOR_COUNT);
287  areq->regs.regs[IDE_REGISTER_SECTOR_COUNT] = *(uint8_t*)argp;
288  break;
289 
290  default:
291  free(areq);
292  return RTEMS_INVALID_NUMBER;
293  break;
294  }
295 
296  rc = rtems_semaphore_create(rtems_build_name('I', 'D', 'E', 'S'),
297  0,
298  RTEMS_FIFO | RTEMS_COUNTING_SEMAPHORE |
299  RTEMS_NO_INHERIT_PRIORITY |
300  RTEMS_NO_PRIORITY_CEILING | RTEMS_LOCAL,
301  0,
302  &(areq->sema));
303  if (rc != RTEMS_SUCCESSFUL)
304  {
305  free(areq);
306  return rc;
307  }
308 
309  ata_add_to_controller_queue(ctrl_minor, areq);
310 
311  /* wait for request processing... */
312  rc = rtems_semaphore_obtain(areq->sema, RTEMS_WAIT, RTEMS_NO_TIMEOUT);
313  if (rc != RTEMS_SUCCESSFUL)
314  {
315  free(areq);
316  return rc;
317  }
318 
319  rtems_semaphore_delete(areq->sema);
320 
321  /*
322  * if no error occurred and if necessary, update internal ata driver data
323  * structures to reflect changes (in device configuration, for example)
324  */
325  if (areq->status == RTEMS_SUCCESSFUL)
326  {
327  switch(cmd)
328  {
329  case ATAIO_SET_MULTIPLE_MODE:
330  /* invalid operation now */
331  default:
332  rc = RTEMS_INVALID_NUMBER;
333  break;
334  }
335  }
336  else
337  {
338  /* XXX: should be correct error processing: for ex, may be
339  * ABRT and then we should return RTEMS_NOT_IMPLEMENTED
340  */
341  rc = RTEMS_IO_ERROR;
342  }
343 
344  /* tell ata driver that controller ready to serve next request */
345  ATA_SEND_EVT(msg, ATA_MSG_SUCCESS_EVT, ctrl_minor, 0);
346 
347  return rc;
348 }

350 /* ata_process_request --
351  * Get first request from controller's queue and process it.
352  *
353  * PARAMETERS:
354  * ctrl_minor - controller identifier
355  *
356  * RETURNS:
357  * NONE
358  */
359 static void
360 ata_process_request(rtems_device_minor_number ctrl_minor)
361 {
362  ata_req_t *areq;
363  uint16_t byte; /* emphasize that only 8 low bits is meaningful */
364  ata_queue_msg_t msg;
365  uint8_t i;
366 #if 0
367  uint8_t dev;
368 #endif
369  uint16_t val;
370  ISR_Level level;
371 
372  /* if no requests to controller then do nothing */
373  if (rtems_chain_is_empty(&ata_ide_ctrls[ctrl_minor].reqs))
374  return;
375 
376  /* get first request in the controller's queue */
377  _ISR_Local_disable(level);
378  areq = (ata_req_t *)rtems_chain_first(&ata_ide_ctrls[ctrl_minor].reqs);
379  _ISR_Local_enable(level);
380 
381 #if 0
382  /* get ATA device identifier (0 or 1) */
383  dev = areq->regs.regs[IDE_REGISTER_DEVICE_HEAD] &
384  IDE_REGISTER_DEVICE_HEAD_DEV;
385 #endif
386 
387  /* execute device select protocol */
388  ide_controller_write_register(ctrl_minor, IDE_REGISTER_DEVICE_HEAD,
389  areq->regs.regs[IDE_REGISTER_DEVICE_HEAD]);
390 
391  do {
392  ide_controller_read_register(ctrl_minor, IDE_REGISTER_STATUS, &byte);
393  } while ((byte & IDE_REGISTER_STATUS_BSY) ||
394  (!(byte & IDE_REGISTER_STATUS_DRDY)));
395 
396  /* fill in all necessary registers on the controller */
397  for (i=0; i< ATA_MAX_CMD_REG_OFFSET; i++)
398  {
399  uint32_t reg = (1 << i);
400  if (areq->regs.to_write & reg)
401  ide_controller_write_register(ctrl_minor, i, areq->regs.regs[i]);
402  }
403 
404 #if ATA_DEBUG
405  ata_printf("ata_process_request: type: %d\n", areq->type);
406 #endif
407 
408  /* continue to execute ATA protocols depending on type of request */
409  if (areq->type == ATA_COMMAND_TYPE_PIO_OUT)
410  {
411  do {
412  ide_controller_read_register(ctrl_minor, IDE_REGISTER_STATUS,
413  &byte);
414  } while (byte & IDE_REGISTER_STATUS_BSY);
415 
416  if (byte & IDE_REGISTER_STATUS_DRQ)
417  {
418  if (areq->cnt)
419  {
420  int ccbuf = areq->cbuf;
421  ide_controller_write_data_block(ctrl_minor,
422  areq->breq->bufs[0].length * areq->cnt,
423  areq->breq->bufs, &areq->cbuf,
424  &areq->pos);
425  ccbuf = areq->cbuf - ccbuf;
426  areq->cnt -= ccbuf;
427  }
428  }
429  else
430  {
431  if (IDE_Controller_Table[ctrl_minor].int_driven == false)
432  {
433  ide_controller_read_register(
434  ctrl_minor,
435  IDE_REGISTER_ALTERNATE_STATUS_OFFSET,
436  &val);
437  ide_controller_read_register(ctrl_minor, IDE_REGISTER_STATUS,
438  &val);
439 
440  ATA_SEND_EVT(msg, ATA_MSG_ERROR_EVT, ctrl_minor,
441  RTEMS_IO_ERROR);
442  }
443  }
444  }
445 
446  if (IDE_Controller_Table[ctrl_minor].int_driven == false)
447  {
448  do {
449  ide_controller_read_register(ctrl_minor, IDE_REGISTER_STATUS,
450  &byte);
451  } while (byte & IDE_REGISTER_STATUS_BSY);
452 
453  ATA_SEND_EVT(msg, ATA_MSG_GEN_EVT, ctrl_minor, 0);
454  }
455 }

779 /* ata_queue_task --
780  * Task which manages ATA driver events queue.
781  *
782  * PARAMETERS:
783  * arg - ignored
784  *
785  * RETURNS:
786  * NONE
787  *
788  * NOTES:
789  * should be non-preemptive
790  */
791 static rtems_task
792 ata_queue_task(rtems_task_argument arg)
793 {
794  ata_queue_msg_t msg;
795  size_t size;
796  ata_req_t *areq;
797  rtems_device_minor_number ctrl_minor;
798  uint16_t val;
799  uint16_t val1;
800  rtems_status_code rc;
801  ISR_Level level;
802 
803  rtems_ata_lock();
804 
805  while (1)
806  {
807  rtems_ata_unlock();
808 
809  /* get event which has happend */
810  rc = rtems_message_queue_receive(ata_queue_id, &msg, &size, RTEMS_WAIT,
811  RTEMS_NO_TIMEOUT);
812  if (rc != RTEMS_SUCCESSFUL)
813  rtems_fatal_error_occurred(RTEMS_INTERNAL_ERROR);
814 
815  /* get controller on which event has happend */
816  ctrl_minor = msg.ctrl_minor;
817 
818  rtems_ata_lock();
819 
820  /* get current request to the controller */
821  _ISR_Local_disable(level);
822  areq = (ata_req_t *)rtems_chain_first(&ata_ide_ctrls[ctrl_minor].reqs);
823  _ISR_Local_enable(level);
824 
825  switch(msg.type)
826  {
827  case ATA_MSG_PROCESS_NEXT_EVT:
828  /* process next request in the controller queue */
829  ata_process_request(ctrl_minor);
830  break;
831 
832  case ATA_MSG_SUCCESS_EVT:
833  /*
834  * finish processing of current request with successful
835  * status and start processing of the next request in the
836  * controller queue
837  */
838  ata_request_done(areq, ctrl_minor, RTEMS_SUCCESSFUL);
839  break;
840 
841  case ATA_MSG_ERROR_EVT:
842  /*
843  * finish processing of current request with error
844  * status and start processing of the next request in the
845  * controller queue
846  */
847  ata_request_done(areq, ctrl_minor, RTEMS_IO_ERROR);
848  break;
849 
850  case ATA_MSG_GEN_EVT:
851  /*
852  * continue processing of the current request to the
853  * controller according to current request state and
854  * ATA protocol
855  */
856  ide_controller_read_register(ctrl_minor, IDE_REGISTER_STATUS,
857  &val);
858  /* process error case */
859  if (val & IDE_REGISTER_STATUS_ERR)
860  {
861  ide_controller_read_register(ctrl_minor,
862  IDE_REGISTER_ERROR,
863  &val);
864  if (val & (IDE_REGISTER_ERROR_UNC |
865  IDE_REGISTER_ERROR_ICRC |
866  IDE_REGISTER_ERROR_IDNF |
867  IDE_REGISTER_ERROR_NM |
868  IDE_REGISTER_ERROR_MED))
869  {
870  if (areq->type == ATA_COMMAND_TYPE_NON_DATA)
871  ata_non_data_request_done(areq, ctrl_minor,
872  RTEMS_UNSATISFIED,
873  RTEMS_IO_ERROR);
874  else
875  ata_request_done(areq, ctrl_minor, RTEMS_IO_ERROR);
876  break;
877  }
878  }
879 
880  switch(areq->type)
881  {
882  case ATA_COMMAND_TYPE_PIO_IN:
883  ata_pio_in_protocol(ctrl_minor, areq);
884  break;
885 
886  case ATA_COMMAND_TYPE_PIO_OUT:
887  ata_pio_out_protocol(ctrl_minor, areq);
888  break;
889 
890  case ATA_COMMAND_TYPE_NON_DATA:
891  ide_controller_read_register(ctrl_minor,
892  IDE_REGISTER_ERROR,
893  &val1);
894  ata_non_data_request_done(areq, ctrl_minor,
895  RTEMS_SUCCESSFUL,
896  val1);
897  break;
898 
899  default:
900 #if ATA_DEBUG
901  ata_printf("ata_queue_task: non-supported command type\n");
902 #endif
903  ata_request_done(areq, ctrl_minor, RTEMS_IO_ERROR);
904  break;
905  }
906  break;
907 
908  default:
909 #if ATA_DEBUG
910  ata_printf("ata_queue_task: internal error\n");
911  rtems_task_exit();
912 #endif
913  rtems_fatal_error_occurred(RTEMS_INTERNAL_ERROR);
914  break;
915  }
916  }
917 }
