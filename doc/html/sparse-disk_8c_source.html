164 static int sparse_disk_write_block(
165  rtems_sparse_disk *sparse_disk,
166  const rtems_blkdev_bnum block,
167  const uint8_t *buffer,
168  const size_t buffer_size )
169 {
170  unsigned int i;
171  bool block_needs_writing = false;
172  rtems_sparse_disk_key *key;
173  rtems_sparse_disk_key block_key = {
174  .block = block,
175  .data = NULL
176  };
177  size_t bytes_to_copy = sparse_disk->media_block_size;
178 
179  if ( buffer_size < bytes_to_copy )
180  bytes_to_copy = buffer_size;
181 
182  /* we only need to write the block if it is different from the fill pattern.
183  * If the read method does not find a block it will deliver the fill pattern anyway.
184  */
185 
186  key = bsearch(
187  &block_key,
188  sparse_disk->key_table,
189  sparse_disk->used_count,
190  sizeof( rtems_sparse_disk_key ),
191  sparse_disk_compare
192  );
193 
194  if ( NULL == key ) {
195  for ( i = 0; ( !block_needs_writing ) && ( i < bytes_to_copy ); ++i ) {
196  if ( buffer[i] != sparse_disk->fill_pattern )
197  block_needs_writing = true;
198  }
199 
200  if ( block_needs_writing ) {
201  key = sparse_disk_get_new_block( sparse_disk, block );
202  }
203  }
204 
205  if ( NULL != key )
206  memcpy( key->data, buffer, bytes_to_copy );
207  else if ( block_needs_writing )
208  return -1;
209 
210  return bytes_to_copy;
211 }

214  * Read/write handling

216 static int sparse_disk_read_write(
217  rtems_sparse_disk *sparse_disk,
218  rtems_blkdev_request *req,
219  const bool read )
220 {
221  int rv = 0;
222  uint32_t req_buffer;
223  rtems_blkdev_sg_buffer *scatter_gather;
224  rtems_blkdev_bnum block;
225  uint8_t *buff;
226  size_t buff_size;
227  unsigned int bytes_handled;
228 
229  rtems_mutex_lock( &sparse_disk->mutex );
230 
231  for ( req_buffer = 0;
232  ( 0 <= rv ) && ( req_buffer < req->bufnum );
233  ++req_buffer ) {
234  scatter_gather = &req->bufs[req_buffer];
235 
236  bytes_handled = 0;
237  buff = (uint8_t *) scatter_gather->buffer;
238  block = scatter_gather->block;
239  buff_size = scatter_gather->length;
240 
241  while ( ( 0 <= rv ) && ( 0 < buff_size ) ) {
242  if ( read )
243  rv = sparse_disk_read_block( sparse_disk,
244  block,
245  &buff[bytes_handled],
246  buff_size );
247  else
248  rv = sparse_disk_write_block( sparse_disk,
249  block,
250  &buff[bytes_handled],
251  buff_size );
252 
253  ++block;
254  bytes_handled += rv;
255  buff_size -= rv;
256  }
257  }
258 
259  rtems_mutex_unlock( &sparse_disk->mutex );
260 
261  if ( 0 > rv )
262  rtems_blkdev_request_done( req, RTEMS_IO_ERROR );
263  else
264  rtems_blkdev_request_done( req, RTEMS_SUCCESSFUL );
265 
266  return 0;
267 }

272 static int sparse_disk_ioctl( rtems_disk_device *dd, uint32_t req, void *argp )
273 {
274  rtems_sparse_disk *sd = rtems_disk_get_driver_data( dd );
275 
276  if ( RTEMS_BLKIO_REQUEST == req ) {
277  rtems_blkdev_request *r = argp;
278 
279  switch ( r->req ) {
280  case RTEMS_BLKDEV_REQ_READ:
281  case RTEMS_BLKDEV_REQ_WRITE:
282  return sparse_disk_read_write( sd, r, r->req == RTEMS_BLKDEV_REQ_READ );
283  default:
284  break;
285  }
286  } else if ( RTEMS_BLKIO_DELETED == req ) {
287  rtems_mutex_destroy( &sd->mutex );
288 
289  if ( NULL != sd->delete_handler )
290  ( *sd->delete_handler )( sd );
291 
292  return 0;
293  } else {
294  return rtems_blkdev_ioctl( dd, req, argp );
295  }
296 
297  errno = EINVAL;
298  return -1;
299 }
