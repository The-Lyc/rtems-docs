107 static int vfcntl(
108  int fd,
109  int cmd,
110  va_list ap
111 )
112 {
113  rtems_libio_t *iop;
114  int fd2;
115  int flags;
116  int mask;
117  int ret = 0;
118 
119  LIBIO_GET_IOP( fd, iop );
120 
121  /*
122  * Now process the fcntl().
123  */
124 
125  /*
126  * This switch should contain all the cases from POSIX.
127  */
128 
129  switch ( cmd ) {
130  case F_DUPFD: /* dup */
131  ret = duplicate_iop( iop );
132  break;
133 
134  case F_DUP2FD: /* dup2 */
135  fd2 = va_arg( ap, int );
136  ret = duplicate2_iop( iop, fd2 );
137  break;
138 
139  case F_GETFD: /* get f_flags */
140  ret = ((rtems_libio_iop_flags(iop) & LIBIO_FLAGS_CLOSE_ON_EXEC) != 0);
141  break;
142 
143  case F_SETFD: /* set f_flags */
144  /*
145  * Interpret the third argument as the "close on exec()" flag.
146  * If this argument is 1, then the file descriptor is to be closed
147  * if a new process is exec()'ed. Since RTEMS does not support
148  * processes, then we can ignore this one except to make
149  * F_GETFD work.
150  */
151 
152  if ( va_arg( ap, int ) )
153  rtems_libio_iop_flags_set( iop, LIBIO_FLAGS_CLOSE_ON_EXEC );
154  else
155  rtems_libio_iop_flags_clear( iop, LIBIO_FLAGS_CLOSE_ON_EXEC );
156  break;
157 
158  case F_GETFL: /* more flags (cloexec) */
159  ret = rtems_libio_to_fcntl_flags( rtems_libio_iop_flags( iop ) );
160  break;
161 
162  case F_SETFL:
163  flags = rtems_libio_fcntl_flags( va_arg( ap, int ) );
164  mask = LIBIO_FLAGS_NO_DELAY | LIBIO_FLAGS_APPEND;
165 
166  /*
167  * XXX If we are turning on append, should we seek to the end?
168  */
169 
170  rtems_libio_iop_flags_clear( iop, mask );
171  rtems_libio_iop_flags_set( iop, flags & mask );
172  break;
173 
174  case F_GETLK:
175  errno = ENOTSUP;
176  ret = -1;
177  break;
178 
179  case F_SETLK:
180  errno = ENOTSUP;
181  ret = -1;
182  break;
183 
184  case F_SETLKW:
185  errno = ENOTSUP;
186  ret = -1;
187  break;
188 
189  case F_SETOWN: /* for sockets. */
190  errno = ENOTSUP;
191  ret = -1;
192  break;
193 
194  case F_GETOWN: /* for sockets. */
195  errno = ENOTSUP;
196  ret = -1;
197  break;
198 
199  default:
200  errno = EINVAL;
201  ret = -1;
202  break;
203  }
204 
205  /*
206  * If we got this far successfully, then we give the optional
207  * filesystem specific handler a chance to process this.
208  */
209 
210  if (ret >= 0) {
211  int err = (*iop->pathinfo.handlers->fcntl_h)( iop, cmd );
212  if (err) {
213  errno = err;
214  ret = -1;
215  }
216  }
217 
218  rtems_libio_iop_drop( iop );
219  return ret;
220 }
