79  /*

133 /*
134  * read bytes from the serial port. We only have stdin.
135  */
136 rtems_device_driver console_read(
137  rtems_device_major_number major,
138  rtems_device_minor_number minor,
139  void * arg
140 )
141 {
142  rtems_libio_rw_args_t *rw_args;
143  char *buffer;
144  int maximum;
145  int count = 0;
146 
147  rw_args = (rtems_libio_rw_args_t *) arg;
148 
149  buffer = rw_args->buffer;
150  maximum = rw_args->count;
151 
152  for (count = 0; count < maximum; count++) {
153  buffer[ count ] = inbyte();
154  if (buffer[ count ] == '\n' || buffer[ count ] == '\r') {
155  buffer[ count++ ] = '\n';
156  break;
157  }
158  }
159 
160  rw_args->bytes_moved = count;
161  return (count >= 0) ? RTEMS_SUCCESSFUL : RTEMS_UNSATISFIED;
162 }

167 rtems_device_driver console_write(
168  rtems_device_major_number major,
169  rtems_device_minor_number minor,
170  void * arg
171 )
172 {
173  int count;
174  int maximum;
175  rtems_libio_rw_args_t *rw_args;
176  char *buffer;
177 
178  rw_args = (rtems_libio_rw_args_t *) arg;
179 
180  buffer = rw_args->buffer;
181  maximum = rw_args->count;
182 
183  for (count = 0; count < maximum; count++) {
184  if ( buffer[ count ] == '\n') {
185  outbyte('\r');

187  outbyte( buffer[ count ] );
188  }
189 
190  rw_args->bytes_moved = maximum;
191  return 0;
192 }
