111 /*
112  * Usage:
113  * xprt = svctcp_create(sock, send_buf_size, recv_buf_size);
114  *
115  * Creates, registers, and returns a (rpc) tcp based transporter.
116  * Once *xprt is initialized, it is registered as a transporter
117  * see (svc.h, xprt_register). This routine returns
118  * a NULL if a problem occurred.
119  *
120  * If sock<0 then a socket is created, else sock is used.
121  * If the socket, sock is not bound to a port then svctcp_create
122  * binds it to an arbitrary port. The routine then starts a tcp
123  * listener on the socket's associated port. In any (successful) case,
124  * xprt->xp_sock is the registered socket number and xprt->xp_port is the
125  * associated port number.
126  *
127  * Since tcp streams do buffered io similar to stdio, the caller can specify
128  * how big the send and receive buffers are via the second and third parms;
129  * 0 => use the system default.
130  */
131 SVCXPRT *
132 svctcp_create(
133  int sock,
134  u_int sendsize,
135  u_int recvsize)
136 {
137  bool_t madesock = FALSE;
138  register SVCXPRT *xprt;
139  register struct tcp_rendezvous *r;
140  struct sockaddr_in addr;
141  socklen_t len = sizeof(struct sockaddr_in);
142  int on;
143 
144  if (sock == RPC_ANYSOCK) {
145  if ((sock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP)) < 0) {
146  perror("svctcp_.c - udp socket creation problem");
147  return ((SVCXPRT *)NULL);
148  }
149  madesock = TRUE;
150  }
151  on = 1;
152  if (ioctl(sock, FIONBIO, &on) < 0) {
153  perror("svc_tcp.c - cannot turn on non-blocking mode");
154  if (madesock)
155  (void)_RPC_close(sock);
156  return ((SVCXPRT *)NULL);
157  }
158  memset(&addr, 0, sizeof (addr));
159  addr.sin_len = sizeof(struct sockaddr_in);
160  addr.sin_family = AF_INET;
161  if (bindresvport(sock, &addr)) {
162  addr.sin_port = 0;
163  (void)bind(sock, (struct sockaddr *)&addr, len);
164  }
165  if ((getsockname(sock, (struct sockaddr *)&addr, &len) != 0) ||
166  (listen(sock, 2) != 0)) {
167  perror("svctcp_.c - cannot getsockname or listen");
168  if (madesock)
169  (void)_RPC_close(sock);
170  return ((SVCXPRT *)NULL);
171  }
172  r = (struct tcp_rendezvous *)mem_alloc(sizeof(*r));
173  if (r == NULL) {
174  (void) fprintf(stderr, "svctcp_create: out of memory\n");
175  return (NULL);
176  }
177  r->sendsize = sendsize;
178  r->recvsize = recvsize;
179  xprt = (SVCXPRT *)mem_alloc(sizeof(SVCXPRT));
180  if (xprt == NULL) {
181  (void) fprintf(stderr, "svctcp_create: out of memory\n");
182  return (NULL);
183  }
184  xprt->xp_p2 = NULL;
185  xprt->xp_p1 = (caddr_t)r;
186  xprt->xp_verf = _null_auth;
187  xprt->xp_ops = &svctcp_rendezvous_op;
188  xprt->xp_port = ntohs(addr.sin_port);
189  xprt->xp_sock = sock;
190  xprt_register(xprt);
191  return (xprt);
192 }

245 rendezvous_request(
246  SVCXPRT *xprt,
247  struct rpc_msg *msg)
248 {
249  int sock;
250  struct tcp_rendezvous *r;
251  struct sockaddr_in addr;
252  socklen_t len;
253  int off;
254 
255  r = (struct tcp_rendezvous *)xprt->xp_p1;
256  again:
257  len = sizeof(struct sockaddr_in);
258  if ((sock = accept(xprt->xp_sock, (struct sockaddr *)&addr,
259  &len)) < 0) {
260  if (errno == EINTR)
261  goto again;
262  return (FALSE);
263  }
264  /*
265  * Guard against FTP bounce attacks.
266  */
267  if (addr.sin_port == htons(20)) {
268  _RPC_close(sock);
269  return (FALSE);
270  }
271  /*
272  * The listening socket is in FIONBIO mode and we inherit it.
273  */
274  off = 0;
275  if (ioctl(sock, FIONBIO, &off) < 0) {
276  _RPC_close(sock);
277  return (FALSE);
278  }
279  /*
280  * make a new transporter (re-uses xprt)
281  */
282  xprt = makefd_xprt(sock, r->sendsize, r->recvsize);
283  xprt->xp_raddr = addr;
284  xprt->xp_addrlen = len;
285  return (FALSE); /* there is never an rpc msg to be processed */
286 }

320 /*
321  * reads data from the tcp conection.
322  * any error is fatal and the connection is closed.
323  * (And a read of zero bytes is a half closed stream => error.)
324  *
325  * Note: we have to be careful here not to allow ourselves to become
326  * blocked too long in this routine. While we're waiting for data from one
327  * client, another client may be trying to connect. To avoid this situation,
328  * some code from svc_run() is transplanted here: the select() loop checks
329  * all RPC descriptors including the one we want and calls svc_getreqset2()
330  * to handle new requests if any are detected.
331  */
332 static int
333 readtcp(
334  char *_xprt,
335  char *buf,
336  int len)
337 {
338  SVCXPRT *xprt = (SVCXPRT*) _xprt;
339  register int sock = xprt->xp_sock;
340  struct timeval start, delta, tv;
341  struct timeval tmp1, tmp2;
342  fd_set *fds;
343 
344  delta = wait_per_try;
345  fds = NULL;
346  gettimeofday(&start, NULL);
347  do {
348  int bytes = sizeof (fd_set);
349  if (fds != NULL)
350  free(fds);
351  fds = (fd_set *)malloc(bytes);
352  if (fds == NULL)
353  goto fatal_err;
354  memcpy(fds, __svc_fdset, bytes);
355 
356  /* XXX we know the other bits are still clear */
357  FD_SET(sock, fds);
358  tv = delta; /* in case select() implements writeback */
359  switch (select(svc_maxfd + 1, fds, NULL, NULL, &tv)) {
360  case -1:
361  if (errno != EINTR)
362  goto fatal_err;
363  gettimeofday(&tmp1, NULL);
364  timersub(&tmp1, &start, &tmp2);
365  timersub(&wait_per_try, &tmp2, &tmp1);
366  if (tmp1.tv_sec < 0 || !timerisset(&tmp1))
367  goto fatal_err;
368  delta = tmp1;
369  continue;
370  case 0:
371  goto fatal_err;
372  default:
373  if (!FD_ISSET(sock, fds)) {
374  svc_getreqset2(fds, svc_maxfd + 1);
375  gettimeofday(&tmp1, NULL);
376  timersub(&tmp1, &start, &tmp2);
377  timersub(&wait_per_try, &tmp2, &tmp1);
378  if (tmp1.tv_sec < 0 || !timerisset(&tmp1))
379  goto fatal_err;
380  delta = tmp1;
381  continue;
382  }
383  }
384  } while (!FD_ISSET(sock, fds));
385  if ((len = _RPC_read(sock, buf, len)) > 0) {
386  if (fds != NULL)
387  free(fds);
388  return (len);
389  }
390 fatal_err:
391  ((struct tcp_conn *)(xprt->xp_p1))->strm_stat = XPRT_DIED;
392  if (fds != NULL)
393  free(fds);
394  return (-1);
395 }
