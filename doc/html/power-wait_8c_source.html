78 BSP_FAST_TEXT_SECTION static void
79 pmc_use_main_rc_osc_4mhz(volatile Pmc *pmc)
80 {
81  uint32_t ckgr_mor;
82 
83  ckgr_mor = pmc->CKGR_MOR;
84  ckgr_mor |= CKGR_MOR_KEY_PASSWD;
85 
86  /* Enable main RC oscillator */
87  ckgr_mor |= CKGR_MOR_MOSCRCEN;
88  PMC->CKGR_MOR = ckgr_mor;
89  pmc_wait_for_main_rc_osc(pmc);
90 
91  /* Set main RC oscillator frequency to 4MHz */
92  ckgr_mor &= ~CKGR_MOR_MOSCRCF_Msk;
93  ckgr_mor |= CKGR_MOR_MOSCRCF_4_MHz;
94  pmc->CKGR_MOR = ckgr_mor;
95  pmc_wait_for_main_rc_osc(pmc);
96 
97  /* Switch to main RC oscillator */
98  ckgr_mor &= ~CKGR_MOR_MOSCSEL;
99  pmc->CKGR_MOR = ckgr_mor;
100  pmc_wait_for_main_osc_selection(pmc);
101  pmc_wait_for_master_clock_ready(pmc);
102 }

121  pmc->CKGR_MOR = ckgr_mor;

126 BSP_FAST_TEXT_SECTION void
127 atsam_power_handler_wait_mode(
128  const atsam_power_control *control,
129  atsam_power_state state
130 )
131 {
132  rtems_interrupt_level level;
133  volatile Pmc *pmc;
134  uint32_t mckr;
135  uint32_t fmr;
136  uint32_t fsmr;
137 
138  (void) control;
139  pmc = PMC;
140 
141  switch (state) {
142  case ATSAM_POWER_OFF:
143  rtems_interrupt_disable(level);
144 
145  mckr = pmc->PMC_MCKR;
146 
147  /* Switch main clock to main RC oscillator (4MHz) */
148  pmc_use_main_rc_osc_4mhz(pmc);
149  pmc_set_master_clock_source(pmc, PMC_MCKR_CSS_MAIN_CLK);
150  pmc_wait_for_master_clock_ready(pmc);
151  pmc_set_master_clock_prescaler(pmc, PMC_MCKR_PRES_CLK_1);
152  pmc_wait_for_master_clock_ready(pmc);
153  pmc_set_master_clock_division(pmc, PMC_MCKR_MDIV_EQ_PCK);
154  pmc_wait_for_master_clock_ready(pmc);
155 
156  /* Set flash wait state to 0 */
157  fmr = EFC->EEFC_FMR;
158  EFC->EEFC_FMR = EEFC_FMR_FWS(0);
159 
160  /* Enter wait mode */
161  fsmr = pmc->PMC_FSMR;
162  pmc->PMC_FSMR = fsmr | PMC_FSMR_LPM;
163  pmc->CKGR_MOR = CKGR_MOR_KEY_PASSWD | CKGR_MOR_WAITMODE;
164  pmc_wait_for_master_clock_ready(pmc);
165  __asm__ volatile ( "wfe" : : : "memory" );
166  pmc->PMC_FSMR = fsmr;
167 
168  /* Restore main clock */
169  pmc_set_master_clock_prescaler(pmc, mckr);
170  pmc_wait_for_master_clock_ready(pmc);
171  pmc_set_master_clock_division(pmc, mckr);
172  pmc_wait_for_master_clock_ready(pmc);
173  pmc_set_master_clock_source(pmc, mckr);
174  pmc_wait_for_master_clock_ready(pmc);
175  pmc_use_main_ext_osc(pmc);
176 
177  /* Restore flash wait state */
178  EFC->EEFC_FMR = fmr;
179 
180  rtems_interrupt_enable(level);
181  break;
182  case ATSAM_POWER_INIT:
183  rtems_interrupt_disable(level);
184 
185  pmc->PMC_FSMR = (pmc->PMC_FSMR & ~PMC_FSMR_FLPM_Msk)
186  | PMC_FSMR_FLPM(PMC_FSMR_FLPM_FLASH_DEEP_POWERDOWN);
187 
188  /* No Cortex-M7 Deep Sleep mode (Backup Mode) */
189  SCB->SCR &= ~SCB_SCR_SLEEPDEEP_Msk;
190 
191  rtems_interrupt_enable(level);
192  default:
193  break;
194  }
195 }
