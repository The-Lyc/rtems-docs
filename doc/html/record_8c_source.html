91 size_t _Record_String_to_items(
92  rtems_record_event event,
93  const char *str,
94  size_t len,
95  rtems_record_item *items,
96  size_t item_count
97 )
98 {
99  size_t s;
100  size_t i;
101 
102  s = 0;
103  i = 0;
104 
105  while ( s < len && i < item_count ) {
106  rtems_record_data data;
107  size_t k;
108 
109  data = 0;
110 
111  for ( k = 0; s < len && k < sizeof( data ); ++k ) {
112  rtems_record_data c;
113 
114  c = (unsigned char) str[ s ];
115  data |= c << ( k * 8 );
116  ++s;
117  }
118 
119  items[ i ].event = event;
120  items[ i ].data = data;
121  ++i;
122  }
123

127 void _Record_Drain(
128  Record_Control *control,
129  uint32_t cpu_index,
130  rtems_record_drain_visitor visitor,
131  void *arg
132 )
133 {
134  unsigned int tail;
135  unsigned int head;
136 
137  tail = _Record_Tail( control );
138  head = _Atomic_Load_uint( &control->head, ATOMIC_ORDER_ACQUIRE );
139 
140  if ( tail == head ) {
141  return;
142  }
143 
144  control->tail = head;
145 
146  control->Header[ 0 ].event = RTEMS_RECORD_PROCESSOR;
147  control->Header[ 0 ].data = cpu_index;
148  control->Header[ 1 ].event = RTEMS_RECORD_PER_CPU_TAIL;
149  control->Header[ 1 ].data = tail;
150  control->Header[ 2 ].event = RTEMS_RECORD_PER_CPU_HEAD;
151  control->Header[ 2 ].data = head;
152  ( *visitor )( control->Header, RTEMS_ARRAY_SIZE( control->Header ), arg );
153 
154  if ( _Record_Is_overflow( control, tail, head ) ) {
155  tail = head + 1;
156  }
157 
158  tail = _Record_Index( control, tail );
159  head = _Record_Index( control, head );
160 
161  if ( tail < head ) {
162  ( *visitor )( &control->Items[ tail ], head - tail, arg );
163  } else {
164  ( *visitor )( &control->Items[ tail ], control->mask + 1 - tail, arg );
165 
166  if ( head > 0 ) {
167  ( *visitor )( &control->Items[ 0 ], head, arg );
168  }
169  }
170 }
