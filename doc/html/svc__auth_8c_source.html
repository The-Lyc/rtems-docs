98 /*
99  * The call rpc message, msg has been obtained from the wire. The msg contains
100  * the raw form of credentials and verifiers. authenticate returns AUTH_OK
101  * if the msg is successfully authenticated. If AUTH_OK then the routine also
102  * does the following things:
103  * set rqst->rq_xprt->verf to the appropriate response verifier;
104  * sets rqst->rq_client_cred to the "cooked" form of the credentials.
105  *
106  * NB: rqst->rq_cxprt->verf must be pre-alloctaed;
107  * its length is set appropriately.
108  *
109  * The caller still owns and is responsible for msg->u.cmb.cred and
110  * msg->u.cmb.verf. The authentication system retains ownership of
111  * rqst->rq_client_cred, the cooked credentials.
112  *
113  * There is an assumption that any flavour less than AUTH_NULL is
114  * invalid.
115  */
116 enum auth_stat
117 _authenticate(
118  struct svc_req *rqst,
119  struct rpc_msg *msg )
120 {
121  register int cred_flavor;
122  register struct authsvc *asp;
123 
124  rqst->rq_cred = msg->rm_call.cb_cred;
125  rqst->rq_xprt->xp_verf.oa_flavor = _null_auth.oa_flavor;
126  rqst->rq_xprt->xp_verf.oa_length = 0;
127  cred_flavor = rqst->rq_cred.oa_flavor;
128  switch (cred_flavor) {
129  case AUTH_NULL:
130  return(_svcauth_null(rqst, msg));
131  case AUTH_UNIX:
132  return(_svcauth_unix(rqst, msg));
133  case AUTH_SHORT:
134  return(_svcauth_short(rqst, msg));
135  /*
136  * We leave AUTH_DES turned off by default because svcauth_des()
137  * needs getpublickey(), which is in librpcsvc, not libc. If we
138  * included AUTH_DES as a built-in flavor, programs that don't
139  * have -lrpcsvc in their Makefiles wouldn't link correctly, even
140  * though they don't use AUTH_DES. And I'm too lazy to go through
141  * the tree looking for all of them.
142  */
143 #ifdef DES_BUILTIN
144  case AUTH_DES:
145  return(_svcauth_des(rqst, msg));
146 #endif
147  }
148 
149  /* flavor doesn't match any of the builtin types, so try new ones */
150  for (asp = Auths; asp; asp = asp->next) {
151  if (asp->flavor == cred_flavor) {
152  enum auth_stat as;
153 
154  as = (*asp->handler)(rqst, msg);
155  return (as);
156  }
157  }
158 
159  return (AUTH_REJECTEDCRED);
160 }

171 /*
172  * Allow the rpc service to register new authentication types that it is
173  * prepared to handle. When an authentication flavor is registered,
174  * the flavor is checked against already registered values. If not
175  * registered, then a new Auths entry is added on the list.
176  *
177  * There is no provision to delete a registration once registered.
178  *
179  * This routine returns:
180  * 0 if registration successful
181  * 1 if flavor already registered
182  * -1 if can't register (errno set)
183  */
184 
185 int
186 svc_auth_reg(
187  int cred_flavor,
188  enum auth_stat (*handler)(struct svc_req *rqst, struct rpc_msg *msg))
189 {
190  register struct authsvc *asp;
191 
192  switch (cred_flavor) {
193  case AUTH_NULL:
194  case AUTH_UNIX:
195  case AUTH_SHORT:
196 #ifdef DES_BUILTIN
197  case AUTH_DES:
198 #endif
199  /* already registered */
200  return (1);
201 
202  default:
203  for (asp = Auths; asp; asp = asp->next) {
204  if (asp->flavor == cred_flavor) {
205  /* already registered */
206  return (1);
207  }
208  }
209 
210  /* this is a new one, so go ahead and register it */
211  asp = (struct authsvc *)mem_alloc(sizeof (*asp));
212  if (asp == NULL) {
213  return (-1);
214  }
215  asp->flavor = cred_flavor;
216  asp->handler = handler;
217  asp->next = Auths;
218  Auths = asp;
219  break;
220  }
221  return (0);
