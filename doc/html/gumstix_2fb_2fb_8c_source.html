64 static void enable_fbskyeye(void)
65 {
66  LCCR1 &= ~LCCR1_PPL;
67  LCCR1 |= (SCREEN_WIDTH -1 ) & LCCR1_PPL;
68  LCCR2 &= ~LCCR2_LPP;
69  LCCR2 |= (SCREEN_HEIGHT/2 -1) & LCCR2_LPP;
70  LCCR3 &= ~LCCR3_BPP;
71  LCCR3 |= 4<<24;
72  FDADR0 = LCD_DMA_POINTER - 0x1000;
73  LCCR0 |= LCCR0_ENB;
