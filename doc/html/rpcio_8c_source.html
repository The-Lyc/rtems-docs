578 rpcUdpServerDestroy(RpcUdpServer s)
579 {
580 RpcUdpServer prev;
581  if (!s)
582  return;
583  /* we should probably verify (but how?) that nobody
584  * (at least: no outstanding XACTs) is using this
585  * server;
586  */
587 
588  /* remove from server list */
589  MU_LOCK(llock);
590  prev = rpcUdpServers;
591  if ( s == prev ) {
592  rpcUdpServers = s->next;
593  } else {
594  for ( ; prev ; prev = prev->next) {
595  if (prev->next == s) {
596  prev->next = s->next;
597  break;
598  }
599  }
600  }
601  MU_UNLOCK(llock);
602 
603  /* MUST have found it */
604  assert(prev);
605 
606  auth_destroy(s->auth);
607 
608  MU_DESTROY(s->authlock);
609  MY_FREE(s);
610 }

636 RpcUdpXact
637 rpcUdpXactCreate(
638  u_long program,
639  u_long version,
640  u_long size
641  )
642 {
643 RpcUdpXact rval=0;
644 struct rpc_msg header;
645 register int i,j;
646 
647  if (!size)
648  size = UDPMSGSIZE;
649  /* word align */
650  size = (size + 3) & ~3;
651 
652  rval = (RpcUdpXact)MY_CALLOC(1,sizeof(*rval) - sizeof(rval->obuf) + size);
653 
654  if (rval) {
655 
656  header.rm_xid = 0;
657  header.rm_direction = CALL;
658  header.rm_call.cb_rpcvers = RPC_MSG_VERSION;
659  header.rm_call.cb_prog = program;
660  header.rm_call.cb_vers = version;
661  xdrmem_create(&(rval->xdrs), rval->obuf.buf, size, XDR_ENCODE);
662 
663  if (!xdr_callhdr(&(rval->xdrs), &header)) {
664  MY_FREE(rval);
665  return 0;
666  }
667  /* pick a free table slot and initialize the XID */
668  MU_LOCK(hlock);
669  rval->obuf.xid = (xidHashSeed++ ^ ((uintptr_t)rval>>10)) & XACT_HASH_MSK;
670  i=j=(rval->obuf.xid & XACT_HASH_MSK);
671  if (msgQ) {
672  /* if there's no message queue, refuse to
673  * give them transactions; we might be in the process to
674  * go away...
675  */
676  do {
677  i=(i+1) & XACT_HASH_MSK; /* cheap modulo */
678  if (!xactHashTbl[i]) {
679 #if (DEBUG) & DEBUG_TRACE_XACT
680  fprintf(stderr,"RPCIO: entering index %i, val %x\n",i,rval);
681 #endif
682  xactHashTbl[i]=rval;
683  j=-1;
684  break;
685  }
686  } while (i!=j);
687  }
688  MU_UNLOCK(hlock);
689  if (i==j) {
690  XDR_DESTROY(&rval->xdrs);
691  MY_FREE(rval);
692  return 0;
693  }
694  rval->obuf.xid = xidUpper[i] | i;
695  rval->xdrpos = XDR_GETPOS(&(rval->xdrs));
696  rval->obufsize = size;
697  }
698  return rval;
699 }

728 /* Send a transaction, i.e. enqueue it to the
729  * RPC daemon who will actually send it.
730  */
731 enum clnt_stat
732 rpcUdpSend(
733  RpcUdpXact xact,
734  RpcUdpServer srvr,
735  struct timeval *timeout,
736  u_long proc,
737  xdrproc_t xres, caddr_t pres,
738  xdrproc_t xargs, caddr_t pargs,
739  ...
740  )
741 {
742 register XDR *xdrs;
743 unsigned long ms;
744 va_list ap;
745 
746  va_start(ap,pargs);
747 
748  if (!timeout)
749  timeout = RPCIOD_DEFAULT_TIMEOUT;
750 
751  ms = 1000 * timeout->tv_sec + timeout->tv_usec/1000;
752 
753  /* round lifetime to closest # of ticks */
754  xact->lifetime = (ms * ticksPerSec + 500) / 1000;
755  if ( 0 == xact->lifetime )
756  xact->lifetime = 1;
757 
758 #if (DEBUG) & DEBUG_TIMEOUT
759  {
760  static int once=0;
761  if (!once++) {
762  fprintf(stderr,
763  "Initial lifetime: %i (ticks)\n",
764  xact->lifetime);
765  }
766  }
767 #endif
768 
769  xact->tolive = xact->lifetime;
770 
771  xact->xres = xres;
772  xact->pres = pres;
773  xact->server = srvr;
774 
775  xdrs = &xact->xdrs;
776  xdrs->x_op = XDR_ENCODE;
777  /* increment transaction ID */
778  xact->obuf.xid += XACT_HASHS;
779  XDR_SETPOS(xdrs, xact->xdrpos);
780  if ( !XDR_PUTLONG(xdrs,(long*)&proc) || !locked_marshal(srvr, xdrs) ||
781  !xargs(xdrs, pargs) ) {
782  va_end(ap);
783  return(xact->status.re_status=RPC_CANTENCODEARGS);
784  }
785  while ((xargs=va_arg(ap,xdrproc_t))) {
786  if (!xargs(xdrs, va_arg(ap,caddr_t)))
787  va_end(ap);
788  return(xact->status.re_status=RPC_CANTENCODEARGS);
789  }
790 
791  va_end(ap);
792 
793  rtems_task_ident(RTEMS_SELF, RTEMS_WHO_AM_I, &xact->requestor);
794  if ( rtems_message_queue_send( msgQ, &xact, sizeof(xact)) ) {
795  return RPC_CANTSEND;
796  }
797  /* wakeup the rpciod */
798  ASSERT( RTEMS_SUCCESSFUL==rtems_event_send(rpciod, RPCIOD_TX_EVENT) );
799 
800  return RPC_SUCCESS;
801 }

1134 /* this code does the work */
1135 static void
1136 rpcio_daemon(rtems_task_argument arg)
1137 {
1138 rtems_status_code stat;
1139 RpcUdpXact xact;
1140 RpcUdpServer srv;
1141 rtems_interval next_retrans, then, unow;
1142 long now; /* need to do signed comparison with age! */
1143 rtems_event_set events;
1144 ListNode newList;
1145 size_t size;
1146 rtems_id q = 0;
1147 ListNodeRec listHead = {0, 0};
1148 unsigned long epoch = RPCIOD_EPOCH_SECS * ticksPerSec;
1149 unsigned long max_period = RPCIOD_RETX_CAP_S * ticksPerSec;
1150 rtems_status_code status;
1151 
1152 
1153  then = rtems_clock_get_ticks_since_boot();
1154 
1155  for (next_retrans = epoch;;) {
1156 
1157  if ( RTEMS_SUCCESSFUL !=
1158  (stat = rtems_event_receive(
1159  RPCIOD_RX_EVENT | RPCIOD_TX_EVENT | RPCIOD_KILL_EVENT,
1160  RTEMS_WAIT | RTEMS_EVENT_ANY,
1161  next_retrans,
1162  &events)) ) {
1163  ASSERT( RTEMS_TIMEOUT == stat );
1164  events = 0;
1165  }
1166 
1167  if (events & RPCIOD_KILL_EVENT) {
1168  int i;
1169 
1170 #if (DEBUG) & DEBUG_EVENTS
1171  fprintf(stderr,"RPCIO: got KILL event\n");
1172 #endif
1173 
1174  MU_LOCK(hlock);
1175  for (i=XACT_HASHS-1; i>=0; i--) {
1176  if (xactHashTbl[i]) {
1177  break;
1178  }
1179  }
1180  if (i<0) {
1181  /* prevent them from creating and enqueueing more messages */
1182  q=msgQ;
1183  /* messages queued after we executed this assignment will fail */
1184  msgQ=0;
1185  }
1186  MU_UNLOCK(hlock);
1187  if (i>=0) {
1188  fprintf(stderr,"RPCIO There are still transactions circulating; I refuse to go away\n");
1189  fprintf(stderr,"(1st in slot %i)\n",i);
1190  rtems_binary_semaphore_post(&fini);
1191  } else {
1192  break;
1193  }
1194  }
1195 
1196  unow = rtems_clock_get_ticks_since_boot();
1197 
1198  /* measure everything relative to then to protect against
1199  * rollover
1200  */
1201  now = unow - then;
1202 
1203  /* NOTE: we don't lock the hash table while we are operating
1204  * on transactions; the paradigm is that we 'own' a particular
1205  * transaction (and hence it's hash table slot) from the
1206  * time the xact was put into the message queue until we
1207  * wake up the requestor.
1208  */
1209 
1210  if (RPCIOD_RX_EVENT & events) {
1211 
1212 #if (DEBUG) & DEBUG_EVENTS
1213  fprintf(stderr,"RPCIO: got RX event\n");
1214 #endif
1215 
1216  while ((xact=sockRcv())) {
1217 
1218  /* extract from the retransmission list */
1219  nodeXtract(&xact->node);
1220 
1221  /* change the ID - there might already be
1222  * a retransmission on the way. When it's
1223  * reply arrives we must not find it's ID
1224  * in the hashtable
1225  */
1226  xact->obuf.xid += XACT_HASHS;
1227 
1228  xact->status.re_status = RPC_SUCCESS;
1229 
1230  /* calculate roundtrip ticks */
1231  xact->trip = now - xact->trip;
1232 
1233  srv = xact->server;
1234 
1235  /* adjust the server's retry period */
1236  {
1237  register TimeoutT rtry = srv->retry_period;
1238  register TimeoutT trip = xact->trip;
1239 
1240  ASSERT( trip >= 0 );
1241 
1242  if ( 0==trip )
1243  trip = 1;
1244 
1245  /* retry_new = 0.75*retry_old + 0.25 * 8 * roundrip */
1246  rtry = (3*rtry + (trip << 3)) >> 2;
1247 
1248  if ( rtry > max_period )
1249  rtry = max_period;
1250 
1251  srv->retry_period = rtry;
1252  }
1253 
1254  /* wakeup requestor */
1255  rtems_event_send(xact->requestor, RTEMS_RPC_EVENT);
1256  }
1257  }
1258 
1259  if (RPCIOD_TX_EVENT & events) {
1260 
1261 #if (DEBUG) & DEBUG_EVENTS
1262  fprintf(stderr,"RPCIO: got TX event\n");
1263 #endif
1264 
1265  while (RTEMS_SUCCESSFUL == rtems_message_queue_receive(
1266  msgQ,
1267  &xact,
1268  &size,
1269  RTEMS_NO_WAIT,
1270  RTEMS_NO_TIMEOUT)) {
1271  /* put to the head of timeout q */
1272  nodeAppend(&listHead, &xact->node);
1273 
1274  xact->age = now;
1275  xact->trip = FIRST_ATTEMPT;
1276  }
1277  }
1278 
1279 
1280  /* work the timeout q */
1281  newList = 0;
1282  for ( xact=(RpcUdpXact)listHead.next;
1283  xact && xact->age <= now;
1284  xact=(RpcUdpXact)listHead.next ) {
1285 
1286  /* extract from the list */
1287  nodeXtract(&xact->node);
1288 
1289  srv = xact->server;
1290 
1291  if (xact->tolive < 0) {
1292  /* this one timed out */
1293  xact->status.re_errno = ETIMEDOUT;
1294  xact->status.re_status = RPC_TIMEDOUT;
1295 
1296  srv->timeouts++;
1297 
1298  /* Change the ID - there might still be
1299  * a reply on the way. When it arrives we
1300  * must not find it's ID in the hash table
1301  *
1302  * Thanks to Steven Johnson for hunting this
1303  * one down.
1304  */
1305  xact->obuf.xid += XACT_HASHS;
1306 
1307 #if (DEBUG) & DEBUG_TIMEOUT
1308  fprintf(stderr,"RPCIO XACT timed out; waking up requestor\n");
1309 #endif
1310  if ( rtems_event_send(xact->requestor, RTEMS_RPC_EVENT) ) {
1311  rtems_panic("RPCIO PANIC: requestor id was 0x%08" PRIx32,
1312  xact->requestor);
1313  }
1314 
1315  } else {
1316  int len;
1317 
1318  len = (int)XDR_GETPOS(&xact->xdrs);
1319 
1320 #ifdef MBUF_TX
1321  xact->refcnt = 1; /* sendto itself */
1322 #endif
1323  if ( len != SENDTO( ourSock,
1324  xact->obuf.buf,
1325  len,
1326  0,
1327  &srv->addr.sa,
1328  sizeof(srv->addr.sin)
1329 #ifdef MBUF_TX
1330  , xact,
1331  paranoia_free,
1332  paranoia_ref
1333 #endif
1334  ) ) {
1335 
1336  xact->status.re_errno = errno;
1337  xact->status.re_status = RPC_CANTSEND;
1338  srv->errors++;
1339 
1340  /* wakeup requestor */
1341  fprintf(stderr,"RPCIO: SEND failure\n");
1342  status = rtems_event_send(xact->requestor, RTEMS_RPC_EVENT);
1343  assert( status == RTEMS_SUCCESSFUL );
1344 
1345  } else {
1346  /* send successful; calculate retransmission time
1347  * and enqueue to temporary list
1348  */
1349  if (FIRST_ATTEMPT != xact->trip) {
1350 #if (DEBUG) & DEBUG_TIMEOUT
1351  fprintf(stderr,
1352  "timed out; tolive is %i (ticks), retry period is %i (ticks)\n",
1353  xact->tolive,
1354  srv->retry_period);
1355 #endif
1356  /* this is a real retry; we backup
1357  * the server's retry interval
1358  */
1359  if ( srv->retry_period < max_period ) {
1360 
1361  /* If multiple transactions for this server
1362  * fail (e.g. because it died) this will
1363  * back-off very agressively (doubling
1364  * the retransmission period for every
1365  * timed out transaction up to the CAP limit)
1366  * which is desirable - single packet failure
1367  * is treated more gracefully by this algorithm.
1368  */
1369 
1370  srv->retry_period<<=1;
1371 #if (DEBUG) & DEBUG_TIMEOUT
1372  fprintf(stderr,
1373  "adjusted to; retry period %i\n",
1374  srv->retry_period);
1375 #endif
1376  } else {
1377  /* never wait longer than RPCIOD_RETX_CAP_S seconds */
1378  fprintf(stderr,
1379  "RPCIO: server '%s' not responding - still trying\n",
1380  srv->name);
1381  }
1382  if ( 0 == ++srv->retrans % 1000) {
1383  fprintf(stderr,
1384  "RPCIO - statistics: already %li retries to server %s\n",
1385  srv->retrans,
1386  srv->name);
1387  }
1388  } else {
1389  srv->requests++;
1390  }
1391  xact->trip = now;
1392  {
1393  long capped_period = srv->retry_period;
1394  if ( xact->lifetime < capped_period )
1395  capped_period = xact->lifetime;
1396  xact->age = now + capped_period;
1397  xact->tolive -= capped_period;
1398  }
1399  /* enqueue to the list of newly sent transactions */
1400  xact->node.next = newList;
1401  newList = &xact->node;
1402 #if (DEBUG) & DEBUG_TIMEOUT
