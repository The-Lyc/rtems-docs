32  Thread_Control *the_thread,

34 )
35 {

40  _Thread_State_acquire( the_thread, &state_lock_context );
41  scheduler = _Thread_Scheduler_get_home( the_thread );
42  _Scheduler_Acquire_critical( scheduler, &scheduler_lock_context );
