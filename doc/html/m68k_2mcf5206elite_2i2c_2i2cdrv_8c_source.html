33 /*
34  * Few I2C transfers may be posted simultaneously, but MBUS driver is able
35  * to process it one-by-one. To serialize transfers, function i2c_transfer
36  * put transfer information to the queue and initiate new transfers if MBUS
37  * driver is not busy. When driver is busy, next transfer is dequeued
38  * when current active transfer is finished.
39  */

80 /* i2cdrv_done --
81  * Callback function which is called from MBus low-level driver when
82  * transfer is finished.
83  */
84 static void
85 i2cdrv_done(void * arg_ptr)
86 {
87  rtems_interrupt_level level;
88  i2c_qel *qel = tqueue + tqueue_tail;
89  qel->done(qel->done_arg_ptr);
90  rtems_interrupt_disable(level);
91  tqueue_tail = (tqueue_tail + 1) % tqueue_size;
92  mbus_busy = false;
93  rtems_interrupt_enable(level);
94  i2cdrv_unload();
95 }

97 /* i2cdrv_unload --
98  * If MBUS controller is not busy and transfer waiting in a queue,
99  * initiate processing of the next transfer in queue.
100  */
101 static void
102 i2cdrv_unload(void)
103 {
104  rtems_interrupt_level level;
105  i2c_qel *qel;
106  rtems_status_code sc;
107  rtems_interrupt_disable(level);
108  if (!mbus_busy && (tqueue_head != tqueue_tail))
109  {
110  mbus_busy = true;
111  rtems_interrupt_enable(level);
112  qel = tqueue + tqueue_tail;
113 
114  I2C_SELECT_BUS(qel->bus);
115  if (i2cdrv_bus_clock_div[qel->bus] != i2cdrv_bus_clock_div_current)
116  {
117  i2cdrv_bus_clock_div_current = i2cdrv_bus_clock_div[qel->bus];
118  mcfmbus_select_clock_divider(&mbus, i2cdrv_bus_clock_div_current);
119  }
120  sc = mcfmbus_i2c_transfer(&mbus, qel->nmsg, qel->msg,
121  i2cdrv_done,qel);
122  if (sc != RTEMS_SUCCESSFUL)
123  {
124  int i;
125  for (i = 0; i < qel->nmsg; i++)
126  {
127  qel->msg[i].status = I2C_RESOURCE_NOT_AVAILABLE;
128  }
129  i2cdrv_done(qel);
130  }
131  }
132  else
133  {
134  rtems_interrupt_enable(level);
135  }
136 }

138 /* i2c_transfer --
139  * Initiate multiple-messages transfer over specified I2C bus or
140  * put request into queue if bus or some other resource is busy. (This
141  * is non-blocking function).
142  *
143  * PARAMETERS:
144  * bus - I2C bus number
145  * nmsg - number of messages
146  * msg - pointer to messages array
147  * done - function which is called when transfer is finished
148  * done_arg_ptr - arbitrary argument pointer passed to done funciton
149  *
150  * RETURNS:
151  * RTEMS_SUCCESSFUL if transfer initiated successfully, or error
152  * code if something failed.
153  */
154 rtems_status_code
155 i2c_transfer(i2c_bus_number bus, int nmsg, i2c_message *msg,
156  i2c_transfer_done done, void * done_arg_ptr)
157 {
158  i2c_qel qel;
159  rtems_interrupt_level level;
160 
161  if (bus >= I2C_NUMBER_OF_BUSES)
162  {
163  return RTEMS_INVALID_NUMBER;
164  }
165 
166  if (msg == NULL)
167  {
168  return RTEMS_INVALID_ADDRESS;
169  }
170 
171  qel.bus = bus;
172  qel.msg = msg;
173  qel.nmsg = nmsg;
174  qel.done = done;
175  qel.done_arg_ptr = done_arg_ptr;
176  rtems_interrupt_disable(level);
177  if ((tqueue_head + 1) % tqueue_size == tqueue_tail)
178  {
179  rtems_interrupt_enable(level);
180  return RTEMS_TOO_MANY;
181  }
182  memcpy(tqueue + tqueue_head, &qel, sizeof(qel));
183  tqueue_head = (tqueue_head + 1) % tqueue_size;
184  rtems_interrupt_enable(level);
185  i2cdrv_unload();
186  return RTEMS_SUCCESSFUL;
187 }

218 /* i2c_select_clock_rate --
219  * select I2C bus clock rate for specified bus. Some bus controller do not
220  * allow to select arbitrary clock rate; in this case nearest possible
221  * slower clock rate is selected.
222  *
223  * PARAMETERS:
224  * bus - I2C bus number
225  * bps - data transfer rate for this bytes in bits per second
226  *
227  * RETURNS:
228  * RTEMS_SUCCESSFUL, if operation performed successfully,
229  * RTEMS_INVALID_NUMBER, if wrong bus number is specified,
230  * RTEMS_UNSATISFIED, if bus do not support data transfer rate selection
231  * or specified data transfer rate could not be used.
232  */
233 rtems_status_code
234 i2c_select_clock_rate(i2c_bus_number bus, int bps)
235 {
236  int div;
237  if (bus >= I2C_NUMBER_OF_BUSES)
238  return RTEMS_INVALID_NUMBER;
239 
240  if (bps == 0)
241  return RTEMS_UNSATISFIED;
242 
243  div = BSP_SYSTEM_FREQUENCY / bps;
244  i2cdrv_bus_clock_div[bus] = div;
245  return RTEMS_SUCCESSFUL;
246 }
