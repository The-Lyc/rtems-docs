44  size_t tokenlen
45 )
46 {
47  if ( rtems_filesystem_is_current_directory( token, tokenlen ) ) {
48  return &dir->Node;
49  } else {

54  rtems_chain_node *current = rtems_chain_first( entries );
55  rtems_chain_node *tail = rtems_chain_tail( entries );
56
