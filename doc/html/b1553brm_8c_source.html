412 int b1553brm_device_init(brm_priv *pDev)
413 {
414  struct amba_dev_info *ambadev;
415  struct ambapp_core *pnpinfo;
416  union drvmgr_key_value *value;
417  unsigned int mem;
418  int size;
419 
420  /* Get device information from AMBA PnP information */
421  ambadev = (struct amba_dev_info *)pDev->dev->businfo;
422  if ( ambadev == NULL ) {
423  return -1;
424  }
425  pnpinfo = &ambadev->info;
426  pDev->irqno = pnpinfo->irq;
427  /* Two versions of the BRM core. One where the registers are accessed using the AHB bus
428  * and one where the APB bus is used
429  */
430  if ( pnpinfo->ahb_slv ) {
431  /* Registers accessed over AHB */
432  pDev->regs = (struct brm_reg *)pnpinfo->ahb_slv->start[0];
433  } else {
434  /* Registers accessed over APB */
435  pDev->regs = (struct brm_reg *)pnpinfo->apb_slv->start;
436  }
437  pDev->minor = pDev->dev->minor_drv;
438 #ifdef DEBUG
439  pDev->log_i = 0;
440  memset(pDev->log,0,sizeof(pDev->log));
441 #endif
442 
443 #ifdef DMA_MEM_128K
444  size = 128 * 1024;
445 #else
446  size = 16 * 1024;
447 #endif
448 
449  /* Get memory configuration from bus resources */
450  value = drvmgr_dev_key_get(pDev->dev, "dmaBaseAdr", DRVMGR_KT_POINTER);
451  if (value)
452  mem = (unsigned int)value->ptr;
453 
454  if (value && (mem & 1)) {
455  /* Remote address, address as BRM looks at it. */
456 
457  /* Translate the base address into an address that the the CPU can understand */
458  pDev->memarea_base_remote = mem & ~1;
459  drvmgr_translate_check(pDev->dev, DMAMEM_TO_CPU,
460  (void *)pDev->memarea_base_remote,
461  (void **)&pDev->memarea_base,
462  size);
463  } else {
464  if (!value) {
465  /* Use dynamically allocated memory + 128k for
466  * alignment
467  */
468  mem = (unsigned int)grlib_malloc(size + 128 * 1024);
469  if (!mem){
470  printk("BRM: Failed to allocate HW memory\n\r");
471  return -1;
472  }
473  /* align memory to 128k boundary */
474  pDev->memarea_base = (mem + 0x1ffff) & ~0x1ffff;
475  } else {
476  pDev->memarea_base = mem;
477  }
478 
479  /* Translate the base address into an address that the BRM core can understand */
480  drvmgr_translate_check(pDev->dev, CPUMEM_TO_DMA,
481  (void *)pDev->memarea_base,
482  (void **)&pDev->memarea_base_remote,
483  size);
484  }
485 
486  /* clear the used memory */
487  memset((char *)pDev->memarea_base, 0, size);
488 
489  /* Set base address of all descriptors */
490  pDev->desc = (struct desc_table *) pDev->memarea_base;
491  pDev->mem = (volatile unsigned short *) pDev->memarea_base;
492  pDev->irq_log = (struct irq_log_list *)(pDev->memarea_base + (0xFFE0<<1)); /* last 64byte */
493 
494  pDev->bm_event = NULL;
495  pDev->rt_event = NULL;
496 
497  pDev->cfg_clksel = 0;
498  pDev->cfg_clkdiv = 0;
499  pDev->cfg_freq = BRM_FREQ_24MHZ;
500 
501  value = drvmgr_dev_key_get(pDev->dev, "clkSel", DRVMGR_KT_INT);
502  if ( value ) {
503  pDev->cfg_clksel = value->i & CLKSEL_MASK;
504  }
505 
506  value = drvmgr_dev_key_get(pDev->dev, "clkDiv", DRVMGR_KT_INT);
507  if ( value ) {
508  pDev->cfg_clkdiv = value->i & CLKDIV_MASK;
509  }
510 
511  value = drvmgr_dev_key_get(pDev->dev, "coreFreq", DRVMGR_KT_INT);
512  if ( value ) {
513  pDev->cfg_freq = value->i & BRM_FREQ_MASK;
514  }
515

517  pDev->regs->w_ctrl = (pDev->cfg_clksel<<9) | (pDev->cfg_clkdiv<<5);
518  /* Reset BRM core */
519  pDev->regs->w_ctrl = 1<<10 | READ_REG(&pDev->regs->w_ctrl);
520 
521  /* RX Semaphore created with count = 0 */
522  if ( rtems_semaphore_create(rtems_build_name('B', 'M', 'R', '0' + pDev->minor),
523  0,
524  RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|RTEMS_NO_PRIORITY_CEILING,
525  0,
526  &pDev->rx_sem) != RTEMS_SUCCESSFUL ) {
527  printk("BRM: Failed to create rx semaphore\n");
528  return RTEMS_INTERNAL_ERROR;
529  }
530 
531  /* TX Semaphore created with count = 1 */
532  if ( rtems_semaphore_create(rtems_build_name('B', 'M', 'T', '0' + pDev->minor),
533  1,
534  RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|RTEMS_NO_PRIORITY_CEILING,
535  0,
536  &pDev->tx_sem) != RTEMS_SUCCESSFUL ){
537  printk("BRM: Failed to create tx semaphore\n");
538  return RTEMS_INTERNAL_ERROR;
539  }
540 
541  /* Device Semaphore created with count = 1 */
542  if ( rtems_semaphore_create(rtems_build_name('B', 'M', 'D', '0' + pDev->minor),
543  1,
544  RTEMS_FIFO|RTEMS_SIMPLE_BINARY_SEMAPHORE|RTEMS_NO_INHERIT_PRIORITY|RTEMS_LOCAL|RTEMS_NO_PRIORITY_CEILING,
545  0,
546  &pDev->dev_sem) != RTEMS_SUCCESSFUL ){
547  printk("BRM: Failed to create device semaphore\n");
548  return RTEMS_INTERNAL_ERROR;
549  }
550 
551  /* Default to RT-mode */
552  rt_init(pDev);
553 
554  return 0;

655  * Each circular buffer has room for 8 messages with up to 34 (32 data + miw + time) words (16b) in each.
656  * The buffers must separated by 34 words.
657  */
658 
659 
660  /* RX Sub-address 0 - 31 */
661  for (i = 0; i < 32; i++) {
662  brm->rtmem->rxsubs[i].ctrl = 0x00E0; /* Interrupt: INTX, IWA, and IBRD */
663  brm->rtmem->rxsubs[i].top = OFS(brm->rtmem->rxsuba_msgs[i]); /* Top address */
664  brm->rtmem->rxsubs[i].cur = OFS(brm->rtmem->rxsuba_msgs[i]); /* Current address */
665  brm->rtmem->rxsubs[i].bot = OFS(brm->rtmem->rxsuba_msgs[i+1]) - sizeof(struct msg)/2; /* Bottom address */
666  brm->last_read[i] = OFS(brm->rtmem->rxsuba_msgs[i]);
667  }
668  /* TX Sub-address 0 - 31 */
669  for (i = 0; i < 32; i++) {
670  brm->rtmem->txsubs[i].ctrl = 0x0060; /* Interrupt: IWA and IBRD */
671  brm->rtmem->txsubs[i].top = OFS(brm->rtmem->txsuba_msgs[i]); /* Top address */
672  brm->rtmem->txsubs[i].cur = OFS(brm->rtmem->txsuba_msgs[i]); /* Current address */
673  brm->rtmem->txsubs[i].bot = OFS(brm->rtmem->txsuba_msgs[i+1]) - sizeof(struct msg)/2; /* Bottom address */
674  brm->last_read[i+32] = OFS(brm->rtmem->txsuba_msgs[i]);
675  brm->written[i] = OFS(brm->rtmem->txsuba_msgs[i]);
676  }
677  /* RX mode code 0 - 31 */
678  for (i = 0; i < 32; i++) {
679  brm->rtmem->rxmodes[i].ctrl = 0x00E0; /* Interrupt: INTX, IWA, and IBRD */
680  brm->rtmem->rxmodes[i].top = OFS(brm->rtmem->rxmode_msgs[i]); /* Top address */
681  brm->rtmem->rxmodes[i].cur = OFS(brm->rtmem->rxmode_msgs[i]); /* Current address */
682  brm->rtmem->rxmodes[i].bot = OFS(brm->rtmem->rxmode_msgs[i+1]) - sizeof(struct msg)/2; /* Bottom address */
683  brm->last_read[i+64] = OFS(brm->rtmem->rxmode_msgs[i]);
684  }
685  /* TX mode code 0 - 31 */
686  for (i = 0; i < 32; i++) {
687  brm->rtmem->txmodes[i].ctrl = 0x0060; /* Interrupt: IWA and IBRD */
688  brm->rtmem->txmodes[i].top = OFS(brm->rtmem->txmode_msgs[i]); /* Top address */
689  brm->rtmem->txmodes[i].cur = OFS(brm->rtmem->txmode_msgs[i]); /* Current address */
690  brm->rtmem->txmodes[i].bot = OFS(brm->rtmem->txmode_msgs[i+1]) - sizeof(struct msg)/2; /* Bottom address */
691  brm->last_read[i+96] = OFS(brm->rtmem->txmode_msgs[i]);
692  }
693 
694 #ifdef DEBUG
695  printk("b1553BRM DMA_AREA: 0x%x\n", (unsigned int)brm->rtmem);
696  printk("LOG: 0x%x\n", &brm->log[0]);
697  printk("LOG_I: 0x%x\n", &brm->log_i);
698 #endif
699 
700  brm->mode = BRM_MODE_RT;
701 
702  return RTEMS_SUCCESSFUL;
703 }

919 static rtems_device_driver brm_write(rtems_device_major_number major, rtems_device_minor_number minor, void *arg)
920 {
921  rtems_libio_rw_args_t *rw_args;
922  struct rt_msg *source;
923  unsigned int count=0, current, next, descriptor, wc, suba;
924  brm_priv *brm;
925  struct drvmgr_dev *dev;
926 
927  if ( drvmgr_get_dev(&b1553brm_drv_info.general, minor, &dev) ) {
928  return RTEMS_UNSATISFIED;
929  }
930  brm = (brm_priv *)dev->priv;
931 
932  if ( ! (brm->mode & BRM_MODE_RT) ){
933  return RTEMS_INVALID_NAME;
934  }
935 
936  rw_args = (rtems_libio_rw_args_t *) arg;
937  source = (struct rt_msg *) rw_args->buffer;
938 
939  FUNCDBG("brm_write [%i,%i]: buf: 0x%x len: %i\n",major, minor, (unsigned int)rw_args->buffer,rw_args->count);
940 
941  do {
942 
943  descriptor = source[count].desc & 0x7F;
944  suba = descriptor-32;
945  wc = source[count].miw >> 11;
946  wc = wc ? wc : 32;
947 
948  /* Only subaddress transmission is allowed with write */
949  if (descriptor < 32 || descriptor >= 64)
950  return RTEMS_INVALID_NAME;
951 
952  current = brm->desc[descriptor].cur;
953  next = brm->written[suba] + 2 + wc;
954 
955  if (brm->written[suba] < current) {
956 
957  if (next > current) {
958 
959  /* No room in transmission buffer */
960  if (brm->tx_blocking && count == 0) {
961  rtems_semaphore_obtain(brm->tx_sem, RTEMS_WAIT, RTEMS_NO_TIMEOUT);
962  } else if ( count > 0 ) {
963  /* return the number of messages sent so far */
964  break;
965  } else {
966  /* Translates to posix EBUSY */
967  return RTEMS_RESOURCE_IN_USE;
968  }
969  }
970  }
971 
972  memcpy((void *)&brm->mem[brm->written[suba]], &source[count], (2+wc)*2);
973 
974  count++;
975 
976  if (next >= brm->desc[descriptor].bot) {
977  next = brm->desc[descriptor].top;
978  }
979  brm->written[suba] = next;
980 
981  } while (count < rw_args->count);
982 
983  rw_args->bytes_moved = count;
984 
985  if (count >= 0) {
986  return RTEMS_SUCCESSFUL;
987  }
988  return RTEMS_UNSATISFIED;
989 }

991 static rtems_device_driver brm_control(rtems_device_major_number major, rtems_device_minor_number minor, void *arg)
992 {
993 
994  unsigned int i=0;
995  unsigned short ctrl, oper, cw1, cw2;
996  rtems_libio_ioctl_args_t *ioarg = (rtems_libio_ioctl_args_t *) arg;
997  unsigned int *data = ioarg->buffer;
998  struct bc_msg *cmd_list = (struct bc_msg *) ioarg->buffer;
999  brm_priv *brm;
1000  struct drvmgr_dev *dev;
1001  rtems_device_driver ret;
1002  int len, msglen;
1003 
1004  FUNCDBG("brm_control[%d]: [%i,%i]\n", minor, major, minor);
1005 
1006  if ( drvmgr_get_dev(&b1553brm_drv_info.general, minor, &dev) ) {
1007  return RTEMS_UNSATISFIED;
1008  }
1009  brm = (brm_priv *)dev->priv;
1010 
1011  if (!ioarg) {
1012  DBG("brm_control: invalid argument\n");
1013  return RTEMS_INVALID_NAME;
1014  }
1015 
1016  ioarg->ioctl_return = 0;
1017  switch (ioarg->command) {
1018 
1019  case BRM_SET_MODE:
1020  if ( data[0] > 2 )
1021  return RTEMS_INVALID_NAME;
1022  stop_operation(brm);
1023  if (data[0] == 0) {
1024  ret = bc_init(brm);
1025  } else if (data[0] == 1) {
1026  ret = rt_init(brm);
1027  } else if (data[0] == 2) {
1028  ret = bm_init(brm);
1029  } else {
1030  ret = RTEMS_INVALID_NAME;
1031  }
1032  if ( ret != RTEMS_SUCCESSFUL)
1033  return ret;
1034 
1035  if ( brm->mode & (BRM_MODE_RT | BRM_MODE_BM ) )
1036  start_operation(brm);
1037  break;
1038 
1039  case BRM_SET_BUS:
1040  stop_operation(brm);
1041  ctrl = READ_REG(&brm->regs->ctrl);
1042  ctrl &= 0xE7FF; /* Clear bit 12-11 ... */
1043  ctrl |= (data[0]&0x3)<<11; /* ... OR in new bus status */
1044  brm->regs->ctrl = ctrl;
1045  start_operation(brm);
1046  break;
1047 
1048  case BRM_SET_MSGTO:
1049  stop_operation(brm);
1050  ctrl = READ_REG(&brm->regs->ctrl);
1051  ctrl &= 0xFDFF; /* Clear bit 9 ... */
1052  ctrl |= (data[0]&1)<<9; /* ... OR in new MSGTO */
1053  brm->regs->ctrl = ctrl;
1054  start_operation(brm);
1055  break;
1056 
1057  case BRM_SET_RT_ADDR:
1058  stop_operation(brm);
1059  oper = READ_REG(&brm->regs->oper);
1060  oper &= 0x03FF; /* Clear bit 15-10 ... */
1061  oper |= (data[0]&0x1f)<<11; /* ... OR in new address */
1062  oper |= odd_parity(data[0]&0x1f)<<10; /* ... OR in parity */
1063  brm->regs->oper = oper;
1064  start_operation(brm);
1065  break;
1066 
1067  case BRM_SET_STD:
1068  stop_operation(brm);
1069  ctrl = READ_REG(&brm->regs->ctrl);
1070  ctrl &= 0xFF7F; /* Clear bit 7 ... */
1071  ctrl |= (data[0]&1)<<7; /* ... OR in new ABSTD (1=A) */
1072  brm->regs->ctrl = ctrl;
1073  start_operation(brm);
1074  break;
1075 
1076  case BRM_SET_BCE:
1077  stop_operation(brm);
1078  ctrl = READ_REG(&brm->regs->ctrl);
1079  ctrl &= 0xFFEF; /* Clear bit 4 ... */
1080  ctrl |= (data[0]&1)<<4; /* ... OR in new BCE */
1081  brm->regs->ctrl = ctrl;
1082  start_operation(brm);
1083  break;
1084 
1085  case BRM_TX_BLOCK:
1086  brm->tx_blocking = data[0];
1087  break;
1088 
1089  case BRM_RX_BLOCK:
1090  brm->rx_blocking = data[0];
1091  break;
1092 
1093  case BRM_DO_LIST:
1094  if ( brm->mode != BRM_MODE_BC ){
1095  return RTEMS_INVALID_NAME;
1096  }
1097 
1098  /* Check if we are bus controller */
1099  if ( ((READ_REG(&brm->regs->oper)>>8) & 3) != 0 ) {
1100  return RTEMS_INVALID_NAME;
1101  }
1102 
1103  /* Already processing list? */
1104  if (is_executing(brm)) {
1105  return RTEMS_RESOURCE_IN_USE;
1106  }
1107 
1108  /* clear any earlier releases */
1109  rtems_semaphore_obtain(brm->tx_sem, RTEMS_NO_WAIT, RTEMS_NO_TIMEOUT);
1110 
1111  brm->bc_list_fail = 0;
1112  brm->cur_list = cmd_list;
1113  brm->regs->dpoint = 0;
1114 
1115  i = 0;
1116  while ( (cmd_list[i].ctrl & BC_EOL) == 0) {
1117 
1118  ctrl = (4<<12) | (((cmd_list[i].ctrl&BC_BUSA)==BC_BUSA)<<9) | (((cmd_list[i].ctrl&BC_RTRT)==BC_RTRT)<<8);
1119 
1120  if (cmd_list[i].ctrl&BC_RTRT) {
1121  cw1 = (cmd_list[i].rtaddr[0]<<11) | (0<<10) | (cmd_list[i].subaddr[0]<<5) | (cmd_list[i].wc & 0x1f); /* receive cw */
1122  cw2 = (cmd_list[i].rtaddr[1]<<11) | (1<<10) | (cmd_list[i].subaddr[1]<<5) | (cmd_list[i].wc & 0x1f); /* transmit cw */
1123  } else {
1124  cw1 = (cmd_list[i].rtaddr[0]<<11) | (((cmd_list[i].ctrl&BC_TR)==BC_TR)<<10) | (cmd_list[i].subaddr[0]<<5) | (cmd_list[i].wc&0x1f);
1125  cw2 = 0;
1126  }
1127 
1128  /* Set up command block */
1129  brm->bcmem->descs[i].ctrl = ctrl;
1130  brm->bcmem->descs[i].cw1 = cw1;
1131  brm->bcmem->descs[i].cw2 = cw2;
1132  /* data pointer:
1133  * (&brm->bcmem->msg_data[i].data[0] & 0x1ffff) / 2
1134  */
1135  brm->bcmem->descs[i].dptr = 1024+i*32; /* data pointer */
1136  brm->bcmem->descs[i].tsw[0] = 0;
1137  brm->bcmem->descs[i].tsw[1] = 0;
1138  brm->bcmem->descs[i].ba = 0;
1139  brm->bcmem->descs[i].timer = 0;
1140 
1141  msglen = cmd_list[i].wc;
1142  if ( msglen == 0 )
1143  msglen = 32;
1144  memcpy((void *)&brm->bcmem->msg_data[i].data[0], &cmd_list[i].data[0], msglen*2);
1145 
1146  i++;
1147  }
1148 
1149  brm->bcmem->descs[i].ctrl = 0; /* end of list */
1150 
1151  start_operation(brm);
1152  break;
1153 
1154  case BRM_LIST_DONE:
1155 
1156  if ( brm->mode != BRM_MODE_BC ){
1157  return RTEMS_INVALID_NAME;
1158  }
1159 
1160  /* Check if we are bus controller */
1161  if ( ((READ_REG(&brm->regs->oper)>>8) & 3) != 0 ) {
1162  return RTEMS_INVALID_NAME;
1163  }
1164 
1165  if (is_executing(brm)) {
1166 
1167  data[0] = 0;
1168  if (brm->tx_blocking) {
1169  rtems_semaphore_obtain(brm->tx_sem, RTEMS_WAIT, RTEMS_NO_TIMEOUT);
1170  data[0] = 1;
1171  if ( brm->bc_list_fail ){
1172  return RTEMS_INVALID_NAME;
1173  }
1174  } else {
1175  return RTEMS_RESOURCE_IN_USE;
1176  }
1177  } else {
1178  data[0] = 1; /* done */
1179  }
1180 
1181  /* copy finished list results back into bc_msg array */
1182  i = 0;
1183  while ( (brm->cur_list[i].ctrl & BC_EOL) == 0) {
1184  if (READ_DMA(&brm->bcmem->descs[i].ctrl) & 1) {
1185  brm->cur_list[i].ctrl |= 0x8000; /* Set BAME */
1186  }
1187  if (brm->cur_list[i].ctrl & BC_TR) {
1188  /* RT Transmit command, copy received data */
1189  len = brm->cur_list[i].wc;
1190  if ( len == 0 )
1191  len = 32;
1192  while ( len-- > 0) {
1193  brm->cur_list[i].data[len] = READ_DMA(&brm->bcmem->msg_data[i].data[len]);
1194  }
