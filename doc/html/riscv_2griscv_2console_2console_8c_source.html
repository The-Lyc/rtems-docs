60 /* AMBA PP find routine. Extract AMBA PnP information into data structure. */
61 static int find_matching_apbuart(struct ambapp_dev *dev, int index, void *arg)
62 {
63  struct ambapp_apb_info *apb = (struct ambapp_apb_info *)dev->devinfo;
64 
65  /* Extract needed information of one APBUART */
66  apbuarts[uarts].regs = (struct apbuart_regs *)apb->start;
67  apbuarts[uarts].irq = apb->irq;
68  /* Get APBUART core frequency, it is assumed that it is the same
69  * as Bus frequency where the UART is situated
70  */
71  apbuarts[uarts].freq_hz = ambapp_freq_get(&ambapp_plb, dev);
72  uarts++;
73 
74  if (uarts >= BSP_NUMBER_OF_TERMIOS_PORTS)
75  return 1; /* Satisfied number of UARTs, stop search */
76  else
77  return 0; /* Continue searching for more UARTs */
78 }

99 #else
100  &apbuart_handler_polled;
101 #endif
102  rtems_status_code status;
103  int i;
104  char console_name[16];
105 
106  rtems_termios_initialize();
107 
108  /* Find UARTs */
109  grlib_console_scan_uarts();
110 
111  /* Update syscon_uart_index to index used as /dev/console
112  * Let user select System console by setting syscon_uart_index. If the
113  * BSP is to provide the default UART (syscon_uart_index==0):
114  * non-MP: APBUART[0] is system console
115  * MP: LEON CPU index select UART
116  */
117  if (syscon_uart_index == 0) {
118 #if defined(RTEMS_MULTIPROCESSING)
119  syscon_uart_index = GRLIB_Cpu_Index;
120 #else
121  syscon_uart_index = 0;
122 #endif
123  } else {
124  syscon_uart_index = syscon_uart_index - 1; /* User selected sys-console */
125  }
126 
127  /* Register Device Names
128  *
129  * 0 /dev/console - APBUART[USER-SELECTED, DEFAULT=APBUART[0]]
130  * 1 /dev/console_a - APBUART[0] (by default not present because is console)
131  * 2 /dev/console_b - APBUART[1]
132  * ...
133  *
134  * On a MP system one should not open UARTs that other OS instances use.
135  */
136  if (syscon_uart_index < uarts) {
137  status = rtems_termios_device_install(
138  CONSOLE_DEVICE_NAME,
139  handler,
140  NULL,
141  grlib_console_get_context(syscon_uart_index)
142  );
143  if (status != RTEMS_SUCCESSFUL)
144  bsp_fatal(LEON3_FATAL_CONSOLE_REGISTER_DEV);
145  }
146  strcpy(console_name,"/dev/console_a");
147  for (i = 0; i < uarts; i++) {
148  if (i == syscon_uart_index)
149  continue; /* skip UART that is registered as /dev/console */
150  console_name[13] = 'a' + i;
151  rtems_termios_device_install(
152  console_name,
153  handler,
154  NULL,
155  grlib_console_get_context(i)
156  );
157  }
158 
159  return RTEMS_SUCCESSFUL;
160 }
