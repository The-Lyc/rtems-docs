160 static int genirq_set_active(
161  struct genirq_priv *priv,
162  int irq,
163  genirq_handler isr,
164  void *arg,
165  int action)
166 {
167  struct genirq_irq_entry *irqentry;
168  struct genirq_handler_entry *isrentry, *e = NULL;
169  int enabled;
170 
171  if ( genirq_check(priv, irq) )
172  return -1;
173 
174  /* Find isr[arg] in ISR list */
175  irqentry = &priv->genirq_table[irq];
176  enabled = 0;
177 
178  isrentry = irqentry->head;
179  while ( isrentry ) {
180  if ( (isrentry->arg == arg) && (isrentry->isr == isr) ) {
181  /* Found ISR */
182  if ( isrentry->enabled == action ) {
183  /* The ISR is already enabled or disabled
184  * depending on request, neccessary actions
185  * were taken last time the same action was
186  * requested.
187  */
188  return 1;
189  }
190  e = isrentry;
191  } else {
192  enabled += isrentry->enabled;
193  }
194  isrentry = isrentry->next;
195  }
196 
197  if ( !e )
198  return -1;
199 
200  e->enabled = action;
201 
202  return enabled;
203 }
