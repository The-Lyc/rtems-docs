103 
104  /* The thermostat */
105  if ( rtems_libi2c_register_drv(
106  BSP_I2C_DS1621_NAME,
107  i2c_ds1621_driver_descriptor,
108  busno,

111  return -1;

113

119  */
120 
121  if ( mknod(

128  }
