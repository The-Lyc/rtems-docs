56 /*
57  * __Restart_Fifo_Full_Timer
58  *
59  * When a character is received, sets the TIMER to raise an interrupt at
60  * TIMEOUT. It's necessary to prevent from not getting n-1 characters
61  * (with n the Uart Fifo size).
62  */
63 static void __Restart_Fifo_Full_Timer (void)
64 {
65  TSR1 |= m340_TO;
66  TCR1 &= ~m340_CPE;
67  WPREL11 = preload;
68  TCR1 |= m340_CPE;
69 }

104 /*
105  * Fifo_Full_benchmark_timer_initialize
106  *
107  * initialize Timer 1 for FIFO full mode
108  */
109 void Fifo_Full_benchmark_timer_initialize (void)
110 {
111  float max_baud_rate;
112  int prescaler_output_tap = -1;
113  int nb_of_clock_ticks = 0;
114  rtems_isr_entry old_handler;
115 
116  /*
117  * USE TIMER 1 for UART FIFO FULL mode
118  */
119  if ( Fifo_Full_on_A || Fifo_Full_on_B ) {
120  /* Disable the timer */
121  TCR1 &= ~m340_SWR;
122 
123  /* Reset the interrupts */
124  TSR1 &= ~(m340_TO | m340_TG | m340_TC);
125 
126  /* Init the stop bit for normal operation, ignore FREEZE, user privileges,
127  * set interrupt arbitration.
128  */
129  TMCR1 = TIMER1_INTERRUPT_ARBITRATION;
130 
131  /* interrupt priority level and interrupt vector */
132  TIR1 = TIMER1_VECTOR | (TIMER1_IRQ_LEVEL << 8);
133 
134  /* compute prescaler */
135  if ( Fifo_Full_on_A && Fifo_Full_on_B) {
136  max_baud_rate = max(
137  m340_uart_config[UART_CHANNEL_A].rx_baudrate,
138  m340_uart_config[UART_CHANNEL_B].rx_baudrate
139  );
140  } else if ( Fifo_Full_on_A ) {
141  max_baud_rate = m340_uart_config[UART_CHANNEL_A].rx_baudrate;
142  } else
143  max_baud_rate = m340_uart_config[UART_CHANNEL_B].rx_baudrate;
144 
145  /* find out config */
146  nb_of_clock_ticks = (10/max_baud_rate)*(CLOCK_SPEED*1000000)*1.2;
147  if (nb_of_clock_ticks < 0xFFFF) {
148  preload = nb_of_clock_ticks;
149  prescaler_output_tap = -1;
150  } else if (nb_of_clock_ticks/2 < 0xFFFF) {
151  preload = nb_of_clock_ticks/2;
152  prescaler_output_tap = m340_Divide_by_2;
153  } else if (nb_of_clock_ticks/4 < 0xFFFF) {
154  preload = nb_of_clock_ticks/4;
155  prescaler_output_tap = m340_Divide_by_4;
156  } else if (nb_of_clock_ticks/8 < 0xFFFF) {
157  preload = nb_of_clock_ticks/8;
158  prescaler_output_tap = m340_Divide_by_16;
159  } else if (nb_of_clock_ticks/16 < 0xFFFF) {
160  preload = nb_of_clock_ticks/16;
161  prescaler_output_tap = m340_Divide_by_16;
162  } else if (nb_of_clock_ticks/32 < 0xFFFF) {
163  preload = nb_of_clock_ticks/32;
164  prescaler_output_tap = m340_Divide_by_32;
165  } else if (nb_of_clock_ticks/64 < 0xFFFF) {
166  preload = nb_of_clock_ticks/64;
167  prescaler_output_tap = m340_Divide_by_64;
168  } else if (nb_of_clock_ticks/128 < 0xFFFF) {
169  preload = nb_of_clock_ticks/128;
170  prescaler_output_tap = m340_Divide_by_128;
171  } else if (nb_of_clock_ticks/256 < 0xFFFF) {
172  preload = nb_of_clock_ticks/256;
173  prescaler_output_tap = m340_Divide_by_256;
174  }
175 
176  /* Input Capture/Output Compare (ICOC) */
177  TCR1 = m340_SWR | m340_TO_Enabled | m340_ICOC;
178  if (prescaler_output_tap!=-1) TCR1 |= prescaler_output_tap | m340_PSE;
179 
180  /* install interrupt vector */
181  rtems_interrupt_catch(InterruptHandler, TIMER1_VECTOR, &old_handler);
182 
183  } /* fifo full mode on a uart */
184 
185  /* install routines */
186  if ( Fifo_Full_on_A ) {
187  Restart_Check_A_Timer = __Restart_Check_Timer;
188  Restart_Fifo_Full_A_Timer = __Restart_Fifo_Full_Timer;
189  } else {
190  Restart_Check_A_Timer = __do_nothing;
191  Restart_Fifo_Full_A_Timer = __do_nothing;
192  }
193 
194  if ( Fifo_Full_on_B ) {
195  Restart_Check_B_Timer = __Restart_Check_Timer;
196  Restart_Fifo_Full_B_Timer = __Restart_Fifo_Full_Timer;
197  } else {
198  Restart_Check_B_Timer = __do_nothing;
199  Restart_Fifo_Full_B_Timer = __do_nothing;
200  }
201 
202  /* start checking timer */
203  Restart_Check_A_Timer();
204  Restart_Check_B_Timer();
205 }
