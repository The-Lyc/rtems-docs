47 __STATIC_INLINE uint32_t _ISI_GetCFG2_Workaround(void)
48 {
49  uint32_t wrongfield;
50  wrongfield = ISI->ISI_CFG2 >> (ISI_CFG2_YCC_SWAP_Pos - 1);
51  return (ISI->ISI_CFG2 & 0x07FFFFFF) | (wrongfield << ISI_CFG2_YCC_SWAP_Pos);
52 }

156 uint8_t ISI_BytesForOnePixel(uint8_t bmpRgb)
157 {
158  uint8_t nbByte_Pixel;
159 
160  if (bmpRgb == RGB) {
161  if ((_ISI_GetCFG2_Workaround() & ISI_CFG2_RGB_MODE) == ISI_CFG2_RGB_MODE) {
162  // RGB: 5:6:5 16bits/pixels
163  nbByte_Pixel = 2;
164  } else {
165  // RGB: 8:8:8 24bits/pixels
166  nbByte_Pixel = 3;
167  }
168  } else {
169  // YUV: 2 pixels for 4 bytes
170  nbByte_Pixel = 2;
171  }
172 
173  return nbByte_Pixel;
174 }

179 void ISI_Reset(void)
180 {
181  uint32_t timeout = 0;
182 
183  // Resets the image sensor interface.
184  // Finish capturing the current frame and then shut down the module.
185  ISI->ISI_CR = ISI_CR_ISI_SRST | ISI_CR_ISI_DIS;
186 
187  // wait Software reset has completed successfully.
188  while ((!(ISI->ISI_SR & ISI_SR_SRST))
189  && (timeout < 0x5000))
190  timeout++;
191 
192  if (timeout == 0x5000)
193  TRACE_ERROR("ISI-Reset timeout\n\r");
194 }

213 void ISI_SetSensorSize(uint32_t hSize, uint32_t vSize)
214 {
215  uint32_t val;
216  val = _ISI_GetCFG2_Workaround();
217  val &= (~ISI_CFG2_IM_VSIZE_Msk);
218  val &= (~ISI_CFG2_IM_HSIZE_Msk);
219  // IM_VSIZE: Vertical size of the Image sensor [0..2047]
220  // Vertical size = IM_VSIZE + 1
221  // IM_HSIZE: Horizontal size of the Image sensor [0..2047]
222  // Horizontal size = IM_HSIZE + 1
223  ISI->ISI_CFG2 &= (~ISI_CFG2_IM_VSIZE_Msk);
224  ISI->ISI_CFG2 &= (~ISI_CFG2_IM_HSIZE_Msk);
225  ISI->ISI_CFG2 = val | ISI_CFG2_IM_VSIZE(vSize - 1) | ISI_CFG2_IM_HSIZE(
226  hSize - 1);
227 }

307 void ISI_calcScalerFactor(void)
308 {
309  uint32_t hLcdSize, hSensorSize;
310  uint32_t hRatio;
311  hLcdSize = ((ISI->ISI_PSIZE & ISI_PSIZE_PREV_HSIZE_Msk) >>
312  ISI_PSIZE_PREV_HSIZE_Pos) + 1;
313  hSensorSize = ((_ISI_GetCFG2_Workaround() & ISI_CFG2_IM_HSIZE_Msk)
314  >> ISI_CFG2_IM_HSIZE_Pos) + 1;
315  hRatio = 1600 * hSensorSize / hLcdSize;
316  ISI->ISI_PDECF = (hRatio / 100);
317 }
