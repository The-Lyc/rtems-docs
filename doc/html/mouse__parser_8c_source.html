95  * Open up the mouse device.
96  * Returns the fd if successful, or negative if unsuccessful.
97  */
98 int mouse_parser_initialize(const char *type)
99 {
100  /* set button bits and parse procedure*/
101  if (!strcmp(type, "pc") || !strcmp(type, "logi")) {
102  /* pc or logitech mouse*/
103  left = PC_LEFT_BUTTON;
104  middle = PC_MIDDLE_BUTTON;
105  right = PC_RIGHT_BUTTON;
106  parse = ParsePC;
107  } else if (strcmp(type, "ms") == 0) {
108  /* microsoft mouse*/
109  left = MS_LEFT_BUTTON;
110  right = MS_RIGHT_BUTTON;
111  middle = 0;
112  parse = ParseMS;
113  } else if (strcmp(type, "ps2") == 0) {
114  /* PS/2 mouse*/
115  left = PS2_LEFT_BUTTON;
116  right = PS2_RIGHT_BUTTON;
117  middle = 0;
118  parse = ParsePS2;
119  } else
120  return -1;
121 
122  printk("Device: /dev/mouse -- mouse type is: %s\n", type );
123 
124  /* initialize data*/
125  availbuttons = left | middle | right;
126  state = IDLE;
127  buttons = 0;
128  xd = 0;
129  yd = 0;
130  return 0;
131 }

133 /*
134  * Attempt to read bytes from the mouse and interpret them.
135  * Returns -1 on error, 0 if either no bytes were read or not enough
136  * was read for a complete state, or 1 if the new state was read.
137  * When a new state is read, the current buttons and x and y deltas
138  * are returned. This routine does not block.
139  */
140 static int MOU_Data( int ch, COORD *dx, COORD *dy, COORD *dz, BUTTON *bptr)
141 {
142  int b;
143 
144  if ( !parse ) {
145  printk( "Mouse parser is not initialized!\n" );
146  return -1;
147  }
148 
149  /*
150  * Loop over all the bytes read in the buffer, parsing them.
151  * When a complete state has been read, return the results,
152  * leaving further bytes in the buffer for later calls.
153  */
154  if ( (*parse)( ch ) ) {
155  *dx = xd;
156  *dy = yd;
157  *dz = 0;
158  b = 0;
159  if (buttons & left)
160  b |= LBUTTON;
161  if (buttons & right)
162  b |= RBUTTON;
163  if (buttons & middle)
164  b |= MBUTTON;
165  *bptr = b;
166  return 1;
167  }
168  return 0;
169 }

171 /*
172  * Input routine for PC mouse.
173  * Returns nonzero when a new mouse state has been completed.
174  */
175 static int ParsePC(int byte)
176 {
177  int sign; /* sign of movement */
178 
179  switch (state) {
180  case IDLE:
181  if ((byte & TOP_FIVE_BITS) == TOP_BIT) {
182  buttons = ~byte & BOTTOM_THREE_BITS;
183  state = XSET;
184  }
185  break;
186 
187  case XSET:
188  sign = 1;
189  if (byte > 127) {
190  byte = 256 - byte;
191  sign = -1;
192  }
193  xd = byte * sign;
194  state = YSET;
195  break;
196 
197  case YSET:
198  sign = 1;
199  if (byte > 127) {
200  byte = 256 - byte;
201  sign = -1;
202  }
203  yd = -byte * sign;
204  state = XADD;
205  break;
206 
207  case XADD:
208  sign = 1;
209  if (byte > 127) {
210  byte = 256 - byte;
211  sign = -1;
212  }
213  xd += byte * sign;
214  state = YADD;
215  break;
216 
217  case YADD:
218  sign = 1;
219  if (byte > 127) {
220  byte = 256 - byte;
221  sign = -1;
222  }
223  yd -= byte * sign;
224  state = IDLE;
225  return 1;
226  }
227  return 0;
228 }

230 /*
231  * Input routine for Microsoft mouse.
232  * Returns nonzero when a new mouse state has been completed.
233  */
234 static int ParseMS(int byte)
235 {
236  switch (state) {
237  case IDLE:
238  if (byte & SIXTH_BIT) {
239  buttons = (byte >> 4) & BOTTOM_TWO_BITS;
240  yd = ((byte & THIRD_FOURTH_BITS) << 4);
241  xd = ((byte & BOTTOM_TWO_BITS) << 6);
242  state = XADD;
243  }
244  break;
245 
246  case XADD:
247  xd |= (byte & BOTTOM_SIX_BITS);
248  state = YADD;
249  break;
250 
251  case YADD:
252  yd |= (byte & BOTTOM_SIX_BITS);
253  state = IDLE;
254  if (xd > 127)
255  xd -= 256;
256  if (yd > 127)
257  yd -= 256;
258  return 1;
259  }
260  return 0;
261 }

264  * Input routine for PS/2 mouse.
265  * Returns nonzero when a new mouse state has been completed.
266  */
267 static int ParsePS2(int byte)
268 {
269  switch (state) {
270  case IDLE:
271  if (byte & PS2_CTRL_BYTE) {
272  buttons = byte &
273  (PS2_LEFT_BUTTON|PS2_RIGHT_BUTTON);
274  state = XSET;
275  }
276  break;
277 
278  case XSET:
279  if(byte > 127)
280  byte -= 256;
281  xd = byte;
282  state = YSET;
283  break;
284 
285  case YSET:
286  if(byte > 127)
287  byte -= 256;
288  yd = -byte;
289  state = IDLE;
290  return 1;
291  }
292  return 0;
