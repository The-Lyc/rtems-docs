315 static int jffs2_block_check_erase(struct jffs2_sb_info *c, struct jffs2_eraseblock *jeb, uint32_t *bad_offset)
316 {
317  void *ebuf;
318  uint32_t ofs;
319  size_t retlen;
320  int ret;
321  unsigned long *wordebuf;
322 
323  ret = mtd_point(c->mtd, jeb->offset, c->sector_size, &retlen,
324  &ebuf, NULL);
325  if (ret != -EOPNOTSUPP) {
326  if (ret) {
327  jffs2_dbg(1, "MTD point failed %d\n", ret);
328  goto do_flash_read;
329  }
330  if (retlen < c->sector_size) {
331  /* Don't muck about if it won't let us point to the whole erase sector */
332  jffs2_dbg(1, "MTD point returned len too short: 0x%zx\n",
333  retlen);
334  mtd_unpoint(c->mtd, jeb->offset, retlen);
335  goto do_flash_read;
336  }
337  wordebuf = ebuf-sizeof(*wordebuf);
338  retlen /= sizeof(*wordebuf);
339  do {
340  if (*++wordebuf != ~0)
341  break;
342  } while(--retlen);
343  mtd_unpoint(c->mtd, jeb->offset, c->sector_size);
344  if (retlen) {
345  pr_warn("Newly-erased block contained word 0x%lx at offset 0x%08tx\n",
346  *wordebuf,
347  jeb->offset +
348  c->sector_size-retlen * sizeof(*wordebuf));
349  return -EIO;
350  }
351  return 0;
352  }
353  do_flash_read:
354  ebuf = kmalloc(PAGE_SIZE, GFP_KERNEL);
355  if (!ebuf) {
356  pr_warn("Failed to allocate page buffer for verifying erase at 0x%08x. Refiling\n",
357  jeb->offset);
358  return -EAGAIN;
359  }
360 
361  jffs2_dbg(1, "Verifying erase at 0x%08x\n", jeb->offset);
362 
363  for (ofs = jeb->offset; ofs < jeb->offset + c->sector_size; ) {
364  uint32_t readlen = min((uint32_t)PAGE_SIZE, jeb->offset + c->sector_size - ofs);
365  int i;
366 
367  *bad_offset = ofs;
368 
369  ret = jffs2_flash_read(c, ofs, readlen, &retlen, ebuf);
370  if (ret) {
371  pr_warn("Read of newly-erased block at 0x%08x failed: %d. Putting on bad_list\n",
372  ofs, ret);
373  ret = -EIO;
374  goto fail;
375  }
376  if (retlen != readlen) {
377  pr_warn("Short read from newly-erased block at 0x%08x. Wanted %d, got %zd\n",
378  ofs, readlen, retlen);
379  ret = -EIO;
380  goto fail;
381  }
382  for (i=0; i<readlen; i += sizeof(unsigned long)) {
383  /* It's OK. We know it's properly aligned */
384  unsigned long *datum = ebuf + i;
385  if (*datum + 1) {
386  *bad_offset += i;
387  pr_warn("Newly-erased block contained word 0x%lx at offset 0x%08x\n",
388  *datum, *bad_offset);
389  ret = -EIO;
390  goto fail;
391  }
392  }
393  ofs += readlen;
394  cond_resched();
395  }
396  ret = 0;
397 fail:
398  kfree(ebuf);
399  return ret;
400 }
