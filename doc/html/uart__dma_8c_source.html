115 static uint8_t _configureUartRxDma(UartDma *pUartd , UartChannel *pUartRx)
116 {
117  sXdmadCfg xdmadRxCfg;
118  uint32_t xdmaCndc, xdmaInt;
119  uint32_t i, LLI_Size;
120  Uart *pUartHwRx = pUartd->pUartHw;
121  sXdmad *pXdmadRx = pUartd->pXdmad;
122  uint8_t *pBuff = 0;
123 
124  /* Setup RX Single block */
125  if (pUartRx->dmaProgrammingMode < XDMAD_LLI) {
126  xdmadRxCfg.mbr_ubc = pUartRx->BuffSize;
127  xdmadRxCfg.mbr_da = (uint32_t)pUartRx->pBuff;
128 
129  xdmadRxCfg.mbr_sa = (uint32_t)&pUartHwRx->UART_RHR;
130  xdmadRxCfg.mbr_cfg = XDMAC_CC_TYPE_PER_TRAN |
131  XDMAC_CC_MBSIZE_SIXTEEN |
132  XDMAC_CC_DSYNC_PER2MEM |
133  XDMAC_CC_CSIZE_CHK_1 |
134  XDMAC_CC_DWIDTH_BYTE |
135  XDMAC_CC_SIF_AHB_IF1 |
136  XDMAC_CC_DIF_AHB_IF1 |
137  XDMAC_CC_SAM_FIXED_AM |
138  XDMAC_CC_DAM_INCREMENTED_AM |
139  XDMAC_CC_PERID(XDMAIF_Get_ChannelNumber
140  (pUartd->uartId, XDMAD_TRANSFER_RX));
141 
142  xdmadRxCfg.mbr_bc = 0;
143 
144  if (pUartRx->dmaProgrammingMode == XDMAD_MULTI)
145  xdmadRxCfg.mbr_bc = pUartRx->dmaBlockSize;
146 
147  xdmadRxCfg.mbr_sus = 0;
148  xdmadRxCfg.mbr_dus = 0;
149  xdmaCndc = 0;
150 
151  /* Put all interrupts on for non LLI list setup of DMA */
152  xdmaInt = (XDMAC_CIE_BIE |
153  XDMAC_CIE_DIE |
154  XDMAC_CIE_FIE |
155  XDMAC_CIE_RBIE |
156  XDMAC_CIE_WBIE |
157  XDMAC_CIE_ROIE);
158 
159  } else if (pUartRx->dmaProgrammingMode == XDMAD_LLI) {
160  /* Setup RX Link List */
161  LLI_Size = pUartRx->dmaBlockSize;
162  pBuff = pUartRx->pBuff;
163 
164  if (pUartRx->pLLIview != NULL) {
165  free(pUartRx->pLLIview);
166  pUartRx->pLLIview = NULL;
167  }
168 
169  pUartRx->pLLIview = malloc(sizeof(LinkedListDescriporView1) * LLI_Size);
170 
171  if (pUartRx->pLLIview == NULL) {
172  TRACE_ERROR(" Can not allocate memory to Rx LLI");
173  return USARTD_ERROR;
174  }
175 
176  xdmadRxCfg.mbr_cfg = XDMAC_CC_TYPE_PER_TRAN |
177  XDMAC_CC_MBSIZE_SIXTEEN |
178  XDMAC_CC_DSYNC_PER2MEM |
179  XDMAC_CC_MEMSET_NORMAL_MODE |
180  XDMAC_CC_CSIZE_CHK_1 |
181  XDMAC_CC_DWIDTH_BYTE |
182  XDMAC_CC_SIF_AHB_IF1 |
183  XDMAC_CC_DIF_AHB_IF1 |
184  XDMAC_CC_SAM_FIXED_AM |
185  XDMAC_CC_DAM_INCREMENTED_AM |
186  XDMAC_CC_PERID(XDMAIF_Get_ChannelNumber(
187  pUartd->uartId, XDMAD_TRANSFER_RX));
188  xdmadRxCfg.mbr_bc = 0;
189 
190  for (i = 0; i < LLI_Size; i++) {
191  pUartRx->pLLIview[i].mbr_ubc = XDMA_UBC_NVIEW_NDV1 |
192  XDMA_UBC_NSEN_UNCHANGED |
193  XDMA_UBC_NDEN_UPDATED |
194  ((i == LLI_Size - 1) ? ((pUartRx->dmaRingBuffer) ?
195  XDMA_UBC_NDE_FETCH_EN : 0) :
196  XDMA_UBC_NDE_FETCH_EN) | pUartRx->BuffSize;
197  pUartRx->pLLIview[i].mbr_sa = (uint32_t)&pUartHwRx->UART_RHR;
198  pUartRx->pLLIview[i].mbr_da = (uint32_t)pBuff;
199  pUartRx->pLLIview[i].mbr_nda = (i == (LLI_Size - 1)) ?
200  ((pUartRx->dmaRingBuffer) ? (uint32_t)pUartRx->pLLIview : 0) :
201  (uint32_t)&pUartRx->pLLIview[ i + 1 ];
202  pBuff += pUartRx->BuffSize;
203  }
204 
205  SCB_CleanDCache_by_Addr((uint32_t *)(pUartRx->pLLIview),
206  sizeof(LinkedListDescriporView1)*LLI_Size);
207 
208  xdmaCndc = XDMAC_CNDC_NDVIEW_NDV1 |
209  XDMAC_CNDC_NDE_DSCR_FETCH_EN |
210  XDMAC_CNDC_NDSUP_SRC_PARAMS_UPDATED |
211  XDMAC_CNDC_NDDUP_DST_PARAMS_UPDATED;
212 
213  xdmaInt = ((pUartRx->dmaRingBuffer) ? XDMAC_CIE_BIE : XDMAC_CIE_LIE);
214 
215  } else
216  return 1;
217 
218  if (XDMAD_ConfigureTransfer(pXdmadRx, pUartRx->ChNum, &xdmadRxCfg,
219  xdmaCndc, (uint32_t)pUartRx->pLLIview, xdmaInt))
220  return USARTD_ERROR;
221 
222  return 0;
223 }

360 uint32_t UARTD_Configure(UartDma *pUartd ,
361  uint8_t uartId,
362  uint32_t uartMode,
363  uint32_t baud,
364  uint32_t clk)
365 {

368 
369  /* Initialize the UART structure */
370  pUartd->uartId = uartId;

372  if (uartId == ID_UART0)
373  pUartd->pUartHw = UART0;
374 
375  if (uartId == ID_UART1)
376  pUartd->pUartHw = UART1;

394  if (!(NVIC_GetActive(XDMAC_IRQn)))
395  NVIC_ClearPendingIRQ(XDMAC_IRQn);

397  return 0;

409 uint32_t UARTD_EnableRxChannels(UartDma *pUartd, UartChannel *pRxCh)
410 {
411  Uart *pUartHw = pUartd->pUartHw;
412  uint32_t Channel;
413 
414  assert(pRxCh);
415  /* Init USART Rx Channel. */
416  pUartd->pRxChannel = pRxCh;
417 
418  /* Enables the USART to receive data. */
419  UART_SetReceiverEnabled (pUartHw , ENABLE);
420 
421 
422  /* Allocate a DMA channel for UART RX. */
423  Channel = XDMAD_AllocateChannel(pUartd->pXdmad, pUartd->uartId,
424  XDMAD_TRANSFER_MEMORY);
425 
426  if (Channel == XDMAD_ALLOC_FAILED)
427  return UARTD_ERROR;
428 
429  pRxCh->ChNum = Channel;
430 
431  /* Setup callbacks for UART RX */
432  if (pRxCh->callback) {
433  XDMAD_SetCallback(pUartd->pXdmad, pRxCh->ChNum,
434  (XdmadTransferCallback)pRxCh->callback, pRxCh->pArgument);
435  } else {
436  XDMAD_SetCallback(pUartd->pXdmad, pRxCh->ChNum,
437  (XdmadTransferCallback)UARTD_Rx_Cb, pUartd);
438  }
439 
440  if (XDMAD_PrepareChannel(pUartd->pXdmad, pRxCh->ChNum))
441  return UARTD_ERROR;
442 
443  if (_configureUartRxDma(pUartd, pRxCh))
444  return UARTD_ERROR_LOCK;
445 
446  /* Check if DMA IRQ is enable; if not Enable it */
447  if (!(NVIC_GetActive(XDMAC_IRQn))) {
448  /* Enable interrupt */
449  NVIC_EnableIRQ(XDMAC_IRQn);
450  }
451 
452  return 0;
453 }
