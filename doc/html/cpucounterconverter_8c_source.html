48 void rtems_counter_initialize_converter( uint32_t frequency )
49 {
50  uint64_t ns_per_s = UINT64_C(1000000000);
51  uint64_t bin_per_s = UINT64_C(1) << 32;
52 
53  to_ns_scaler = ((ns_per_s << 32) + frequency - 1) / frequency;
54  from_ns_scaler = ((UINT64_C(1) << 32) * frequency + ns_per_s - 1) / ns_per_s;
55 
56  to_sbt_scaler = ((bin_per_s << 31) + frequency - 1) / frequency;
57  from_sbt_scaler = ((UINT64_C(1) << 31) * frequency + bin_per_s - 1) / bin_per_s;
58 }
