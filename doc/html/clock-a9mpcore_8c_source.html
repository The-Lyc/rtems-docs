60 static uint64_t a9mpcore_clock_get_counter(volatile a9mpcore_gt *gt)
61 {
62  uint32_t cl;
63  uint32_t cu1;
64  uint32_t cu2;
65 
66  do {
67  cu1 = gt->cntrupper;
68  cl = gt->cntrlower;
69  cu2 = gt->cntrupper;
70  } while (cu1 != cu2);
71 
72  return ((uint64_t) cu2 << 32) | cl;
73 }

133 static void a9mpcore_clock_initialize(void)
134 {
135  volatile a9mpcore_gt *gt = A9MPCORE_GT;
136  uint64_t periphclk = a9mpcore_clock_periphclk();
137  uint64_t us_per_tick = rtems_configuration_get_microseconds_per_tick();
138  uint32_t interval = (uint32_t) ((periphclk * us_per_tick) / 1000000);
139  uint64_t cmpval;
140 
141  gt->ctrl &= A9MPCORE_GT_CTRL_TMR_EN;
142  gt->irqst = A9MPCORE_GT_IRQST_EFLG;
143 
144  cmpval = a9mpcore_clock_get_counter(gt);
145  cmpval += interval;
146 
147  a9mpcore_clock_gt_init(gt, cmpval, interval);
148  a9mpcore_clock_secondary_initialization(gt, cmpval, interval);
149 
150  a9mpcore_tc.tc_get_timecount = a9mpcore_clock_get_timecount;
151  a9mpcore_tc.tc_counter_mask = 0xffffffff;
152  a9mpcore_tc.tc_frequency = periphclk;
153  a9mpcore_tc.tc_quality = RTEMS_TIMECOUNTER_QUALITY_CLOCK_DRIVER;
154  rtems_timecounter_install(&a9mpcore_tc);
155 }
