109 static rtems_status_code tmu_finalize(void)
110 {
111  rtems_status_code sc;
112 
113  sc = rtems_semaphore_obtain(done_sem, RTEMS_WAIT, 100);
114  if (sc != RTEMS_SUCCESSFUL)
115  return sc;
116 
117  if (invalidate_after) {
118  invalidate_l2();
119  __asm__ volatile( /* Invalidate Level-1 data cache */
120  "wcsr DCC, r0\n"
121  "nop\n"
122  );
123  }
124 
125  return RTEMS_SUCCESSFUL;
126 }

128 rtems_device_driver tmu_control(

130  rtems_device_minor_number minor,
131  void *arg
132 )
133 {

135  struct tmu_td *td = (struct tmu_td *)args->buffer;
136  rtems_status_code sc;
137 
138  switch (args->command) {
139  case TMU_EXECUTE:
140  tmu_start(td);
141  sc = tmu_finalize();
142  break;
143  case TMU_EXECUTE_NONBLOCK:
144  tmu_start(td);
145  sc = RTEMS_SUCCESSFUL;
146  break;
147  case TMU_EXECUTE_WAIT:
148  sc = tmu_finalize();
149  break;
150  default:
151  sc = RTEMS_UNSATISFIED;
152  break;
153  }
154 
155  if (sc == RTEMS_SUCCESSFUL)
156  args->ioctl_return = 0;
157  else
158  args->ioctl_return = -1;
159 
160  return sc;
