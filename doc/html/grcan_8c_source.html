356 static rtems_device_driver grcan_hw_start(struct grcan_priv *pDev)
357 {
358  /*
359  * tmp is set but never used. GCC gives a warning for this
360  * and we need to tell GCC not to complain.
361  */
362  unsigned int tmp RTEMS_UNUSED;
363 
364  SPIN_IRQFLAGS(oldLevel);
365 
366  FUNCDBG();
367 
368  /* Check that memory has been allocated successfully */
369  if (!pDev->tx || !pDev->rx)
370  return RTEMS_NO_MEMORY;
371 
372  /* Configure FIFO configuration register
373  * and Setup timing
374  */
375  if (pDev->config_changed) {
376  grcan_hw_config(pDev->regs, &pDev->config);
377  pDev->config_changed = 0;
378  }
379 
380  /* Setup receiver */
381  pDev->regs->rx0addr = (unsigned int)pDev->_rx_hw;
382  pDev->regs->rx0size = pDev->rxbuf_size;
383 
384  /* Setup Transmitter */
385  pDev->regs->tx0addr = (unsigned int)pDev->_tx_hw;
386  pDev->regs->tx0size = pDev->txbuf_size;
387 
388  /* Setup acceptance filters */
389  grcan_hw_accept(pDev->regs, &pDev->afilter);
390 
391  /* Sync filters */
392  grcan_hw_sync(pDev->regs, &pDev->sfilter);
393 
394  /* Clear status bits */
395  tmp = READ_REG(&pDev->regs->stat);
396  pDev->regs->stat = 0;
397 
398  /* Setup IRQ handling */
399 
400  /* Clear all IRQs */
401  tmp = READ_REG(&pDev->regs->pir);
402  pDev->regs->picr = 0x1ffff;
403 
404  /* unmask TxLoss|TxErrCntr|RxErrCntr|TxAHBErr|RxAHBErr|OR|OFF|PASS */
405  pDev->regs->imr = 0x1601f;
406 
407  /* Enable routing of the IRQs */
408  SPIN_LOCK_IRQ(&pDev->devlock, oldLevel);
409  IRQ_UNMASK(pDev->irq + GRCAN_IRQ_TXSYNC);
410  IRQ_UNMASK(pDev->irq + GRCAN_IRQ_RXSYNC);
411  IRQ_UNMASK(pDev->irq + GRCAN_IRQ_IRQ);
412  SPIN_UNLOCK_IRQ(&pDev->devlock, oldLevel);
413 
414  /* Enable receiver/transmitter */
415  pDev->regs->rx0ctrl = GRCAN_RXCTRL_ENABLE;
416  pDev->regs->tx0ctrl = GRCAN_TXCTRL_ENABLE;
417 
418  /* Enable HurriCANe core */
419  pDev->regs->ctrl = GRCAN_CTRL_ENABLE;
420 
421  /* Leave transmitter disabled, it is enabled when
422  * trying to send something.
423  */
424  return RTEMS_SUCCESSFUL;
425 }

572 static int grcan_calc_timing(
573  unsigned int baud, /* The requested BAUD to calculate timing for */
574  unsigned int core_hz, /* Frequency in Hz of GRCAN Core */
575  unsigned int sampl_pt,
576  struct grcan_timing *timing /* result is placed here */
577 )
578 {
579  int best_error = 1000000000;
580  int error;
581  int best_tseg = 0, best_brp = 0, brp = 0;
582  int tseg = 0, tseg1 = 0, tseg2 = 0;
583  int sjw = 1;
584 
585  /* Default to 90% */
586  if ((sampl_pt < 50) || (sampl_pt > 99)) {
587  sampl_pt = GRCAN_SAMPLING_POINT;
588  }
589 
590  if ((baud < 5000) || (baud > 1000000)) {
591  /* invalid speed mode */
592  return -1;
593  }
594 
595  /* find best match, return -2 if no good reg
596  * combination is available for this frequency
597  */
598 
599  /* some heuristic specials */
600  if (baud > ((1000000 + 500000) / 2))
601  sampl_pt = 75;
602 
603  if (baud < ((12500 + 10000) / 2))
604  sampl_pt = 75;
605 
606  /* tseg even = round down, odd = round up */
607  for (
608  tseg = (MIN_TSEG1 + MIN_TSEG2 + 2) * 2;
609  tseg <= (MAX_TSEG2 + MAX_TSEG1 + 2) * 2 + 1;
610  tseg++
611  ) {
612  brp = core_hz / ((1 + tseg / 2) * baud) + tseg % 2;
613  if (
614  (brp <= 0) ||
615  ((brp > 256 * 1) && (brp <= 256 * 2) && (brp & 0x1)) ||
616  ((brp > 256 * 2) && (brp <= 256 * 4) && (brp & 0x3)) ||
617  ((brp > 256 * 4) && (brp <= 256 * 8) && (brp & 0x7)) ||
618  (brp > 256 * 8)
619  )
620  continue;
621 
622  error = baud - core_hz / (brp * (1 + tseg / 2));
623  if (error < 0) {
624  error = -error;
625  }
626 
627  if (error <= best_error) {
628  best_error = error;
629  best_tseg = tseg / 2;
630  best_brp = brp - 1;
631  }
632  }
633 
634  if (best_error && (baud / best_error < 10)) {
635  return -2;
636  } else if (!timing)
637  return 0; /* nothing to store result in, but a valid bitrate can be calculated */
638 
639  tseg2 = best_tseg - (sampl_pt * (best_tseg + 1)) / 100;
640 
641  if (tseg2 < MIN_TSEG2) {
642  tseg2 = MIN_TSEG2;
643  }
644 
645  if (tseg2 > MAX_TSEG2) {
646  tseg2 = MAX_TSEG2;
647  }
648 
649  tseg1 = best_tseg - tseg2 - 2;
650 
651  if (tseg1 > MAX_TSEG1) {
652  tseg1 = MAX_TSEG1;
653  tseg2 = best_tseg - tseg1 - 2;
654  }
655 
656  /* Get scaler and BRP from pseudo BRP */
657  if (best_brp <= 256) {
658  timing->scaler = best_brp;
659  timing->bpr = 0;
660  } else if (best_brp <= 256 * 2) {
661  timing->scaler = ((best_brp + 1) >> 1) - 1;
662  timing->bpr = 1;
663  } else if (best_brp <= 256 * 4) {
664  timing->scaler = ((best_brp + 1) >> 2) - 1;
665  timing->bpr = 2;
666  } else {
667  timing->scaler = ((best_brp + 1) >> 3) - 1;
668  timing->bpr = 3;
669  }
670 
671  timing->ps1 = tseg1 + 1;
672  timing->ps2 = tseg2;
673  timing->rsj = sjw;
674 
675  return 0;
676 }

697  * Due to hardware wrap around simplification write pointer will
698  * never reach the read pointer, at least a gap of 8 bytes.
699  * The only time they are equal is when the read pointer has
700  * reached the write pointer (empty buffer)
701  *
702  */
703  if (wp != rp) {
704  /* Not empty, we have received chars...
705  * Read as much as possible from DMA buffer
706  */
707  size = READ_REG(&regs->rx0size);
708 
709  /* Get number of bytes available in RX buffer */
710  trunk_msg_cnt = grcan_hw_rxavail(rp, wp, size);
711 
712  /* truncate size if user space buffer hasn't room for
713  * all received chars.
714  */
715  if (trunk_msg_cnt > max)
716  trunk_msg_cnt = max;
717 
718  /* Read until i is 0 */
719  i = trunk_msg_cnt;
720 
721  addr = (unsigned int)pDev->rx;
722  source = (struct grcan_msg *)(addr + rp);
723  dest = buffer;
724  rxmax = addr + (size - GRCAN_MSG_SIZE);
725 
726  /* Read as many can messages as possible */
727  while (i > 0) {
728  /* Read CAN message from DMA buffer */
729  tmp.head[0] = READ_DMA_WORD(&source->head[0]);
730  tmp.head[1] = READ_DMA_WORD(&source->head[1]);
731  if (tmp.head[1] & 0x4) {
732  DBGC(DBG_RX, "overrun\n");
733  }
734  if (tmp.head[1] & 0x2) {
735  DBGC(DBG_RX, "bus-off mode\n");
736  }
737  if (tmp.head[1] & 0x1) {
738  DBGC(DBG_RX, "error-passive mode\n");
739  }
740  /* Convert one grcan CAN message to one "software" CAN message */
741  dest->extended = tmp.head[0] >> 31;
742  dest->rtr = (tmp.head[0] >> 30) & 0x1;
743  if (dest->extended) {
744  dest->id = tmp.head[0] & 0x3fffffff;
745  } else {
746  dest->id = (tmp.head[0] >> 18) & 0xfff;
747  }
748  dest->len = tmp.head[1] >> 28;
749  for (j = 0; j < dest->len; j++)
750  dest->data[j] = READ_DMA_BYTE(&source->data[j]);
751 
752  /* wrap around if neccessary */
753  source =
754  ((unsigned int)source >= rxmax) ?
755  (struct grcan_msg *)addr : source + 1;
756  dest++; /* straight user buffer */
757  i--;
758  }
759  {
760  /* A bus off interrupt may have occured after checking pDev->started */
761  SPIN_IRQFLAGS(oldLevel);
762 
763  SPIN_LOCK_IRQ(&pDev->devlock, oldLevel);
764  if (pDev->started == STATE_STARTED) {
765  regs->rx0rd = (unsigned int) source - addr;
766  regs->rx0ctrl = GRCAN_RXCTRL_ENABLE;
767  } else {
768  DBGC(DBG_STATE, "cancelled due to a BUS OFF error\n");
769  trunk_msg_cnt = state2err[pDev->started];
770  }
771  SPIN_UNLOCK_IRQ(&pDev->devlock, oldLevel);
772  }
773  return trunk_msg_cnt;
774  }
775  return 0;
776 }

854 static int grcan_wait_rxdata(struct grcan_priv *pDev, int min)
855 {
856  unsigned int wp, rp, size, irq;
857  unsigned int irq_trunk, dataavail;
858  int wait, state;
859  SPIN_IRQFLAGS(oldLevel);
860 
861  FUNCDBG();
862 
863  /*** block until receive IRQ received
864  * Set up a valid IRQ point so that an IRQ is received
865  * when one or more messages are received
866  */
867  SPIN_LOCK_IRQ(&pDev->devlock, oldLevel);
868  state = pDev->started;
869 
870  /* A bus off interrupt may have occured after checking pDev->started */
871  if (state != STATE_STARTED) {
872  SPIN_UNLOCK_IRQ(&pDev->devlock, oldLevel);
873  if (state == STATE_BUSOFF) {
874  DBGC(DBG_STATE, "cancelled due to a BUS OFF error\n");
875  } else if (state == STATE_AHBERR) {
876  DBGC(DBG_STATE, "cancelled due to a AHB error\n");
877  } else {
878  DBGC(DBG_STATE, "cancelled due to STOP (unexpected) \n");
879  }
880  return state2err[state];
881  }
882 
883  size = READ_REG(&pDev->regs->rx0size);
884  rp = READ_REG(&pDev->regs->rx0rd);
885  wp = READ_REG(&pDev->regs->rx0wr);
886 
887  /**** Calculate IRQ Pointer ****/
888  irq = wp + min * GRCAN_MSG_SIZE;
889  /* wrap irq around */
890  if (irq >= size) {
891  irq_trunk = irq - size;
892  } else
893  irq_trunk = irq;
894 
895  /* init IRQ HW */
896  pDev->regs->rx0irq = irq_trunk;
897 
898  /* Clear pending Rx IRQ */
899  pDev->regs->picr = GRCAN_RXIRQ_IRQ;
900 
901  wp = READ_REG(&pDev->regs->rx0wr);
902

904  dataavail = grcan_hw_rxavail(rp, wp, size);
905 
906  if (dataavail < min) {
907  /* Still empty, proceed with sleep - Turn on IRQ (unmask irq) */
908  pDev->regs->imr = READ_REG(&pDev->regs->imr) | GRCAN_RXIRQ_IRQ;
909  wait = 1;
910  } else {
911  /* enough message has been received, abort sleep - don't unmask interrupt */
912  wait = 0;
913  }
914  SPIN_UNLOCK_IRQ(&pDev->devlock, oldLevel);
915 
916  /* Wait for IRQ to fire only if has been triggered */
917  if (wait) {
918  rtems_semaphore_obtain(pDev->rx_sem, RTEMS_WAIT, RTEMS_NO_TIMEOUT);
919  /*
920  * The semaphore is released either due to the expected IRQ
921  * condition or by BUSOFF, AHBERROR or another thread calling
922  * grcan_stop(). In either case, state2err[] has the correnct
923  * return value.
924  */
925  return state2err[pDev->started];
926  }
927 
928  return 0;
929 }

932  * point the location of the CAN message corresponding to min.
933  *
934  * min must be at least WRAP_AROUND_TX_MSGS less than max buffer capacity
935  * (pDev->txbuf_size/GRCAN_MSG_SIZE) for this algo to work.
936  */
937 static int grcan_wait_txspace(struct grcan_priv *pDev, int min)
938 {

940  unsigned int irq, rp, wp, size, space_left;
941  unsigned int irq_trunk;
942  SPIN_IRQFLAGS(oldLevel);
943 
944  DBGC(DBG_TX, "\n");
945  /*FUNCDBG(); */
946 
947  SPIN_LOCK_IRQ(&pDev->devlock, oldLevel);
948  state = pDev->started;
949  /* A bus off interrupt may have occured after checking pDev->started */
950  if (state != STATE_STARTED) {
951  SPIN_UNLOCK_IRQ(&pDev->devlock, oldLevel);
952  if (state == STATE_BUSOFF) {
953  DBGC(DBG_STATE, "cancelled due to a BUS OFF error\n");
954  } else if (state == STATE_AHBERR) {
955  DBGC(DBG_STATE, "cancelled due to a AHB error\n");
956  } else {
957  DBGC(DBG_STATE, "cancelled due to STOP (unexpected)\n");
958  }
959  return state2err[state];
960  }
961 
962  pDev->regs->tx0ctrl = GRCAN_TXCTRL_ENABLE;
963 
964  size = READ_REG(&pDev->regs->tx0size);
965  wp = READ_REG(&pDev->regs->tx0wr);
966 
967  rp = READ_REG(&pDev->regs->tx0rd);
968 
969  /**** Calculate IRQ Pointer ****/
970  irq = rp + min * GRCAN_MSG_SIZE;
971  /* wrap irq around */
972  if (irq >= size) {
973  irq_trunk = irq - size;
974  } else
975  irq_trunk = irq;
976 
977  /* trigger HW to do a IRQ when enough room in buffer */
978  pDev->regs->tx0irq = irq_trunk;
979 
980  /* Clear pending Tx IRQ */
981  pDev->regs->picr = GRCAN_TXIRQ_IRQ;
982 
983  /* One problem, if HW already gone past IRQ place the IRQ will
984  * never be received resulting in a thread hang. We check if so
985  * before proceeding.
986  *
987  * has the HW already gone past the IRQ generation place?
988  * == does min fit info tx buffer?
989  */
990  rp = READ_REG(&pDev->regs->tx0rd);
991 
992  space_left = grcan_hw_txspace(rp, wp, size);
993 
994  if (space_left < min) {
995  /* Still too full, proceed with sleep - Turn on IRQ (unmask irq) */
996  pDev->regs->imr = READ_REG(&pDev->regs->imr) | GRCAN_TXIRQ_IRQ;
997  wait = 1;
998  } else {
999  /* There are enough room in buffer, abort wait - don't unmask interrupt */
1000  wait = 0;
1001  }
1002  SPIN_UNLOCK_IRQ(&pDev->devlock, oldLevel);
1003 
1004  /* Wait for IRQ to fire only if it has been triggered */
1005  if (wait) {
1006  rtems_semaphore_obtain(pDev->tx_sem, RTEMS_WAIT, RTEMS_NO_TIMEOUT);
1007  return state2err[pDev->started];
1008  }
1009 
1010  /* At this point the TxIRQ has been masked, we ned not to mask it */
1011  return 0;
1012 }

1015 {
1016  int wait, state;
1017  unsigned int rp, wp;
1018  SPIN_IRQFLAGS(oldLevel);
1019  FUNCDBG();

1028  (rp = READ_REG(&pDev->regs->tx0rd))

1033

1040  DBGC(DBG_STATE, "cancelled due to a AHB error\n");
1041  } else {
1042  DBGC(DBG_STATE, "cancelled due to STOP (unexpected)\n");
1043  }
1044  return state2err[state];
1045  }
1046 
1047  /* Clear pending TXEmpty IRQ */
1048  pDev->regs->picr = GRCAN_TXEMPTY_IRQ;

1051  /* Still not empty, proceed with sleep - Turn on IRQ (unmask irq) */

1053  READ_REG(&pDev->regs->imr) | GRCAN_TXEMPTY_IRQ;

1061  break;

1064  rtems_semaphore_obtain(pDev->txempty_sem, RTEMS_WAIT, RTEMS_NO_TIMEOUT);

1066  if (state != STATE_STARTED) {
1067  return state2err[state];
1068  }
1069  }
1070  return 0;
1071 }
