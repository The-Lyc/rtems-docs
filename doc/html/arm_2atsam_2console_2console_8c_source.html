60 static void atsam_usart_interrupt(void *arg)
61 {
62  rtems_termios_tty *tty = arg;
63  atsam_usart_context *ctx = rtems_termios_get_device_context(tty);
64  Usart *regs = ctx->regs;
65  uint32_t csr = regs->US_CSR;
66 
67  while ((csr & US_CSR_RXRDY) != 0) {
68  char c = (char) regs->US_RHR;
69 
70  rtems_termios_enqueue_raw_characters(tty, &c, 1);
71 
72  csr = regs->US_CSR;
73  }
74 
75  if (ctx->transmitting && (csr & US_CSR_TXEMPTY) != 0) {
76  rtems_termios_dequeue_characters(tty, 1);
77  }
78 }
79 #endif

81 static bool atsam_usart_set_attributes(
82  rtems_termios_device_context *base,
83  const struct termios *term
84 )
85 {
86  atsam_usart_context *ctx = (atsam_usart_context *) base;
87  Usart *regs = ctx->regs;
88  rtems_termios_baud_t baud;
89  uint32_t mr;
90 
91  baud = rtems_termios_baud_to_number(term->c_ospeed);
92  regs->US_BRGR = (BOARD_MCK / baud) / 16;
93 
94  if ((term->c_cflag & CREAD) != 0) {
95  regs->US_CR = US_CR_RXEN | US_CR_TXEN;
96  } else {
97  regs->US_CR = US_CR_TXEN;
98  }
99 
100  mr = US_MR_USART_MODE_NORMAL | US_MR_USCLKS_MCK;
101 
102  switch (term->c_cflag & CSIZE) {
103  case CS5:
104  mr |= US_MR_CHRL_5_BIT;
105  break;
106  case CS6:
107  mr |= US_MR_CHRL_6_BIT;
108  break;
109  case CS7:
110  mr |= US_MR_CHRL_7_BIT;
111  break;
112  default:
113  mr |= US_MR_CHRL_8_BIT;
114  break;
115  }
116 
117  if ((term->c_cflag & PARENB) != 0) {
118  if ((term->c_cflag & PARODD) != 0) {
119  mr |= US_MR_PAR_ODD;
120  } else {
121  mr |= US_MR_PAR_EVEN;
122  }
123  } else {
124  mr |= US_MR_PAR_NO;
125  }
126 
127  if ((term->c_cflag & CSTOPB) != 0) {
128  mr |= US_MR_NBSTOP_2_BIT;
129  } else {
130  mr |= US_MR_NBSTOP_1_BIT;
131  }
132 
133  regs->US_MR = mr;
134 
135  return true;
136 }

193 static void atsam_usart_write(
194  rtems_termios_device_context *base,
195  const char *buf,
196  size_t len
197 )
198 {
199  atsam_usart_context *ctx = (atsam_usart_context *) base;
200  Usart *regs = ctx->regs;
201 
202 #ifdef ATSAM_CONSOLE_USE_INTERRUPTS
203  if (len > 0) {
204  ctx->transmitting = true;
205  regs->US_THR = buf[0];
206  regs->US_IER = US_IDR_TXEMPTY;
207  } else {
208  ctx->transmitting = false;
209  regs->US_IDR = US_IDR_TXEMPTY;
210  }
211 #else
212  size_t i;
213 
214  for (i = 0; i < len; ++i) {
215  while ((regs->US_CSR & US_CSR_TXEMPTY) == 0) {
216  /* Wait */
217  }
218 
219  regs->US_THR = buf[i];
220  }
221 #endif
222 }
