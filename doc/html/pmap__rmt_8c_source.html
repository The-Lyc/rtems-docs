208  ifc.ifc_buf = buf;
209  if (ioctl(sock, SIOCGIFCONF, (char *)&ifc) < 0) {
210  perror("broadcast: ioctl (get interface configuration)");
211  return (0);
212  }
213 #define max(a, b) (a > b ? a : b)
214 #define size(p) max((p).sa_len, sizeof(p))
215  cplim = buf + ifc.ifc_len; /*skip over if's with big ifr_addr's */
216  for (cp = buf; cp < cplim;

218  ifr = (struct ifreq *)cp;
219  if (ifr->ifr_addr.sa_family != AF_INET)
220  continue;
221  ifreq = *ifr;
222  if (ioctl(sock, SIOCGIFFLAGS, (char *)&ifreq) < 0) {
223  perror("broadcast: ioctl (get interface flags)");
224  continue;
225  }
226  if ((ifreq.ifr_flags & IFF_BROADCAST) &&
227  (ifreq.ifr_flags & IFF_UP)) {
228  sin = (struct sockaddr_in *)&ifr->ifr_addr;
229 #ifdef SIOCGIFBRDADDR /* 4.3BSD */
230  if (ioctl(sock, SIOCGIFBRDADDR, (char *)&ifreq) < 0) {
231  addr =
232  inet_makeaddr(inet_netof(sin->sin_addr),
233  INADDR_ANY);
234  } else {
235  addr = ((struct sockaddr_in*)
236  &ifreq.ifr_addr)->sin_addr;
237  }
238 #else /* 4.2 BSD */
239  addr = inet_makeaddr(inet_netof(sin->sin_addr),
240  INADDR_ANY);
241 #endif
242  for (n=i-1; n>=0; n--) {
243  if (addr.s_addr == addrs[n].s_addr)
244  break;
245  }
246  if (n<0) {
247  addrs[i++] = addr;
248  }
249  }
250  }
251  return (i);

352  /*

354  * The response timeout grows larger per iteration.
355  *
356  * XXX This will loop about 5 times the stop. If there are
357  * lots of signals being received by the process it will quit
358  * send them all in one quick burst, not paying attention to
359  * the intended function of sending them slowly over half a
360  * minute or so
361  */
362  for (t.tv_sec = 4; t.tv_sec <= 14; t.tv_sec += 2) {
363  for (i = 0; i < nets; i++) {
364  baddr.sin_addr = addrs[i];

366  (struct sockaddr *)&baddr,
367  sizeof (struct sockaddr)) != outlen) {
368  perror("Cannot send broadcast packet");

371  }
372  }

380  msg.acpted_rply.ar_results.proc = (xdrproc_t) xdr_rmtcallres;
381  /* XXX we know the other bits are still clear */
382  FD_SET(sock, fds);
383  tv = t; /* for select() that copies back */
384  switch (select(sock + 1, fds, NULL, NULL, &tv)) {

401  (struct sockaddr *)&raddr, &fromlen);
402  if (inlen < 0) {

406  stat = RPC_CANTRECV;
407  goto done_broad;
408  }

410  goto recv_again;
411  /*
412  * see if reply transaction id matches sent id.
413  * If so, decode the results.
414  */
415  xdrmem_create(xdrs, inbuf, (u_int)inlen, XDR_DECODE);
416  if (xdr_replymsg(xdrs, &msg)) {
417  if ((msg.rm_xid == xid) &&
418  (msg.rm_reply.rp_stat == MSG_ACCEPTED) &&
419  (msg.acpted_rply.ar_stat == SUCCESS)) {
420  raddr.sin_port = htons((u_short)port);
421  done = (*eachresult)(resultsp, &raddr);
422  }
423  /* otherwise, we just ignore the errors ... */
424  }
425  xdrs->x_op = XDR_FREE;
426  msg.acpted_rply.ar_results.proc = (xdrproc_t) xdr_void;
427  (void)xdr_replymsg(xdrs, &msg);
428  (void)(*xresults)(xdrs, resultsp);

432  goto done_broad;

435  }
436  }

442  AUTH_DESTROY(unix_auth);
