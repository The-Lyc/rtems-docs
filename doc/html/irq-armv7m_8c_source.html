54 {
55  ARMV7M_Exception_handler *vector_table;
56  int i;
57 
58  vector_table = (ARMV7M_Exception_handler *) bsp_vector_table_begin;
59 
60  if (bsp_vector_table_begin != bsp_start_vector_table_begin) {
61  memcpy(
62  vector_table,
63  bsp_start_vector_table_begin,
64  (size_t) bsp_vector_table_size
65  );
66  }
67 
68  _ARMV7M_SCB->icsr = ARMV7M_SCB_ICSR_PENDSVCLR | ARMV7M_SCB_ICSR_PENDSTCLR;
69 
70  for (i = BSP_INTERRUPT_VECTOR_MIN; i <= BSP_INTERRUPT_VECTOR_MAX; ++i) {
71  _ARMV7M_NVIC_Clear_enable(i);
72  _ARMV7M_NVIC_Clear_pending(i);
73  _ARMV7M_NVIC_Set_priority(i, BSP_ARMV7M_IRQ_PRIORITY_DEFAULT);
74  }
75 
76  _ARMV7M_SCB->vtor = vector_table;
77 
78  return RTEMS_SUCCESSFUL;
79 }
