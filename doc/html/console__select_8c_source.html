83 static bool parse_printk_or_console(
84  const char *param,
85  rtems_device_minor_number *minor_out
86 )
87 {
88  static const char *opt;
89  const char *option;
90  const char *comma;
91  size_t length;
92  size_t index;
93  rtems_device_minor_number minor;
94  console_tbl *conscfg;
95 
96  /*
97  * Check the command line for the type of mode the console is.
98  */
99  opt = bsp_cmdline_arg(param);
100  if ( !opt ) {
101  return false;
102  }
103 
104  /*
105  * Fine the length, there can be more command line visible.
106  */
107  length = 0;
108  while ((opt[length] != ' ') && (opt[length] != '\0')) {
109  ++length;
110  if (length > NAME_MAX) {
111  printk("invalid option (%s): too long\n", param);
112  return false;
113  }
114  }
115 
116  /*
117  * Only match up to a comma or NULL
118  */
119  index = 0;
120  while ((opt[index] != '=') && (index < length)) {
121  ++index;
122  }
123 
124  if (opt[index] != '=') {
125  printk("invalid option (%s): no equals\n", param);
126  return false;
127  }
128 
129  ++index;
130  option = &opt[index];
131 
132  while ((opt[index] != ',') && (index < length)) {
133  ++index;
134  }
135 
136  if (opt[index] == ',')
137  comma = &opt[index];
138  else
139  comma = NULL;
140 
141  length = &opt[index] - option;
142 
143  conscfg = console_find_console_entry( option, length, &minor );
144  if ( conscfg == NULL ) {
145  return false;
146  }
147 
148  *minor_out = minor;
149  if (comma) {
150  option = comma + 1;
151  if (strncmp (option, "115200", sizeof ("115200") - 1) == 0)
152  conscfg->pDeviceParams = (void *)115200;
153  else if (strncmp (option, "57600", sizeof ("57600") - 1) == 0)
154  conscfg->pDeviceParams = (void *)57600;
155  else if (strncmp (option, "38400", sizeof ("38400") - 1) == 0)
156  conscfg->pDeviceParams = (void *)38400;
157  else if (strncmp (option, "19200", sizeof ("19200") - 1) == 0)
158  conscfg->pDeviceParams = (void *)19200;
159  else if (strncmp (option, "9600", sizeof ("9600") - 1) == 0)
160  conscfg->pDeviceParams = (void *)9600;
161  else if (strncmp (option, "4800", sizeof ("4800") - 1) == 0)
162  conscfg->pDeviceParams = (void *)4800;
163  }
164 
165  return true;
166 }

178 /*
179  * Parse the arguments early so the printk and console ports are
180  * set appropriately.
181  */
182 void pc386_parse_console_arguments(void)
183 {
184  rtems_device_minor_number minor;
185  rtems_device_minor_number minor_console = 0;
186  rtems_device_minor_number minor_printk = 0;
187 
188  /*
189  * Assume that if only --console is specified, that printk() should
190  * follow that selection by default.
191  */
192  if ( parse_printk_or_console( "--console=", &minor ) ) {
193  minor_console = minor;
194  minor_printk = minor;
195  }
196 
197  /*
198  * But if explicitly specified, attempt to honor it.
199  */
200  if ( parse_printk_or_console( "--printk=", &minor ) ) {
201  minor_printk = minor;
202  }
203 
204  printk( "Console: %s printk: %s\n",
205  get_name(minor_console),get_name(minor_printk) );
206 
207  /*
208  * Any output after this can cause problems until termios is initialised.
209  */
210  Console_Port_Minor = minor_console;
211  BSPPrintkPort = minor_printk;
212 }
