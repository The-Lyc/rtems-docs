105 svcudp_bufcreate(
106  int sock,
107  u_int sendsz,
108  u_int recvsz )
109 {
110  bool_t madesock = FALSE;
111  register SVCXPRT *xprt;
112  register struct svcudp_data *su;
113  struct sockaddr_in addr;
114  socklen_t len = sizeof(struct sockaddr_in);
115 
116  if (sock == RPC_ANYSOCK) {
117  if ((sock = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP)) < 0) {
118  perror("svcudp_create: socket creation problem");
119  return ((SVCXPRT *)NULL);
120  }
121  madesock = TRUE;
122  }
123  memset((char *)&addr, 0, sizeof (addr));

131  perror("svcudp_create - cannot getsockname");

139  return (NULL);

180 svcudp_recv(
181  SVCXPRT *xprt,
182  struct rpc_msg *msg)
183 {
184  register struct svcudp_data *su = su_data(xprt);
185  register XDR *xdrs = &(su->su_xdrs);
186  register int rlen;
187  char *reply;
188  u_long replylen;
189 
190  again:
191  xprt->xp_addrlen = sizeof(struct sockaddr_in);
192  rlen = recvfrom(xprt->xp_sock, rpc_buffer(xprt), (int) su->su_iosz,
193  0, (struct sockaddr *)&(xprt->xp_raddr), &(xprt->xp_addrlen));
194  if (rlen == -1 && errno == EINTR)
195  goto again;
196  if (rlen == -1 || rlen < 4*sizeof(u_int32_t))
197  return (FALSE);
198  xdrs->x_op = XDR_DECODE;
199  XDR_SETPOS(xdrs, 0);
200  if (! xdr_callmsg(xdrs, msg))
201  return (FALSE);
202  su->su_xid = msg->rm_xid;
203  if (su->su_cache != NULL) {
204  if (cache_get(xprt, msg, &reply, &replylen)) {
205  (void) sendto(xprt->xp_sock, reply, (int) replylen, 0,
206  (struct sockaddr *) &xprt->xp_raddr, xprt->xp_addrlen);
207  return (TRUE);
208  }
209  }
210  return (TRUE);

344 /*
345  * Enable use of the cache.
346  * Note: there is no disable.
347  */
348 int svcudp_enablecache(
349  SVCXPRT *transp,
350  u_long size)
351 {
352  struct svcudp_data *su = su_data(transp);
353  struct udp_cache *uc;
354 
355  if (su->su_cache != NULL) {
356  CACHE_PERROR("enablecache: cache already enabled");
357  return(0);
358  }
359  uc = ALLOC(struct udp_cache, 1);
360  if (uc == NULL) {
361  CACHE_PERROR("enablecache: could not allocate cache");
362  return(0);
363  }
364  uc->uc_size = size;
365  uc->uc_nextvictim = 0;
366  uc->uc_entries = ALLOC(cache_ptr, size * SPARSENESS);
367  if (uc->uc_entries == NULL) {
368  CACHE_PERROR("enablecache: could not allocate cache data");
369  return(0);
370  }
371  BZERO(uc->uc_entries, cache_ptr, size * SPARSENESS);
372  uc->uc_fifo = ALLOC(cache_ptr, size);
373  if (uc->uc_fifo == NULL) {
374  CACHE_PERROR("enablecache: could not allocate cache fifo");
375  return(0);
376  }
377  BZERO(uc->uc_fifo, cache_ptr, size);
378  su->su_cache = (char *) uc;
379  return(1);
380 }

384  * Set an entry in the cache
385  */
386 static void
387 cache_set(
388  SVCXPRT *xprt,
389  u_long replylen)
390 {
391  register cache_ptr victim;
392  register cache_ptr *vicp;
393  register struct svcudp_data *su = su_data(xprt);
394  struct udp_cache *uc = (struct udp_cache *) su->su_cache;
395  u_int loc;
396  char *newbuf;
397 
398  /*
399  * Find space for the new entry, either by
400  * reusing an old entry, or by mallocing a new one
401  */
402  victim = uc->uc_fifo[uc->uc_nextvictim];
403  if (victim != NULL) {
404  loc = CACHE_LOC(xprt, victim->cache_xid);
405  for (vicp = &uc->uc_entries[loc];
406  *vicp != NULL && *vicp != victim;
407  vicp = &(*vicp)->cache_next)
408  ;
409  if (*vicp == NULL) {
410  CACHE_PERROR("cache_set: victim not found");
411  return;
412  }
413  *vicp = victim->cache_next; /* remote from cache */
414  newbuf = victim->cache_reply;
415  } else {
416  victim = ALLOC(struct cache_node, 1);
417  if (victim == NULL) {
418  CACHE_PERROR("cache_set: victim alloc failed");
419  return;
420  }
421  newbuf = mem_alloc(su->su_iosz);
422  if (newbuf == NULL) {
423  CACHE_PERROR("cache_set: could not allocate new rpc_buffer");
424  return;
425  }
426  }
427 
428  /*
429  * Store it away
430  */
431  victim->cache_replylen = replylen;
432  victim->cache_reply = rpc_buffer(xprt);
433  rpc_buffer(xprt) = newbuf;
434  xdrmem_create(&(su->su_xdrs), rpc_buffer(xprt), su->su_iosz, XDR_ENCODE);
435  victim->cache_xid = su->su_xid;
436  victim->cache_proc = uc->uc_proc;
437  victim->cache_vers = uc->uc_vers;
438  victim->cache_prog = uc->uc_prog;
439  victim->cache_addr = uc->uc_addr;
440  loc = CACHE_LOC(xprt, victim->cache_xid);
441  victim->cache_next = uc->uc_entries[loc];
442  uc->uc_entries[loc] = victim;
443  uc->uc_fifo[uc->uc_nextvictim++] = victim;
444  uc->uc_nextvictim %= uc->uc_size;
445 }

451 static int
452 cache_get(
453  SVCXPRT *xprt,
454  struct rpc_msg *msg,
455  char **replyp,
456  u_long *replylenp)
457 {

460  register struct svcudp_data *su = su_data(xprt);
461  register struct udp_cache *uc = (struct udp_cache *) su->su_cache;
462 
463 # define EQADDR(a1, a2) (memcmp(&a1, &a2, sizeof(a1)) == 0)
464 
465  loc = CACHE_LOC(xprt, su->su_xid);
466  for (ent = uc->uc_entries[loc]; ent != NULL; ent = ent->cache_next) {
467  if (ent->cache_xid == su->su_xid &&
468  ent->cache_proc == uc->uc_proc &&
469  ent->cache_vers == uc->uc_vers &&
470  ent->cache_prog == uc->uc_prog &&
471  EQADDR(ent->cache_addr, uc->uc_addr)) {
472  *replyp = ent->cache_reply;
473  *replylenp = ent->cache_replylen;

484  uc->uc_addr = xprt->xp_raddr;
485  return(0);
