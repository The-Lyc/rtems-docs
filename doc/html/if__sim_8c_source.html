123 static void *
124 alloc_mbuf_rx(int *psz, uintptr_t *paddr)
125 {
126 struct mbuf *m;
127 unsigned long l,o;
128 
129  MGETHDR(m, M_DONTWAIT, MT_DATA);
130  if ( !m )
131  return 0;
132  MCLGET(m, M_DONTWAIT);
133  if ( ! (m->m_flags & M_EXT) ) {
134  m_freem(m);
135  return 0;
136  }
137 
138  o = mtod(m, unsigned long);
139  l = ALIGNTO(o, RX_BUF_ALIGNMENT) - o;
140 
141  /* align start of buffer */
142  m->m_data += l;
143 
144  /* reduced length */
145  l = MCLBYTES - l;
146 
147  m->m_len = m->m_pkthdr.len = l;
148  *psz = m->m_len;
149  *paddr = mtod(m, unsigned long);
150 
151  return (void*) m;
152 }

201 static void
202 ifsim_start(struct ifnet *ifp)
203 {
204 struct ifsim_softc *sc = ifp->if_softc;
205 struct mbuf *m, *mh, *m1;
206 
207  while ( ifp->if_snd.ifq_head ) {
208  IF_DEQUEUE( &ifp->if_snd, mh );
209  for ( m=mh, m1 = m->m_next ; m1 ; m1 = m1->m_next ) {
210  ifsim_out(sc, IFSIM_TX_BUF_REG, mtod(m, uint32_t));
211  ifsim_out(sc, IFSIM_TX_STA_REG, m->m_len);
212  /* dummy-busywait; the emulated hardware DMAs our
213  * data away 'immediately' i.e., this loop is
214  * never executed
215  */
216  while ( ! (IFSIM_TX_STA_DONE & ifsim_in(sc, IFSIM_TX_STA_REG)) )
217  /* Loop */;
218  m = m1;
219  }
220  ifsim_out(sc, IFSIM_TX_BUF_REG, mtod(m, uint32_t));
221  ifsim_out(sc, IFSIM_TX_STA_REG, m->m_len | IFSIM_TX_STA_LST);
222 
223  /* dummy-busywait; the emulated hardware DMAs our
224  * data away 'immediately' i.e., this loop is
225  * never executed
226  */
227  while ( ! (IFSIM_TX_STA_DONE & ifsim_in(sc, IFSIM_TX_STA_REG)) )
228  /* Loop */;
229 
230  m_freem(mh);
231  }
232 }

310 static void
311 ifsim_task(void *arg)
312 {
313 struct ifsim_softc *sc;
314 uint32_t sta;
315 struct ifnet *ifp;
316 unsigned len;
317 rtems_event_set evs;
318 
319  while (1) {
320 
321  rtems_bsdnet_event_receive(
322  ((1<<IFSIM_SLOTS)-1),
323  RTEMS_WAIT | RTEMS_EVENT_ANY,
324  RTEMS_NO_TIMEOUT,
325  &evs);
326 
327  evs &= ((1<<IFSIM_SLOTS)-1);
328 
329 #ifdef IRQ_DEBUG
330  printk("Task got evs %u\n", evs);
331 #endif
332 
333  for ( sc = theIfSims; evs; evs>>=1, sc++ ) {
334 
335  if ( ! ( evs & 1 ) )
336  continue;
337 
338  ifp = &sc->arpcom.ac_if;
339 
340  while ( ifsim_in(sc, IFSIM_IRQ_REG) & IFSIM_RX_IRQ ) {
341  struct mbuf *m = sc->pvt.rbuf;
342 
343  sta = ifsim_in(sc, IFSIM_RX_STA_REG);
344 
345 
346  if ( (sta & IFSIM_RX_STA_DONE) ) {
347 
348  if ( (ifp->if_flags & IFF_RUNNING) ) {
349  if ( 0 == get_rxbuf(sc) ) {
350  /* enqueue packet */
351  struct ether_header *eh;
352  uint32_t crc_net, crc;
353  int crc_len;
354 
355  crc_len = (IFSIM_CSR_CRCEN & ifsim_in(sc, IFSIM_CSR_REG)) ? sizeof(crc_net) : 0;
356 
357  len = (sta & IFSIM_RX_CNT_MSK) - crc_len;
358 
359  eh = (struct ether_header*)(mtod(m,unsigned long) + ETH_RX_OFFSET);
360 
361  m->m_len = m->m_pkthdr.len = len - sizeof(struct ether_header) - ETH_RX_OFFSET;
362  m->m_data += sizeof(struct ether_header) + ETH_RX_OFFSET;
363  m->m_pkthdr.rcvif = ifp;
364 
365 #ifdef DEBUG_WO_BSDNET
366  {
367  int i;
368  for ( i=0; i<len + crc_len; ) {
369  printk("%02X ",((char*)eh)[i]);
370  if ( 0 == (++i & 0xf) )
371  fputc('\n',stdout);
372  }
373  if ( i & 0xf )
374  fputc('\n', stdout);
375  printk("*****\n");
376  }
377 #endif
378 
379  if ( crc_len
380  && (memcpy(&crc_net, (char*)eh + len, crc_len),
381  (crc = (ether_crc32_le((uint8_t *)eh, len) ^ 0xffffffff)) != crc_net) ) {
382  printk("CSUM: me 0x%08" PRIx32 ", them 0x%08" PRIx32 "\n", crc, crc_net);
383  sc->pvt.rx_cserrs++;
384  } else {
385 
386  ifp->if_ipackets++;
387  ifp->if_ibytes += len;
388 
389 #ifdef DEBUG_WO_BSDNET
390  m_freem(m);
391 #else
392  ether_input(ifp, eh, m);
393 #endif
394 
395  m = 0;
396  }
397 
398  } else {
399  /* throw away and reuse buffer */
400  sc->pvt.rx_nobufs++;
401  }
402  }
403  } else {
404  /* throw away and reuse buffer */
405  if ( (sta & IFSIM_RX_ERR_NOSPC) )
406  sc->pvt.rx_err_nospc++;
407  if ( (sta & IFSIM_RX_ERR_DMA) )
408  sc->pvt.rx_err_dma++;
409  if ( (sta & IFSIM_RX_ERR_RD) )
410  sc->pvt.rx_err_rd++;
411  }
412 
413  if ( m ) {
414  /* reuse */
415  ifsim_out(sc, IFSIM_RX_STA_REG, m->m_pkthdr.len);
416  }
417  }
418  /* re-enable interrupt */
419  ifsim_out(sc, IFSIM_IEN_REG, IFSIM_RX_IRQ);
420  }
421  }
