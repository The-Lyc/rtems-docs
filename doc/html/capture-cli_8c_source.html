58 static void
59 rtems_capture_cli_open (int argc,
60  char** argv,
61  const rtems_monitor_command_arg_t* command_arg RC_UNUSED,
62  bool verbose RC_UNUSED)
63 {
64  uint32_t size = 0;
65  bool enable = false;
66  rtems_status_code sc;
67  int arg;
68 
69  if (argc <= 1)
70  {
71  fprintf (stdout, open_usage);
72  return;
73  }
74 
75  for (arg = 1; arg < argc; arg++)
76  {
77  if (argv[arg][0] == '-')
78  {
79  if (argv[arg][1] == 'i')
80  enable = true;
81  else
82  fprintf (stdout, "warning: option -%c ignored\n", argv[arg][1]);
83  }
84  else
85  {
86  size = strtoul (argv[arg], 0, 0);
87 
88  if (size < 100)
89  {
90  fprintf (stdout, "error: size must be greater than or equal to 100\n");
91  return;
92  }
93  }
94  }
95 
96  sc = rtems_capture_open (size, capture_timestamp);
97 
98  if (sc != RTEMS_SUCCESSFUL)
99  {
100  fprintf (stdout, "error: open failed: %s\n", rtems_status_text (sc));
101  return;
102  }
103 
104  fprintf (stdout, "capture engine opened.\n");
105 
106  if (!enable)
107  return;
108 
109  sc = rtems_capture_set_control (enable);
110 
111  if (sc != RTEMS_SUCCESSFUL)
112  {
113  fprintf (stdout, "error: open enable failed: %s\n", rtems_status_text (sc));
114  return;
115  }
116 
117  fprintf (stdout, "capture engine enabled.\n");
118 }

195 static bool
196 rtems_capture_cli_print_task (rtems_tcb *tcb, void *arg)
197 {
198  rtems_task_priority ceiling = rtems_capture_watch_get_ceiling ();
199  rtems_task_priority floor = rtems_capture_watch_get_floor ();
200  rtems_task_priority priority;
201  int length;
202  uint32_t flags = rtems_capture_task_control_flags (tcb);
203 
204  priority = rtems_capture_task_real_priority (tcb);
205 
206  fprintf (stdout, " ");
207  rtems_monitor_dump_id (rtems_capture_task_id (tcb));
208  fprintf (stdout, " ");
209  if (rtems_capture_task_api (rtems_capture_task_id (tcb)) != OBJECTS_POSIX_API)
210  {
211  rtems_monitor_dump_name (rtems_capture_task_id (tcb));
212  fprintf (stdout, " ");
213  }
214  else
215  {
216  fprintf (stdout, " ");
217  }
218  rtems_monitor_dump_priority (rtems_capture_task_start_priority (tcb));
219  fprintf (stdout, " ");
220  rtems_monitor_dump_priority (rtems_capture_task_real_priority (tcb));
221  fprintf (stdout, " ");
222  rtems_monitor_dump_priority (rtems_capture_task_curr_priority (tcb));
223  fprintf (stdout, " ");
224  length = rtems_monitor_dump_state (rtems_capture_task_state (tcb));
225  fprintf (stdout, "%*c", 14 - length, ' ');
226  fprintf (stdout, " %c%c",
227  'a',
228  flags & RTEMS_CAPTURE_TRACED ? 't' : '-');
229 
230  if ((floor > ceiling) && (ceiling > priority))
231  fprintf (stdout, "--");
232  else
233  {
234  fprintf (stdout, "%c%c",
235  rtems_capture_task_control (tcb) ?
236  (flags & RTEMS_CAPTURE_WATCH ? 'w' : '+') : '-',
237  rtems_capture_watch_global_on () ? 'g' : '-');
238  }
239  fprintf (stdout, "\n");
240  return false;
241 }

306 static bool
307 rtems_capture_cli_get_name_id (char* arg,
308  bool* valid_name,
309  bool* valid_id,
310  rtems_name* name,
311  rtems_id* id)
312 {
313  size_t l;
314  size_t i;
315 
316  if (*valid_name && *valid_id)
317  {
318  fprintf (stdout, "error: too many arguments\n");
319  return 0;
320  }
321 
322  /*
323  * See if the arg is all hex digits.
324  */
325 
326  l = strlen (arg);
327 
328  for (i = 0; i < l; i++)
329  if (!isxdigit ((unsigned char)arg[i]))
330  break;
331 
332  if (i == l)
333  {
334  *id = strtoul (arg, 0, 16);
335  *valid_id = true;
336  }
337  else
338  {
339  /*
340  * This is a bit of hack but it should work on all platforms
341  * as it is what the score does with names.
342  *
343  * @warning The extra assigns play with the byte order so do not
344  * remove unless the score has been updated.
345  */
346  rtems_name rname;
347 
348  rname = rtems_build_name(arg[0], arg[1], arg[2], arg[3]);
349  *name = rname;
350  *valid_name = true;
351  }
352 
353  return 1;
354 }

741 static void
742 rtems_capture_cli_trigger_worker (int set, int argc, char** argv)
743 {
744  rtems_status_code sc;
745  int arg;
746  int trigger = 0; /* switch */
747  rtems_capture_trigger_mode trigger_mode = rtems_capture_from_any;
748  bool trigger_set = false;
749  bool is_from = false;
750  bool is_to = false;
751  rtems_name name = 0;
752  rtems_id id = 0;
753  bool valid_name = false;
754  bool valid_id = false;
755  rtems_name from_name = 0;
756  rtems_id from_id = 0;
757  bool from_valid_name = false;
758  bool from_valid_id = false;
759  rtems_name to_name = 0;
760  rtems_id to_id = 0;
761  bool to_valid_name = false;
762  bool to_valid_id = false;
763 
764  for (arg = 1; arg < argc; arg++)
765  {
766  if (argv[arg][0] == '-')
767  {
768  switch (argv[arg][1])
769  {
770  case '?':
771  fprintf (stdout, trigger_set_usage, set ? "ctset" : "ctclear");
772  fprintf (stdout, trigger_set_types);
773  return;
774  default:
775  fprintf (stdout, "warning: option -%c ignored\n", argv[arg][1]);
776  break;
777  }
778  }
779  else
780  {
781  if (!trigger_set)
782  {
783  bool found = false;
784  int t;
785 
786  for (t = 0; t < RTEMS_CAPTURE_CLI_TRIGGERS_NUM; t++)
787  if (strcmp (argv[arg], rtems_capture_cli_trigger[t].name) == 0)
