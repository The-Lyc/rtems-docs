144 /* NOTE: RTEMS (ss-20020301) uses a 'short st_ino' type :-( but the
145  * NFS fileid is 32 bit. [Later versions of RTEMS have fixed this;
146  * nfsInit() issues a warning if you run a version with 'short st_ino'.]
147  *
148  * As a workarount, we merge the upper 16bits of the fileid into the
149  * minor device no. Hence, it is still possible to uniquely identify
150  * a file by looking at its device number (major = nfs, minor = part
151  * of the fileid + our 'nfs-id' identifier).
152  *
153  * This has an impact on performance, as e.g. getcwd() stats() all
154  * directory entries when it believes it has crossed a mount point
155  * (a.st_dev != b.st_dev).
156  *
157  * OTOH, it also might cause node comparison failure! E.g. 'getcwd()'
158  * assumes that two nodes residing in the same directory must be located
159  * on the same device and hence compares 'st_ino' only.
160  * If two files in the same directory have the same inode number
161  * modulo 2^16, they will be considered equal (although their device
162  * number doesn't match - getcwd doesn't look at it).
163  *
164  * Other software might or might not be affected.
165  *
166  * The only clean solution to this problem is bumping up the size of
167  * 'ino_t' at least to 'long'.
168  * Note that this requires _all_ software (libraries etc.) to be
169  * recompiled.
170  */

285 static bool_t
286 xdr_dir_info_entry(XDR *xdrs, DirInfo di)
287 {
288 union {
289  char nambuf[NFS_MAXNAMLEN+1];
290  nfscookie cookie;
291 } dummy;
292 struct dirent *pde = (struct dirent *)di->ptr;
293 u_int fileid;
294 char *name;
295 register int nlen = 0,len,naligned = 0;
296 nfscookie *pcookie;
297 
298  len = di->len;
299 
300  if ( !xdr_u_int(xdrs, &fileid) )
301  return FALSE;
302 
303  /* we must pass the address of a char* */
304  name = (len > NFS_MAXNAMLEN) ? pde->d_name : dummy.nambuf;
305 
306  if ( !xdr_filename(xdrs, &name) ) {
307  return FALSE;
308  }
309 
310  if (len >= 0) {
311  nlen = strlen(name);
312  naligned = nlen + 1 /* string delimiter */ + 3 /* alignment */;
313  naligned &= ~3;
314  len -= naligned;
315  }
316 
317  /* if the cookie goes into the DirInfo, we hope this doesn't fail
318  * - the caller ends up with an invalid readdirargs cookie otherwise...
319  */
320  pcookie = (len >= 0) ? &di->readdirargs.cookie : &dummy.cookie;
321  if ( !xdr_nfscookie(xdrs, pcookie) ) {
322  return FALSE;
323  }
324 
325  di->len = len;
326  /* adjust the buffer pointer */
327  if (len >= 0) {
328  pde->d_ino = fileid;
329  pde->d_namlen = nlen;
330  pde->d_off = di->ptr - di->buf;
331 #ifdef DT_UNKNOWN
332  pde->d_type = DT_UNKNOWN;
333 #endif
334  if (name == dummy.nambuf) {
335  memcpy(pde->d_name, dummy.nambuf, nlen + 1);
336  }
337  pde->d_reclen = DIRENT_HEADER_SIZE + naligned;
338  di->ptr += pde->d_reclen;
339  }
340 
341  return TRUE;
342 }

345 static bool_t
346 xdr_dir_info(XDR *xdrs, DirInfo di)
347 {
348 DirInfo dip;
349 
350  if ( !xdr_nfsstat(xdrs, &di->status) )
351  return FALSE;
352 
353  if ( NFS_OK != di->status )
354  return TRUE;
355 
356  dip = di;
357 
358  while (dip) {
359  /* reserve space for the dirent 'header' - we assume it's word aligned! */
360 #ifdef DEBUG
361  assert( DIRENT_HEADER_SIZE % 4 == 0 );
362 #endif
363  dip->len -= DIRENT_HEADER_SIZE;
364 
365  /* we pass a 0 size - size is unused since
366  * we always pass a non-NULL pointer
367  */
368  if ( !xdr_pointer(xdrs, (void*)&dip, 0 /* size */, (xdrproc_t)xdr_dir_info_entry) )
369  return FALSE;
370  }
371 
372  if ( ! xdr_bool(xdrs, &di->eofreached) )
373  return FALSE;
374 
375  /* if everything fits into the XDR buffer but not the user's buffer,
376  * they must resume reading from where xdr_dir_info_entry() started
377  * skipping and 'eofreached' needs to be adjusted
378  */
379  if ( di->len < 0 && di->eofreached )
380  di->eofreached = FALSE;
381 
382  return TRUE;
383 }

1762 int rtems_nfs_initialize(
1763  rtems_filesystem_mount_table_entry_t *mt_entry,
1764  const void *data
1765 )
1766 {
1767 char *host;
1768 struct sockaddr_in saddr;
1769 enum clnt_stat stat;
1770 fhstatus fhstat;
1771 u_long uid,gid;
1772 #ifdef NFS_V2_PORT
1773 int retry;
1774 #endif
1775 Nfs nfs = 0;
1776 NfsNode rootNode = 0;
1777 RpcUdpServer nfsServer = 0;
1778 int e = -1;
1779 char *path = mt_entry->dev;
1780 
1781  if (rpcUdpInit () < 0) {
1782  fprintf (stderr, "error: initialising RPC\n");
1783  return -1;
1784  }
1785 
1786  if (nfsInit(0, 0) != 0) {
1787  fprintf (stderr, "error: initialising NFS\n");
1788  return -1;
1789  };
1790 
1791 #if 0
1792  printf("Trying to mount %s on %s\n",path,mntpoint);
1793 #endif
1794 
1795  if ( buildIpAddr(&uid, &gid, &host, &saddr, &path) )
1796  return -1;
1797 
1798 #ifdef NFS_V2_PORT
1799  /* if the portmapper fails, retry a fixed port */
1800  for (retry = 1, saddr.sin_port = 0, stat = RPC_FAILED;
1801  retry >= 0 && stat;
1802  stat && (saddr.sin_port = htons(NFS_V2_PORT)), retry-- )
1803 #endif
1804  stat = rpcUdpServerCreate(
1805  &saddr,
1806  NFS_PROGRAM,
1807  NFS_VERSION_2,
1808  uid,
1809  gid,
1810  &nfsServer
1811  );
1812 
1813  if ( RPC_SUCCESS != stat ) {
1814  fprintf(stderr,
1815  "Unable to contact NFS server - invalid port? (%s)\n",
1816  clnt_sperrno(stat));
1817  e = EPROTONOSUPPORT;
1818  goto cleanup;
1819  }
1820 
1821 
1822  /* first, try to ping the NFS server by
1823  * calling the NULL proc.
1824  */
1825  if ( nfscall(nfsServer,
1826  NFSPROC_NULL,
1827  (xdrproc_t)xdr_void, 0,
1828  (xdrproc_t)xdr_void, 0) ) {
1829 
1830  fputs("NFS Ping ",stderr);
1831  fwrite(host, 1, path-host-1, stderr);
1832  fprintf(stderr," failed: %s\n", strerror(errno));
1833 
1834  e = errno ? errno : EIO;
1835  goto cleanup;
1836  }
1837 
1838  /* that seemed to work - we now try the
1839  * actual mount
1840  */
1841 
1842  /* reuse server address but let the mntcall()
1843  * search for the mountd's port
1844  */
1845  saddr.sin_port = 0;
1846 
1847  stat = mntcall( &saddr,
1848  MOUNTPROC_MNT,
1849  (xdrproc_t)xdr_dirpath,
1850  &path,
1851  (xdrproc_t)xdr_fhstatus,
1852  &fhstat,
1853  uid,
1854  gid );
1855 
1856  if (stat) {
1857  fprintf(stderr,"MOUNT -- %s\n",clnt_sperrno(stat));
1858  if ( e<=0 )
1859  e = EIO;
1860  goto cleanup;
1861  } else if (NFS_OK != (e=fhstat.fhs_status)) {
1862  fprintf(stderr,"MOUNT: %s\n",strerror(e));
1863  goto cleanup;
1864  }
1865 
1866  nfs = nfsCreate(nfsServer);
1867  assert( nfs );
1868  nfsServer = 0;
1869 
1870  nfs->uid = uid;
1871  nfs->gid = gid;
1872 
1873  /* that seemed to work - we now create the root node
1874  * and we also must obtain the root node attributes
1875  */
1876  rootNode = nfsNodeCreate(nfs, &fhstat.fhstatus_u.fhs_fhandle);
1877  assert( rootNode );
1878 
1879  if ( updateAttr(rootNode, 1 /* force */) ) {
1880  e = errno;
1881  goto cleanup;
1882  }
1883 
1884  /* looks good so far */
1885 
1886  mt_entry->mt_fs_root->location.node_access = rootNode;
1887 
1888  rootNode = 0;
1889 
1890  mt_entry->ops = &nfs_fs_ops;
1891  mt_entry->mt_fs_root->location.handlers = &nfs_dir_file_handlers;
1892  mt_entry->pathconf_limits_and_options = &nfs_limits_and_options;
1893 
1894  LOCK(nfsGlob.llock);
1895  nfsGlob.num_mounted_fs++;
1896  /* allocate a new ID for this FS */
1897  nfs->id = nfsGlob.fs_ids++;
1898  UNLOCK(nfsGlob.llock);
1899 
1900  mt_entry->fs_info = nfs;
1901  nfs->mt_entry = mt_entry;
1902  nfs = 0;
1903 
1904  e = 0;
1905 
1906 cleanup:
1907  if (nfs)
1908  nfsDestroy(nfs);
1909  if (nfsServer)
1910  rpcUdpServerDestroy(nfsServer);
1911  if (rootNode)
1912  nfsNodeDestroy(rootNode);
1913  if (e)
1914  rtems_set_errno_and_return_minus_one(e);
1915  else
1916  return 0;
1917 }
