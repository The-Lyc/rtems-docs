32  void *arg)

34  rtems_status_code sc;

36  i2c_bus_number bus = DS1307_I2C_BUS_NUMBER;
37  i2c_address addr = DS1307_I2C_ADDRESS;
38  int try = 0;
39  do {
40  status = i2c_wrbyte(bus, addr, 0);
41  if (status == I2C_NO_DEVICE)
42  break;
43  try++;
44  } while ((try < 15) && (status != I2C_SUCCESSFUL));
45 
46  if (status == I2C_SUCCESSFUL)
47  {
48  sc = rtems_io_register_name("/dev/nvram", major, 0);

51  errno = EIO;

86 /* nvram_driver_read --
87  * Non-volatile memory device driver read primitive.
88  */
89 rtems_device_driver
90 nvram_driver_read(rtems_device_major_number major,
91  rtems_device_minor_number minor,
92  void *arg)
93 {
94  rtems_libio_rw_args_t *args = arg;
95  uint32_t count;
96  i2c_bus_number bus = DS1307_I2C_BUS_NUMBER;
97  i2c_address addr = DS1307_I2C_ADDRESS;
98  i2c_message_status status;
99  if (args->offset >= DS1307_NVRAM_SIZE)
100  {
101  count = 0;
102  }
103  else if (args->offset + args->count >= DS1307_NVRAM_SIZE)
104  {
105  count = DS1307_NVRAM_SIZE - args->offset;
106  }
107  else
108  {
109  count = args->count;
110  }
111  if (count > 0)
112  {
113  int try = 0;
114  do {
115  status = i2c_wbrd(bus, addr, DS1307_NVRAM_START + args->offset,
116  args->buffer, count);
117  try++;
118  } while ((try < 15) && (status != I2C_SUCCESSFUL));
119  if (status != I2C_SUCCESSFUL)
120  {
121  errno = EIO;
122  return RTEMS_UNSATISFIED;
123  }
124  }
125  args->bytes_moved = count;
126  return RTEMS_SUCCESSFUL;
127 }
