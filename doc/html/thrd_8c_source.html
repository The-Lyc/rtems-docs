51 thrd_create(thrd_t *thr, thrd_start_t func, void *arg)
52 {
53  struct thrd_param *tp;
54 
55  /*
56  * Work around return type inconsistency. Wrap execution using
57  * a function conforming to pthread_create()'s start_routine.
58  */
59  tp = malloc(sizeof(*tp));
60  if (tp == NULL)
61  return (thrd_nomem);
62  tp->func = func;
63  tp->arg = arg;
64  if (pthread_create(thr, NULL, thrd_entry, tp) != 0) {
65  free(tp);
66  return (thrd_error);
67  }
68  return (thrd_success);
69 }
