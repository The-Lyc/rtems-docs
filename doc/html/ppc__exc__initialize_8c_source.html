35 static void ppc_exc_initialize_booke(void *vector_base)
36 {
37  /* Interrupt vector prefix register */
38  MTIVPR((uintptr_t) vector_base);
39 
40  if (
41  ppc_cpu_is_specific_e200(PPC_e200z0)
42  || ppc_cpu_is_specific_e200(PPC_e200z1)
43  ) {
44  /*
45  * These cores have hard wired IVOR registers. An access will case a
46  * program exception.
47  */
48  return;
49  }

83 void ppc_exc_initialize_with_vector_base(
84  uintptr_t interrupt_stack_begin,
85  void *vector_base
86 )
87 {
88  rtems_status_code sc = RTEMS_SUCCESSFUL;
89  const ppc_exc_categories *const categories = ppc_exc_current_categories();
90  unsigned vector = 0;
91  uint32_t sda_base = 0;
92  uint32_t r13 = 0;
93 
94  if (categories == NULL) {
95  ppc_exc_fatal_error();
96  }
97 
98  /* Assembly code needs SDA_BASE in r13 (SVR4 or EABI). Make sure
99  * early init code put it there.
100  */
101  __asm__ volatile (
102  "lis %0, _SDA_BASE_@h\n"
103  "ori %0, %0, _SDA_BASE_@l\n"
104  "mr %1, 13\n"
105  : "=r" (sda_base), "=r"(r13)
106  );
107 
108  if (sda_base != r13) {
109  ppc_exc_fatal_error();
110  }
111 
112  ppc_exc_initialize_interrupt_stack(interrupt_stack_begin);
113 
114 #ifndef PPC_EXC_CONFIG_BOOKE_ONLY
115 
116  /* Use current MMU / RI settings when running C exception handlers */
117  ppc_exc_msr_bits = ppc_machine_state_register() & (MSR_DR | MSR_IR | MSR_RI);
118 
119 #ifdef __ALTIVEC__
120  /* Need vector unit enabled to save/restore altivec context */
121  ppc_exc_msr_bits |= MSR_VE;
122 #endif
123 
124 #endif /* PPC_EXC_CONFIG_BOOKE_ONLY */
125 
126  if (ppc_cpu_is_bookE() == PPC_BOOKE_STD || ppc_cpu_is_bookE() == PPC_BOOKE_E500) {
127  ppc_exc_initialize_booke(vector_base);
128  }
129 
130  for (vector = 0; vector <= LAST_VALID_EXC; ++vector) {
131  ppc_exc_category category = ppc_exc_category_for_vector(categories, vector);
132 
133  if (category != PPC_EXC_INVALID) {
134  void *const vector_address = ppc_exc_vector_address(vector, vector_base);
135  uint32_t prologue [16];
136  size_t prologue_size = sizeof(prologue);
137 
138  sc = ppc_exc_make_prologue(
139  vector,
140  vector_base,
141  category,
142  prologue,
143  &prologue_size
144  );
145  if (sc != RTEMS_SUCCESSFUL) {
146  ppc_exc_fatal_error();
147  }
148 
149  ppc_code_copy(vector_address, prologue, prologue_size);
150  }
151  }
152 
153 #ifndef PPC_EXC_CONFIG_BOOKE_ONLY
154  /* If we are on a classic PPC with MSR_DR enabled then
155  * assert that the mapping for at least this task's
156  * stack is write-back-caching enabled (see README/CAVEATS)
157  * Do this only if the cache is physically enabled.
158  * Since it is not easy to figure that out in a
159  * generic way we need help from the BSP: BSPs
160  * which run entirely w/o the cache may set
161  * ppc_exc_cache_wb_check to zero prior to calling
162  * this routine.
163  *
164  * We run this check only after exception handling is
165  * initialized so that we have some chance to get
166  * information printed if it fails.
167  *
168  * Note that it is unsafe to ignore this issue; if
169  * the check fails, do NOT disable it unless caches
170  * are always physically disabled.
171  */
172  if (ppc_exc_cache_wb_check && (MSR_DR & ppc_exc_msr_bits)) {
173  /* The size of 63 assumes cache lines are at most 32 bytes */
174  uint8_t dummy[63];
175  uintptr_t p = (uintptr_t) dummy;
176  /* If the dcbz instruction raises an alignment exception
177  * then the stack is mapped as write-thru or caching-disabled.
178  * The low-level code is not capable of dealing with this
179  * ATM.
180  */
181  p = (p + 31U) & ~31U;
182  __asm__ volatile ("dcbz 0, %0"::"b" (p));
183  /* If we make it thru here then things seem to be OK */
184  }
185 #endif /* PPC_EXC_CONFIG_BOOKE_ONLY */
186 }
