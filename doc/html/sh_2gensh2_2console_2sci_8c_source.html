96 /* Translate termios' tcflag_t into sci settings */
97 static int _sci_set_cflags(
98  struct scidev_t *sci_dev,
99  tcflag_t c_cflag,
100  speed_t spd
101 )
102 {
103  uint8_t smr;
104  uint8_t brr;
105 
106  if ( spd )
107  {
108  if ( _sci_get_brparms( spd, &smr, &brr ) != 0 )
109  return -1;
110  }
111 
112  if ( c_cflag & CSIZE )
113  {
114  if ( c_cflag & CS8 )
115  smr &= ~SCI_SEVEN_BIT_DATA;
116  else if ( c_cflag & CS7 )
117  smr |= SCI_SEVEN_BIT_DATA;
118  else
119  return -1;
120  }
121 
122  if ( c_cflag & CSTOPB )
123  smr |= SCI_STOP_BITS_2;
124  else
125  smr &= ~SCI_STOP_BITS_2;
126 
127  if ( c_cflag & PARENB )
128  smr |= SCI_PARITY_ON;
129  else
130  smr &= ~SCI_PARITY_ON;
131 
132  if ( c_cflag & PARODD )
133  smr |= SCI_ODD_PARITY;
134  else
135  smr &= ~SCI_ODD_PARITY;
136 
137  write8( smr, sci_dev->addr + SCI_SMR );
138  write8( brr, sci_dev->addr + SCI_BRR );
139 
140  return 0;
141 }

200 static bool rdSCI0(unsigned char *ch)
201 {
202  uint8_t temp;
203  bool result = false;
204 
205  if ((read8(SCI_SSR0) & SCI_RDRF) != 0x00) {
206  /* read input */
207  *ch = read8(SCI_RDR0);
208  /* Clear RDRF flag */
209  temp = read8(SCI_SSR0) & ~SCI_RDRF;
210  write8(temp, SCI_SSR0);
211  /* Check for transmission errors */
212  if (temp & (SCI_ORER | SCI_FER | SCI_PER)) {
213  /* TODO: report to RTEMS transmission error */
214 
215  /* clear error flags*/
216  temp &= ~(SCI_ORER | SCI_FER | SCI_PER);
217  write8(temp, SCI_SSR0);

219  result = true;

269 /* sh_sci_initialize
270  *
271  * This routine initializes (registers) the sh_sci IO drivers.
272  *
273  * Input parameters: ignored
274  *
275  * Output parameters: NONE
276  *
277  * Return values: RTEMS_SUCCESSFUL
278  * if all sci[...] register, else calls
279  * rtems_fatal_error_occurred(status)
280  */
281 rtems_device_driver sh_sci_initialize(
282  rtems_device_major_number major,
283  rtems_device_minor_number minor,
284  void *arg )
285 {
286  rtems_device_driver status;
287  rtems_device_minor_number i;
288 
289  /*
290  * register all possible devices.
291  * the initialization of the hardware is done by sci_open
292  *
293  * One of devices could be previously registered by console
294  * initialization therefore we check it everytime
295  */
296  for ( i = 0 ; i < SCI_MINOR_DEVICES ; i++ ) {
297  /* OK. We assume it is not registered yet. */
298  status = rtems_io_register_name(
299  sci_device[i].name,
300  major,
301  sci_device[i].minor
302  );
303  if (status != RTEMS_SUCCESSFUL)
304  rtems_fatal_error_occurred(status);
305  }
306 
307  /* non-default hardware setup occurs in sh_sci_open() */
308  return RTEMS_SUCCESSFUL;
309 }

315 rtems_device_driver sh_sci_open(
316  rtems_device_major_number major,
317  rtems_device_minor_number minor,
318  void * arg )
319 {
320  uint8_t temp8;
321  uint16_t temp16;
322 
323  unsigned a;
324 
325  /* check for valid minor number */
326  if (( minor > ( SCI_MINOR_DEVICES -1 )) || ( minor < 0 )) {
327  return RTEMS_INVALID_NUMBER;
328  }
329 
330  /* device already opened */
331  if ( sci_device[minor].opened > 0 ) {
332  sci_device[minor].opened++;
333  return RTEMS_SUCCESSFUL;
334  }
335 
336  /* set PFC registers to enable I/O pins */
337 
338  if ((minor == 0)) {
339  temp16 = read16(PFC_PACRL2); /* disable SCK0, DMA, IRQ */
340  temp16 &= ~(PA2MD1 | PA2MD0);
341  temp16 |= (PA_TXD0 | PA_RXD0); /* enable pins for Tx0, Rx0 */
342  write16(temp16, PFC_PACRL2);
343 
344  } else if (minor == 1) {
345  temp16 = read16(PFC_PACRL2); /* disable SCK1, DMA, IRQ */
346  temp16 &= ~(PA5MD1 | PA5MD0);
347  temp16 |= (PA_TXD1 | PA_RXD1); /* enable pins for Tx1, Rx1 */
348  write16(temp16, PFC_PACRL2);
349 
350  } /* add other devices and pins as req'd. */
351 
352  /* set up SCI registers */
353  write8(0x00, sci_device[minor].addr + SCI_SCR); /* Clear SCR */
354  /* set SMR and BRR */
355  _sci_set_cflags( &sci_device[minor], sci_device[minor].cflags, sci_device[minor].spd );
356 
357  for (a=0; a < 10000L; a++) { /* Delay */
358  __asm__ volatile ("nop");
359  }
360 
361  write8((SCI_RE | SCI_TE), /* enable async. Tx and Rx */
362  sci_device[minor].addr + SCI_SCR);
363 
364  /* clear error flags */
365  temp8 = read8(sci_device[minor].addr + SCI_SSR);
366  while (temp8 & (SCI_RDRF | SCI_ORER | SCI_FER | SCI_PER)) {
367  temp8 = read8(sci_device[minor].addr + SCI_RDR); /* flush input */
368  temp8 = read8(sci_device[minor].addr + SCI_SSR); /* clear some flags */
369  write8(temp8 & ~(SCI_RDRF | SCI_ORER | SCI_FER | SCI_PER),
370  sci_device[minor].addr + SCI_SSR);
371  temp8 = read8(sci_device[minor].addr + SCI_SSR); /* check if everything is OK */
372  }
373  /* Clear RDRF flag */
374  write8(0x00, sci_device[minor].addr + SCI_TDR); /* force output */
375  /* Clear the TDRE bit */
376  temp8 = read8(sci_device[minor].addr + SCI_SSR) & ~SCI_TDRE;
377  write8(temp8, sci_device[minor].addr + SCI_SSR);
378 
379  /* add interrupt setup if required */
380 
381 
382  sci_device[minor].opened++;
383 
384  return RTEMS_SUCCESSFUL;
385 }
