286 static rtems_status_code mpc55xx_dspi_send_addr( rtems_libi2c_bus_t *bus, uint32_t addr, int rw)
287 {
288  mpc55xx_dspi_bus_entry *e = (mpc55xx_dspi_bus_entry *) bus;
289  union DSPI_SR_tag sr = MPC55XX_ZERO_FLAGS;
290 
291  /* Flush transmit and receive FIFO */
292  e->regs->MCR.B.CLR_TXF = 1;
293  e->regs->MCR.B.CLR_RXF = 1;
294 
295  /* Clear status flags */
296  sr.B.EOQF = 1;
297  sr.B.TFFF = 1;
298  sr.B.RFDF = 1;
299  e->regs->SR.R = sr.R;
300 
301  /* Frame command */
302  e->push_data.R = 0;
303  e->push_data.B.CONT = 0;
304  e->push_data.B.CTAS = MPC55XX_DSPI_CTAR_DEFAULT;
305  e->push_data.B.EOQ = 0;
306  e->push_data.B.CTCNT = 0;
307  switch (addr) {
308  case 0:
309  e->push_data.B.PCS0 = 1;
310  break;
311  case 1:
312  e->push_data.B.PCS1 = 1;
313  break;
314  case 2:
315  e->push_data.B.PCS2 = 1;
316  break;
317  case 3:
318  e->push_data.B.PCS3 = 1;
319  break;
320  case 4:
321  e->push_data.B.PCS4 = 1;
322  break;
323  case 5:
324  e->push_data.B.PCS5 = 1;
325  break;
326  default:
327  return -RTEMS_INVALID_ADDRESS;
328  }
329  mpc55xx_dspi_store_push_data( e);
330 
331  return RTEMS_SUCCESSFUL;
332 }

337 static int mpc55xx_dspi_set_transfer_mode( rtems_libi2c_bus_t *bus, const rtems_libi2c_tfr_mode_t *mode)
338 {
339  mpc55xx_dspi_bus_entry *e = (mpc55xx_dspi_bus_entry *) bus;
340  union DSPI_CTAR_tag ctar = MPC55XX_ZERO_FLAGS;
341 
342  if (mode->bits_per_char != 8) {
343  return -RTEMS_INVALID_NUMBER;
344  }
345 
346  e->idle_char = mode->idle_char;
347 
348  ctar.R = e->regs->CTAR [MPC55XX_DSPI_CTAR_DEFAULT].R;
349 
350  ctar.B.PCSSCK = 0;
351  ctar.B.CSSCK = 0;
352  ctar.B.PASC = 0;
353  ctar.B.ASC = 0;
354 
355  ctar.B.LSBFE = mode->lsb_first ? 1 : 0;
356  ctar.B.CPOL = mode->clock_inv ? 1 : 0;
357  ctar.B.CPHA = mode->clock_phs ? 1 : 0;
358 
359  if (mode->baudrate != e->baud) {
360  uint32_t scaler = bsp_clock_speed / mode->baudrate;
361  mpc55xx_dspi_baudrate_scaler_entry bse = mpc55xx_dspi_search_baudrate_scaler( scaler, 0, MPC55XX_DSPI_BAUDRATE_SCALER_TABLE_SIZE / 2, MPC55XX_DSPI_BAUDRATE_SCALER_TABLE_SIZE);
362 
363  ctar.B.PBR = bse.pbr;
364  ctar.B.BR = bse.br;
365 
366  e->baud = mode->baudrate;
367  }
368 
369  e->regs->CTAR [MPC55XX_DSPI_CTAR_DEFAULT].R = ctar.R;
370 
371  return 0;
372 }

410 static int mpc55xx_dspi_read_write( rtems_libi2c_bus_t *bus, unsigned char *in, const unsigned char *out, int n)
411 {
412  mpc55xx_dspi_bus_entry *e = (mpc55xx_dspi_bus_entry *) bus;
413 
414  /* Non cache aligned characters */
415  int n_nc = n;
416 
417  /* Cache aligned characters */
418  int n_c = 0;
419 
420  /* Register addresses */
421  volatile void *push = &e->regs->PUSHR.R;
422  volatile void *pop = &e->regs->POPR.R;
423  volatile union DSPI_SR_tag *status = &e->regs->SR;
424 
425  /* Push and pop data */
426  union DSPI_PUSHR_tag push_data = e->push_data;
427  union DSPI_POPR_tag pop_data;
428 
429  /* Status register */
430  union DSPI_SR_tag sr;
431 
432  /* Read and write indices */
433  int r = 0;
434  int w = 0;
435 
436  if (n == 0) {
437  return 0;
438  } else if (in == NULL && out == NULL) {
439  return -RTEMS_INVALID_ADDRESS;
440  }
441 
442  if (n > MPC55XX_DSPI_EDMA_MAGIC_SIZE) {
443  n_nc = (int) mpc55xx_non_cache_aligned_size( in);
444  n_c = (int) mpc55xx_cache_aligned_size( in, (size_t) n);
445  if (n_c > EDMA_TCD_BITER_LINKED_SIZE) {
446  RTEMS_SYSLOG_WARNING( "buffer size out of range, cannot use eDMA\n");
447  n_nc = n;
448  n_c = 0;
449  } else if (n_nc + n_c != n) {
450  RTEMS_SYSLOG_WARNING( "input buffer not proper cache aligned, cannot use eDMA\n");
451  n_nc = n;
452  n_c = 0;
453  }
454  }
455 
456 #ifdef DEBUG
457  if (e->regs->SR.B.TXCTR != e->regs->SR.B.RXCTR) {
458  RTEMS_SYSLOG_WARNING( "FIFO counter not equal\n");
459  }
460 #endif /* DEBUG */
461 
462  /* Direct IO */
463  if (out == NULL) {
464  push_data.B.TXDATA = e->idle_char;
465  while (r < n_nc || w < n_nc) {
466  /* Wait for available FIFO */
467  do {
468  sr.R = status->R;
469  } while (sr.B.TXCTR == MPC55XX_DSPI_FIFO_SIZE && sr.B.RXCTR == 0);
470 
471  /* Write */
472  if (w < n_nc && (w - r) < MPC55XX_DSPI_FIFO_SIZE && sr.B.TXCTR != MPC55XX_DSPI_FIFO_SIZE) {
473  ++w;
474  ppc_write_word( push_data.R, push);
475  }
476 
477  /* Read */
478  if (r < n_nc && sr.B.RXCTR != 0) {
479  pop_data.R = ppc_read_word( pop);
480  in [r] = (unsigned char) pop_data.B.RXDATA;
481  ++r;
482  }
483  }
484  } else if (in == NULL) {
485  while (r < n_nc || w < n_nc) {
486  /* Wait for available FIFO */
487  do {
488  sr.R = status->R;
489  } while (sr.B.TXCTR == MPC55XX_DSPI_FIFO_SIZE && sr.B.RXCTR == 0);
490 
491  /* Write */
492  if (w < n_nc && (w - r) < MPC55XX_DSPI_FIFO_SIZE && sr.B.TXCTR != MPC55XX_DSPI_FIFO_SIZE) {
493  push_data.B.TXDATA = out [w];
494  ++w;
495  ppc_write_word( push_data.R, push);
496  }
497 
498  /* Read */
499  if (r < n_nc && sr.B.RXCTR != 0) {
500  pop_data.R = ppc_read_word( pop);
501  ++r;
502  }
503  }
504  } else {
505  while (r < n_nc || w < n_nc) {
506  /* Wait for available FIFO */
507  do {
508  sr.R = status->R;
509  } while (sr.B.TXCTR == MPC55XX_DSPI_FIFO_SIZE && sr.B.RXCTR == 0);
510 
511  /* Write */
512  if (w < n_nc && (w - r) < MPC55XX_DSPI_FIFO_SIZE && sr.B.TXCTR != MPC55XX_DSPI_FIFO_SIZE) {
513  push_data.B.TXDATA = out [w];
514  ++w;
515  ppc_write_word( push_data.R, push);
516  }
517 
518  /* Read */
519  if (r < n_nc && sr.B.RXCTR != 0) {
520  pop_data.R = ppc_read_word( pop);
521  in [r] = (unsigned char) pop_data.B.RXDATA;
522  ++r;
523  }
524  }
525  }
526 
527  /* eDMA transfers */
528  if (n_c > 0) {
529  rtems_status_code sc = RTEMS_SUCCESSFUL;
530  unsigned char *in_c = in + n_nc;
531  const unsigned char *out_c = out + n_nc;
532  struct tcd_t tcd_transmit = EDMA_TCD_DEFAULT;
533  struct tcd_t tcd_receive = EDMA_TCD_DEFAULT;
534 
535  /* Cache operations */
536  rtems_cache_flush_multiple_data_lines( out_c, (size_t) n_c);
537  rtems_cache_invalidate_multiple_data_lines( in_c, (size_t) n_c);
538 
539  /* Set transmit TCD */
540  if (out == NULL) {
541  e->push_data.B.TXDATA = e->idle_char;
542  mpc55xx_dspi_store_push_data( e);
543  tcd_transmit.SADDR = mpc55xx_dspi_push_data_address( e);
544  tcd_transmit.SDF.B.SSIZE = 2;
545  tcd_transmit.SDF.B.SOFF = 0;
546  tcd_transmit.DADDR = (uint32_t) push;
547  tcd_transmit.SDF.B.DSIZE = 2;
548  tcd_transmit.CDF.B.DOFF = 0;
549  tcd_transmit.NBYTES = 4;
550  tcd_transmit.CDF.B.CITER = n_c;
551  tcd_transmit.BMF.B.BITER = n_c;
552  } else {
553  unsigned push_channel = mpc55xx_edma_channel_by_tcd( e->edma_push.edma.edma_tcd);
554  mpc55xx_edma_clear_done( e->edma_transmit.edma.edma_tcd);
555  tcd_transmit.SADDR = (uint32_t) out_c;
556  tcd_transmit.SDF.B.SSIZE = 0;
557  tcd_transmit.SDF.B.SOFF = 1;
558  tcd_transmit.DADDR = mpc55xx_dspi_push_data_address( e) + 3;
559  tcd_transmit.SDF.B.DSIZE = 0;
560  tcd_transmit.CDF.B.DOFF = 0;
561  tcd_transmit.NBYTES = 1;
562  tcd_transmit.CDF.B.CITERE_LINK = 1;
563  tcd_transmit.BMF.B.BITERE_LINK = 1;
564  tcd_transmit.BMF.B.MAJORLINKCH = push_channel;
565  tcd_transmit.CDF.B.CITER = EDMA_TCD_LINK_AND_BITER( push_channel, n_c);
566  tcd_transmit.BMF.B.BITER = EDMA_TCD_LINK_AND_BITER( push_channel, n_c);
567  tcd_transmit.BMF.B.MAJORE_LINK = 1;
568  }
569  tcd_transmit.BMF.B.D_REQ = 1;
570  tcd_transmit.BMF.B.INT_MAJ = 1;
571  *e->edma_transmit.edma.edma_tcd = tcd_transmit;
572 
573  /* Set receive TCD */
574  if (in == NULL) {
575  tcd_receive.CDF.B.DOFF = 0;
576  tcd_receive.DADDR = mpc55xx_dspi_nirvana_address( e);
577  } else {
578  tcd_receive.CDF.B.DOFF = 1;
579  tcd_receive.DADDR = (uint32_t) in_c;
580  }
581  tcd_receive.SADDR = (uint32_t) pop + 3;
582  tcd_receive.SDF.B.SSIZE = 0;
583  tcd_receive.SDF.B.SOFF = 0;
584  tcd_receive.SDF.B.DSIZE = 0;
585  tcd_receive.NBYTES = 1;
586  tcd_receive.BMF.B.D_REQ = 1;
587  tcd_receive.BMF.B.INT_MAJ = 1;
588  tcd_receive.CDF.B.CITER = n_c;
589  tcd_receive.BMF.B.BITER = n_c;
590  *e->edma_receive.edma.edma_tcd = tcd_receive;
591 
592  /* Clear request flags */
593  sr.R = 0;
594  sr.B.TFFF = 1;
595  sr.B.RFDF = 1;
596  status->R = sr.R;
597 
598  /* Enable hardware requests */
599  mpc55xx_edma_enable_hardware_requests( e->edma_receive.edma.edma_tcd);
600  mpc55xx_edma_enable_hardware_requests( e->edma_transmit.edma.edma_tcd);
601
