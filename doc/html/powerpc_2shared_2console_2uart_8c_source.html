117 /*
118  * Uart initialization, it is hardcoded to 8 bit, no parity,
119  * one stop bit, FIFO, things to be changed
120  * are baud rate and nad hw flow control,
121  * and longest rx fifo setting
122  */
123 void
124 BSP_uart_init(int uart, int baud, int hwFlow)
125 {
126  unsigned char tmp;
127 
128  /* Sanity check */
129  SANITY_CHECK(uart);
130 
131  /* Make sure any printk activity drains before
132  * re-initializing.
133  */
134  while ( ! (uread(uart, LSR) & TEMT) )
135  ;
136 
137  switch(baud)
138  {
139  case 50:
140  case 75:
141  case 110:
142  case 134:
143  case 300:
144  case 600:
145  case 1200:
146  case 2400:
147  case 9600:
148  case 19200:
149  case 38400:
150  case 57600:
151  case 115200:
152  break;
153  default:
154  assert(0);
155  return;
156  }
157 
158  /* Set DLAB bit to 1 */
159  uwrite(uart, LCR, DLAB);
160 
161  if ( (int)BSPBaseBaud <= 0 ) {
162  /* Use current divisor assuming BSPBaseBaud gives us the current speed */
163  BSPBaseBaud = BSPBaseBaud ? -BSPBaseBaud : 9600;
164  BSPBaseBaud *= ((uread(uart, DLM) << 8) | uread(uart, DLL));
165  }
166 
167  /* Set baud rate */
168  uwrite(uart, DLL, (BSPBaseBaud/baud) & 0xff);
169  uwrite(uart, DLM, ((BSPBaseBaud/baud) >> 8) & 0xff);
170 
171  /* 8-bit, no parity , 1 stop */
172  uwrite(uart, LCR, CHR_8_BITS);
173 
174  /* Set DTR, RTS and OUT2 high */
175  uwrite(uart, MCR, DTR | RTS | OUT_2);
176 
177  /* Enable FIFO */
178  uwrite(uart, FCR, FIFO_EN | XMIT_RESET | RCV_RESET | RECEIVE_FIFO_TRIGGER12);
179 
180  /* Disable Interrupts */
181  uwrite(uart, IER, 0);
182 
183  /* Read status to clear them */
184  tmp = uread(uart, LSR);
185  tmp = uread(uart, RBR);
186  tmp = uread(uart, MSR);
187  (void) tmp; /* avoid set but not used warning */
188 
189  /* Remember state */
190  uart_data[uart].hwFlow = hwFlow;
191  uart_data[uart].baud = baud;
192  return;
193 }

335 /*
336  * Status function, -1 if error
337  * detected, 0 if no received chars available,
338  * 1 if received char available, 2 if break
339  * is detected, it will eat break and error
340  * chars. It ignores overruns - we cannot do
341  * anything about - it execpt count statistics
342  * and we are not counting it.
343  */
344 int
345 BSP_uart_polled_status(int uart)
346 {
347  unsigned char val;
348 
349  SANITY_CHECK(uart);
350 
351  val = uread(uart, LSR);
352 
353  if(val & BI)
354  {
355  /* BREAK found, eat character */
356  uread(uart, RBR);
357  return BSP_UART_STATUS_BREAK;
358  }
359 
360  if((val & (DR | OE | FE)) == 1)
361  {
362  /* No error, character present */
363  return BSP_UART_STATUS_CHAR;
364  }
365 
366  if((val & (DR | OE | FE)) == 0)
367  {
368  /* Nothing */
369  return BSP_UART_STATUS_NOCHAR;
370  }
371 
372  /*
373  * Framing or parity error
374  * eat character
375  */
376  uread(uart, RBR);
377 
378  return BSP_UART_STATUS_ERROR;

701  )
702  );
703  termios_tx_active_com[uart] = 0;
704  }
705  break;
706  case RECEIVER_DATA_AVAIL :
707  case CHARACTER_TIMEOUT_INDICATION:

709  {

718  }

722  assert(off < sizeof(buf));

724  buf[off++] = uread(uart, RBR);
725  }

730  break;
731  default:
732  /* Should not happen */
733  assert(0);
734  return;
735  }
736  }
