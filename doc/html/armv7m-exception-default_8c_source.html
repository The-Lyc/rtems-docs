23 void __attribute__((naked)) _ARMV7M_Exception_default( void )
24 {
25  /* On exception entry, ARMv7M saves context state onto a stack pointed to
26  * by either MSP or PSP. The value stored in LR indicates whether we were
27  * in Thread or Handler mode, whether we were using the FPU (if any),
28  * and which stack pointer we were using.
29  * In particular, bit 2 of LR will be 0 if we were using MSP.
30  *
31  * For a more detailed explanation, see the Exception Entry Behavior
32  * section of the ARMv7M Architecture Reference Manual.
33  */
34 
35  /* As we're in Handler mode here, we'll always operate on MSP.
36  * However, we need to store the right SP in our CPU_Exception_frame.
37  */
38  __asm__ volatile (
39  "sub sp, %[cpufsz]\n" /* Allocate space for a CPU_Exception_frame. */
40  "stm sp, {r0-r12}\n"
41  "tst lr, #4\n" /* Check if bit 2 of LR is zero. If so, PSR.Z = 1 */
42  "itte eq\n" /* IF bit 2 of LR is zero... (PSR.Z == 1) */
43  "mrseq r3, msp\n" /* THEN we were using MSP */
44  "addeq r3, %[cpufsz]\n" /* THEN, set r3 = old MSP value */
45  "mrsne r3, psp\n" /* ELSE it is not zero; we were using PSP */
46  "add r2, r3, %[v7mlroff]\n"
47  "add r1, sp, %[cpuspoff]\n"
48  "ldm r2, {r4-r6}\n" /* Grab LR, PC and xPSR from the stack */
49  "tst lr, #16\n" /* Check if we have an FP state on the frame */
50  "ite eq\n"
51  "addeq r3, #104\n" /* Back to previous SP with FP state */
52  "addne r3, #32\n" /* Back to previous SP without FP state */
53  "tst r6, #512\n" /* Check xPSR if the SP was aligned */
54  "it ne\n"
55  "addne r3, #4\n" /* Undo alignment */
56  "stm r1, {r3-r6}\n" /* Store to CPU_Exception_frame */
57  "mrs r1, ipsr\n"
58  "str r1, [sp, %[cpuvecoff]]\n"
59 
60  /* Argument for high level handler */
61  "mov r0, sp\n"
62 
63  /* Clear VFP context pointer */
64  "add r3, sp, %[cpuvfpoff]\n"
65  "mov r1, #0\n"
66  "str r1, [r3]\n"
67 
68 #ifdef ARM_MULTILIB_VFP
69  /* Ensure that the FPU is enabled */
70  "ldr r4, =%[cpacr]\n"
71  "tst r4, #(0xf << 20)\n"
72  "bne 1f\n"
73 
74  /* Save VFP context */
75  "sub sp, %[vfpsz]\n"
76  "add r4, sp, #4\n"
77  "bic r4, r4, #7\n"
78  "str r4, [r3]\n"
79  "vmrs r2, FPSCR\n"
80  "stmia r4!, {r1-r2}\n"
81  "vstmia r4!, {d0-d15}\n"
82  "mov r1, #0\n"
83  "mov r2, #0\n"
84  "adds r3, r4, #128\n"
85  "2:\n"
86  "stmia r4!, {r1-r2}\n"
87  "cmp r4, r3\n"
88  "bne 2b\n"
89  "1:\n"
90 #endif
91 
92  "b _ARM_Exception_default\n"
93  :
94  : [cpufsz] "i" (sizeof(CPU_Exception_frame)),
95  [cpuspoff] "i" (offsetof(CPU_Exception_frame, register_sp)),
96  [v7mlroff] "i" (offsetof(ARMV7M_Exception_frame, register_lr)),
97  [cpuvecoff] "J" (offsetof(CPU_Exception_frame, vector)),
98  [cpuvfpoff] "i" (ARM_EXCEPTION_FRAME_VFP_CONTEXT_OFFSET),
99  [cpacr] "i" (ARMV7M_CPACR),
100  [vfpsz] "i" (ARM_VFP_CONTEXT_SIZE)
101  );
102 }
